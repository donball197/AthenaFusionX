use crate::agent_protocol::{AgentDirective, AgentStatus};
use crate::agents::NexusAgent;
use crate::kernel;
use anyhow::Result;
use async_trait::async_trait;

pub struct ScribeAgent;

impl ScribeAgent {
    pub fn enshrine(handle: u32, offset: usize, data: &str) -> bool {
        let bytes = data.as_bytes();
        unsafe {
            kernel::zx_vmo_write_native(handle, offset, bytes.as_ptr(), bytes.len()) == 0
        }
    }

    pub fn recall(handle: u32, offset: usize, size: usize) -> Option<String> {
        let mut buffer = vec![0u8; size];
        unsafe {
            if kernel::zx_vmo_read_native(handle, offset, buffer.as_mut_ptr(), size) == 0 {
                String::from_utf8(buffer).ok()
            } else {
                None
            }
        }
    }

    pub fn scry_vault(handle: u32, offset: usize, size: usize) {
        let mut buffer = vec![0u8; size];
        unsafe {
            if kernel::zx_vmo_read_native(handle, offset, buffer.as_mut_ptr(), size) == 0 {
                println!("\x1b[1;30m--- [SCRY: 0x{:x} | Offset: {}] ---\x1b[0m", handle, offset);
                for chunk in buffer.chunks(16) {
                    let hex: String = chunk.iter().map(|b| format!("{:02x} ", b)).collect();
                    let ascii: String = chunk
                        .iter()
                        .map(|&b| if b >= 32 && b <= 126 { b as char } else { '.' })
                        .collect();
                    println!("\x1b[35m{:32} \x1b[1;37m| {}\x1b[0m", hex, ascii);
                }
            } else {
                println!(
                    "\x1b[31m[SCRY] Access Denied: Handle 0x{:x} is opaque or invalid.\x1b[0m",
                    handle
                );
            }
        }
    }
}

    }

    }

fn extract_last_json(text: &str) -> Option<String> {
    let bytes = text.as_bytes();
    let mut end = None;
    for (i, &b) in bytes.iter().enumerate().rev() {
        if b == b'}' {
            end = Some(i);
            break;
        }
    }
    let end = end?;
    let mut depth: i32 = 0;
    let mut start = None;
    for (i, &b) in bytes[..=end].iter().enumerate().rev() {
        match b {
            b'}' => depth += 1,
            b'{' => {
                depth -= 1;
                if depth == 0 {
                    start = Some(i);
                    break;
                }
            }
            _ => {}
        }
    }
    let start = start?;
    let candidate = &text[start..=end];
    match serde_json::from_str::<serde_json::Value>(candidate) {
        Ok(v) if v.is_object() => {
            Some(serde_json::to_string(&v).unwrap_or_else(|_| candidate.to_string()))
        }
        _ => None,
    }
}

use anyhow::Result;
use nexus_singularity::neural_agent::{ContextPacket, MicroAgent};
use std::fs;
use std::sync::Arc;
use tokio::sync::broadcast;
use walkdir::WalkDir;
use std::path::PathBuf;

struct Domain {
    name: &'static str,
    paths: Vec<&'static str>,
}

async fn analyze_domain_hierarchical(
    domain: &Domain,
    agent: &Arc<MicroAgent>,
    client: &reqwest::Client,
    api_key: &str,
) -> Result<String> {
    let mut file_stack: Vec<PathBuf> = Vec::new();
    let mut current_batch = String::new();
    let mut batch_summaries = Vec::new();
    let max_batch_chars = 30_000;

    println!("\x1b[36m[*] Hierarchical Scan: {}\x1b[0m", domain.name);
    for path in &domain.paths {
        for entry in WalkDir::new(path).into_iter().flatten() {
            let p = entry.path();
            if p.is_file() {
                // ⚡ FIX: Explicitly typed closure parameter for extension matching
                if let Some(ext) = p.extension().and_then(|s: &std::ffi::OsStr| s.to_str()) {
                    if matches!(ext, "rs" | "svelte" | "js" | "ts" | "zig" | "json") {
                        if p.to_string_lossy().contains("node_modules") || p.to_string_lossy().contains("-lock.json") {
                            continue;
                        }
                        file_stack.push(p.to_owned());
                    }
                }
            }
        }
    }

    for file_path in file_stack {
        // ⚡ FIX: Specify the generic path type for read_to_string
        if let Ok(content) = fs::read_to_string::<&PathBuf>(&file_path) {
            let entry_text = format!("\n--- FILE: {} ---\n{}\n", file_path.display(), content);
            if current_batch.len() + entry_text.len() > max_batch_chars {
                let summary = request_batch_summary(agent, client, api_key, &current_batch, domain.name).await?;
                batch_summaries.push(summary);
                current_batch.clear();
                tokio::time::sleep(tokio::time::Duration::from_secs(5)).await;
            }
            current_batch.push_str(&entry_text);
        }
    }

    if !current_batch.is_empty() {
        let summary = request_batch_summary(agent, client, api_key, &current_batch, domain.name).await?;
        batch_summaries.push(summary);
    }

use crate::agent_protocol::AgentAction;
use serde::{Deserialize, Serialize};
use std::collections::hash_map::DefaultHasher;
use std::hash::{Hash, Hasher};
use std::time::{SystemTime, UNIX_EPOCH};

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct MerkleLeaf {
    pub action: AgentAction,
    pub actor_id: String,
    pub timestamp: u64,
    pub hash: u64,
}

impl MerkleLeaf {
    pub fn new(action: AgentAction, actor_id: &str) -> Self {
        let ts = SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_secs();
        let mut hasher = DefaultHasher::new();
        serde_json::to_vec(&action).unwrap_or_default().hash(&mut hasher);
        actor_id.hash(&mut hasher);
        ts.hash(&mut hasher);
        Self { action, actor_id: actor_id.to_string(), timestamp: ts, hash: hasher.finish() }
    }
}

#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct GovernanceLog {
    pub history: Vec<MerkleLeaf>,
    pub root_hash: u64,
}

impl GovernanceLog {
    pub fn init() -> Self { Self::default() }

    /// record: Incremental update of the system root hash.
    pub fn record(&mut self, action: AgentAction, actor_id: &str) {
        let leaf = MerkleLeaf::new(action, actor_id);
        
        // Incremental: Root = Hash(OldRoot + NewLeaf)
        let mut hasher = DefaultHasher::new();
        self.root_hash.hash(&mut hasher);
        leaf.hash.hash(&mut hasher);
        self.root_hash = hasher.finish();

        self.history.push(leaf);
        
        if self.history.len() % 10 == 0 {
            info!("\x1b[34m[GOVERNANCE] Checkpoint: Root=0x{:x} (Size: {})\x1b[0m", self.root_hash, self.history.len());
        }
    }
}

pub struct SafeBuffer {
    data: Vec<u8>,
    max_capacity: usize,
}

impl SafeBuffer {
    pub fn new(max_capacity: usize) -> Self {
        Self { data: Vec::with_capacity(max_capacity), max_capacity }
    }

    pub fn write_safe(&mut self, payload: &[u8]) -> Result<(), String> {
        if self.data.len() + payload.len() > self.max_capacity {
            return Err(format!(
                "MEMORY FAULT: Buffer overflow prevented. Tried to write {} bytes, only {} remaining.",
                payload.len(),
                self.max_capacity.saturating_sub(self.data.len())
            ));
        }
        self.data.extend_from_slice(payload);
        Ok(())
    }

    /// Returns current length.
    pub fn len(&self) -> usize {
        self.data.len()
    }

    /// Returns true if buffer is empty.
    pub fn is_empty(&self) -> bool {
        self.data.is_empty()
    }

    /// Clears the buffer, resetting it for reuse.
    pub fn clear(&mut self) {
        self.data.clear();
    }

    /// Returns a read-only slice of the buffer contents.
    pub fn as_slice(&self) -> &[u8] {
        &self.data
    }
}

    pub refcount: u32,
    ops: Arc<dyn KernelOps>,
}

impl Vmo {
    pub fn new(ops: Arc<dyn KernelOps>, size: usize) -> Result<Self, KernelError> {
        let handle = ops.vmo_create(size);
        if handle == 0 {
            Err(KernelError::VmoCloneFailed { parent: 0, size })
        } else {
            Ok(Self { handle, size, refcount: 1, ops })
        }
    }

    /// Zircon-style clone with Snapshot capability.
    /// _options bit 0 = ZX_VMO_CHILD_SNAPSHOT (reserved for future CoW enforcement)
    pub fn clone_vmo(&self, _options: u32) -> Result<Self, KernelError> {
        let child_handle = self.ops.vmo_clone(self.handle, self.size);
        if child_handle == 0 {
            Err(KernelError::VmoCloneFailed { parent: self.handle, size: self.size })
        } else {
            println!(
                "\x1b[36m[VMO] COW Clone Created — Handle {} → {} [RESONANCE: 1.00]\x1b[0m",
                self.handle, child_handle
            );
            Ok(Self {
                handle: child_handle,
                size: self.size,
                refcount: 1,
                ops: self.ops.clone(),
            })
        }
    }
}

impl Drop for Vmo {
    fn drop(&mut self) {
        self.ops.handle_close(self.handle);
    }
}

/// AverageConsensus: Distributed thermal averaging for swarm_sim binary.
#[derive(Debug, Clone)]
pub struct ConsensusNode {
    pub name: String,
    pub local_state: f64,
}

pub struct AverageConsensus {
    pub nodes: Vec<ConsensusNode>,
}

impl AverageConsensus {
    pub fn new(initial: Vec<(&str, f64)>) -> Self {
        Self {
            nodes: initial
                .into_iter()
                .map(|(name, state)| ConsensusNode { name: name.to_string(), local_state: state })
                .collect(),
        }
    }

    /// iterate: One step of the average-consensus algorithm (epsilon = mixing rate).
    pub fn iterate(&mut self, epsilon: f64) {
        let mean: f64 =
            self.nodes.iter().map(|n| n.local_state).sum::<f64>() / self.nodes.len() as f64;
        for node in self.nodes.iter_mut() {
            node.local_state += epsilon * (mean - node.local_state);
        }
    }

    /// converge: Run up to max_iterations until all nodes are within 0.01 of each other.
    pub fn converge(&mut self, epsilon: f64, max_iterations: usize) {
        for _ in 0..max_iterations {
            let min = self.nodes.iter().map(|n| n.local_state).fold(f64::INFINITY, f64::min);
            let max = self.nodes.iter().map(|n| n.local_state).fold(f64::NEG_INFINITY, f64::max);
            if (max - min).abs() < 0.01 {
                break;
            }
            self.iterate(epsilon);
        }
    }

    /// persist_to_vmo: Stub — writes consensus result to kernel VMO handle.
    pub fn persist_to_vmo(&self, handle: usize) {
        let result = self.nodes.first().map(|n| n.local_state).unwrap_or(0.0);

const COMMAND_ALLOWLIST: &[&str] = &[
    "cargo check", "cargo build", "cargo test", "cargo clippy",
    "git status", "git log", "git diff", "ls", "find src", "cat src/",
];

impl ArbiterAgent {
    pub fn new(agent: Arc<MicroAgent>) -> Self { Self { agent } }

    pub fn validate_action(action_type: &str, command: &str) -> Result<(), String> {
        // ⚡ FIX: ALWAYS run the danger-pattern scan first.
        let dangerous_patterns = [
            "rm -rf /", "rm -rf ~", "mkfs", "> /dev/sda", "> /dev/nvme",
            "chmod 777 /", "/etc/shadow", "mv /etc/", "mv /usr/",
            ":(){ :|:& };:", "dd if=/dev/zero", "fork bomb",
            "sudo su", "sudo -i", "sudo bash", "sudo sh", "passwd root", "chown root",
            "$'\\x", "\\u0072\\u006d", "base64 -d |", "| bash", "| sh", "eval $(", "exec(",
            "curl | bash", "wget | bash", "curl | sh",
            "history -c", "unset HISTFILE",
        ];

        let normalized = command.to_lowercase();

        for pattern in dangerous_patterns.iter() {
            if normalized.contains(pattern) {
                return Err(format!("SANDBOX_VIOLATION: Threat pattern '{}' detected in {}.", pattern, action_type));
            }
        }

        if normalized.contains("../../../") || normalized.contains("/etc/passwd") {
            return Err("SANDBOX_VIOLATION: Directory traversal attempt blocked.".to_string());
        }

        if (normalized.contains("| bash") || normalized.contains("| sh") || normalized.contains("|bash") || normalized.contains("|sh")) && !normalized.starts_with("echo") {
            return Err("SANDBOX_VIOLATION: Pipe-to-shell pattern blocked.".to_string());
        }

        // ⚡ FIX: Allowlist is enforced ONLY AFTER sandbox patterns are cleared.
        if std::env::var("NEXUS_STRICT_COMMANDS").is_ok() {
            let permitted = COMMAND_ALLOWLIST.iter().any(|&allowed| {
                command.trim().starts_with(allowed)
            });
            if !permitted {
                return Err(format!("ALLOWLIST_VIOLATION: '{}' is not in the permitted command set.", command));
            }
        }

        Ok(())
    }

    pub fn evaluate_entropy_delta(base_entropy: f32, new_entropy: f32) -> Result<(), String> {
        let delta = (new_entropy - base_entropy).abs();
        if delta > 0.5 { return Err(format!("ENTROPY_ANOMALY: Delta of {:.4} exceeds 0.5 threshold.", delta)); }
        Ok(())
    }

    pub fn resolve_consensus(alpha_vector: &[f32], beta_vector: &[f32]) -> bool {
        if alpha_vector.len() != beta_vector.len() { return false; }
        let diff: f32 = alpha_vector.iter().zip(beta_vector.iter()).map(|(a, b)| (a - b).abs()).sum();
        diff < 0.5
    }

    pub fn bft_consensus(vectors: &[Vec<f32>], epsilon: f32) -> Option<Vec<f32>> {
        if vectors.is_empty() { return None; }
        let n = vectors.len();
        let threshold = (2 * n / 3) + 1;
        let dim = vectors[0].len();
        if vectors.iter().any(|v| v.len() != dim) { return None; }

        let (best_idx, neighbor_count) = (0..n)
            .map(|i| {
                let neighbors = (0..n).filter(|&j| { vectors[i].iter().zip(vectors[j].iter()).map(|(a, b)| (a - b).abs()).sum::<f32>() < epsilon }).count();
                (i, neighbors)
            }).max_by_key(|&(_, count)| count)?;
            
        if neighbor_count < threshold { return None; }

        let agreeing: Vec<&Vec<f32>> = (0..n).filter(|&j| { vectors[best_idx].iter().zip(vectors[j].iter()).map(|(a, b)| (a - b).abs()).sum::<f32>() < epsilon }).map(|j| &vectors[j]).collect();
        let centroid: Vec<f32> = (0..dim).map(|k| agreeing.iter().map(|v| v[k]).sum::<f32>() / agreeing.len() as f32).collect();
        Some(centroid)
    }
}

const MIN_KEY_BYTES: usize = 32;
const DEFAULT_TTL_SECS: u64 = 86_400;

#[derive(Debug, Serialize, Deserialize)]
pub struct Claims {
    pub sub: String,
    pub exp: usize,
    pub iat: usize,
}

fn load_key() -> Result<String> {
    let key = std::env::var("NEXUS_AUTH_KEY")
        .map_err(|_| anyhow!("NEXUS_AUTH_KEY env var is not set"))?;
    if key.len() < MIN_KEY_BYTES {
        return Err(anyhow!(
            "NEXUS_AUTH_KEY is too short ({} bytes). Minimum is {} bytes.",
            key.len(),
            MIN_KEY_BYTES
        ));
    }
    Ok(key)
}

fn token_ttl() -> u64 {
    std::env::var("NEXUS_TOKEN_TTL_SECS")
        .ok()
        .and_then(|v| v.parse().ok())
        .unwrap_or(DEFAULT_TTL_SECS)
}

pub fn issue_sovereign_token() -> Result<String> {
    let key = load_key()?;
    let now = SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs();
    let claims = Claims {
        sub: "architect".to_owned(),
        iat: now as usize,
        exp: (now + token_ttl()) as usize,
    };
    encode(
        &Header::default(),
        &claims,
        &EncodingKey::from_secret(key.as_bytes()),
    )
    .map_err(|e| anyhow!("JWT_GEN_FAIL: {}", e))
}

/// TokenVerdict: Distinguishes between infrastructure failures and policy rejections.
/// Use verify_token_detailed() when you need to log *why* a token was rejected.
#[derive(Debug, PartialEq)]
pub enum TokenVerdict {
    Valid,
    Expired,
    InvalidSignature,
    MissingKey,
}

/// verify_token_detailed: Returns the specific reason for rejection.
/// Err(_) means the auth key is misconfigured — not that the token is bad.
pub fn verify_token_detailed(token: &str) -> Result<TokenVerdict> {
    let key = load_key()?;
    let mut validation = Validation::new(Algorithm::HS256);
    validation.leeway = 0;
    match decode::<Claims>(token, &DecodingKey::from_secret(key.as_bytes()), &validation) {
        Ok(_) => Ok(TokenVerdict::Valid),
        Err(e) => Ok(match e.kind() {
            jsonwebtoken::errors::ErrorKind::ExpiredSignature => TokenVerdict::Expired,
            _ => TokenVerdict::InvalidSignature,
        }),
    }
}

/// verify_token: Backward-compatible shim — returns true only if Valid.
/// Err(_) indicates a key configuration problem, not a bad token.
pub fn verify_token(token: &str) -> Result<bool> {
    match verify_token_detailed(token)? {
        TokenVerdict::Valid => Ok(true),
        TokenVerdict::Expired => {
            tracing::warn!("[AUTH] Token rejected: expired");
            Ok(false)
        }
        TokenVerdict::InvalidSignature => {
            tracing::warn!("[AUTH] Token rejected: invalid signature");
            Ok(false)
        }
        TokenVerdict::MissingKey => Ok(false), // already an Err from load_key
    }
}

#[derive(Clone, Serialize, Deserialize, Debug)]
pub struct ClusterNode {
    pub id: String,
    pub host: String,
    pub port: u16,
    pub cpu: f32,
    pub ram: f32,
    pub temp: f32,
    pub last_seen: u64,
}

#[derive(Clone)]
pub struct ClusterMesh {
    nodes: Arc<RwLock<Vec<ClusterNode>>>,
}

fn now_secs() -> u64 {
    SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap_or_default()
        .as_secs()
}

impl ClusterMesh {
    pub fn new() -> Self {
        Self { nodes: Arc::new(RwLock::new(vec![])) }
    }

    pub fn register(&self, mut node: ClusterNode) {
        node.last_seen = now_secs();
        let mut guard = self.nodes.write();
        if let Some(existing) = guard.iter_mut().find(|n| n.id == node.id) {
            *existing = node;
        } else {
            guard.push(node);
        }
    }

    pub fn get_all_nodes(&self) -> Vec<ClusterNode> { self.nodes.read().clone() }

    pub fn best_node(&self) -> Option<ClusterNode> {
        let now = now_secs();
        self.nodes
            .read()
            .iter()
            .filter(|n| now.saturating_sub(n.last_seen) < 40)
            .cloned()
            .min_by(|a, b| {
                (a.cpu + a.ram)
                    .partial_cmp(&(b.cpu + b.ram))
                    .unwrap_or(std::cmp::Ordering::Equal)
            })
    }
}

/**
 * NEXUS Diff Engine — v2.0 [SURGICAL + FUZZY]
 * Copyright 2026 Donald Ball
 * Handles precise edits even when LLM mangles whitespace or line order.
 */

use anyhow::{anyhow, Result};

pub struct DiffEngine;

impl DiffEngine {
    /// Primary entry point: tries exact match first, then fuzzy/indentation-agnostic.
    pub fn apply(original: &str, search: &str, replace: &str) -> Result<String> {
        if search.trim().is_empty() {
            return Ok(original.to_string());
        }

        // Tier 1: Exact match (fastest, most precise)
        if let Some(pos) = original.find(search) {
            return Ok(replace_first_occurrence(original, pos, search.len(), replace));
        }

        // Tier 2: Indentation-agnostic matching (handles LLM whitespace issues)
        if let Some(result) = Self::apply_agnostic(original, search, replace) {
            return Ok(result);
        }

        // Tier 3: Fuzzy line-based matching (for larger blocks)
        if let Some(result) = Self::apply_fuzzy_lines(original, search, replace) {
            return Ok(result);
        }

        Err(anyhow!(
            "PATTERN_NOT_FOUND: Could not locate search block.\n\
            Search preview (first 120 chars):\n{}\n\n\
            Hint: Try making the search string more unique or shorter.",
            search.chars().take(120).collect::<String>()
        ))
    }

    fn apply_agnostic(original: &str, search: &str, replace: &str) -> Option<String> {
        let search_lines: Vec<&str> = search.lines().map(|l| l.trim_start()).collect();
        let orig_lines: Vec<&str> = original.lines().collect();
        let search_len = search_lines.len();

        if search_len == 0 || orig_lines.len() < search_len {
            return None;
        }

        for i in 0..=(orig_lines.len() - search_len) {
            let window = &orig_lines[i..i + search_len];
            let window_trimmed: Vec<&str> = window.iter().map(|l| l.trim_start()).collect();

            if window_trimmed == search_lines {
                let mut result = orig_lines[..i].join("\n");
                if !result.is_empty() { result.push('\n'); }
                result.push_str(replace);
                if i + search_len < orig_lines.len() {
                    result.push('\n');
                    result.push_str(&orig_lines[i + search_len..].join("\n"));
                }
                return Some(result);
            }
        }
        None
    }

    fn apply_fuzzy_lines(original: &str, search: &str, replace: &str) -> Option<String> {
        // Very coarse fallback: if search is long, try to match by first and last line
        let search_lines: Vec<&str> = search.lines().collect();
        if search_lines.len() < 3 {
            return None;
        }

        let first = search_lines[0].trim();
        let last = search_lines.last().unwrap().trim();

        let orig_lines: Vec<&str> = original.lines().collect();

        for i in 0..orig_lines.len() {
            if orig_lines[i].trim() == first {
                for j in (i + 2)..orig_lines.len() {
                    if orig_lines[j].trim() == last {
                        let mut result = orig_lines[..i].join("\n");
                        if !result.is_empty() { result.push('\n'); }
                        result.push_str(replace);
                        if j + 1 < orig_lines.len() {
                            result.push('\n');
                            result.push_str(&orig_lines[j + 1..].join("\n"));
                        }
                        return Some(result);
                    }
                }
            }
        }
        None
    }

    pub fn unified_diff(_old: &str, _new: &str, path: &str) -> String {
        format!("--- a/{}\n+++ b/{}\n[MODIFICATION APPLIED]", path, path)
    }
}

fn replace_first_occurrence(s: &str, pos: usize, old_len: usize, new: &str) -> String {
    let mut result = String::with_capacity(s.len() - old_len + new.len());
    result.push_str(&s[..pos]);
    result.push_str(new);
    result.push_str(&s[pos + old_len..]);
    result
}

// src/lazarus_vault.rs — v1.4 [LEDGER_FIX]
// FIX: auto_snapshot() now appends every snapshot to ledger.json.
//      Previously it wrote the file to .nexus/vault/<hash> but never
//      touched the ledger, so list_snapshots() returned [] permanently
//      and SelfHealingEngine rollback was always blind.
// FIX: list_snapshots() returns entries sorted oldest→newest so
//      .last() in the rollback path gives the most recent stable state.
use serde::{Deserialize, Serialize};
use std::collections::hash_map::DefaultHasher;
use std::fs;
use std::hash::{Hash, Hasher};
use std::path::Path;
use std::time::SystemTime;

#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct Snapshot {
    pub id:        String,
    pub path:      String,
    pub hash:      String,
    pub timestamp: String,
    pub directive: String,
}

pub struct LazarusVault;

impl LazarusVault {
    const VAULT_DIR:  &'static str = ".nexus/vault";
    const LEDGER_PATH: &'static str = ".nexus/vault/ledger.json";

    pub async fn auto_snapshot(path: &str, tag: &str) -> Result<Snapshot, String> {
        // Reject directories — vault only snapshots files
        let metadata = fs::metadata(path)
            .map_err(|e| format!("PATH_NOT_FOUND: {}", e))?;
        if metadata.is_dir() {
            return Err("DIRECTORY_SNAPSHOT_NOT_SUPPORTED: Specify a file.".into());
        }

        let content = fs::read_to_string(path)
            .map_err(|e| format!("FILE_READ_FAULT: {}", e))?;

        // Hash the content for deduplication
        let mut hasher = DefaultHasher::new();
        content.hash(&mut hasher);
        let hash = format!("{:x}", hasher.finish());

        // Ensure vault directory exists
        fs::create_dir_all(Self::VAULT_DIR)
            .map_err(|e| format!("DIR_FAULT: {}", e))?;

        // Write snapshot file
        let dest = format!("{}/{}", Self::VAULT_DIR, hash);
        fs::write(&dest, &content)
            .map_err(|e| format!("FS_WRITE_FAULT: {}", e))?;

        let snapshot = Snapshot {
            id:        hash.clone(),
            path:      path.to_string(),
            hash:      hash.clone(),
            timestamp: format!("{:?}", SystemTime::now()),
            directive: tag.to_string(),
        };

        // FIX: Persist snapshot to ledger.json
        Self::append_to_ledger(&snapshot)?;

        println!(
            "\x1b[32m[LAZARUS] Snapshot {} → {} [tag: {}]\x1b[0m",
            &hash[..8], path, tag
        );

        Ok(snapshot)
    }

    /// Appends a snapshot entry to ledger.json (creates it if missing).
    fn append_to_ledger(snapshot: &Snapshot) -> Result<(), String> {
        let mut entries = Self::list_snapshots();

        // Deduplicate by hash — don't write the same content twice
        if entries.iter().any(|e| e.hash == snapshot.hash) {
            return Ok(());
        }

        entries.push(snapshot.clone());

        let json = serde_json::to_string_pretty(&entries)
            .map_err(|e| format!("LEDGER_SERIALIZE_FAULT: {}", e))?;

        fs::write(Self::LEDGER_PATH, json)
            .map_err(|e| format!("LEDGER_WRITE_FAULT: {}", e))?;

        Ok(())
    }

    /// Returns all snapshots sorted oldest → newest.
    /// .last() gives the most recent stable snapshot for rollback.
    pub fn list_snapshots() -> Vec<Snapshot> {
        if !Path::new(Self::LEDGER_PATH).exists() {
            return vec![];
        }
        let data = fs::read_to_string(Self::LEDGER_PATH).unwrap_or_default();
        let mut entries: Vec<Snapshot> = serde_json::from_str(&data).unwrap_or_default();
        // Sort by timestamp string ascending (ISO-like strings sort correctly)
        entries.sort_by(|a, b| a.timestamp.cmp(&b.timestamp));
        entries
    }

    /// Returns the most recent snapshot for a given file path.
    pub fn latest_for_path(path: &str) -> Option<Snapshot> {
        Self::list_snapshots()
            .into_iter()
            .filter(|s| s.path == path)
            .last()
    }
}

use tracing::Subscriber;
use tracing_subscriber::prelude::*;
use tracing_subscriber::{Layer, EnvFilter};
use tokio::sync::broadcast;
use std::sync::Mutex;

pub struct NexusTraceLayer {
    pub tx: broadcast::Sender<String>,
}

impl<S: Subscriber> Layer<S> for NexusTraceLayer {
    fn on_event(&self, event: &tracing::Event<'_>, _ctx: tracing_subscriber::layer::Context<'_, S>) {
        let mut payload = String::new();
        let mut visitor = StringVisitor(&mut payload);
        event.record(&mut visitor);

        let level = *event.metadata().level();
        let target = event.metadata().target();
        let clean_payload = payload.trim_matches('"');

        let log_json = serde_json::json!({
            "type": "LOG_EVENT",
            "payload": format!("[{}] {}: {}", level, target, clean_payload)
        }).to_string();

        let _ = self.tx.send(log_json);
        
        if let Ok(mut guard) = TURN_BUFFER.lock() {
            if let Some(buffer) = guard.as_mut() {
                buffer.push(format!("[{}] {}", level, clean_payload));
            }
        }
    }
}

static TURN_BUFFER: Mutex<Option<Vec<String>>> = Mutex::new(None);

pub fn begin_turn_capture() {
    if let Ok(mut guard) = TURN_BUFFER.lock() {
        *guard = Some(Vec::new());
    }
}

pub fn end_turn_capture() -> Vec<String> {
    if let Ok(mut guard) = TURN_BUFFER.lock() {
        guard.take().unwrap_or_default()
    } else {
        Vec::new()
    }
}

struct StringVisitor<'a>(&'a mut String);
impl<'a> tracing::field::Visit for StringVisitor<'a> {
    fn record_debug(&mut self, field: &tracing::field::Field, value: &dyn std::fmt::Debug) {
        if field.name() == "message" {
            self.0.push_str(&format!("{:?}", value));
        }
    }
}

pub fn init_tracing(tx: broadcast::Sender<String>) {
    let layer = NexusTraceLayer { tx };
    
    // ⚡ FIX: Silences the dependency noise
    let filter = EnvFilter::try_from_default_env()
        .unwrap_or_else(|_| EnvFilter::new("info,nexus_singularity=debug,sqlx=warn,hyper=off,neli=off"));

    tracing_subscriber::registry()
        .with(filter)
        .with(layer)
        .with(tracing_subscriber::fmt::layer().with_ansi(true))
        .init();
    
    tracing::info!(">> [TRACING] Sovereign Observability Linked.");
}

// src/memory_bridge.rs — v1.1 [TRUNCATION_COMPRESSION]
//
// FIX: summarize_memory_strand now truncates strands exceeding
//      MAX_STRAND_CHARS to prevent unbounded memory DB growth.
//      A truncation marker is appended so the LLM knows the
//      strand was compressed.
// NOTE: Full Zig pruner integration remains a future upgrade.

const MAX_STRAND_CHARS: usize = 2_000;

/// summarize_memory_strand: Compress a raw memory strand for storage.
/// Currently applies a hard character cap with a truncation marker.
/// Future: delegate to the Zig pruner substrate for semantic compression.
pub fn summarize_memory_strand(raw_text: &str) -> String {
    if raw_text.len() <= MAX_STRAND_CHARS {
        return raw_text.to_string();
    }

    // Find a clean word boundary near the limit
    let boundary = raw_text[..MAX_STRAND_CHARS]
        .rfind(|c: char| c.is_whitespace())
        .unwrap_or(MAX_STRAND_CHARS);

    format!("{}… [TRUNCATED: {} → {} chars]", &raw_text[..boundary], raw_text.len(), boundary)
}

pub struct NexusMetrics {

    pub task_failures:     AtomicU64,
    pub arbiter_blocks:    AtomicU64,
    pub heal_successes:    AtomicU64,
    pub heal_failures:     AtomicU64,
    pub api_latency_ms_sum: AtomicU64,
    pub api_calls:         AtomicU64,
    pub circuit_opens:     AtomicU64,
}

impl NexusMetrics {
    const fn new() -> Self {
        Self {
            tasks_total: AtomicU64::new(0),
            task_failures: AtomicU64::new(0),
            arbiter_blocks: AtomicU64::new(0),
            heal_successes: AtomicU64::new(0),
            heal_failures: AtomicU64::new(0),
            api_latency_ms_sum: AtomicU64::new(0),
            api_calls: AtomicU64::new(0),
            circuit_opens: AtomicU64::new(0),
        }
    }

    /// Increment total tasks processed
    pub fn inc_task(&self) {
        self.tasks_total.fetch_add(1, Ordering::Relaxed);
    }

    /// Increment task failures
    pub fn inc_failure(&self) {
        self.task_failures.fetch_add(1, Ordering::Relaxed);
    }

    /// Increment arbiter blocks (safety interventions)
    pub fn inc_arbiter_block(&self) {
        self.arbiter_blocks.fetch_add(1, Ordering::Relaxed);
    }

    /// Increment successful self-healing sutures
    pub fn inc_heal_success(&self) {
        self.heal_successes.fetch_add(1, Ordering::Relaxed);
    }

    /// Increment failed self-healing attempts
    pub fn inc_heal_failure(&self) {
        self.heal_failures.fetch_add(1, Ordering::Relaxed);
    }

    /// Increment circuit breaker trips
    pub fn inc_circuit_open(&self) {
        self.circuit_opens.fetch_add(1, Ordering::Relaxed);
    }

    /// Record API call with latency (used by Orchestrator)
    pub fn record_api_call(&self, latency_ms: u128) {
        self.api_calls.fetch_add(1, Ordering::Relaxed);
        self.api_latency_ms_sum.fetch_add(latency_ms as u64, Ordering::Relaxed);
    }

    /// Compute average latency (integer ms)
    pub fn avg_latency_ms(&self) -> u64 {
        let calls = self.api_calls.load(Ordering::Relaxed);
        if calls == 0 {
            0
        } else {
            self.api_latency_ms_sum.load(Ordering::Relaxed) / calls
        }
    }
}

use crate::agent_protocol::AgentAction;

#[derive(Serialize, Deserialize, Clone, Debug, PartialEq)]
pub enum StepStatus {
    Pending,
    Running,
    Done,
    Failed(String),
}

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct MissionStep {
    pub id: usize,
    pub description: String,
    pub action: Option<AgentAction>,
    pub status: StepStatus,
    pub result: Option<String>,
}

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct Mission {
    pub id: String,
    pub goal: String,
    pub steps: Vec<MissionStep>,
    pub created_at: String,
    pub completed: bool,
}

pub struct MissionPlanner;

impl MissionPlanner {
    pub async fn execute(state: &Arc<AppState>, mut mission: Mission) -> anyhow::Result<Mission> {
        let (lifecycle_tx, _) = tokio::sync::broadcast::channel::<String>(16);

        // SUTURE: Added state.memory.clone() to satisfy new spawn signature
        let handle: AgentHandle = crate::component_runner::AgentComponent::spawn_legacy(
            &format!("mission-{}", mission.id),
            Capabilities::READ | Capabilities::WRITE,
            state.agent.clone(),
            lifecycle_tx,
            state.memory.clone(),
        );

        for step in mission.steps.iter_mut() {
            if step.status != StepStatus::Pending {
                continue;
            }
            step.status = StepStatus::Running;

            if let Some(action) = step.action.clone() {
                match handle.execute_actions(vec![action]).await {
                    Ok(results) => {
                        step.status = StepStatus::Done;
                        step.result = Some(results.join("\n"));
                    }
                    Err(e) => {
                        step.status = StepStatus::Failed(e.to_string());
                        let _ = state.tx.send(format!("Mission step {} failed: {}", step.id, e));
                        break;
                    }
                }
            }
        }

        mission.completed = mission.steps.iter().all(|s| matches!(s.status, StepStatus::Done));
        Ok(mission)
    }
}

use anyhow::Result;
use serde::{Deserialize, Serialize};
use sqlx::{sqlite::SqlitePoolOptions, Row, SqlitePool};

#[derive(Serialize, Deserialize, Debug, Clone)]
pub struct PersistentMission {
    pub id: String,
    pub goal: String,
    pub status: String,
    pub created_at: i64,
    pub payload: String,
}

#[derive(Clone)]
pub struct MissionQueue {
    pool: SqlitePool,
}

impl MissionQueue {
    pub async fn new(db_path: &str) -> Result<Self> {
        let pool = SqlitePoolOptions::new()
            .max_connections(5)
            .connect(&format!("sqlite:{}", db_path))
            .await?;
        let queue = Self { pool };
        queue.init_schema().await?;
        Ok(queue)
    }

    async fn init_schema(&self) -> Result<()> {
        sqlx::query(
            "CREATE TABLE IF NOT EXISTS missions (
                id TEXT PRIMARY KEY,
                goal TEXT NOT NULL,
                status TEXT NOT NULL,
                created_at INTEGER NOT NULL,
                payload TEXT NOT NULL
            )",
        )
        .execute(&self.pool)
        .await?;
        Ok(())
    }

    pub async fn enqueue(&self, mission: PersistentMission) -> Result<()> {
        sqlx::query(
            "INSERT OR REPLACE INTO missions (id, goal, status, created_at, payload) VALUES (?, ?, ?, ?, ?)",
        )
        .bind(&mission.id)
        .bind(&mission.goal)
        .bind(&mission.status)
        .bind(mission.created_at)
        .bind(&mission.payload)
        .execute(&self.pool)
        .await?;
        Ok(())
    }

    pub async fn get_pending(&self) -> Result<Vec<PersistentMission>> {
        let rows = sqlx::query(
            "SELECT id, goal, status, created_at, payload FROM missions WHERE status = 'pending' ORDER BY created_at",
        )
        .fetch_all(&self.pool)
        .await?;

        Ok(rows
            .into_iter()
            .map(|row| PersistentMission {
                id: row.get("id"),
                goal: row.get("goal"),
                status: row.get("status"),
                created_at: row.get("created_at"),
                payload: row.get("payload"),
            })
            .collect())
    }

    pub async fn update_status(&self, id: &str, status: &str) -> Result<()> {
        sqlx::query("UPDATE missions SET status = ? WHERE id = ?")
            .bind(status)
            .bind(id)
            .execute(&self.pool)
            .await?;
        Ok(())
    }
}

use anyhow::{anyhow, Context, Result};
use std::env;
use std::path::{Path, PathBuf};

const DIAGNOSTIC_ALLOWLIST: &[&str] = &[
    "/proc/loadavg",
    "/proc/meminfo",
    "/sys/class/thermal/thermal_zone0/temp",
];

pub fn validate_path(requested_path: &str) -> Result<PathBuf> {
    if requested_path.contains('\0') {
        return Err(anyhow!("SECURITY VIOLATION: Null byte in path"));
    }

    if DIAGNOSTIC_ALLOWLIST.binary_search(&requested_path).is_ok() {
        return Ok(PathBuf::from(requested_path));
    }

    let workspace_root = env::current_dir().context("Failed to resolve workspace root")?;

    if requested_path.contains("..") || requested_path.starts_with('/') {
        return Err(anyhow!("SECURITY VIOLATION: Absolute paths and traversal components forbidden"));
    }

    let target   = Path::new(requested_path);
    let absolute = workspace_root.join(target);

    let mut existing_base = absolute.clone();
    let mut non_existent_components = Vec::new();

    while !existing_base.exists() {
        if let Some(comp) = existing_base.file_name() {
            non_existent_components.push(comp.to_owned());
        }
        if !existing_base.pop() {
            return Err(anyhow!("Failed to resolve path ancestry"));
        }
    }

    let canonical_base = existing_base.canonicalize().context("Failed to canonicalize base path")?;
    let canonical_root = workspace_root.canonicalize().context("Failed to canonicalize workspace root")?;

    if !canonical_base.starts_with(&canonical_root) {
        return Err(anyhow!("[CRITICAL] Path escape detected. Substrate breach prevented."));
    }

    let mut safe_path = canonical_base;
    for comp in non_existent_components.into_iter().rev() {
        safe_path.push(comp);
    }
    Ok(safe_path)
}

pub fn normalize_and_check(path: &str) -> Result<PathBuf> {
    if path.contains('\0') {
        return Err(anyhow!("SECURITY VIOLATION: Null byte in path"));
    }
    if path.contains("..") || path.starts_with('/') {
        return Err(anyhow!("SECURITY VIOLATION: Path must be relative and within workspace"));
    }
    let workspace = env::current_dir().context("Failed to resolve workspace root")?;
    let candidate = workspace.join(path);
    if candidate.exists() {
        let canonical = candidate.canonicalize().context("Failed to canonicalize")?;
        let canonical_root = workspace.canonicalize().context("Failed to canonicalize root")?;
        if !canonical.starts_with(&canonical_root) {
            return Err(anyhow!("[CRITICAL] Path escape detected."));
        }
        Ok(canonical)
    } else {
        Ok(candidate)
    }
}

pub async fn execute_mission(url: &str, memory: Arc<Mutex<MemoryPalace>>) -> Result<String> {
    let mut headers = reqwest::header::HeaderMap::new();
    headers.insert("Accept", "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8".parse().unwrap());
    let client = reqwest::Client::builder()
        .user_agent("Mozilla/5.0")
        .default_headers(headers)
        .build()?;

    let response = client.get(url).send().await?.text().await?;

    let (title, full_strand) = {
        let document = Html::parse_document(&response);
        let title_selector = Selector::parse("title, h1").unwrap();
        let body_selector = Selector::parse("p, pre, code, article, .content").unwrap();

        let title_inner = document.select(&title_selector).next()
            .map(|e| e.text().collect::<String>()).unwrap_or_else(|| url.to_string());

        let mut content = Vec::new();
        for element in document.select(&body_selector).take(50) {
            let text: String = element.text().collect();
            let trimmed = text.trim();
            if trimmed.len() > 10 { content.push(trimmed.to_string()); }
        }
        (title_inner, content.join("\n\n"))
    };

    let result_summary = format!("TITLE: {}\nCONTENT_DENSITY: {} chars\nDATA: {}", title, full_strand.len(), full_strand);

    {
        let palace = memory.lock().await;
        palace.store_memory("ALPHA-SCOUT", &title, &result_summary, None).await?;
    }

    Ok(format!("SCRAPE_SUCCESS: Secured {} bytes", full_strand.len()))

use axum::{body::Body, http::{Request, StatusCode, header}, middleware::Next, response::Response};

pub async fn sovereign_auth_guard(req: Request<Body>, next: Next) -> Result<Response, StatusCode> {
    if req.uri().path() == "/api/auth/login" { return Ok(next.run(req).await); }

    let auth_header = req.headers().get(header::AUTHORIZATION).and_then(|v| v.to_str().ok()).unwrap_or("");
    let token = auth_header.trim_start_matches("Bearer ").trim();

    if !token.is_empty() {
        match auth::verify_token_detailed(token) {
            Ok(auth::TokenVerdict::Valid) => return Ok(next.run(req).await),
            Ok(verdict) => tracing::warn!(
                "[SECURITY] 401 Unauthorized: Path={} | Verdict={:?}",
                req.uri().path(), verdict
            ),
            Err(e) => tracing::error!(
                "[SECURITY] Auth key misconfiguration at {}: {}",
                req.uri().path(), e
            ),
        }
    } else {
        tracing::warn!("[SECURITY] 401 Unauthorized: Path={} | No token provided",
            req.uri().path());
    }
    Err(StatusCode::UNAUTHORIZED)
}

use bytes::{Bytes, BytesMut};
use futures_util::Stream;
use std::pin::Pin;
use std::task::{Context, Poll};

pub struct NeuralChunker<S> {
    stream: S,
    buffer: BytesMut,
    chunk_size: usize,
}

impl<S> NeuralChunker<S> {
    pub fn new(stream: S, chunk_size: usize) -> Self {
        Self { stream, buffer: BytesMut::with_capacity(chunk_size * 2), chunk_size }
    }
}

impl<S> Stream for NeuralChunker<S>
where
    S: Stream<Item = Result<Bytes, reqwest::Error>> + Unpin,
{
    type Item = Result<String, String>;

    fn poll_next(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
        // Continue polling the inner stream until we reach chunk_size or EOF
        while self.buffer.len() < self.chunk_size {
            match Pin::new(&mut self.stream).poll_next(cx) {
                Poll::Ready(Some(Ok(bytes))) => self.buffer.extend_from_slice(&bytes),
                Poll::Ready(Some(Err(e))) => return Poll::Ready(Some(Err(e.to_string()))),
                Poll::Ready(None) => break,
                Poll::Pending => return Poll::Pending,
            }
        }

        if self.buffer.is_empty() {
            return Poll::Ready(None);
        }

        // Split the buffer at the chunk boundary or remaining length
        let split_at = std::cmp::min(self.buffer.len(), self.chunk_size);
        let chunk = self.buffer.split_to(split_at);

        Poll::Ready(Some(Ok(String::from_utf8_lossy(&chunk).to_string())))
    }
}

use std::fs;
use std::path::{Path, PathBuf};

pub struct VoidWalker;

impl VoidWalker {
    pub fn identify_orphans() -> Vec<String> {
        let mut orphans = Vec::new();
        let mut declared_mods = Vec::new();

        let root_files = ["src/main.rs", "src/lib.rs"];
        for root in &root_files {
            if let Ok(content) = fs::read_to_string(root) {
                Self::trace_module_tree(Path::new(root), &content, &mut declared_mods, 0);
            }
        }

        let mut all_files = Vec::new();
        let _ = Self::visit_dirs_shallow(Path::new("src"), &mut all_files, 0);

        let ignore_dirs = ["target", "ziggy-llm", "node_modules", ".lazarus_vault", "ui/dist"];

        for file_path in all_files {
            let path_str = file_path.to_string_lossy().to_string();

            if ignore_dirs.iter().any(|d| path_str.contains(d)) { continue; }
            if path_str.ends_with("mod.rs") || path_str.contains("/bin/") { continue; }
            if !declared_mods.iter().any(|m| path_str.ends_with(m)) {
                orphans.push(path_str);
            }
        }
        orphans
    }

    fn trace_module_tree(
        current_file: &Path,
        content: &str,
        whitelist: &mut Vec<String>,
        depth: u8,
    ) {
        const MAX_DEPTH: u8 = 8;
        if depth > MAX_DEPTH { return; }
        
        for line in content.lines() {
            let trimmed = line.trim();
            if trimmed.starts_with("mod ") || trimmed.starts_with("pub mod ") {
                if let Some(mod_name) =
                    trimmed.split_whitespace().nth(if trimmed.starts_with("pub") { 2 } else { 1 })
                {
                    let name = mod_name.trim_end_matches(';');
                    let parent = current_file.parent().unwrap_or(Path::new(""));
                    let possible_paths = vec![
                        parent.join(format!("{}.rs", name)),
                        parent.join(name).join("mod.rs"),
                        parent.join(current_file.file_stem().unwrap_or_default()).join(format!("{}.rs", name)),
                    ];
                    
                    for path in possible_paths {
                        if path.exists() {
                            let path_key = path.to_string_lossy().to_string();
                            if !whitelist.contains(&path_key) {
                                whitelist.push(path_key);
                                if let Ok(sub_content) = fs::read_to_string(&path) {
                                    Self::trace_module_tree(&path, &sub_content, whitelist, depth + 1);
                                }
                            }
                        }
                    }
                }
            }
        }
    }

    fn visit_dirs_shallow(dir: &Path, files: &mut Vec<PathBuf>, depth: u8) -> std::io::Result<()> {
        if depth > 4 { return Ok(()); }
        if dir.is_dir() {
            for entry in fs::read_dir(dir)? {
                let entry = entry?;
                let path = entry.path();
                if path.is_dir() {
                    Self::visit_dirs_shallow(&path, files, depth + 1)?;
                } else if path.extension().and_then(|s| s.to_str()) == Some("rs") {
                    files.push(path);
                }
            }
        }
        Ok(())
    }
}

// src/wasm_registry.rs — v1.0 Named WASM Cell Registry
// Stores compiled WASM cells by name so they can be reused across
// sessions without re-synthesis. Each entry tracks creation time,
// the original Rust source, and execution hit count.
use anyhow::Result;
use serde::{Deserialize, Serialize};
use sqlx::{sqlite::SqlitePoolOptions, Row, SqlitePool};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct WasmCell {
    pub name: String,
    pub wasm_b64: String,
    pub source_code: String,
    pub created_at: String,
    pub hit_count: i64,
}

pub struct WasmRegistry {
    pool: SqlitePool,
}

impl WasmRegistry {
    pub async fn new(db_path: &str) -> Result<Self> {
        if !std::path::Path::new(db_path).exists() {
            std::fs::File::create(db_path).ok();
        }
        let pool = SqlitePoolOptions::new()
            .max_connections(3)
            .connect(&format!("sqlite:{}", db_path))
            .await?;

        sqlx::query("PRAGMA journal_mode=WAL;").execute(&pool).await?;

        sqlx::query(
            "CREATE TABLE IF NOT EXISTS wasm_cells (
                name        TEXT PRIMARY KEY,
                wasm_b64    TEXT NOT NULL,
                source_code TEXT NOT NULL,
                created_at  DATETIME DEFAULT CURRENT_TIMESTAMP,
                hit_count   INTEGER DEFAULT 0
            )",
        )
        .execute(&pool)
        .await?;

        Ok(Self { pool })
    }

    /// store: Upserts a named WASM cell.
    pub async fn store(&self, name: &str, wasm_b64: &str, source: &str) -> Result<()> {
        sqlx::query(
            "INSERT INTO wasm_cells (name, wasm_b64, source_code)
             VALUES (?, ?, ?)
             ON CONFLICT(name) DO UPDATE SET
               wasm_b64=excluded.wasm_b64,
               source_code=excluded.source_code,
               created_at=CURRENT_TIMESTAMP,
               hit_count=0",
        )
        .bind(name)
        .bind(wasm_b64)
        .bind(source)
        .execute(&self.pool)
        .await?;
        Ok(())
    }

    /// get: Retrieves a cell by name and increments its hit count.
    pub async fn get(&self, name: &str) -> Result<Option<WasmCell>> {
        let row = sqlx::query(
            "SELECT name, wasm_b64, source_code, created_at, hit_count
             FROM wasm_cells WHERE name = ?",
        )
        .bind(name)
        .fetch_optional(&self.pool)
        .await?;

        if let Some(r) = row {
            sqlx::query("UPDATE wasm_cells SET hit_count = hit_count + 1 WHERE name = ?")
                .bind(name)
                .execute(&self.pool)
                .await?;
            Ok(Some(WasmCell {
                name: r.get("name"),
                wasm_b64: r.get("wasm_b64"),
                source_code: r.get("source_code"),
                created_at: r.get("created_at"),
                hit_count: r.get("hit_count"),
            }))
        } else {
            Ok(None)
        }
    }

    /// list_all: Returns all stored cells (name + metadata, no wasm_b64).
    pub async fn list_all(&self) -> Result<Vec<WasmCell>> {
        let rows = sqlx::query(
            "SELECT name, '' as wasm_b64, source_code, created_at, hit_count
             FROM wasm_cells ORDER BY created_at DESC",
        )
        .fetch_all(&self.pool)
        .await?;

        Ok(rows
            .into_iter()
            .map(|r| WasmCell {
                name: r.get("name"),
                wasm_b64: String::new(),
                source_code: r.get("source_code"),
                created_at: r.get("created_at"),
                hit_count: r.get("hit_count"),
            })
            .collect())
    }

    /// delete: Removes a named cell.
    pub async fn delete(&self, name: &str) -> Result<()> {
        sqlx::query("DELETE FROM wasm_cells WHERE name = ?").bind(name).execute(&self.pool).await?;
        Ok(())
    }
}

// src/wasm_runtime.rs — v1.1 Hardened Wasmtime Runtime
// FIX: Added explicit fuel cap to prevent runaway WASM cells from
//      monopolising CPU on the 4GB Duet.
// FIX: Better error context — includes func_name in all error messages.
// NEW: run_f64 for double-precision cells.
// NEW: run_named for registry-backed execution.
use anyhow::{anyhow, Result};
use wasmtime::*;

/// FUEL_CAP: Max instructions before forceful termination.
/// 10M instructions ≈ ~10ms of compute on ARM64 Crostini at 1GHz.
const FUEL_CAP: u64 = 50_000_000;

pub struct WasmRuntime;

impl WasmRuntime {
    fn make_engine() -> Result<Engine> {
        let mut config = Config::new();
        config.consume_fuel(true);
        Ok(Engine::new(&config)?)
    }

    /// run_bytes: Execute a compiled WASM cell with f32 I/O.
    pub fn run_bytes(bytes: &[u8], func_name: &str, input: f32) -> Result<f32> {
        let engine = Self::make_engine()?;
        let module = Module::from_binary(&engine, bytes)
            .map_err(|e| anyhow!("WASM_LOAD_FAULT [{}]: {}", func_name, e))?;
        let mut store = Store::new(&engine, ());
        store.set_fuel(FUEL_CAP)?;
        let instance = Instance::new(&mut store, &module, &[])
            .map_err(|e| anyhow!("WASM_INST_FAULT [{}]: {}", func_name, e))?;
        let func = instance
            .get_typed_func::<f32, f32>(&mut store, func_name)
            .map_err(|e| anyhow!("WASM_FUNC_MISSING [{}]: {}", func_name, e))?;
        func.call(&mut store, input).map_err(|e| anyhow!("WASM_EXEC_FAULT [{}]: {}", func_name, e))
    }

    /// run_f64: f64 variant for higher-precision cells.
    pub fn run_f64(bytes: &[u8], func_name: &str, input: f64) -> Result<f64> {
        let engine = Self::make_engine()?;
        let module = Module::from_binary(&engine, bytes)?;
        let mut store = Store::new(&engine, ());
        store.set_fuel(FUEL_CAP)?;
        let instance = Instance::new(&mut store, &module, &[])?;
        let func = instance.get_typed_func::<f64, f64>(&mut store, func_name)?;
        Ok(func.call(&mut store, input)?)
    }

    /// run_i32: Integer variant for flag/status cells.
    pub fn run_i32(bytes: &[u8], func_name: &str, input: i32) -> Result<i32> {
        let engine = Self::make_engine()?;
        let module = Module::from_binary(&engine, bytes)?;
        let mut store = Store::new(&engine, ());
        store.set_fuel(FUEL_CAP)?;
        let instance = Instance::new(&mut store, &module, &[])?;
        let func = instance.get_typed_func::<i32, i32>(&mut store, func_name)?;
        Ok(func.call(&mut store, input)?)
    }

    /// list_exports: Returns names of all exported functions in a WASM module.
    /// Useful for discovery without executing.
    pub fn list_exports(bytes: &[u8]) -> Result<Vec<String>> {
        let engine = Engine::default();
        let module = Module::from_binary(&engine, bytes)?;
        Ok(module
            .exports()
            .filter(|e| matches!(e.ty(), ExternType::Func(_)))
            .map(|e| e.name().to_string())
            .collect())
    }

    /// validate: Returns Ok(()) if bytes are a valid WASM module.
    pub fn validate(bytes: &[u8]) -> Result<()> {
        let engine = Engine::default();
        Module::from_binary(&engine, bytes).map(|_| ()).map_err(|e| anyhow!("WASM_INVALID: {}", e))
    }
}

use crate::agent_protocol::AgentAction;
use serde_json::{json, Value};
use std::path::Path;
use tokio::fs;
use async_process::{Command, Stdio};
use futures_lite::{io::BufReader, prelude::*};
use tokio::sync::broadcast;

pub struct NexusShell;

impl NexusShell {
    pub async fn eval(action: &AgentAction, tx: &broadcast::Sender<String>) -> Value {
        match action {
            AgentAction::ReadFile { path, .. } => Self::builtin_read_file(path).await,
            AgentAction::WriteFile { path, content, .. } => Self::builtin_write_file(path, content).await,
            AgentAction::ExecuteCommand { command, .. } => Self::builtin_exec(command, tx).await,
            _ => json!({ "status": "unsupported", "action": format!("{:?}", action) }),
        }
    }

    async fn builtin_read_file(path: &str) -> Value {
        let p = Path::new(path);
        match fs::read_to_string(p).await {
            Ok(content) => json!({
                "status": "success", "operation": "read_file", "path": path,
                "content": content.chars().take(4096).collect::<String>()
            }),
            Err(e) => json!({ "status": "error", "path": path, "reason": e.to_string() })
        }
    }

    async fn builtin_write_file(path: &str, content: &str) -> Value {
        let p = Path::new(path);
        if let Some(parent) = p.parent() { let _ = fs::create_dir_all(parent).await; }
        match fs::write(p, content).await {
            Ok(_) => json!({ "status": "success", "operation": "write_file", "path": path, "bytes_written": content.len() }),
            Err(e) => json!({ "status": "error", "path": path, "reason": e.to_string() })
        }
    }

    async fn builtin_exec(command: &str, tx: &broadcast::Sender<String>) -> Value {
        let mut child = match Command::new("sh")
            .arg("-c")
            .arg(command)
            .stdout(Stdio::piped())
            .stderr(Stdio::piped())
            .spawn() {
                Ok(child) => child,
                Err(e) => return json!({ "status": "error", "command": command, "reason": e.to_string() }),
            };

        let mut stdout_final = String::new();
        let mut stderr_final = String::new();

        let stdout = child.stdout.take().unwrap();
        let stderr = child.stderr.take().unwrap();

        let mut stdout_reader = BufReader::new(stdout).lines();
        let mut stderr_reader = BufReader::new(stderr).lines();

        loop {
            tokio::select! {
                Some(Ok(line)) = stdout_reader.next() => {
                    // ⚡ BROADCAST: Send live line to Svelte IDE
                    let _ = tx.send(json!({"type": "TERMINAL_OUT", "payload": line}).to_string());
                    stdout_final.push_str(&line);
                    stdout_final.push('\n');
                }
                Some(Ok(line)) = stderr_reader.next() => {
                    let _ = tx.send(json!({"type": "TERMINAL_ERR", "payload": line}).to_string());
                    stderr_final.push_str(&line);
                    stderr_final.push('\n');
                }
                status = child.status() => {
                    let exit_status = status.map(|s| s.success()).unwrap_or(false);
                    return json!({
                        "status": if exit_status { "success" } else { "failed" },
                        "command": command,
                        "stdout": stdout_final,
                        "stderr": stderr_final.trim()
                    });
                }
            }
        }
    }
}

    /// This is the "Semantic Precision" upgrade to replace simple string diffs.
    pub fn update_function(path: &str, function_name: &str, new_body: &str) -> Result<String> {
        let source_code = fs::read_to_string(path)?;
        let mut parser = Parser::new();
        parser.set_language(&tree_sitter_rust::LANGUAGE.into())?;
        
        let tree = parser.parse(&source_code, None)
            .ok_or_else(|| anyhow!("AST_PARSE_FAIL: Could not parse {}", path))?;
            
        // Query to find the specific function node
        let query_str = format!(
            "(function_item name: (identifier) @name (#eq? @name \"{}\")) @func",
            function_name
        );
        let query = Query::new(&tree_sitter_rust::LANGUAGE.into(), &query_str)?;
        let mut cursor = QueryCursor::new();
        let mut matches = cursor.matches(&query, tree.root_node(), source_code.as_bytes());
        while let Some(m) = matches.next() {
            for capture in m.captures {
                if capture.index == 1 { // The @func capture
                    let node = capture.node;
                    let start = node.start_byte();
                    let end = node.end_byte();
                    
                    let mut new_code = source_code.clone();
                    new_code.replace_range(start..end, new_body);
                    return Ok(new_code);
                }
            }
        }
        
        Err(anyhow!("FUNCTION_NOT_FOUND: Could not find function '{}' in {}", function_name, path))
    }
}

use anyhow::{anyhow, Result};
use crate::agent_protocol::AgentAction;
use serde_json::{json, Value};

pub fn extract_actions(response: &str) -> Result<Vec<AgentAction>> {
    let stripped = response.trim()
        .replace("```json", "")
        .replace("```", "");

    let renamed = stripped
        .replace("\"action\":", "\"type\":")
        .replace("\"action\" :", "\"type\":")
        .replace("\"file_path\":", "\"path\":")
        .replace("\"fn_name\":", "\"function_name\":");

    let start = renamed.find('[')
        .ok_or_else(|| anyhow!("PARSER_FRACTURE: No JSON array start '[' found"))?;
    let end = renamed.rfind(']')
        .ok_or_else(|| anyhow!("PARSER_FRACTURE: No JSON array end ']' found"))?;

    if start >= end {
        return Err(anyhow!("PARSER_FRACTURE: Invalid JSON array bounds"));
    }

    let json_text = &renamed[start..=end];

    if let Ok(mut parsed_json) = serde_json::from_str::<Value>(json_text) {
        if let Some(arr) = parsed_json.as_array_mut() {
            for obj in arr.iter_mut() {
                if let Some(map) = obj.as_object_mut() {
                    if !map.contains_key("type") {
                        if map.contains_key("search") && map.contains_key("replace") {
                            map.insert("type".to_string(), json!("diff_patch"));
                        } else if map.contains_key("command") {
                            map.insert("type".to_string(), json!("execute_command"));
                        } else if map.contains_key("function_name") && map.contains_key("new_body") {
                            map.insert("type".to_string(), json!("patch_function"));
                        } else if map.contains_key("path") && map.contains_key("content") {
                            map.insert("type".to_string(), json!("write_file"));
                        } else if map.contains_key("path") {
                            map.insert("type".to_string(), json!("read_file"));
                        }
                    }
                }
            }
        }

        if let Ok(actions) = serde_json::from_value::<Vec<AgentAction>>(parsed_json) {
            if !actions.is_empty() { return Ok(actions); }
        }
    }

    Err(anyhow!("PARSER_FRACTURE: Malformed or unrecoverable array"))
}

use std::collections::HashMap;
use tracing::warn;

pub struct LogicEngine;

impl LogicEngine {
    pub fn is_action_safe(directive: &str) -> bool {
        let forbidden = [
            "rm -rf /", "mkfs", "shutdown", "chmod 777",
            "dd if=/dev/zero", "> /dev/sda", "fork bomb", ":(){ :|:& };:"
        ];
        let d_lower = directive.to_lowercase();
        !forbidden.iter().any(|&p| d_lower.contains(p))
    }

    pub fn detect_deadlock(observations: &[String]) -> bool {
        if observations.len() < 3 { return false; }

        let mut read_counts: HashMap<String, usize> = HashMap::new();
        let mut progress = false;

        for obs in observations {
            if obs.contains("[WRITE_OK]") || obs.contains("[DIFF_PATCH_OK]") || obs.contains("[PATCH_OK]") {
                progress = true;
            }
            if obs.contains("[READ_OK]") {
                if let Some(after) = obs.strip_prefix("[READ_OK] ") {
                    let path = after.split_whitespace().next().unwrap_or("").to_string();
                    if !path.is_empty() {
                        *read_counts.entry(path).or_insert(0) += 1;
                    }
                }
            }
        }

        if !progress && read_counts.values().any(|&count| count >= 3) {
            warn!("[LOGIC] Deadlock: Same file read 3+ times with zero progress.");
            return true;
        }

        let mut last_cmd_sig = String::new();
        let mut cmd_repeat_count = 0;

        for obs in observations {
            if let Some(start) = obs.find("[CMD:") {
                let sig = obs[start..].to_string();
                if sig == last_cmd_sig {
                    cmd_repeat_count += 1;
                } else {
                    last_cmd_sig = sig;
                    cmd_repeat_count = 1;
                }
            }
        }

        if cmd_repeat_count >= 2 {
            warn!("[LOGIC] Deadlock: Command repetition detected.");
            return true;
        }
        false
    }
}

// src/executor.rs — v2.6 [IRONCLAD_FSM + CLEAN_STATE]

use crate::agent_protocol::AgentAction;
use crate::arbiter_agent::ArbiterAgent;
use crate::distributed_scheduler::DistributedScheduler;
use crate::server::VmoState;
use anyhow::Result;
use std::sync::Arc;
use tokio::sync::Mutex;

#[derive(Debug, PartialEq, Clone)]
pub enum FsmState { Idle, Executing, Complete, Fault }

pub struct RoboticExecutor {
    pub current_state: FsmState,
    tx: tokio::sync::broadcast::Sender<String>,
}

impl RoboticExecutor {
    pub fn new(_api_key: String, tx: tokio::sync::broadcast::Sender<String>) -> Self {
        Self { current_state: FsmState::Idle, tx }
    }

    pub async fn execute_handle(
        &mut self,
        actions: Vec<AgentAction>,
        memory: Arc<Mutex<nexus_substrate::memory::MemoryPalace>>,
        vmo_state: Arc<Mutex<VmoState>>,
        dist: Arc<DistributedScheduler>,
    ) -> Result<Vec<String>> {
        self.current_state = FsmState::Executing;
        let mut observations = Vec::new();
        let total = actions.len();
        let mut had_halt = false;

        for (idx, action) in actions.into_iter().enumerate() {
            let _ = self.tx.send(serde_json::json!({
                "type": "TASK_STATUS",
                "payload": format!("Executing → {:?}", action)
            }).to_string());

            if matches!(action, AgentAction::Halt { .. }) {
                had_halt = true;
            }

            match self.execute_single_action(action, memory.clone(), vmo_state.clone(), dist.clone()).await {
                Ok(obs) => {
                    observations.push(obs.clone());
                    tracing::info!("Action observation: {}", obs);

                    if obs.contains("[READ_OK]")
                        || obs.contains("[WRITE_OK]")
                        || obs.contains("[CMD:")
                        || obs.contains("[DIFF_PATCH_OK]")
                        || obs.contains("[PATCH_OK]")
                        || obs.contains("[REPORT_OK]")
                        || obs.contains("[HALT]") {
                        
                        let skipped = total.saturating_sub(idx + 1);
                        if skipped > 0 {
                            let yield_msg = format!(">> [YIELD] Action completed. {} action(s) suspended for next turn.", skipped);
                            observations.push(yield_msg.clone());
                            tracing::info!("{}", yield_msg);
                        }
                        break;
                    }
                }
                Err(e) => {
                    let err_obs = format!("ACTION_FAILED: {}", e);
                    observations.push(err_obs.clone());
                    tracing::warn!("Action failed: {}", e);
                    self.current_state = FsmState::Fault;
                    return Ok(observations);
                }
            }
        }

        // Final FSM state
        if had_halt || observations.iter().any(|o| o.contains("[HALT]")) {
            self.current_state = FsmState::Complete;
        } else {
            self.current_state = FsmState::Complete;   // Most stable for now
        }

        Ok(observations)
    }

    // execute_single_action remains exactly as you wrote it (v2.5)
    async fn execute_single_action(
        &mut self,
        action: AgentAction,
        _memory: Arc<Mutex<nexus_substrate::memory::MemoryPalace>>,
        _vmo_state: Arc<Mutex<VmoState>>,
        _dist: Arc<DistributedScheduler>,
    ) -> Result<String> {
        match action {
            AgentAction::ExecuteCommand { command, reasoning } => {
                let _ = self.tx.send(serde_json::json!({"type": "NEURAL_THOUGHT", "payload": format!("Reasoning: {}", reasoning)}).to_string());
                if let Err(e) = ArbiterAgent::validate_action("execute_command", &command) {
                    return Ok(format!("SHIELD_BLOCKED: {}", e));
                }
                match crate::podman::PodmanShield::execute(&command) {
                    Ok(output) => {
                        let stdout = String::from_utf8_lossy(&output.stdout).trim().to_string();
                        let stderr = String::from_utf8_lossy(&output.stderr).trim().to_string();
                        let status = if output.status.success() { "SUCCESS" } else { "FAILED" };
                        Ok(format!("[CMD:{}] STDOUT: {}\nSTDERR: {}\nReason: {}", status, stdout, stderr, reasoning))
                    }
                    Err(e) => Ok(format!("PODMAN_SHIELD_ERROR: {}", e)),
                }
            }

            AgentAction::WriteFile { path, content, reasoning } => {
                let _ = self.tx.send(serde_json::json!({"type": "NEURAL_THOUGHT", "payload": format!("Writing file: {} | Reason: {}", path, reasoning)}).to_string());
                let content_len = content.len();
                let preview: String = content.chars().take(80).collect();
                tokio::fs::write(&path, content).await
                    .map(|_| format!("[WRITE_OK] {} bytes → {} | Preview: {}…", content_len, path, preview))
                    .map_err(|e| anyhow::anyhow!("WRITE_FAILED: {}", e))
            }

            AgentAction::ReadFile { path, reasoning } => {
                let _ = self.tx.send(serde_json::json!({"type": "NEURAL_THOUGHT", "payload": format!("Reading file: {} | Reason: {}", path, reasoning)}).to_string());
                match tokio::fs::read_to_string(&path).await {
                    Ok(content) => {
                        let byte_len = content.len();
                        const MAX_CHARS: usize = 4096;
                        let char_count = content.chars().count();
                        let injected: String = content.chars().take(MAX_CHARS).collect();
                        let note = if char_count > MAX_CHARS {
                            format!("\n[TRUNCATED: showing {}/{} chars — use diff_patch for large files]", MAX_CHARS, char_count)
                        } else {
                            String::new()
                        };
                        Ok(format!(
                            "[READ_OK] {} ({} bytes){}\n--- CONTENT ---\n{}\n--- END CONTENT ---",
                            path, byte_len, note, injected
                        ))
                    }
                    Err(e) => anyhow::bail!("READ_FAILED: {}", e),
                }
            }

            AgentAction::DiffPatch { path, search, replace, reasoning } => {
                let _ = self.tx.send(serde_json::json!({"type": "NEURAL_THOUGHT", "payload": format!("Applying diff patch to {} | Reason: {}", path, reasoning)}).to_string());
                match tokio::fs::read_to_string(&path).await {
                    Ok(original) => {
                        let orig_lines = original.lines().count();
                        match crate::diff::DiffEngine::apply(&original, &search, &replace) {
                            Ok(new_content) => {
                                let new_lines: usize = new_content.lines().count();
                                match tokio::fs::write(&path, &new_content).await {
                                    Ok(_) => Ok(format!("[DIFF_PATCH_OK] {} | Lines: {} → {}", path, orig_lines, new_lines)),
                                    Err(e) => Ok(format!("[DIFF_PATCH_WRITE_FAILED] {}", e)),
                                }
                            }
                            Err(e) => Ok(format!("[DIFF_PATCH_FAILED] {}", e)),
                        }
                    }
                    Err(e) => Ok(format!("[DIFF_PATCH_READ_FAILED] {}", e)),
                }
            }

            AgentAction::PatchFunction { path, function_name, new_body, reasoning } => {
                let _ = self.tx.send(serde_json::json!({"type": "NEURAL_THOUGHT", "payload": format!("Patching function {} in {} | Reason: {}", function_name, path, reasoning)}).to_string());
                match crate::ast_editor::AstEditor::update_function(&path, &function_name, &new_body) {
                    Ok(full_code) => {
                        match tokio::fs::write(&path, full_code).await {
                            Ok(_) => Ok(format!("[PATCH_OK] Function '{}' updated in {}", function_name, path)),
                            Err(e) => Ok(format!("PATCH_WRITE_FAILED: {}", e)),
                        }
                    },
                    Err(e) => Ok(format!("PATCH_FAILED: {}", e)),
                }
            }

            AgentAction::ExportReport { title, content, reasoning } => {
                let _ = self.tx.send(serde_json::json!({"type": "NEURAL_THOUGHT", "payload": format!("Exporting report: {} | Reason: {}", title, reasoning)}).to_string());
                let safe_name = title.to_lowercase().replace(|c: char| !c.is_alphanumeric() && c != ' ' && c != '-', "_");
                let path = format!("src/{}.md", safe_name);
                tokio::fs::write(&path, content).await
                    .map(|_| format!("[REPORT_OK] Exported → {}", path))
                    .map_err(|e| anyhow::anyhow!("EXPORT_FAILED: {}", e))
            }

            AgentAction::FetchUiSnapshot { reasoning } => {
                let _ = self.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("fetch_ui_snapshot triggered — {}", reasoning)}).to_string());
                Ok("[UI_SNAPSHOT_REQUESTED] Ghost DOM state capture requested.".to_string())
            }

            AgentAction::Halt { reasoning } => {
                self.current_state = FsmState::Complete;
                let clean_reason = reasoning.chars().take(100).collect::<String>();
                let _ = self.tx.send(serde_json::json!({
                    "type": "TASK_STATUS",
                    "payload": format!("HALT: {}", clean_reason)
                }).to_string());
                Ok(format!("[HALT] {}", clean_reason))
            }

            _ => Ok(format!("[STUB_OK] Action acknowledged: {:?}", action))
        }
    }
}

pub async fn start_thermal_bridge(monitor: Arc<Mutex<SystemMonitor>>) {
    println!("\x1b[35m>> [TELEMETRY_BRIDGE] Mapping Synthetic Thermal VMO (Handle: 0x7E1)\x1b[0m");
    
    // Create a 4KB page for hardware metrics
    let handle = unsafe { nexus_substrate::kernel::ffi::zx_vmo_create_native(4096) };
    kernel::register_handle(handle, kernel::HandleType::Vmo(handle));

    loop {
        let vitals = {
            let mut m = monitor.lock().await;
            m.get_vitals()
        };

        // Format: "TEMP:XX.X|CPU:0.XX|RAM:X.XX"
        let payload = format!("TEMP:{:.1}|CPU:{:.2}|RAM:{:.2}", 
            vitals.temperature, vitals.cpu_load, vitals.ram_gb);
        
        let bytes = payload.as_bytes();
        unsafe {
            nexus_substrate::kernel::ffi::zx_vmo_write_native(handle, 0, bytes.as_ptr(), bytes.len());
        }

        sleep(Duration::from_secs(2)).await;
    }
}

use sysinfo::System;

pub struct Vitals {
    pub ram_gb: f64,
    pub total_ram_gb: f64,
    pub temperature: f32,
    pub cpu_load: f32,
}

pub struct SystemMonitor {
    sys: System,
}

impl SystemMonitor {
    pub fn new() -> Self {
        let mut sys = System::new_all();
        sys.refresh_all();
        Self { sys }
    }

    pub fn get_vitals(&mut self) -> Vitals {
        self.sys.refresh_memory();
        self.sys.refresh_cpu_all();

        let total_mem = self.sys.total_memory() as f64 / 1024.0 / 1024.0 / 1024.0;
        let used_mem = self.sys.used_memory() as f64 / 1024.0 / 1024.0 / 1024.0;
        let load = self.sys.global_cpu_usage() / 100.0;

        // ⚡ PUMP: Push real CPU load into the Zig kernel for synthetic thermal model
        unsafe {
            nexus_substrate::kernel::ffi::set_substrate_load(load);
        }

        // Synthetic temperature: 30°C idle → 70°C full load
        let temp = (load * 40.0) + 30.0;

        Vitals { ram_gb: used_mem, total_ram_gb: total_mem, temperature: temp, cpu_load: load }
    }
}

const FORMAT_CORRECTION: &str = "\
[FORMAT_VIOLATION] Your previous response was REJECTED — it contained prose \
before the JSON array. You MUST output ONLY a raw JSON array. \
Start your response with `[` and end with `]`. \
No reasoning text outside JSON. Retry the same action now.";

fn truncate_obs(obs: String, max_chars: usize) -> String {
    let char_count = obs.chars().count();
    if char_count <= max_chars { return obs; }
    let truncated: String = obs.chars().take(max_chars).collect();
    format!("{}... [TRUNCATED {}/{} chars]", truncated, max_chars, char_count)
}

pub async fn send_task(
    orchestrator: Arc<Orchestrator>,
    client: &Client,
    api_key: &str,
    raw_directive: &str,
    memory: Arc<Mutex<MemoryPalace>>,
    vmo_state: Arc<Mutex<VmoState>>,
    tx: broadcast::Sender<String>,
    dist: Arc<DistributedScheduler>,
) -> String {
    METRICS.inc_task();

    let mut session_obs: Vec<String> = vec![];
    let mut turn = 0usize;
    let mut consecutive_fractures = 0usize;
    let mut mission_outcome = "INCOMPLETE".to_string();

    const MAX_TURNS: usize = 12;
    const MAX_CONSECUTIVE_FRACTURES: usize = 3;
    const OBS_WINDOW: usize = 5;
    const MAX_OBS_CHARS: usize = 2500;

    let execution_law = cli::get_execution_law_v6().await;
    let mut executor = RoboticExecutor::new(api_key.to_string(), tx.clone());

    'mission: while turn < MAX_TURNS {
        turn += 1;
        info!("=== Turn {} / {} ===", turn, MAX_TURNS);

        if LogicEngine::detect_deadlock(&session_obs) {
            warn!("[DEADLOCK] Cortex Loop detected. Aborting mission.");
            METRICS.inc_failure();
            let _ = tx.send(serde_json::json!({"type": "TASK_STATUS", "payload": "DEADLOCK_ABORT"}).to_string());
            break 'mission;
        }

        if consecutive_fractures >= MAX_CONSECUTIVE_FRACTURES {
            warn!("[FRACTURE_STORM] {} consecutive failures. Aborting.", consecutive_fractures);
            METRICS.inc_failure();
            let _ = tx.send(serde_json::json!({"type": "TASK_STATUS", "payload": "FRACTURE_STORM_ABORT"}).to_string());
            break 'mission;
        }

        let window = session_obs.iter().rev().take(OBS_WINDOW).rev().cloned().collect::<Vec<_>>();

        let ctx = crate::neural_agent::ContextPacket {
            system_prompt: Some(execution_law.clone()),
            recent_logs: window,
            ..Default::default()
        };

        let response = match orchestrator.run(raw_directive.to_string(), ctx, client, api_key, Some(tx.clone())).await {
            Ok(resp) => resp,
            Err(ref e) if e.contains("CONTEXT_OVERFLOW") => {
                warn!("[CONTEXT_OVERFLOW] Pruning observation history and retrying...");
                let keep = session_obs.len() / 2;
                session_obs = session_obs.split_off(session_obs.len() - keep);
                continue 'mission;
            }
            Err(e) => {
                error!("Uplink failed: {}", e);
                break 'mission;
            }
        };

        match json_parser::extract_actions(&response) {
            Ok(actions) => {
                if actions.is_empty() { break 'mission; }
                consecutive_fractures = 0;

                match executor.execute_handle(actions.clone(), memory.clone(), vmo_state.clone(), dist.clone()).await {
                    Ok(observations) => {
                        for obs in observations {
                            let clean = truncate_obs(obs, MAX_OBS_CHARS);
                            session_obs.push(clean.clone());
                            if clean.contains("_OK") || clean.contains("[CMD:") { break; }
                        }
                    }
                    Err(e) => session_obs.push(format!("EXECUTOR_FAULT: {}", e)),
                }

                if actions.iter().any(|a| matches!(a, AgentAction::Halt { .. })) {
                    info!(">> [MISSION COMPLETE] Agent halted after {} turns.", turn);
                    mission_outcome = "COMPLETE".to_string();
                    break 'mission;
                }
            }
            Err(e) => {
                consecutive_fractures += 1;
                warn!("Turn {} | Parser fracture ({}/{}): {}", turn, consecutive_fractures, MAX_CONSECUTIVE_FRACTURES, e);
                session_obs.retain(|o| !o.starts_with("[FORMAT_VIOLATION]"));
                session_obs.push(FORMAT_CORRECTION.to_string());
            }
        }
        tokio::time::sleep(tokio::time::Duration::from_millis(500)).await;
    }

    let obs_summary = session_obs.iter().filter(|o| o.contains("_OK") || o.contains("FAULT") || o.contains("[CMD:")).cloned().collect::<Vec<_>>().join(" | ");
    let mem = memory.lock().await;
    let _ = mem.store_memory("TASK_MANAGER", raw_directive, &format!("[{}] turns={} | {}", mission_outcome, turn, obs_summary), None).await;

    "TASK_COMPLETE".to_string()
}

fn safe_path(workspace_root: &PathBuf, user_path: &str) -> Result<PathBuf, (StatusCode, Json<serde_json::Value>)> {
    if user_path.is_empty() { 
        return Err((StatusCode::BAD_REQUEST, Json(json!({ "error": "PATH_EMPTY" })))); 
    }
    
    let path = std::path::Path::new(user_path);
    // Lexical Guard: Reject absolute paths or any directory traversal '..' without invoking filesystem
    if path.is_absolute() || path.components().any(|c| matches!(c, std::path::Component::ParentDir)) {
        return Err((StatusCode::FORBIDDEN, Json(json!({ "error": "PATH_TRAVERSAL_DENIED" }))));
    }
    
    Ok(workspace_root.join(path))
}

pub async fn login_handler(Json(payload): Json<LoginRequest>) -> impl IntoResponse {
    let server_key = std::env::var("NEXUS_AUTH_KEY").unwrap_or_default();
    if !server_key.is_empty() && payload.key == server_key {
        match auth::issue_sovereign_token() {
            Ok(token) => (StatusCode::OK, Json(json!({ "token": token }))).into_response(),
            Err(_) => (StatusCode::INTERNAL_SERVER_ERROR, "Token generation failed").into_response(),
        }
    } else {
        (StatusCode::UNAUTHORIZED, "Invalid DNA Signature").into_response()
    }
}

async fn bootstrap_handler(ConnectInfo(addr): ConnectInfo<SocketAddr>) -> impl IntoResponse {
    if !addr.ip().is_loopback() { return (StatusCode::FORBIDDEN, "Loopback only").into_response(); }
    match std::fs::read_to_string(".nexus_ui_token") {
        Ok(token) => (StatusCode::OK, Json(json!({ "token": token.trim() }))).into_response(),
        Err(_) => (StatusCode::SERVICE_UNAVAILABLE, "Token not yet generated").into_response(),
    }
}

pub async fn unified_gateway(ConnectInfo(addr): ConnectInfo<SocketAddr>, State(state): State<Arc<AppState>>, Json(intent): Json<Intent>) -> impl IntoResponse {
    let ip = addr.ip().to_string();
    if let Err(retry_ms) = token_ledger::http_check(&ip) {
        return (StatusCode::TOO_MANY_REQUESTS, [(header::RETRY_AFTER, format!("{}", retry_ms / 1000 + 1))], Json(json!({ "status": "RATE_LIMITED", "retry_after_ms": retry_ms }))).into_response();
    }
    let directive_str = intent.directive.or_else(|| intent.payload.as_ref()?.get("directive")?.as_str().map(|s| s.to_string())).unwrap_or_default();
    if !LogicEngine::is_action_safe(&directive_str) { return (StatusCode::FORBIDDEN, Json(json!({ "status": "LOGIC_VIOLATION" }))).into_response(); }
    let s = state.clone();
    let d_clone = directive_str.clone();
    tokio::spawn(async move {
        let d = crate::cli::inject_workspace_dna(&d_clone).await;

async fn fs_read_handler(State(state): State<Arc<AppState>>, Query(q): Query<FsQuery>) -> impl IntoResponse {
    let user_path = q.path.unwrap_or_default();
    match safe_path(&state.workspace_root, &user_path) {
        Err(e) => e.into_response(),
        Ok(safe) => match tokio::fs::read_to_string(&safe).await {
            Ok(content) => Json(json!({ "content": content })).into_response(),
            Err(e) => (StatusCode::NOT_FOUND, Json(json!({ "error": e.to_string() }))).into_response(),
        },
    }
}

async fn fs_write_handler(State(state): State<Arc<AppState>>, Json(payload): Json<serde_json::Value>) -> impl IntoResponse {
    let user_path = payload["path"].as_str().unwrap_or_default();
    let content   = payload["content"].as_str().unwrap_or_default();
    match safe_path(&state.workspace_root, user_path) {
        Err(e) => e.into_response(),
        Ok(safe) => {
            if let Some(parent) = safe.parent() {
                if let Err(e) = tokio::fs::create_dir_all(parent).await { return (StatusCode::INTERNAL_SERVER_ERROR, Json(json!({ "error": e.to_string() }))).into_response(); }
            }
            match tokio::fs::write(&safe, content).await {
                Ok(_) => Json(json!({ "status": "success" })).into_response(),
                Err(e) => (StatusCode::INTERNAL_SERVER_ERROR, Json(json!({ "error": e.to_string() }))).into_response(),
            }
        }
    }
}

async fn fs_list_handler(State(state): State<Arc<AppState>>, Query(q): Query<FsQuery>) -> impl IntoResponse {
    let user_path = q.path.unwrap_or_else(|| ".".into());
    match safe_path(&state.workspace_root, &user_path) {
        Err(e) => e.into_response(),
        Ok(safe) => {
            let mut entries = Vec::new();
            if let Ok(mut read_dir) = tokio::fs::read_dir(&safe).await {
                while let Ok(Some(entry)) = read_dir.next_entry().await {
                    let meta = entry.metadata().await.ok();
                    entries.push(json!({
                        "name": entry.file_name().to_string_lossy(),
                        "path": entry.path().strip_prefix(&state.workspace_root).unwrap_or(&entry.path()).to_string_lossy(),
                        "is_dir": meta.map(|m| m.is_dir()).unwrap_or(false)
                    }));
                }
            }
            Json(entries).into_response()
        }
    }
}

export fn zx_vmo_calculate_entropy(h: u32) f32 {
    const shard = shard_for(h);
    shard_mutexes[shard].lock();
    defer shard_mutexes[shard].unlock();
    const meta = lookup_valid(h) orelse return 0.0;
    var counts = [_]u32{0} ** 256;
    for (meta.ptr[0..meta.size]) |byte| counts[byte] += 1;
    var entropy: f32 = 0.0;
    const size_f: f32 = @floatFromInt(meta.size);
    for (counts) |count| {
        if (count > 0) {
            const p = @as(f32, @floatFromInt(count)) / size_f;
            entropy -= p * std.math.log2(p);
        }
    }
    return entropy / 8.0;

        if (maybe_meta) |meta| {
            if (meta.magic == SENTINEL_MAGIC and count < max_count) {
                const slot: u32 = @intCast(i);
                out_handles[count] = slot | (meta.generation << GENERATION_SHIFT);
                count += 1;
            }
        }
    }
    return count;
}

use axum::{response::IntoResponse, Json};
use serde_json::json;

pub async fn handle_get_vmos() -> impl IntoResponse {
    let mut handles = [0u32; 1024];
    let count = unsafe { crate::kernel::ffi::zx_vmo_get_all_handles(handles.as_mut_ptr(), 1024) };

    let mut vmo_list = Vec::new();
    for i in 0..count {
        let mut info = crate::kernel::ffi::VmoInfo {
            handle: 0,
            size: 0,
            committed_pages: 0,
            is_cow: 0,
            parent_handle: 0,
        };
        let res = unsafe { crate::kernel::ffi::zx_vmo_get_info(handles[i], &mut info) };

        if res == 0 {
            // ⚡ V17: Real-time Entropy Pull
            let entropy = unsafe { crate::kernel::ffi::zx_vmo_calculate_entropy(handles[i]) };

            vmo_list.push(json!({
                "handle": format!("0x{:x}", info.handle),
                "size_bytes": info.size,
                "committed_pages": info.committed_pages,
                "memory_used_kb": (info.committed_pages * 4096) / 1024,
                "is_cow": info.is_cow,
                "entropy": format!("{:.4}", entropy),
                "parent_handle": if info.parent_handle > 0 { format!("0x{:x}", info.parent_handle) } else { "None".to_string() }
            }));
        }
    }

    Json(json!({ "vmos": vmo_list, "active_count": count }))
}

#[derive(Debug, Serialize, Deserialize)]
pub struct AgentMemory {

pub struct MemoryPalace {
    pub pool: SqlitePool,
}

impl MemoryPalace {
    pub async fn new(db_path: &str) -> Result<Self, sqlx::Error> {
        if !Path::new(db_path).exists() { std::fs::File::create(db_path).ok(); }
        let pool = SqlitePoolOptions::new()
            .max_connections(5)
            .connect(&format!("sqlite:{}", db_path))
            .await?;

        sqlx::query("PRAGMA journal_mode=WAL;").execute(&pool).await?;

        let palace = Self { pool };
        palace.init_full_schema().await?;
        Ok(palace)
    }

    pub async fn init_full_schema(&self) -> Result<(), sqlx::Error> {
        sqlx::query(
            "CREATE TABLE IF NOT EXISTS neural_context (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            origin TEXT NOT NULL,
            directive TEXT NOT NULL,
            result TEXT NOT NULL,
            embedding TEXT,
            timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
        );",
        ).execute(&self.pool).await?;

        sqlx::query(
            "CREATE VIRTUAL TABLE IF NOT EXISTS memory_search USING fts5(
            directive, result, content='neural_context', content_rowid='id'
        );",
        ).execute(&self.pool).await?;

        sqlx::query(
            "CREATE TABLE IF NOT EXISTS system_insights (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            summary TEXT NOT NULL,
            timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
        );",
        ).execute(&self.pool).await?;

        Ok(())
    }

    /// SUTURE: Atomic Transaction over FTS and Base Table
    pub async fn store_memory(
        &self,
        origin: &str,
        directive: &str,
        result: &str,
        embedding: Option<Vec<f32>>,
    ) -> Result<(), sqlx::Error> {
        let emb_str = embedding.map(|v| serde_json::to_string(&v).unwrap_or_default());
        
        let mut tx = self.pool.begin().await?;

        let res = sqlx::query("INSERT INTO neural_context (origin, directive, result, embedding) VALUES (?, ?, ?, ?)")
            .bind(origin)
            .bind(directive)
            .bind(result)
            .bind(emb_str)
            .execute(&mut *tx)
            .await?;

        let row_id = res.last_insert_rowid();
        sqlx::query("INSERT INTO memory_search (rowid, directive, result) VALUES (?, ?, ?)")
            .bind(row_id)
            .bind(directive)
            .bind(result)
            .execute(&mut *tx)
            .await?;
            
        tx.commit().await?;
        Ok(())
    }

    pub async fn store_insight(&self, summary: &str) -> Result<(), sqlx::Error> {
        sqlx::query("INSERT INTO system_insights (summary) VALUES (?)").bind(summary).execute(&self.pool).await?;
        Ok(())
    }

    pub async fn fetch_all(&self) -> Result<Vec<AgentMemory>, sqlx::Error> {
        let rows = sqlx::query("SELECT id, origin, directive, result, timestamp FROM neural_context ORDER BY id DESC LIMIT 50")
            .fetch_all(&self.pool).await?;
        Ok(rows.into_iter().map(|row| AgentMemory {
            id: row.get("id"),
            origin: row.get("origin"),
            directive: row.get("directive"),
            result: row.get("result"),
            timestamp: row.get("timestamp"),
        }).collect())
    }

    pub async fn search_memories(&self, query: &str) -> Result<Vec<AgentMemory>, sqlx::Error> {
        let rows = sqlx::query(
            "SELECT n.id, n.origin, n.directive, n.result, n.timestamp 
             FROM neural_context n JOIN memory_search m ON n.id = m.rowid 
             WHERE memory_search MATCH ? ORDER BY rank LIMIT 10",
        ).bind(query).fetch_all(&self.pool).await?;

        Ok(rows.into_iter().map(|row| AgentMemory {
            id: row.get(0), origin: row.get(1), directive: row.get(2), result: row.get(3), timestamp: row.get(4),
        }).collect())
    }

    pub async fn semantic_search(&self, query_vec: &[f32]) -> Result<Vec<AgentMemory>, sqlx::Error> {
        let rows = sqlx::query("SELECT id, origin, directive, result, timestamp, embedding FROM neural_context WHERE embedding IS NOT NULL")
            .fetch_all(&self.pool).await?;

        let mut scored_mems = Vec::new();
        for row in rows {
            let emb_str: Option<String> = row.get("embedding");
            if let Some(s) = emb_str {
                if let Ok(vec) = serde_json::from_str::<Vec<f32>>(&s) {
                    let score = crate::vector_bridge::cosine_similarity(query_vec, &vec);
                    scored_mems.push((score, row));
                }
            }
        }

        scored_mems.sort_by(|a, b| b.0.partial_cmp(&a.0).unwrap_or(std::cmp::Ordering::Equal));
        scored_mems.truncate(3); 

        Ok(scored_mems.into_iter().map(|(_, row)| AgentMemory {
            id: row.get("id"), origin: row.get("origin"), directive: row.get("directive"),
            result: row.get("result"), timestamp: row.get("timestamp"),
        }).collect())
    }

    pub async fn purge_all(&self) -> Result<(), sqlx::Error> {
        sqlx::query("DELETE FROM neural_context").execute(&self.pool).await?;
        sqlx::query("DELETE FROM memory_search").execute(&self.pool).await?;
        sqlx::query("DELETE FROM system_insights").execute(&self.pool).await?;
        Ok(())
    }

    pub async fn fetch_random_strands(&self, limit: i64) -> Result<Vec<AgentMemory>, sqlx::Error> {
        let rows = sqlx::query("SELECT id, origin, directive, result, timestamp FROM neural_context ORDER BY RANDOM() LIMIT ?")
            .bind(limit).fetch_all(&self.pool).await?;
        Ok(rows.into_iter().map(|row| AgentMemory {
            id: row.get("id"), origin: row.get("origin"), directive: row.get("directive"), result: row.get("result"), timestamp: row.get("timestamp"),
        }).collect())
    }
}

pub fn cosine_similarity(a: &[f32], b: &[f32]) -> f32 {
    if a.len() != b.len() || a.is_empty() { return 0.0; }
    let dot: f32 = a.iter().zip(b.iter()).map(|(x, y)| x * y).sum();
    let norm_a: f32 = a.iter().map(|x| x * x).sum::<f32>().sqrt();
    let norm_b: f32 = b.iter().map(|x| x * x).sum::<f32>().sqrt();
    if norm_a == 0.0 || norm_b == 0.0 { return 0.0; }
    dot / (norm_a * norm_b)
}

use once_cell::sync::Lazy;
use std::collections::HashMap;
use std::sync::Mutex;
use std::time::{Duration, Instant, SystemTime};

#[derive(Debug, Clone)]
pub struct ModelQuota {
    pub rpm_limit: u32,
    pub tpm_limit: u32,
}

impl ModelQuota {
    fn for_model(model: &str) -> Self {
        match model {
            m if m.contains("flash") => Self { rpm_limit: 15, tpm_limit: 1_000_000 },
            m if m.contains("31b")   => Self { rpm_limit: 20, tpm_limit: 2_000_000 },
            _                        => Self { rpm_limit: 15, tpm_limit: 1_000_000 },
        }
    }
}

#[derive(Debug)]
struct WindowEntry {
    timestamp: Instant,
    tokens:    u32,
}

#[derive(Debug, Default)]
struct ModelWindow {
    entries: Vec<WindowEntry>,
}

impl ModelWindow {
    fn purge_stale(&mut self) {
        let cutoff = Instant::now() - Duration::from_secs(60);
        self.entries.retain(|e| e.timestamp > cutoff);
    }

    fn token_sum(&self) -> u32 {
        self.entries.iter().map(|e| e.tokens).sum()
    }

    fn ms_until_slot_free(&self) -> u64 {
        if self.entries.is_empty() { return 0; }
        let oldest = self.entries.iter().map(|e| e.timestamp).min().unwrap();
        (Duration::from_secs(60).saturating_sub(oldest.elapsed())).as_millis() as u64 + 100
    }
}

pub struct TokenLedger {
    windows: Mutex<HashMap<String, ModelWindow>>,
}

impl TokenLedger {
    pub fn new() -> Self {
        Self { windows: Mutex::new(HashMap::new()) }
    }

    pub fn check_and_reserve(&self, model: &str, estimated_tokens: u32) -> Result<(), u64> {
        let quota = ModelQuota::for_model(model);
        let mut map = self.windows.lock().unwrap();
        let window = map.entry(model.to_string()).or_default();
        window.purge_stale();

        if window.entries.len() as u32 >= quota.rpm_limit { return Err(window.ms_until_slot_free()); }
        if window.token_sum() + estimated_tokens > quota.tpm_limit { return Err(window.ms_until_slot_free()); }

        window.entries.push(WindowEntry { timestamp: Instant::now(), tokens: estimated_tokens });
        Ok(())
    }
}

pub static LEDGER: Lazy<TokenLedger> = Lazy::new(TokenLedger::new);

pub async fn wait_for_quota(model: &str, estimated_tokens: u32) {
    let mut attempts = 0;
    loop {
        match LEDGER.check_and_reserve(model, estimated_tokens) {
            Ok(()) => return,
            Err(wait_ms) => {
                let jitter = (generate_seed() % 500) as u64;
                println!(
                    "\x1b[33m[LEDGER] {} ceiling hit. Waiting {}ms (attempt {})...\x1b[0m",
                    model, wait_ms + jitter, attempts + 1
                );
                tokio::time::sleep(Duration::from_millis(wait_ms + jitter)).await;
                attempts += 1;
                if attempts > 10 { break; }
            }
        }
    }
}

fn http_rpm_limit() -> u32 {
    std::env::var("NEXUS_HTTP_RPM").ok().and_then(|v| v.parse().ok()).unwrap_or(60)
}

static HTTP_WINDOWS: Lazy<Mutex<HashMap<String, Vec<Instant>>>> = Lazy::new(|| Mutex::new(HashMap::new()));

pub fn http_check(ip: &str) -> Result<(), u64> {
    let limit = http_rpm_limit();
    let cutoff = Instant::now() - Duration::from_secs(60);

    let mut map = HTTP_WINDOWS.lock().unwrap();
    let window = map.entry(ip.to_string()).or_default();
    window.retain(|&ts| ts > cutoff);

    if window.len() as u32 >= limit {
        let oldest = *window.first().unwrap();
        let wait = (Duration::from_secs(60).saturating_sub(oldest.elapsed())).as_millis() as u64;
        return Err(wait + 100);
    }

    window.push(Instant::now());
    Ok(())
}

/// generate_seed: OS-entropy random u64 with timestamp mixing for per-call uniqueness.
/// RandomState seeds from getrandom() syscall; writing the current nanos ensures
/// two calls within the same nanosecond still return different values.
fn generate_seed() -> u64 {
    use std::hash::{BuildHasher, Hasher};
    let t = SystemTime::now()
        .duration_since(SystemTime::UNIX_EPOCH)
        .unwrap_or_default()
        .as_nanos() as u64;
    let mut h = std::collections::hash_map::RandomState::new().build_hasher();
    h.write_u64(t);
    h.finish()
}
