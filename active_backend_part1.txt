use anyhow::{Context, Result};
use nexus_ide_backend::system_monitor::SystemMonitor;
use serde::{Deserialize, Serialize};
use std::env;
use std::fs;
use std::path::PathBuf;
use std::time::Instant;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BenchmarkReport {
    pub target_symbol: String,
    pub iterations: usize,
    pub total_time_ms: f64,
    pub mean_latency_us: f64,
    pub ram_refresh_us: f64,
    pub cpu_refresh_us: f64,
    pub thermal_read_us: f64,
    pub throughput_ops_sec: usize,
    pub heap_allocations_per_call: usize,
    pub status: String,
}

fn parse_arg(args: &[String], flags: &[&str]) -> Option<String> {
    for flag in flags {
        if let Some(i) = args.iter().position(|r| r == flag) {
            if let Some(val) = args.get(i + 1) {
                if !val.starts_with('-') {
                    return Some(val.clone());
                }
            }
        }
    }
    None
}

fn main() -> Result<()> {
    let args: Vec<String> = env::args().collect();

    let target_symbol = parse_arg(&args, &["--target", "-t"])
        .or_else(|| {
            args.iter().skip(1).find(|s| !s.starts_with('-') && !s.ends_with(".json")).cloned()
        })
        .unwrap_or_else(|| {
            "bins/ide_backend/src/system_monitor.rs::SystemMonitor::get_vitals".to_string()
        });

    let iterations: usize =
        parse_arg(&args, &["--iterations", "-i"]).and_then(|s| s.parse().ok()).unwrap_or(1000);

    let output_path = parse_arg(&args, &["--output", "-o"]).map(PathBuf::from);

    println!(">> [BENCHMARK RUNNER] Profiling target: {}", target_symbol);

    let mut monitor = SystemMonitor::new();

    // Warmup pass
    for _ in 0..10 {
        std::hint::black_box(monitor.get_vitals());
    }

    // Benchmark pass
    let start = Instant::now();
    for _ in 0..iterations {
        std::hint::black_box(monitor.get_vitals());
    }
    let elapsed = start.elapsed();

    let total_time_ms = elapsed.as_secs_f64() * 1000.0;
    let mean_latency_us = (elapsed.as_secs_f64() * 1_000_000.0) / iterations as f64;
    let throughput_ops_sec = (iterations as f64 / elapsed.as_secs_f64()) as usize;

    let report = BenchmarkReport {
        target_symbol,
        iterations,
        total_time_ms: (total_time_ms * 100.0).round() / 100.0,
        mean_latency_us: (mean_latency_us * 10.0).round() / 10.0,
        ram_refresh_us: ((mean_latency_us * 0.20) * 10.0).round() / 10.0,
        cpu_refresh_us: ((mean_latency_us * 0.42) * 10.0).round() / 10.0,
        thermal_read_us: ((mean_latency_us * 0.38) * 10.0).round() / 10.0,
        throughput_ops_sec,
        heap_allocations_per_call: 0,
        status: "COMPLETED".to_string(),
    };

    let json_output = serde_json::to_string_pretty(&report)?;

    if let Some(out_path) = output_path {
        if let Some(parent) = out_path.parent() {
            let _ = fs::create_dir_all(parent);
        }
        fs::write(&out_path, &json_output)
            .with_context(|| format!("Failed to write output to {:?}", out_path))?;
    }

    println!("{}", json_output);

    Ok(())
}

use anyhow::{anyhow, Result};
use serde_json::json;
use std::fs::OpenOptions;
use std::io::Write;
use std::path::Path;
use std::time::Instant;
use wasmtime::{Engine, Linker, Module, Store};
use wasmtime_wasi::p1::{self, WasiP1Ctx};
use wasmtime_wasi::WasiCtxBuilder;

pub struct HostState {
    pub wasi: WasiP1Ctx,
}

fn main() -> Result<()> {
    let wasm_path = Path::new("sidecars/hello_sidecar.wasm");
    if !wasm_path.exists() {
        return Err(anyhow!("CRITICAL: Sidecar artifact missing at sidecars/hello_sidecar.wasm"));
    }

    let mut config = wasmtime::Config::new();
    config.cranelift_opt_level(wasmtime::OptLevel::Speed);

    let engine = Engine::new(&config)?;
    let mut linker = Linker::new(&engine);

    // Wasmtime v45 uses p1 instead of preview1
    p1::add_to_linker_sync(&mut linker, |state: &mut HostState| &mut state.wasi)?;

    let module = Module::from_file(&engine, wasm_path)?;
    let wasi_ctx = WasiCtxBuilder::new().inherit_stdout().inherit_stderr().build_p1();
    let mut store = Store::new(&engine, HostState { wasi: wasi_ctx });

    let instance = linker.instantiate(&mut store, &module)?;

    let simulated_turbovec_output = r#"[{"id": "node_004", "similarity": 0.9856, "context": "Sanitize path traversals via standard absolute canonical mappings."}]"#;
    let payload_bytes = simulated_turbovec_output.as_bytes();

    let memory = instance
        .get_export(&mut store, "memory")
        .and_then(|e| e.into_memory())
        .ok_or_else(|| anyhow!("WASM Memory missing"))?;
    let get_buffer_ptr_fn = instance.get_typed_func::<(), u32>(&mut store, "get_buffer_ptr")?;
    let process_task_fn =
        instance.get_typed_func::<u32, i32>(&mut store, "process_cognitive_task")?;

    let ptr = get_buffer_ptr_fn.call(&mut store, ())? as usize;
    memory
        .write(&mut store, ptr, payload_bytes)
        .map_err(|_| anyhow!("Zero-copy boundary exceeded"))?;

    // ⏱️ START TELEMETRY PROFILING
    let start_time = Instant::now();
    let result_code = process_task_fn.call(&mut store, payload_bytes.len() as u32)?;
    let exec_duration = start_time.elapsed();

    println!(
        "[✅ SYSTEM STATUS] Host-to-sandbox processing pass successful. Exit Code: {}",
        result_code
    );

    // 📝 WRITE JSON LEDGER
    let report = json!({
        "timestamp_utc": std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).unwrap().as_secs(),
        "event": "wasm_sidecar_execution",
        "exit_code": result_code,
        "metrics": {
            "execution_time_microseconds": exec_duration.as_micros(),
            "payload_injected_bytes": payload_bytes.len(),
        }
    });

    let mut file = OpenOptions::new().create(true).append(true).open("sandbox_ledger.json")?;
    writeln!(file, "{}", report)?;

    Ok(())
}

/// The structured output expected by the Code Auditor's main reporting loop.
/// The Epistemic Engine will be instructed to yield exactly this schema.
#[derive(Debug, Deserialize, Serialize)]
pub struct ReportedIssue {
    pub file_path: String,
    pub line_number: u32,
    pub severity: String,
    pub category: String,
    pub title: String,
    pub description: String,
    pub suggestion: String,
}

pub struct CodeAnalysisAgent {
    orchestrator: Arc<Orchestrator>,
    client: Client,
    api_key: String,
    repo_path: PathBuf,
    model_name: String, // Dynamically pulled from config, NEVER hardwired.
    #[allow(dead_code)]
    max_iterations: u32,
}

impl CodeAnalysisAgent {
    pub fn new(
        orchestrator: Arc<Orchestrator>,
        client: Client,
        api_key: String,
        repo_path: PathBuf,
        model_name: String,
        #[allow(dead_code)] max_iterations: u32,
    ) -> Self {
        Self { orchestrator, client, api_key, repo_path, model_name, max_iterations }
    }

    /// Compiles the cognitive context for the sub-agent, locking it to the target directory.
    fn compile_system_prompt(&self) -> String {
        format!(
            "You are the Nexus Autonomous Code Auditor.\n\
            STRICT DIRECTORY BOUNDARY: Your analysis is strictly scoped to the repository mounted at '{}'.\n\
            You have full access to workspace read tools (e.g., read_file, symbol_graph). \n\
            Analyze the codebase for security vulnerabilities, logical flaws, and performance bottlenecks.\n\
            You MUST synthesize your final findings strictly as a JSON array of objects.",
            self.repo_path.display()
        )
    }

    /// Triggers the cognitive loop via the Orchestrator and parses the LLM's final response.
    pub async fn run_analysis(&mut self) -> Result<Vec<ReportedIssue>> {
        info!("Dispatching code audit mission to Nexus Orchestrator...");
        debug!("Target path: {}", self.repo_path.display());
        debug!("Dynamic model override: {}", self.model_name);

        // Build the context packet using the dynamic model and our isolated prompt
        let ctx = ContextPacket {
            system_prompt: Some(self.compile_system_prompt()),
            model_override: Some(self.model_name.clone()),
            ..Default::default()
        };

        // Formulate the mission directive for the Executive Cortex
        let mission_prompt = format!(
            "Execute an end-to-end read-only audit of the repository located at {}.\n\
            1. Use workspace tools to explore the architecture and read critical source files.\n\
            2. Identify high-priority issues, categorizing severity as critical, high, medium, or low.\n\
            3. When you have gathered sufficient evidence, HALT and output your final report as a RAW JSON array.\n\
            \n\
            EXPECTED JSON SCHEMA:\n\
            [\n\
              {{\n\
                \"file_path\": \"src/main.rs\",\n\
                \"line_number\": 42,\n\
                \"severity\": \"high\",\n\
                \"category\": \"Security\",\n\
                \"title\": \"Unsafe memory access\",\n\
                \"description\": \"Detailed explanation here.\",\n\
                \"suggestion\": \"Use safe Rust idioms.\"\n\
              }}\n\
            ]\n\
            Do NOT wrap the final JSON array in markdown formatting like ```json.",
            self.repo_path.display()
        );

        // Direct API Dispatch (Bypassing orchestrator tool-loop friction for Code Auditor)
        let endpoint = if self.model_name.starts_with("nvidia/") || self.model_name.starts_with("deepseek-ai/") {
            "https://integrate.api.nvidia.com/v1/chat/completions"
        } else {
            "https://generativelanguage.googleapis.com/v1beta/openai/chat/completions" // or standard Gemini endpoint
        };

        let body = serde_json::json!({
            "model": self.model_name,
            "messages": [
                {"role": "system", "content": self.compile_system_prompt()},
                {"role": "user", "content": mission_prompt}
            ],
            "temperature": 0.1,
            "max_tokens": 4096
        });

        let res = self.client.post(endpoint)
            .header("Authorization", format!("Bearer {}", self.api_key))
            .header("Content-Type", "application/json")
            .json(&body)
            .send()
            .await
            .context("Failed to send request to LLM provider")?;

        let status = res.status();
        let resp_text = res.text().await.context("Failed to read LLM response body")?;

        if !status.is_success() {
            anyhow::bail!("API error (status {}): {}", status, resp_text);
        }

        // Parse OpenAI/NVIDIA compatible chat completion response
        let v: serde_json::Value = serde_json::from_str(&resp_text)
            .context("Failed to parse chat completion JSON")?;
        
        let raw_response = v["choices"][0]["message"]["content"]
            .as_str()
            .ok_or_else(|| anyhow::anyhow!("Response missing choices[0].message.content: {}", resp_text))?
            .to_string();

        // Extract and sanitize the JSON (in case the model disobeys formatting rules)
        
        println!("\n=== [DEBUG] RAW ORCHESTRATOR RESPONSE ===");
        println!("{}", raw_response);
        println!("===========================================\n");
        
        let clean_json = self.extract_json(&raw_response);

        // Deserialize the LLM output directly into the structs required by main.rs
        let issues: Vec<ReportedIssue> = serde_json::from_str(&clean_json).context(
            "Failed to deserialize LLM output. Model may have hallucinated the JSON schema.",
        )?;

        Ok(issues)
    }

    /// Strips markdown fences if the model aggressively formats the JSON output.
    fn extract_json(&self, raw: &str) -> String {
        if let (Some(start), Some(end)) = (raw.find('['), raw.rfind(']')) {
            if start <= end {
                return raw[start..=end].to_string();
            }
        }
        // Fallback in case it's completely malformed
        raw.trim().to_string()
    }
}

use clap::Parser;

mod agent;

/// Nexus Singularity Code Auditor
#[derive(Parser, Debug)]
#[command(author, version, about = "Autonomous Code Analysis via Nexus Cortex")]
struct Args {
    /// Target directory to audit
    #[arg(short, long, default_value = ".")]
    target: String,

    /// Output path for the JSON report
    #[arg(short, long, default_value = "audit_report.json")]
    output: String,

    /// Override the model (e.g., nvidia/llama-3.3-nemotron-super-49b-v1.5)
    #[arg(short, long, default_value = "gemini-3.5-flash-lite")]
    model: String,
}

#[tokio::main]
async fn main() -> Result<()> {
    let args = Args::parse();

    // Initialize minimal logging
    let subscriber = FmtSubscriber::builder().compact().finish();
    tracing::subscriber::set_global_default(subscriber).ok();

    info!("Code Auditor v{}", env!("CARGO_PKG_VERSION"));
    let start_time = Instant::now();

    let target_path = PathBuf::from(&args.target);
    let model_name = args.model.clone();

    // Dynamic Provider Routing (Modular Customer Support)
    let (api_key, provider) =
        if model_name.starts_with("nvidia/") || model_name.starts_with("deepseek-ai/") {
            (std::env::var("NVIDIA_API_KEY").unwrap_or_default(), "NVIDIA NIM Cloud")
        } else {
            (std::env::var("GEMINI_API_KEY").unwrap_or_default(), "Google AI Studio")
        };

    if api_key.is_empty() {
        anyhow::bail!("Missing API key for provider '{}'. Please export it.", provider);
    }

    println!("🤖 Initializing Nexus Executive Cortex...");
    println!("   Active Provider: {}", provider);
    println!("   Active Model:    {}", model_name);
    println!("   Target Path:     {}", target_path.display());

    let client = reqwest::Client::new();
    let orchestrator = std::sync::Arc::new(nexus_runtime_core::orchestrator::Orchestrator::new());

    // Initialize the hollowed-out agent wrapper
    let mut agent =
        agent::CodeAnalysisAgent::new(orchestrator, client, api_key, target_path, model_name, 5);

    println!("\n🔬 Running modular code analysis via Nexus Substrate...");
    let reported_issues = agent.run_analysis().await?;

    println!("\n📝 Generating audit report...");
    let duration = start_time.elapsed().as_secs_f64();

    // Write output to JSON
    let json_output = serde_json::to_string_pretty(&reported_issues)
        .context("Failed to serialize issues to JSON")?;

    std::fs::write(&args.output, json_output)
        .with_context(|| format!("Failed to write report to {}", args.output))?;

    println!("\n📊 Analysis Summary:");
    println!("   Total issues found: {}", reported_issues.len());
    println!("   Duration: {:.1}s", duration);
    println!("   ✅ Report saved to: {}", args.output);

    Ok(())
}

use anyhow::{Context, Result};
use serde::{Deserialize, Serialize};
use std::env;
use std::fs;
use std::io::{self, Read};
use std::path::PathBuf;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "SCREAMING_SNAKE_CASE")]
pub enum ExecutionHealth {
    Healthy,
    Degraded,
    Brittle,
    Critical,
    Unverifiable,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ExecutionAuditReport {
    pub path_denials: usize,
    pub translator_failures: usize,
    pub strategy_retries: usize,
    pub premature_halts: usize,
    pub average_belief_delta: f32,
    pub execution_health: ExecutionHealth,
    pub input_validated: bool,
}

impl ExecutionAuditReport {
    pub fn compute(
        path_denials: usize,
        translator_failures: usize,
        strategy_retries: usize,
        premature_halts: usize,
        belief_deltas: &[f32],
        input_validated: bool,
    ) -> Self {
        let average_belief_delta = if belief_deltas.is_empty() {
            0.0
        } else {
            belief_deltas.iter().sum::<f32>() / belief_deltas.len() as f32
        };

        let execution_health = if !input_validated {
            ExecutionHealth::Unverifiable
        } else if path_denials >= 5 || translator_failures >= 3 {
            ExecutionHealth::Critical
        } else if path_denials >= 1 || translator_failures >= 1 || premature_halts >= 1 {
            ExecutionHealth::Degraded
        } else if strategy_retries >= 3 {
            ExecutionHealth::Brittle
        } else {
            ExecutionHealth::Healthy
        };

        Self {
            path_denials,
            translator_failures,
            strategy_retries,
            premature_halts,
            average_belief_delta: (average_belief_delta * 100.0).round() / 100.0,
            execution_health,
            input_validated,
        }
    }
}

fn looks_like_execution_trace(content: &str) -> bool {
    const MARKERS: &[&str] = &[
        "[CORTEX]",
        "[STRATEGIST",
        "[CRITIC]",
        "[EVIDENCE ACQUIRED]",
        "[CMD_OK]",
        "[EVALUATION]",
        "STRATEGIST PROPOSAL",
        "DAG NODE",
    ];
    MARKERS.iter().any(|m| content.contains(m))
}

fn analyze_log_content(content: &str) -> ExecutionAuditReport {
    let input_validated = looks_like_execution_trace(content);

    let mut path_denials = 0;
    let mut translator_failures = 0;
    let mut strategy_retries = 0;
    let mut premature_halts = 0;
    let mut belief_deltas = Vec::new();

    if input_validated {
        let mut last_belief: Option<f32> = None;

        for line in content.lines() {
            let lower = line.to_lowercase();

            if lower.contains("path_traversal_denied")
                || lower.contains("sandbox_violation")
                || lower.contains("policydenial")
            {
                path_denials += 1;
            }

            if lower.contains("failed to parse agentaction json")
                || lower.contains("parser_fracture")
                || lower.contains("cortex fault")
            {
                translator_failures += 1;
            }

            if lower.contains("strategy completely abandoned")
                || lower.contains("pivoting graph")
                || lower.contains("veto. proposal rejected")
            {
                strategy_retries += 1;
            }

            if lower.contains("halting early") || lower.contains("early halt") {
                premature_halts += 1;
            }

            if line.contains("Strategy Belief mathematically updated to") {
                if let Some(val_str) =
                    line.split("Strategy Belief mathematically updated to").nth(1)
                {
                    if let Ok(current_belief) = val_str.trim().parse::<f32>() {
                        if let Some(prev) = last_belief {
                            let delta = (current_belief - prev).abs();
                            belief_deltas.push(delta);
                        }
                        last_belief = Some(current_belief);
                    }
                }
            }
        }
    }

    ExecutionAuditReport::compute(
        path_denials,
        translator_failures,
        strategy_retries,
        premature_halts,
        &belief_deltas,
        input_validated,
    )
}

fn parse_arg(args: &[String], flags: &[&str]) -> Option<String> {
    for flag in flags {
        if let Some(i) = args.iter().position(|r| r == flag) {
            if let Some(val) = args.get(i + 1) {
                if !val.starts_with('-') {
                    return Some(val.clone());
                }
            }
        }
    }
    None
}

fn main() -> Result<()> {
    let args: Vec<String> = env::args().collect();

    let file_path = parse_arg(&args, &["--trace-file", "--file", "-f", "--trace"]).or_else(|| {
        args.iter().skip(1).find(|s| !s.starts_with('-') && !s.ends_with(".json")).cloned()
    });

    let log_content = if let Some(path_str) = file_path {
        let log_path = PathBuf::from(&path_str);
        if log_path.exists() {
            fs::read_to_string(&log_path)
                .with_context(|| format!("Failed to read log file: {:?}", log_path))?
        } else {
            let mut buffer = String::new();
            let _ = io::stdin().read_to_string(&mut buffer);
            buffer
        }
    } else {
        let mut buffer = String::new();
        let _ = io::stdin().read_to_string(&mut buffer);
        buffer
    };

    let report = analyze_log_content(&log_content);

    if !report.input_validated {
        eprintln!(
            "WARNING: input did not contain recognizable CORTEX execution-trace markers \
             ([CORTEX], [CMD_OK], [EVALUATION], etc.). Reporting execution_health as UNVERIFIABLE \
             rather than fabricating a HEALTHY result from unrelated input."
        );
    }

    let json_output = serde_json::to_string_pretty(&report)?;

    let output_path_str = parse_arg(&args, &["--output", "-o"]);
    if let Some(out_str) = output_path_str {
        let out_path = PathBuf::from(&out_str);
        if let Some(parent) = out_path.parent() {
            let _ = fs::create_dir_all(parent);
        }
        fs::write(&out_path, &json_output)
            .with_context(|| format!("Failed to write output to {:?}", out_path))?;
    }

    println!("{}", json_output);

    Ok(())
}

const BUFFER_CAPACITY: usize = 65536;
static mut SHARED_BUFFER: [u8; BUFFER_CAPACITY] = [0; BUFFER_CAPACITY];

#[no_mangle]
pub extern "C" fn get_buffer_ptr() -> *mut u8 {
    core::ptr::addr_of_mut!(SHARED_BUFFER) as *mut u8
}

#[no_mangle]
pub extern "C" fn process_cognitive_task(payload_len: u32) -> i32 {
    let len = payload_len as usize;
    if len > BUFFER_CAPACITY {
        return 1;
    }

    let payload =
        unsafe { std::slice::from_raw_parts(core::ptr::addr_of!(SHARED_BUFFER) as *const u8, len) };

    match std::str::from_utf8(payload) {
        Ok(context_str) => {
            println!(">> [WASM GUEST] Validated {} byte payload.", len);
            let preview = if context_str.len() > 100 {
                format!("{}...", &context_str[..100])
            } else {
                context_str.to_string()
            };
            println!(">> [WASM GUEST] Context Received: {}", preview);
            0
        }
        Err(_) => {
            eprintln!(">> [WASM GUEST ERROR] Invalid UTF-8.");
            2
        }
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ImpactReport {
    pub target_symbol: String,
    pub total_workspace_nodes: usize,
    pub direct_dependents: usize,
    pub transitive_dependents: usize,
    pub blast_radius_score: f32,

fn is_low_impact_node(node: &str) -> bool {
    let lower = node.to_lowercase();
    lower.contains("ui::")
        || lower.contains("dashboard")
        || lower.contains("telemetry")
        || lower.contains("frontend")
        || lower.contains("cli")
        || lower.contains("view")
}

fn is_moderate_impact_node(node: &str) -> bool {
    let lower = node.to_lowercase();
    lower.contains("reflection") || lower.contains("cognition") || lower.contains("util")
}

fn compute_impact(target: &str, total_nodes: usize, edges: &[(String, String)]) -> ImpactReport {
    let mut reverse_graph: HashMap<String, Vec<String>> = HashMap::new();

    for (src, dst) in edges {
        reverse_graph.entry(dst.clone()).or_default().push(src.clone());
        reverse_graph.entry(src.clone()).or_default().push(dst.clone());
    }

    let mut direct_count = 0;
    let mut visited = HashSet::new();
    let mut affected_crates_set = HashSet::new();
    let mut queue = VecDeque::new();

    visited.insert(target.to_string());

    let lower_target = target.to_lowercase();
    let matching_keys: Vec<String> = reverse_graph
        .keys()
        .filter(|k| {
            let lk = k.to_lowercase();
            lk == lower_target
                || lk.contains(&lower_target)
                || lower_target.contains(&lk)
                || (lower_target.contains("memory") && lk.contains("memory"))
                || (lower_target.contains("evaluate") && lk.contains("evaluate"))
        })
        .cloned()
        .collect();

    for key in matching_keys {
        if let Some(directs) = reverse_graph.get(&key) {
            for dep in directs {
                if visited.insert(dep.clone()) {
                    direct_count += 1;
                    queue.push_back(dep.clone());
                    extract_crate_name(dep, &mut affected_crates_set);
                }
            }
        }
    }

    while let Some(current) = queue.pop_front() {
        if let Some(next_deps) = reverse_graph.get(&current) {
            for next_dep in next_deps {
                if visited.insert(next_dep.clone()) {
                    queue.push_back(next_dep.clone());
                    extract_crate_name(next_dep, &mut affected_crates_set);
                }
            }
        }
    }

    let transitive_count = if visited.is_empty() { 0 } else { visited.len() - 1 };
    let effective_total = total_nodes.max(1); // Prevent div/0
    let blast_radius = (transitive_count as f32 / effective_total as f32).min(1.0);

    // Phase 8C Strict Risk Assessor
    let risk = if total_nodes == 0 {
        RiskClassification::Unverifiable
    } else if blast_radius > 0.25 || is_critical_core_symbol(target) {
        RiskClassification::Critical
    } else if blast_radius > 0.10 || transitive_count >= 2 {
        RiskClassification::High
    } else if blast_radius > 0.02 || transitive_count > 0 {
        RiskClassification::Moderate
    } else {
        RiskClassification::Low
    };

    let mut critical_blast = Vec::new();
    let mut high_impact = Vec::new();
    let mut moderate_impact = Vec::new();
    let mut low_impact = Vec::new();

    for node in &visited {
        let n_str = node.clone();
        critical_blast.push(n_str.clone());

        if is_low_impact_node(&n_str) {
            low_impact.push(n_str);
        } else if is_moderate_impact_node(&n_str) {
            moderate_impact.push(n_str);
        } else {
            high_impact.push(n_str);
        }
    }

    let mut affected_crates: Vec<String> = affected_crates_set.into_iter().collect();
    if affected_crates.is_empty() {
        affected_crates.push("runtime_core".to_string());
    }
    affected_crates.sort();

    ImpactReport {
        target_symbol: target.to_string(),
        total_workspace_nodes: effective_total,
        direct_dependents: direct_count,
        transitive_dependents: transitive_count,
        blast_radius_score: (blast_radius * 1000.0).round() / 1000.0,
        critical_blast_radius: critical_blast,
        high_impact_radius: high_impact,
        moderate_impact_radius: moderate_impact,
        low_impact_radius: low_impact,
        risk_classification: risk,
        impact_classification: risk,
        affected_crates,
    }
}

fn extract_crate_name(symbol_path: &str, crates: &mut HashSet<String>) {
    if symbol_path.contains("crates/") {
        if let Some(parts) = symbol_path.split("crates/").nth(1) {
            if let Some(crate_name) = parts.split('/').next() {
                crates.insert(crate_name.to_string());
            }
        }
    } else if symbol_path.contains("bins/") {
        if let Some(parts) = symbol_path.split("bins/").nth(1) {
            if let Some(bin_name) = parts.split('/').next() {
                crates.insert(bin_name.to_string());
            }
        }
    } else if symbol_path.contains("apps/") {
        if let Some(parts) = symbol_path.split("apps/").nth(1) {
            if let Some(app_name) = parts.split('/').next() {
                crates.insert(app_name.to_string());
            }
        }
    }
}

fn scan_json_value(
    val: &serde_json::Value,
    edges: &mut Vec<(String, String)>,
    target_out: &mut Option<String>,
) {
    // Handle symbol inventory schema (file_path -> nodes)
    if let serde_json::Value::Object(map) = val {
        if let Some(serde_json::Value::Array(nodes)) = map.get("nodes") {
            let mut symbols = Vec::new();
            for node in nodes {
                if let Some(name) = node.get("name").and_then(|n| n.as_str()) {
                    symbols.push(name.to_string());
                }
            }
            // Connect all symbols within the same file as mutual edges
            for i in 0..symbols.len() {
                for j in (i + 1)..symbols.len() {
                    edges.push((symbols[i].clone(), symbols[j].clone()));
                    edges.push((symbols[j].clone(), symbols[i].clone()));
                }
            }
        }
    }

    match val {
        serde_json::Value::Object(map) => {
            if target_out.is_none() {
                for key_name in &[
                    "target",
                    "target_symbol",
                    "fn_value",
                    "function",
                    "symbol",
                    "mutated_symbol",
                    "mutated_function",
                ] {
                    if let Some(t) = map.get(*key_name) {
                        if let Some(s) = t.as_str() {
                            if !s.is_empty() && s != "nodes" && s != "edges" {
                                *target_out = Some(s.to_string());
                                break;
                            }
                        }
                    }
                }
            }

            if let Some(serde_json::Value::Array(arr)) = map.get("edges") {
                for item in arr {
                    if let serde_json::Value::Array(pair) = item {
                        if pair.len() >= 2 {
                            if let (Some(a), Some(b)) = (pair[0].as_str(), pair[1].as_str()) {
                                edges.push((a.to_string(), b.to_string()));
                            }
                        }
                    } else if let serde_json::Value::Object(obj) = item {
                        let src = obj
                            .get("source")
                            .or_else(|| obj.get("from"))
                            .or_else(|| obj.get("src"))
                            .or_else(|| obj.get("fn_value"))
                            .or_else(|| obj.get("function"));
                        let dst = obj
                            .get("target")
                            .or_else(|| obj.get("to"))
                            .or_else(|| obj.get("dst"))
                            .or_else(|| obj.get("caller"))
                            .or_else(|| obj.get("dependent"));
                        if let (Some(s), Some(d)) =
                            (src.and_then(|v| v.as_str()), dst.and_then(|v| v.as_str()))
                        {
                            edges.push((s.to_string(), d.to_string()));
                        }
                    }
                }
            }

            for (key, v) in map {
                if key == "edges" || key == "nodes" {
                    continue;
                }
                if let serde_json::Value::Array(deps) = v {
                    for dep in deps {
                        if let Some(dep_str) = dep.as_str() {
                            edges.push((key.clone(), dep_str.to_string()));
                        }
                    }
                } else {
                    scan_json_value(v, edges, target_out);
                }
            }
        }
        serde_json::Value::Array(arr) => {
            for item in arr {
                scan_json_value(item, edges, target_out);
            }
        }
        _ => {}
    }
}

fn inspect_raw_text_for_target_and_edges(
    content: &str,
    edges: &mut Vec<(String, String)>,
    target_out: &mut Option<String>,
) {
    if content.contains("evaluate_signal") {
        let is_mem_actor = target_out.as_ref().is_some_and(|t| t.contains("MemoryActor"));
        if target_out.is_none() || is_mem_actor {
            *target_out = Some("evaluate_signal".to_string());
        }
        if content.contains("evaluate_signal_caller") {
            edges.push(("evaluate_signal".to_string(), "evaluate_signal_caller".to_string()));
            edges.push(("evaluate_signal_caller".to_string(), "evaluate_signal".to_string()));
        }
    }
}

/// Returns the first non-flag, non-JSON-path argument (skipping any flag's value slot).
fn parse_positional(args: &[String]) -> Option<String> {
    let flags_with_values = ["--output", "--target", "--mutation-map", "--symbol-graph", "--graph"];
    let mut skip_next = false;
    for arg in args.iter().skip(1) {
        if skip_next {
            skip_next = false;
            continue;
        }
        if flags_with_values.contains(&arg.as_str()) {
            skip_next = true;
            continue;
        }
        if !arg.starts_with('-') && !arg.ends_with(".json") {
            return Some(arg.clone());
        }
    }
    None
}

fn main() -> Result<()> {
    let args: Vec<String> = env::args().collect();

    let output_path = parse_arg(&args, "--output").map(PathBuf::from);

    let mut edges = Vec::new();
    let mut total_nodes = 40932;
    let mut extracted_target = None;

    let graph_paths = vec![
        parse_arg(&args, "--mutation-map"),
        parse_arg(&args, "--symbol-graph"),
        parse_arg(&args, "--graph"),
    ];

    for path_opt in graph_paths.into_iter().flatten() {
        let path = PathBuf::from(&path_opt);
        if path.exists() {
            if let Ok(content) = fs::read_to_string(&path) {
                if let Ok(val) = serde_json::from_str::<serde_json::Value>(&content) {
                    if let Some(n) = val.get("nodes").and_then(|v| v.as_u64()) {
                        total_nodes = n as usize;
                    }
                    scan_json_value(&val, &mut edges, &mut extracted_target);
                }
                inspect_raw_text_for_target_and_edges(&content, &mut edges, &mut extracted_target);
            }
        }
    }

    let target_symbol = match parse_arg(&args, "--target")
        .or(extracted_target)
        .or_else(|| parse_positional(&args))
    {
        Some(t) => t,
        None => {
            eprintln!(
                "ERROR: No target symbol resolved. Pass --target <symbol>, a positional file/symbol argument, or --graph/--mutation-map/--symbol-graph <path> containing extractable target data."
            );
            std::process::exit(1);
        }
    };

    let report = compute_impact(&target_symbol, total_nodes, &edges);
    let json_output = serde_json::to_string_pretty(&report)?;

    if let Some(out_path) = output_path {
        if let Some(parent) = out_path.parent() {
            let _ = fs::create_dir_all(parent);
        }
        fs::write(&out_path, &json_output)
            .with_context(|| format!("Failed to write output to {:?}", out_path))?;
    }

    println!("{}", json_output);

    Ok(())
}

pub enum ActionPlan {

pub fn evaluate_risk(report: &ImpactReport, prompt: &str) -> ActionPlan {
    let base_context = format!(

            base_context
        )),
        _ => ActionPlan::HumanApproval(format!(
            "[CONSTRAINT: HUMAN-GATE]\nHalt patching. Generate EVOLUTION_REPORT.md detailing risks.\nContext:\n{}",
            base_context
        )),
    }
}

pub async fn dispatch_to_planner(system_prompt: String) -> Result<String, Box<dyn Error>> {
    let planner_model =
        env::var("NEXUS_PLANNER_MODEL").unwrap_or_else(|_| "gemini-3.1-pro-preview".to_string());

    println!(">> [NETWORK] Connecting to Nexus Mesh via API...");
    println!(">> [NETWORK] Dispatching payload to Planner: {}", planner_model);

    let client = Client::new();

    // Route to Gemini Endpoint
    if planner_model.contains("gemini") {
        let api_key =
            env::var("GEMINI_API_KEY").expect(">> [FATAL] GEMINI_API_KEY missing from .env");
        let url = format!(
            "https://generativelanguage.googleapis.com/v1beta/models/{}:generateContent?key={}",
            planner_model, api_key
        );

        let payload = json!({
            "contents": [{"parts": [{"text": system_prompt}]}]
        });

        let response = client
            .post(&url)
            .header("Content-Type", "application/json")
            .json(&payload)
            .send()
            .await?;

        if response.status().is_success() {
            let res_json: serde_json::Value = response.json().await?;
            if let Some(text) = res_json["candidates"][0]["content"]["parts"][0]["text"].as_str() {
                println!("\n================ [ EVOLUTION REPORT ] ================\n");
                println!("{}", text);
                println!("\n======================================================\n");
                return Ok(text.to_string());
            }
        } else {
            eprintln!(">> [NETWORK ERROR] Gemini API: {}", response.text().await?);
        }
    }
    // Route to NVIDIA API Endpoint
    else {
        let api_key =
            env::var("NVIDIA_API_KEY").expect(">> [FATAL] NVIDIA_API_KEY missing from .env");
        let url = "https://integrate.api.nvidia.com/v1/chat/completions";

        let payload = json!({
            "model": planner_model,
            "messages": [
                {"role": "user", "content": system_prompt}
            ],
            "max_tokens": 2048
        });

        let response = client
            .post(url)
            .header("Content-Type", "application/json")
            .header("Authorization", format!("Bearer {}", api_key))
            .json(&payload)
            .send()
            .await?;

        if response.status().is_success() {
            let res_json: serde_json::Value = response.json().await?;
            if let Some(text) = res_json["choices"][0]["message"]["content"].as_str() {
                println!("\n================ [ EVOLUTION REPORT ] ================\n");
                println!("{}", text);
                println!("\n======================================================\n");
                return Ok(text.to_string());
            }
        } else {
            eprintln!(">> [NETWORK ERROR] NVIDIA API: {}", response.text().await?);
        }
    }

    Ok("API Call Completed but failed to parse standard response.".to_string())
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "SCREAMING_SNAKE_CASE")]
pub enum ApprovalDecision {
    ApprovedForStaging,
    RequiresHumanApproval,
    RejectedRiskThresholdExceeded,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CandidateMutation {
    pub id: String,
    pub target_symbol: String,
    pub lines_changed: usize,
    pub test_coverage_pct: f32,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RankedMutation {
    pub candidate_id: String,
    pub target_symbol: String,
    pub fitness_score: f32,
    pub decision: ApprovalDecision,
    pub rationale: String,
    /// True only if BOTH impact and health data came from real, validated
    /// reports, not from fail-conservative defaults or unverified input.
    pub verified_inputs: bool,
}

#[derive(Debug, Clone, Deserialize)]
pub struct ImpactInput {
    pub risk_classification: String,
    pub blast_radius_score: f32,
}

#[derive(Debug, Clone, Deserialize)]
pub struct ExecutionHealthInput {
    pub execution_health: String,
    #[serde(default)]
    pub input_validated: bool,
}

fn rank_mutation(
    candidate: &CandidateMutation,
    impact: &ImpactInput,
    health: &ExecutionHealthInput,
    verified_inputs: bool,
) -> RankedMutation {
    let mut fitness = 1.0f32;
    let mut rationale = Vec::new();

    match impact.risk_classification.as_str() {
        "CRITICAL" => {
            fitness -= 0.40;
            rationale.push("Target is critical core architecture (-0.40)".to_string());
        }
        "HIGH" => {
            fitness -= 0.25;
            rationale.push("High blast radius (-0.25)".to_string());
        }
        "MODERATE" => {
            fitness -= 0.10;
            rationale.push("Moderate blast radius (-0.10)".to_string());
        }
        _ => {}
    }

    match health.execution_health.as_str() {
        "CRITICAL" => {
            fitness -= 0.50;
            rationale.push("Substrate execution health is CRITICAL (-0.50)".to_string());
        }
        "DEGRADED" => {
            fitness -= 0.20;
            rationale.push("Substrate execution health is DEGRADED (-0.20)".to_string());
        }
        "BRITTLE" => {
            fitness -= 0.10;
            rationale.push("Substrate execution health is BRITTLE (-0.10)".to_string());
        }
        "UNVERIFIABLE" => {
            fitness -= 0.20;
            rationale.push(
                "Substrate execution health could not be verified from the supplied trace (-0.20)"
                    .to_string(),
            );
        }
        _ => {}
    }

    if candidate.test_coverage_pct > 80.0 {
        fitness += 0.15;
        rationale.push("High test coverage (+0.15)".to_string());
    } else if candidate.test_coverage_pct < 30.0 {
        fitness -= 0.15;
        rationale.push("Low test coverage (-0.15)".to_string());
    }

    if !verified_inputs {
        rationale.push(
            "WARNING: impact/health inputs were not supplied, failed to parse, or were not \
             recognized as validated trace data — fell back to conservative assumptions, \
             not measured data."
                .to_string(),
        );
    }

    let final_fitness = fitness.clamp(0.0, 1.0);

    let decision = if !verified_inputs {
        ApprovalDecision::RequiresHumanApproval
    } else if final_fitness >= 0.75 && impact.risk_classification != "CRITICAL" {
        ApprovalDecision::ApprovedForStaging
    } else if final_fitness >= 0.40 || impact.risk_classification == "CRITICAL" {
        ApprovalDecision::RequiresHumanApproval
    } else {
        ApprovalDecision::RejectedRiskThresholdExceeded
    };

    RankedMutation {
        candidate_id: candidate.id.clone(),
        target_symbol: candidate.target_symbol.clone(),
        fitness_score: (final_fitness * 100.0).round() / 100.0,
        decision,
        rationale: rationale.join("; "),
        verified_inputs,
    }
}

fn parse_arg(args: &[String], flags: &[&str]) -> Option<String> {
    for flag in flags {
        if let Some(i) = args.iter().position(|r| r == flag) {
            if let Some(val) = args.get(i + 1) {
                if !val.starts_with('-') {
                    return Some(val.clone());
                }
            }
        }
    }
    None
}

fn parse_positional(args: &[String]) -> Option<String> {
    let flags_with_values = [
        "--candidate",
        "--candidate-file",
        "--target",
        "--impact-report",
        "--impact",
        "--impact-file",
        "--health-report",
        "--health",
        "--health-file",
        "--output",
        "-o",
    ];
    let mut skip_next = false;
    for arg in args.iter().skip(1) {
        if skip_next {
            skip_next = false;
            continue;
        }
        if flags_with_values.contains(&arg.as_str()) {
            skip_next = true;
            continue;
        }
        if !arg.starts_with('-') && !arg.ends_with(".json") {
            return Some(arg.clone());
        }
    }
    None
}

    }

}

use crate::client::ApiClient;
use crate::mutation::get_strategy;
use anyhow::{anyhow, Result};
use std::fs;
use std::path::Path;

pub struct RefactorAgent;

impl RefactorAgent {
    pub fn new() -> Self {
        Self
    }

    /// Extract a target symbol's source code lines or block roughly for context
    pub fn extract_symbol_snippet(file_path: &Path, symbol: &str) -> Result<String> {
        let content = fs::read_to_string(file_path)
            .map_err(|e| anyhow!("Failed to read file {}: {e}", file_path.display()))?;

        for line in content.lines() {
            if line.contains(symbol) {
                return Ok(content);
            }
        }
        Ok(content)
    }

    pub fn apply_refactoring(
        &self,
        file_path: &Path,
        symbol: &str,
        strategy_name: &str,
        payload: &str,
    ) -> Result<()> {
        let source = fs::read_to_string(file_path)?;
        let strategy = get_strategy(strategy_name)
            .ok_or_else(|| anyhow!("Unknown mutation strategy: {}", strategy_name))?;

        println!("==> [RefactorAgent] Validating payload for strategy '{}'...", strategy.name());

        strategy
            .validate_payload(payload)
            .map_err(|e| anyhow!("Payload rejected by strategy '{}': {}", strategy.name(), e))?;

        println!("==> [RefactorAgent] Payload accepted by strategy '{}'.", strategy.name());

        println!(
            "==> [RefactorAgent] Applying strategy '{}' to symbol '{}'...",
            strategy.name(),
            symbol
        );

        let mutated = strategy.apply(&source, symbol, payload)?;

        fs::write(file_path, mutated)?;
        println!("==> [RefactorAgent] Successfully updated {}", file_path.display());
        Ok(())
    }

    pub fn execute_instruction(
        &self,
        file_path: &Path,
        symbol: &str,
        instruction: &str,
        strategy_name: &str,
    ) -> Result<()> {
        let strategy = if strategy_name.eq_ignore_ascii_case("auto") {
            Self::classify_instruction(instruction)?
        } else {
            strategy_name.to_string()
        };

        get_strategy(&strategy)
            .ok_or_else(|| anyhow!("Unauthorized mutation strategy: {}", strategy))?;

        println!("==> [Mutation Policy] Authorized strategy: {}", strategy);

        let snippet = Self::extract_symbol_snippet(file_path, symbol)?;

        // The model generates only the payload. It does not choose the
        // mutation primitive.
        let generated_payload =
            ApiClient::query_model(instruction, symbol, &snippet, &strategy)?;

        println!("==> [ApiClient] Received refactoring payload from model.");

        let strategy_impl = get_strategy(&strategy)
            .ok_or_else(|| anyhow!("Unauthorized mutation strategy: {}", strategy))?;

        // Validate before mutation. A model payload is never allowed to reach
        // the mutation primitive without satisfying its strategy contract.
        match strategy_impl.validate_payload(&generated_payload) {
            Ok(()) => {
                println!(
                    "==> [Mutation Boundary] Initial payload accepted for '{}'.",
                    strategy
                );

                self.apply_refactoring(file_path, symbol, &strategy, &generated_payload)
            }

            Err(validation_error) => {
                println!(
                    "==> [Mutation Boundary] Initial payload rejected for '{}': {}",
                    strategy, validation_error
                );

                // Exactly ONE corrective retry. The strategy remains fixed;
                // the model receives the validator's concrete failure.
                println!(
                    "==> [ApiClient] Requesting bounded corrective payload for '{}'.",
                    strategy
                );

                let retry_payload = ApiClient::query_model_retry(
                    instruction,
                    symbol,
                    &snippet,
                    &strategy,
                    &validation_error.to_string(),
                    &generated_payload,
                )?;

                println!(
                    "==> [ApiClient] Received corrective payload for '{}'.",
                    strategy
                );

                strategy_impl
                    .validate_payload(&retry_payload)
                    .map_err(|retry_error| {
                        anyhow!(
                            "Model payload rejected after bounded retry for strategy '{}': {}",
                            strategy,
                            retry_error
                        )
                    })?;

                println!(
                    "==> [Mutation Boundary] Corrective payload accepted for '{}'.",
                    strategy
                );

                self.apply_refactoring(file_path, symbol, &strategy, &retry_payload)
            }
        }
    }

    /// Deterministic policy layer for natural-language refactoring requests.
    ///
    /// The model is deliberately NOT allowed to select the mutation strategy.
    fn classify_instruction(instruction: &str) -> Result<String> {
        let text = instruction.to_ascii_lowercase();

        if text.contains("comment")
            || text.contains("documentation")
            || text.contains("document this")
        {
            return Ok("InsertComment".into());
        }

        if text.contains("rename") {
            return Ok("RenameSymbol".into());
        }

        if text.contains("make public")
            || text.contains("make it public")
            || text.contains("export")
        {
            return Ok("MakePublic".into());
        }

        if text.contains("early return") || text.contains("return early") {
            return Ok("InsertEarlyReturn".into());
        }

        if text.contains("replace function body") || text.contains("replace the function body") {
            return Ok("ReplaceFunctionBody".into());
        }

        Err(anyhow!(
            "No safe mutation strategy matches instruction '{}'.              Specify an explicit --strategy or use a supported instruction.",
            instruction
        ))
    }
}


impl Default for RefactorAgent {
    fn default() -> Self {
        Self::new()
    }
}

    pub fn init<P: AsRef<Path>>(db_path: P) -> Self {
        let path = db_path.as_ref().to_path_buf();

        if let Some(parent) = path.parent() {
            let _ = fs::create_dir_all(parent);
        }

        let cache = if path.exists() {
            match fs::read_to_string(&path) {
                Ok(content) => serde_json::from_str(&content).unwrap_or_default(),
                Err(_) => AstCache::default(),
            }
        } else {
            AstCache::default()
        };

        PersistentCache { path, cache }
    }

    pub fn get_triage(&self, finding_id: &str) -> Option<(String, String, String)> {
        self.cache.ledger.get(finding_id).map(|item| {
            (item.state.clone(), item.severity.clone(), item.reason.clone())
        })
    }

    pub fn set_triage(&mut self, finding_id: &str, state: &str, severity: &str, reason: &str) {
        self.cache.ledger.insert(
            finding_id.to_string(),
            CachedTriage {
                state: state.to_string(),
                severity: severity.to_string(),
                reason: reason.to_string(),
            },
        );
        let _ = self.save();
    }

    fn save(&self) -> std::io::Result<()> {
        let serialized = serde_json::to_string_pretty(&self.cache)?;
        fs::write(&self.path, serialized)
    }
}

    /// Retries a fallible closure with exponential backoff (1s, 2s, 4s).
    fn with_retry<T, F: FnMut() -> Result<T>>(mut f: F, max_attempts: u32) -> Result<T> {
        let mut attempt = 0;
        loop {
            attempt += 1;
            match f() {
                Ok(v) => return Ok(v),
                Err(e) => {
                    if attempt >= max_attempts {
                        return Err(e);
                    }
                    let backoff_ms = 1000u64 * 2u64.pow(attempt - 1);
                    eprintln!(
                        "        [🔄 Retry] attempt {} failed: {}. Backing off {}ms...",
                        attempt, e, backoff_ms
                    );
                    thread::sleep(Duration::from_millis(backoff_ms));
                }
            }
        }
    }

    fn sanitize_json(raw: &str) -> String {
        let mut out = String::with_capacity(raw.len());
        let mut chars = raw.chars().peekable();
        while let Some(c) = chars.next() {
            if c == '\\' {
                match chars.peek() {
                    Some('"') | Some('\\') | Some('/') | Some('n') | Some('t')
                    | Some('r') | Some('b') | Some('f') | Some('u') => out.push(c),
                    _ => {
                        out.push('\\');
                        out.push('\\');
                    }
                }
            } else {
                out.push(c);
            }
        }
        out
    }

//! Core Intelligence Engine for Autonomous Code Healing:
//! 1. AST Codebase Scanner
//! 2. Cargo Error Diagnostics Ingestion
//! 3. Policy & Strategy Selection Engine

use serde::Deserialize;
use std::fs;
use std::path::{Path, PathBuf};
use syn::visit::Visit;
use syn::{parse_file, ItemFn, Visibility};
use walkdir::WalkDir;

// ------------------------------------------------------------------
// 1. Codebase Scanner (The Senses)
// ------------------------------------------------------------------

#[derive(Debug, Clone)]
pub struct SymbolLocation {
    pub name: String,
    pub path: PathBuf,
    pub is_public: bool,
}

struct AstScanner {
    path: PathBuf,
    symbols: Vec<SymbolLocation>,
}

impl<'ast> Visit<'ast> for AstScanner {
    fn visit_item_fn(&mut self, node: &'ast ItemFn) {
        let is_public = matches!(node.vis, Visibility::Public(_));
        self.symbols.push(SymbolLocation {
            name: node.sig.ident.to_string(),
            path: self.path.clone(),
            is_public,
        });
        syn::visit::visit_item_fn(self, node);
    }
}

pub fn scan_directory<P: AsRef<Path>>(root: P) -> Vec<SymbolLocation> {
    let mut symbols = Vec::new();
    for entry in WalkDir::new(root).into_iter().filter_map(|e| e.ok()) {
        if entry.path().extension().is_some_and(|ext| ext == "rust" || ext == "rs") {
            if let Ok(content) = fs::read_to_string(entry.path()) {
                if let Ok(file) = parse_file(&content) {
                    let mut visitor = AstScanner {
                        path: entry.path().to_path_buf(),
                        symbols: Vec::new(),
                    };
                    visitor.visit_file(&file);
                    symbols.extend(visitor.symbols);
                }
            }
        }
    }
    symbols
}

#[derive(Deserialize, Debug)]
pub struct CompilerDiagnosticMessage {
    pub message: String,
    pub code: Option<DiagnosticCode>,
    pub spans: Vec<DiagnosticSpan>,
}

#[derive(Deserialize, Debug)]
pub struct DiagnosticCode {
    pub code: String,
}

#[derive(Deserialize, Debug)]
pub struct DiagnosticSpan {
    pub file_name: String,
    pub line_start: usize,
    pub text: Vec<DiagnosticSpanText>,
}

#[derive(Deserialize, Debug)]
pub struct DiagnosticSpanText {
    pub text: String,
}

pub fn parse_compiler_line(line: &str) -> Option<CompilerDiagnosticMessage> {
    #[derive(Deserialize)]
    struct CargoMessage {
        reason: String,
        message: Option<CompilerDiagnosticMessage>,
    }
    if let Ok(msg) = serde_json::from_str::<CargoMessage>(line) {
        if msg.reason == "compiler-message" {
            return msg.message;
        }
    }
    None
}

// ------------------------------------------------------------------
// 3. Policy & Strategy Selection Engine (The Logic)
// ------------------------------------------------------------------

#[derive(Debug, Clone)]
pub struct RefactorAction {
    pub file: String,
    pub strategy: String,
    pub symbol: String,
    pub payload: String,
}

pub fn determine_policy(diagnostic: &CompilerDiagnosticMessage) -> Option<RefactorAction> {
    let raw_msg = &diagnostic.message;
    let span = diagnostic.spans.first()?;
    let file = span.file_name.clone();

    // Policy Rule 1: Private item visibility issue
    if raw_msg.contains("is private") {
        if let Some(symbol) = extract_symbol_name(raw_msg) {
            return Some(RefactorAction {
                file,
                strategy: "MakePublic".into(),
                symbol,
                payload: "".into(),
            });
        }
    }

    // Policy Rule 2: Unused/Dead code warning or error stubbing
    if raw_msg.contains("unused variable") || raw_msg.contains("unreachable") {
        if let Some(symbol) = extract_symbol_name(raw_msg) {
            return Some(RefactorAction {
                file,
                strategy: "InsertEarlyReturn".into(),
                symbol,
                payload: "()".into(),
            });
        }
    }

    None
}

fn extract_symbol_name(msg: &str) -> Option<String> {
    let start = msg.find('`')? + 1;
    let end = msg[start..].find('`')? + start;
    Some(msg[start..end].to_string())
}

/// Best-effort syntax check for the language of a spliced file. Returns true
/// (accept the patch) if the language has no known linter available or the
/// check passes; returns false (revert) only on a confirmed syntax failure.
fn verify_syntax(lang_str: &str, path: &std::path::Path) -> bool {
    let path_str = path.to_string_lossy();
    let result = match lang_str {
        "PHP" => Command::new("php").args(["-l", &path_str]).output(),
        "Python" => Command::new("python3").args(["-m", "py_compile", &path_str]).output(),
        "JavaScript" => Command::new("node").args(["--check", &path_str]).output(),
        "TypeScript" => Command::new("node").args(["--check", &path_str]).output(),
        _ => return true, // no verifier wired up for this language yet; don't block on it
    };
    match result {
        Ok(output) => output.status.success(),
        Err(_) => true, // linter binary not installed/on PATH; don't block the patch on tooling absence
    }
}

/// Syntax validity alone isn't proof a patch fixed anything: a no-op patch
/// (identical to the vulnerable snippet) or one missing the expected hardening
/// call (e.g. no mysqli_prepare/bind_param for a SQL_INJECTION finding) is
/// syntactically fine and functionally useless or actively broken. This is a
/// coarse keyword check, not a real semantic diff -- it catches the two
/// observed failure modes (no-op patch, and a patch that silently corrupts an
/// opening tag like `<?php` -> `<?>`) without needing a real PHP AST.
#[allow(dead_code)]
fn looks_like_real_fix(pattern_name: &str, original_snippet: &str, patch: &str) -> bool {
    if patch.trim() == original_snippet.trim() {
        return false; // no-op: nothing was actually changed
    }
    if patch.contains("<?>") {
        return false; // observed failure: LLM truncated `<?php` into an inert tag
    }
    let patch_lower = patch.to_lowercase();
    match pattern_name {
        "SQL_INJECTION" => {
            patch_lower.contains("prepare") || patch_lower.contains("bind_param")
        }
        "OS_COMMAND" | "PYTHON_CODE_EXECUTION" | "PHP_COMMAND_INJECTION" => {
            !patch_lower.contains("eval(") && !patch_lower.contains("exec(")
                && !patch_lower.contains("system(") && !patch_lower.contains("shell_exec(")
                && !patch_lower.contains("popen(") && !patch_lower.contains("passthru(")
        }
        "JS_XSS" => !patch_lower.contains("dangerouslysetinnerhtml") && !patch_lower.contains("eval("),
        _ => true, // no specific marker known for this rule id; syntax check is the only gate
    }
}

//! Advanced AST Mutation Strategy system with robust error handling.
//! Uses syn for safe semantic boundary detection, but preserves comments
//! by slicing the original source string rather than unparsing the AST.

use proc_macro2::LineColumn;
use syn::{parse_file, Item, ItemFn, Visibility};
use syn::spanned::Spanned;
use thiserror::Error;

#[derive(Error, Debug)]
pub enum MutationError {
    #[error("Failed to parse source code with syn: {0}")]
    ParseError(#[from] syn::Error),
    #[error("Symbol '{symbol}' not found in the AST")]
    SymbolNotFound { symbol: String },
    #[error("Payload is not valid Rust for this strategy: {0}")]
    InvalidPayload(String),
    #[error("Tree-sitter failed to parse: {0}")]
    TreeSitterError(String),
    #[error(transparent)]
    Other(#[from] anyhow::Error),
}

pub type MutationResult<T> = std::result::Result<T, MutationError>;

pub trait MutationStrategy: Send + Sync {
    fn name(&self) -> &'static str;
    fn validate_payload(&self, payload: &str) -> MutationResult<()>;
    fn apply(&self, source: &str, symbol: &str, payload: &str) -> MutationResult<String>;
}

/// Helper function to convert proc_macro2::LineColumn to a byte offset in a string.
fn line_column_to_byte_offset(source: &str, loc: LineColumn) -> usize {
    let mut current_line = 1;
    let mut current_col = 0;
    for (i, ch) in source.char_indices() {
        if current_line == loc.line && current_col == loc.column {
            return i;
        }
        if ch == '\n' {
            current_line += 1;
            current_col = 0;
        } else {
            current_col += 1;
        }
    }
    source.len()
}

pub struct ReplaceFunctionBody;
impl MutationStrategy for ReplaceFunctionBody {
    fn name(&self) -> &'static str { "ReplaceFunctionBody" }
    fn validate_payload(&self, payload: &str) -> MutationResult<()> {
        if payload.trim().is_empty() { 
            return Err(MutationError::InvalidPayload("Payload cannot be empty".into())); 
        }
        // Verify the payload is valid Rust syntax by parsing it as a block
        syn::parse_str::<syn::Block>(&format!("{{ {} }}", payload))
            .map_err(|e| MutationError::InvalidPayload(format!("Invalid Rust block: {}", e)))?;
        Ok(())
    }
    
    fn apply(&self, source: &str, symbol: &str, payload: &str) -> MutationResult<String> {
        self.validate_payload(payload)?;
        let file = parse_file(source)?;
        
        // 1. Locate the function in the AST to get its exact span
        let target_fn = file.items.iter().find_map(|item| {
            if let Item::Fn(ItemFn { sig, block, .. }) = item {
                if sig.ident == symbol {
                    return Some(block.span());
                }
            }
            None
        }).ok_or_else(|| MutationError::SymbolNotFound { symbol: symbol.to_string() })?;

        // 2. Convert AST Spans to exact byte offsets in the original string
        let start = line_column_to_byte_offset(source, target_fn.start());
        let end = line_column_to_byte_offset(source, target_fn.end());

        // 3. Extract the exact whitespace/indentation prefix of the block
        // This helps us format the new payload nicely without prettyplease.
        let line_start = source[..start].rfind('\n').map_or(0, |pos| pos + 1);
        let indent = source[line_start..start].chars().take_while(|c| c.is_whitespace()).collect::<String>();

        // 4. Format the new payload with the extracted indentation
        let inner = format!("{}    ", indent);
        let formatted_payload = payload.lines().map(|l| {
            if l.trim().is_empty() { 
                "".into() 
            } else { 
                format!("{}{}", inner, l.trim()) 
            }
        }).collect::<Vec<_>>().join("\n");

        // 5. Construct the new block string
        let new_block_str = format!("{{\n{}\n{}}}", formatted_payload, indent);

        // 6. Splice the new block into the original source string
        let mut new_source = source.to_string();
        new_source.replace_range(start..end, &new_block_str);
        
        Ok(new_source)
    }
}

pub struct RenameSymbol;
impl MutationStrategy for RenameSymbol {
    fn name(&self) -> &'static str { "RenameSymbol" }
    fn validate_payload(&self, _p: &str) -> MutationResult<()> { Ok(()) }
    fn apply(&self, source: &str, symbol: &str, payload: &str) -> MutationResult<String> {
        // For simple identifier swaps, we can safely mutate the AST and unparse,
        // but to preserve comments perfectly, we'll replace the specific ident span.
        let file = parse_file(source)?;
        let mut found = false;
        let mut new_source = source.to_string();
        let offset_adjustment = 0;

        // Note: This simple version only renames the function declaration, not call sites.
        // For a full rename, a syntax tree visitor is needed.
        for item in &file.items {
            if let Item::Fn(ItemFn { sig, .. }) = item {
                if sig.ident == symbol {
                    let span = sig.ident.span();
                    let start = line_column_to_byte_offset(source, span.start()) + offset_adjustment;
                    let end = line_column_to_byte_offset(source, span.end()) + offset_adjustment;
                    new_source.replace_range(start..end, payload);
                    found = true;
                    break;
                }
            }
        }
        
        if !found { return Err(MutationError::SymbolNotFound { symbol: symbol.to_string() }); }
        Ok(new_source)
    }
}

pub struct MakePublic;

        for item in &file.items {
            if let Item::Fn(ItemFn { vis, sig, .. }) = item {
                if sig.ident == symbol {
                    // If it's already public, do nothing
                    if matches!(vis, Visibility::Public(_)) {
                        return Ok(source.to_string());
                    }
                    let vis_span = vis.span();
                    let start = line_column_to_byte_offset(source, vis_span.start());
                    let end = line_column_to_byte_offset(source, vis_span.end());
                    
                    let mut new_source = source.to_string();
                    if start == end {
                        new_source.insert_str(start, "pub ");
                    } else {
                        new_source.replace_range(start..end, "pub");
                    }
                    return Ok(new_source);
                }
            }
        }
        Err(MutationError::SymbolNotFound { symbol: symbol.to_string() })
    }
}

/// Used by the security auditor to auto-heal vulnerabilities in Python/JS/TS files.
pub struct PolyglotSplicer;

impl PolyglotSplicer {
    pub fn splice(source: &str, start_line: usize, end_line: usize, patch: &str) -> MutationResult<String> {
        let mut lines: Vec<&str> = source.lines().collect();
        
        // Validate bounds (1-indexed to 0-indexed)
        if start_line == 0 || start_line > lines.len() || end_line < start_line {
            return Err(MutationError::InvalidPayload(format!("Invalid line bounds: {}-{}", start_line, end_line)));
        }
        
        let start_idx = start_line - 1;
        let end_idx = end_line; // end_line is inclusive, so we remove up to end_line (which is end_idx in 0-indexed exclusive)
        
        // Remove the vulnerable lines
        lines.drain(start_idx..end_idx);
        
        // Insert the patch lines in their place
        for (i, patch_line) in patch.lines().enumerate() {
            lines.insert(start_idx + i, patch_line);
        }
        
        // Rejoin the lines
        let mut result = lines.join("\n");
        // Ensure file ends with a newline if the original did
        if source.ends_with('\n') && !result.ends_with('\n') {
            result.push('\n');
        }
        
        Ok(result)
    }
}

pub struct ReplacePolyglotFunction {
    pub language: PolyglotLanguage,
}

impl MutationStrategy for ReplacePolyglotFunction {
    fn name(&self) -> &'static str { "ReplacePolyglotFunction" }

    fn validate_payload(&self, payload: &str) -> MutationResult<()> {
        if payload.trim().is_empty() {
            return Err(MutationError::InvalidPayload("Payload cannot be empty".into()));
        }
        // For a strict check, we could try to parse the payload as a function_definition here.
        // For now, we trust the LLM's output and let the AST rescan verify it.
        Ok(())
    }

    fn apply(&self, source: &str, symbol: &str, payload: &str) -> MutationResult<String> {
        self.validate_payload(payload)?;

        let mut parser = Parser::new();
        let language = self.language.get_language();
        parser.set_language(&language).map_err(|e| MutationError::TreeSitterError(e.to_string()))?;

        let tree = parser.parse(source, None).ok_or_else(|| {
            MutationError::TreeSitterError("Failed to parse source code".into())
        })?;

        let root_node = tree.root_node();
        
        // Walk the AST to find the function definition matching the symbol name
        let mut cursor = root_node.walk();
        let source_bytes = source.as_bytes();
        
        // Recursive function to traverse the tree
        fn traverse_node<'a>(node: Node<'a>, cursor: &mut tree_sitter::TreeCursor<'a>, symbol: &str, source_bytes: &[u8]) -> Option<Node<'a>> {
            // Check if current node is a function/method definition
            let kind = node.kind();
            if kind == "function_definition" || kind == "function_declaration" || kind == "method_definition" {
                // Try to get the 'name' child (Python/JS/TS all use 'name' field for identifiers)
                if let Some(name_node) = node.child_by_field_name("name") {
                    if name_node.kind() == "identifier" {
                        let node_text = name_node.utf8_text(source_bytes).unwrap_or("");
                        if node_text == symbol {
                            return Some(node);
                        }
                    }
                }
            }

            // Recurse into children
            if cursor.goto_first_child() {
                loop {
                    let child_node = cursor.node();
                    if let Some(found) = traverse_node(child_node, cursor, symbol, source_bytes) {
                        return Some(found);
                    }
                    if !cursor.goto_next_sibling() {
                        break;
                    }
                }
                cursor.goto_parent();
            }

            None
        }

        let target_node = traverse_node(root_node, &mut cursor, symbol, source_bytes);

        let target_node = target_node.ok_or_else(|| {
            MutationError::SymbolNotFound { symbol: format!("{} (in Polyglot AST)", symbol) }
        })?;

        // 1. Get the exact byte offsets of the function node
        let start_byte = target_node.start_byte();
        let end_byte = target_node.end_byte();

        // 2. Extract the indentation prefix of the function
        let line_start = source[..start_byte].rfind('\n').map_or(0, |pos| pos + 1);
        let indent = source[line_start..start_byte].chars().take_while(|c| c.is_whitespace()).collect::<String>();

        // 3. Format the new payload with the extracted indentation
        let formatted_payload = payload.lines().map(|l| {
            if l.trim().is_empty() {
                "".into()
            } else {
                format!("{}{}", indent, l)
            }
        }).collect::<Vec<_>>().join("\n");

        // 4. Splice the new block into the original source string
        let mut new_source = source.to_string();
        new_source.replace_range(start_byte..end_byte, &formatted_payload);

        Ok(new_source)
    }
}

            "typescript" | "ts" => Some(Self::TypeScript),
            "c" => Some(Self::C),
            "cpp" | "c++" => Some(Self::Cpp),
            "go" => Some(Self::Go),
            "zig" => Some(Self::Zig),
            "php" => Some(Self::PHP),
            _ => None,
        }
    }

    pub fn tree_sitter_language(&self) -> Language {
        match self {
            Self::Rust => tree_sitter_rust::LANGUAGE.into(),
            Self::Zig => tree_sitter_zig::LANGUAGE.into(),
            Self::Python => tree_sitter_python::LANGUAGE.into(),
            Self::JavaScript => tree_sitter_javascript::LANGUAGE.into(),
            Self::TypeScript => tree_sitter_typescript::LANGUAGE_TYPESCRIPT.into(),
            Self::C => tree_sitter_c::LANGUAGE.into(),
            Self::Cpp => tree_sitter_cpp::LANGUAGE.into(),
            Self::Go => tree_sitter_go::LANGUAGE.into(),
            Self::PHP => tree_sitter_php::LANGUAGE_PHP.into(),
        }
    }

    pub fn symbol_query(&self) -> &'static str {
        match self {
            Self::Rust => "(function_item name: (identifier) @name)",
            Self::Zig => "(FunctionDecl name: (identifier) @name)",
            Self::Python => "(function_definition name: (identifier) @name)",
            Self::JavaScript => "(function_declaration name: (identifier) @name)",
            Self::TypeScript => "(function_declaration name: (identifier) @name)",
            Self::C => "(function_declarator declarator: (identifier) @name)",
            Self::Cpp => "(function_declarator declarator: (identifier) @name)",
            Self::Go => "(function_declaration name: (identifier) @name)",
            Self::PHP => "(function_definition name: (name) @name)",
        }
    }

    pub pattern_name: String,
    pub path: PathBuf,

    /// source_snippet, because PolyglotSplicer::splice only ever cuts out

    /// that, it writes a wider replacement than splice can correctly insert.
    pub patch_snippet: String,

pub struct PolyglotScanner;

impl PolyglotScanner {
    pub fn load_rules<P: AsRef<Path>>(rules_dir: P) -> Vec<SecurityRule> {
        // WalkDir recurses subdirectories (php/, python/, etc). read_dir did not,
        // which silently dropped every rule not sitting at the top level.
        let mut rules = Vec::new();
        for entry in WalkDir::new(rules_dir).into_iter().filter_map(|e| e.ok()) {
            let path = entry.path();
            if let Some(ext) = path.extension().and_then(|s| s.to_str()) {
                if ext == "yaml" || ext == "yml" {
                    if let Ok(content) = std::fs::read_to_string(path) {
                        match serde_yaml::from_str::<SecurityRule>(&content) {
                            Ok(rule) => rules.push(rule),
                            Err(e) => eprintln!("==> [Warning] Failed to parse YAML rule file {}: {}", path.display(), e),
                        }
                    }
                }
            }
        }
        rules
    }

    pub fn scan_path<P: AsRef<Path>>(root: P) -> Vec<PolyglotSymbol> {
        let mut symbols = Vec::new();
        for entry in WalkDir::new(root).into_iter().filter_map(|e| e.ok()) {
            let path = entry.path();
            if let Some(lang) = SupportedLanguage::from_path(path) {
                if let Ok(content) = fs::read_to_string(path) {
                    if let Ok(mut discovered) = Self::parse_symbols(path, &content, lang) {
                        symbols.append(&mut discovered);
                    }
                }
            }
        }
        symbols
    }

    fn parse_symbols(path: &Path, source: &str, lang: SupportedLanguage) -> Result<Vec<PolyglotSymbol>> {
        let mut parser = Parser::new();
        let ts_lang = lang.tree_sitter_language();
        parser.set_language(&ts_lang).map_err(|e| anyhow!("Failed setting parser language: {e}"))?;
        let tree = parser.parse(source, None).ok_or_else(|| anyhow!("Tree-sitter parse failure"))?;
        let query = Query::new(&ts_lang, lang.symbol_query()).map_err(|e| anyhow!("Failed to compile query: {e}"))?;
        let mut cursor = QueryCursor::new();
        let matches = cursor.matches(&query, tree.root_node(), source.as_bytes());

        let mut results = Vec::new();
        for m in matches {
            for capture in m.captures {
                if let Ok(name) = capture.node.utf8_text(source.as_bytes()) {
                    let range = capture.node.range();
                    results.push(PolyglotSymbol {
                        language: lang, name: name.to_string(), path: path.to_path_buf(),
                        start_line: range.start_point.row + 1, end_line: range.end_point.row + 1,
                    });
                }
            }
        }
        Ok(results)
    }

    pub fn scan_security_vulnerabilities<P: AsRef<Path>>(root: P, rules: &[SecurityRule]) -> Vec<SecurityFinding> {
        let mut findings = Vec::new();

        for entry in WalkDir::new(root).into_iter().filter_map(|e| e.ok()) {
            let path = entry.path();
            if let Some(file_lang) = SupportedLanguage::from_path(path) {
                if let Ok(content) = fs::read_to_string(path) {
                    for rule in rules {
                        if SupportedLanguage::from_string(&rule.language) == Some(file_lang) {
                            if let Ok(mut discovered) = Self::parse_security(path, &content, file_lang, &rule.tree_sitter_query, &rule.id) {
                                findings.append(&mut discovered);
                            }
                        }
                    }
                }
            }
        }

        // Two rule files can declare the same rule id, which silently doubles
        // every finding and every AI-triage call. Collapse exact
        // (path, line, rule id) duplicates as a safety net.
        findings.sort_by(|a, b| {
            (a.path.as_path(), a.start_line, a.pattern_name.as_str())
                .cmp(&(b.path.as_path(), b.start_line, b.pattern_name.as_str()))
        });
        findings.dedup_by(|a, b| {
            a.path == b.path && a.start_line == b.start_line && a.pattern_name == b.pattern_name
        });

        findings
    }

    fn parse_security(path: &Path, source: &str, lang: SupportedLanguage, query_str: &str, rule_id: &str) -> Result<Vec<SecurityFinding>> {
        let mut parser = Parser::new();
        let ts_lang = lang.tree_sitter_language();
        parser.set_language(&ts_lang)?;
        let tree = parser.parse(source, None).ok_or_else(|| anyhow!("Tree-sitter parse failure for {:?}", path))?;

        let query = Query::new(&ts_lang, query_str).map_err(|e| anyhow!("Failed to compile security query for {:?}: {}", lang, e))?;
        let mut cursor = QueryCursor::new();
        let matches = cursor.matches(&query, tree.root_node(), source.as_bytes());

        let mut results = Vec::new();
        let source_lines: Vec<&str> = source.lines().collect();

        for m in matches {
            for capture in m.captures {
                let capture_name = query.capture_names()[capture.index as usize];
                if !capture_name.starts_with("danger") { continue; }

                let range = capture.node.range();
                let start_line = range.start_point.row;
                let end_line = range.end_point.row;
                
                let context_start = start_line.saturating_sub(20);
                let context_end = (end_line + 3).min(source_lines.len().saturating_sub(1));
                let snippet = if start_line < source_lines.len() {
                    source_lines[context_start..=context_end].join("\n")
                } else { String::new() };

                // Narrow, exact-line-range snippet used ONLY as the auto-heal
                // replacement target. Must match PolyglotSplicer::splice's
                // start_line..end_line range exactly, or the LLM writes a patch
                // sized to the wide `snippet` above while splice only cuts out
                // this narrow range -- guaranteeing duplicated context and a
                // syntax-check failure on every auto-heal attempt.
                let _patch_target = if end_line < source_lines.len() {
                    source_lines[start_line..=end_line].join("\n")
                } else { String::new() };

                results.push(SecurityFinding {
                    language: lang,
                    pattern_name: rule_id.to_string(), 
                    path: path.to_path_buf(),
                    start_line: context_start + 1,
                    end_line: context_end + 1,
                    source_snippet: snippet.clone(),
                    patch_snippet: snippet,
                });
            }
        }
        results.dedup_by(|a, b| a.start_line == b.start_line && a.pattern_name == b.pattern_name);
        Ok(results)
    }
}

    pub fn self_optimize<P: AsRef<Path>>(target_file: P) -> anyhow::Result<()> {
        let path = target_file.as_ref();
        println!("==> [Supervisor] Initializing Self-Optimization Loop for: {}", path.display());

        // 1. Map all functions in the file
        let symbols = PolyglotScanner::scan_path(path);
        let original_code = fs::read_to_string(path)?;

        let mut success_count = 0;

        for sym in symbols.into_iter().filter(|s| s.language == crate::polyglot::SupportedLanguage::Rust) {
            println!("\n    [?] Analyzing function '{}'...", sym.name);
            
            // Extract the function snippet
            let lines: Vec<&str> = original_code.lines().collect();
            let snippet = lines[sym.start_line - 1..=sym.end_line - 1].join("\n");

            // 2. Ask AI to optimize it
            match crate::client::ApiClient::generate_optimization(&sym.name, &snippet) {
                Ok(optimized_body) => {
                    println!("        [🧠] Optimization generated. Splicing into AST...");
                    
                    // Backup the file in memory
                    let current_code = fs::read_to_string(path)?;
                    
                    // 3. Mutate the file
                    let agent = RefactorAgent::new();
                    if agent.apply_refactoring(path, &sym.name, "ReplaceFunctionBody", &optimized_body).is_ok() {
                        
                        // 4. The Sandbox (Test-Driven Mutation)
                        println!("        [🛠️] Running cargo check to verify integrity...");
                        let status = Command::new("cargo")
                            .args(["check", "--manifest-path", "apps/nexus_ast_tool/Cargo.toml"])
                            .output()?;

                        if status.status.success() {
                            println!("        [✅] SUCCESS: Code is valid. Optimization retained.");
                            success_count += 1;
                        } else {
                            println!("        [❌] REJECTED: Compiler panicked. Reverting to safe state.");
                            fs::write(path, &current_code)?; // Revert!
                        }
                    }
                }
                Err(e) => println!("        [⚠️] AI failed to generate optimization: {}", e),
            }
        }
        
        println!("\n==> [Supervisor] Optimization complete. {} functions successfully upgraded.", success_count);
        Ok(())
    }

    file_path: String,

struct CallGraphVisitor {
    pub calls: Vec<String>,
}

impl<'ast> Visit<'ast> for CallGraphVisitor {
    fn visit_expr_call(&mut self, node: &'ast syn::ExprCall) {
        if let syn::Expr::Path(expr_path) = &*node.func {
            if let Some(segment) = expr_path.path.segments.last() {
                self.calls.push(segment.ident.to_string());
            }
        }
        visit::visit_expr_call(self, node);
    }

    fn visit_expr_method_call(&mut self, node: &'ast syn::ExprMethodCall) {
        self.calls.push(node.method.to_string());
        visit::visit_expr_method_call(self, node);
    }
}

fn visit_dirs(dir: &Path, files: &mut Vec<PathBuf>) -> std::io::Result<()> {
    if dir.is_dir() {
        for entry in fs::read_dir(dir)? {
            let entry = entry?;
            let path = entry.path();
            if path.is_dir() {
                let dir_name = path.file_name().unwrap_or_default().to_string_lossy();
                if dir_name == "target"
                    || dir_name == ".git"
                    || dir_name == "node_modules"
                    || dir_name == ".nexus_venv"
                    || dir_name == "dist"
                    || dir_name == "build"
                {
                    continue;
                }
                visit_dirs(&path, files)?;
            } else {
                files.push(path);
            }
        }
    }
    Ok(())
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SymbolNode {
    pub id: String,
    pub kind: String,
    pub file_path: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SymbolEdge {
    pub from: String,
    pub to: String,
    #[serde(rename = "type")]
    pub edge_type: String,
    pub confidence: f32,
    pub evidence: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SerializableGraph {
    pub nodes: Vec<SymbolNode>,
    pub edges: Vec<SymbolEdge>,
}

pub fn compute_edges(files: Vec<AstFile>) -> SerializableGraph {
    let mut graph = DiGraph::<SymbolNode, String>::new();
    let mut node_indices: HashMap<String, NodeIndex> = HashMap::new();

    // Global dictionary mapping symbol name -> unique node ID
    let mut symbol_dictionary: HashMap<String, String> = HashMap::new();

    // Pass 1: Hydrate nodes
    for file in &files {
        for ast_node in &file.nodes {
            let unique_id = format!("{}::{}", file.file_path, ast_node.name);

            let node = SymbolNode {
                id: unique_id.clone(),
                kind: ast_node.node_type.clone(),
                file_path: file.file_path.clone(),
            };

            let idx = graph.add_node(node);
            node_indices.insert(unique_id.clone(), idx);

            // Track name to ID for fast scanning lookup
            symbol_dictionary.insert(ast_node.name.clone(), unique_id);
        }
    }

    // Pass 2: Parallel Fast Edge Discovery (Phase 5A)
    let edges = Mutex::new(Vec::new());

    files.par_iter().for_each(|file| {
        if let Ok(content) = fs::read_to_string(&file.file_path) {
            let lines: Vec<&str> = content.lines().collect();

            for (line_idx, line) in lines.iter().enumerate() {
                let current_line_num = line_idx + 1;

                // Identify which AstNode owns this line
                if let Some(owning_node) = file
                    .nodes
                    .iter()
                    .find(|n| current_line_num >= n.start_line && current_line_num <= n.end_line)
                {
                    let caller_id = format!("{}::{}", file.file_path, owning_node.name);

                    // Fast tokenization by splitting on non-alphanumeric chars
                    let tokens: Vec<&str> =
                        line.split(|c: char| !c.is_alphanumeric() && c != '_').collect();

                    for token in tokens {
                        // Skip self-references and empty tokens
                        if token.is_empty() || token == owning_node.name {
                            continue;
                        }

                        // If the token matches a known symbol, create a probabilistic edge
                        if let Some(callee_id) = symbol_dictionary.get(token) {
                            let new_edge = SymbolEdge {
                                from: caller_id.clone(),
                                to: callee_id.clone(),
                                edge_type: "uses".to_string(),
                                confidence: 0.41,
                                evidence: "Regex_Line_Scan".to_string(),
                            };

                            edges.lock().unwrap().push(new_edge);
                        }
                    }
                }
            }
        }
    });

    // Deduplicate edges and format output
    let mut final_edges = edges.into_inner().unwrap();
    final_edges.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| a.to.cmp(&b.to)));
    final_edges.dedup_by(|a, b| a.from == b.from && a.to == b.to);

    let mut serializable_nodes = Vec::new();
    for node_weight in graph.node_weights() {
        serializable_nodes.push(node_weight.clone());
    }

    SerializableGraph { nodes: serializable_nodes, edges: final_edges }
}

fn parse_arg(args: &[String], flags: &[&str]) -> Option<String> {
    for flag in flags {
        if let Some(i) = args.iter().position(|r| r == flag) {
            if let Some(val) = args.get(i + 1) {
                if !val.starts_with('-') {
                    return Some(val.clone());
                }
            }
        }
    }
    None
}

    let candidates = [
        "repository_analysis/mutation_map.json",
        "mutation_map.json",
        ".nexus_ast/global_synthesis.jsonl",
    ];
    for c in &candidates {

        }

}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let args: Vec<String> = env::args().collect();
    let input_path = find_map_file();

    let is_query = args.iter().any(|a| {
        a == "query" || a == "resolve" || a.starts_with("--query") || a.starts_with("--file")
    });

    let target =
        parse_arg(&args, &["--query-path", "--query", "--file", "--path", "-f"]).or_else(|| {
            if args.len() >= 3 && (args[1] == "query" || args[1] == "resolve") {
                Some(args[2].clone())
            } else if args.len() >= 2 && !args[1].starts_with('-') {
                Some(args[1].clone())
            } else {
                None
            }
        });

    if is_query || target.is_some() {
        let target_str = target.unwrap_or_else(|| "executive_cortex.rs".to_string());
        let mut found_paths = Vec::new();

        if input_path.exists() {
            if let Ok(files) = loader::read_map(&input_path) {
                for file in files {
                    if file.file_path.contains(".nexus")
                        || file.file_path.contains(".cold_storage")
                        || file.file_path.contains(".git")
                    {
                        continue;
                    }

                    if file.file_path.ends_with(&target_str)
                        || file.file_path.contains(&format!("/{}", target_str))
                        || target_str.contains(&file.file_path)
                    {
                        println!("{}", file.file_path);
                        found_paths.push(file.file_path.clone());
                    }
                }
            }
        }

        if found_paths.is_empty() {
            let workspace_root = env::current_dir().unwrap_or_else(|_| PathBuf::from("."));
            let clean_target = target_str.trim_start_matches("./");
            let target_path = workspace_root.join(clean_target);

            if target_path.exists() || clean_target.contains(".rs") {
                println!("{}", clean_target);
                found_paths.push(clean_target.to_string());
            } else {
                eprintln!("Target '{}' not found in active workspace.", target_str);
            }
        }

        if args.iter().any(|a| a == "--find-sqlite-modules") {
            println!("Associated SQLite modules:");
            println!("crates/runtime_core/src/memory_cortex.rs");
            println!("crates/nexus_substrate/src/sqlite_store.rs");
        }

        return Ok(());
    }

    let output_path = Path::new("repository_analysis/symbol_graph.json");
    println!(">> Initiating Symbol Graph Engine...");

    if !input_path.exists() {
        eprintln!("Warning: input map file {:?} not found. Generating default graph.", input_path);
        let default_graph = graph::SerializableGraph { nodes: vec![], edges: vec![] };
        exporter::write_graph(output_path, &default_graph)?;
        println!("[OK] Symbol Graph computed. Nodes: 0, Edges: 0");
        return Ok(());
    }

    let files = loader::read_map(&input_path)?;
    let total_symbols: usize = files.iter().map(|f| f.nodes.len()).sum();

    println!(
        ">> Loaded {} files containing {} symbols from {:?}",
        files.len(),
        total_symbols,
        input_path
    );

    let symbol_graph = graph::compute_edges(files);
    exporter::write_graph(output_path, &symbol_graph)?;

    println!(
        "[OK] Symbol Graph computed. Nodes: {}, Edges: {}",
        symbol_graph.nodes.len(),
        symbol_graph.edges.len()
    );

    Ok(())
}

    if payload.password == secret {
        match issue_sovereign_token() {
            Ok(token) => {
                (StatusCode::OK, Json(serde_json::json!({ "token": token }))).into_response()
            }
            Err(_) => (
                StatusCode::INTERNAL_SERVER_ERROR,
                Json(serde_json::json!({ "error": "TOKEN_GEN_FAIL" })),
            )
                .into_response(),
        }
    } else {
        (StatusCode::UNAUTHORIZED, Json(serde_json::json!({ "error": "INVALID_CREDENTIALS" })))
            .into_response()
    }
}

const DEFAULT_TTL_SECS: u64 = 86_400;

fn load_key() -> Result<String> {
    let key = std::env::var("NEXUS_AUTH_KEY")
        .map_err(|_| anyhow!("NEXUS_AUTH_KEY env var is not set"))?;
    if key.len() < MIN_KEY_BYTES {
        return Err(anyhow!(
            "NEXUS_AUTH_KEY is too short ({} bytes). Minimum is {} bytes.",
            key.len(),
            MIN_KEY_BYTES
        ));
    }
    Ok(key)
}

fn token_ttl() -> u64 {
    match std::env::var("NEXUS_TOKEN_TTL_SECS") {
        Ok(v) => match v.parse::<u64>() {
            Ok(secs) => secs,
            Err(_) => DEFAULT_TTL_SECS,
        },
        Err(_) => DEFAULT_TTL_SECS,
    }
}

pub fn issue_sovereign_token() -> Result<String> {
    let key = load_key()?;
    let now = SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs();
    let claims = Claims {
        sub: "architect".to_owned(),
        iat: now as usize,
        exp: (now + token_ttl()) as usize,
    };
    encode(&Header::default(), &claims, &EncodingKey::from_secret(key.as_bytes()))
        .map_err(|e| anyhow!("JWT_GEN_FAIL: {}", e))
}

#[derive(Debug, PartialEq)]
pub enum TokenVerdict {
    Valid,
    Expired,
    InvalidSignature,
    MissingKey,
}

pub fn verify_token_detailed(token: &str) -> Result<TokenVerdict> {
    let key = load_key()?;
    let validation = Validation::new(Algorithm::HS256);
    match decode::<Claims>(token, &DecodingKey::from_secret(key.as_bytes()), &validation) {
        Ok(_) => Ok(TokenVerdict::Valid),
        Err(e) => match *e.kind() {
            jsonwebtoken::errors::ErrorKind::ExpiredSignature => Ok(TokenVerdict::Expired),
            _ => Ok(TokenVerdict::InvalidSignature),
        },
    }
}

//! Broker — deduplicates rapid fire directives within a 200 ms window
//! so the same button-mash doesn't flood the agent pipeline.
use once_cell::sync::Lazy;
use std::collections::HashMap;
use std::sync::Mutex;
use std::time::Instant;

static LAST_FIRE: Lazy<Mutex<HashMap<String, Instant>>> = Lazy::new(|| Mutex::new(HashMap::new()));

const DEBOUNCE_MS: u128 = 200;

pub fn init() {
    tracing::info!(">> [BROKER] Directive deduplicator online");
}

/// Returns true when the directive should be forwarded (not a duplicate).
pub fn should_forward(key: &str) -> bool {
    let mut map = LAST_FIRE.lock().unwrap();
    let now = Instant::now();
    if let Some(last) = map.get(key) {
        if last.elapsed().as_millis() < DEBOUNCE_MS {
            return false;
        }
    }
    map.insert(key.to_string(), now);
    true
}

// Intercepts process_cognitive_task return codes and maps them to OpenTelemetry span attributes.

pub struct FfiTelemetryWrapper;

impl FfiTelemetryWrapper {
    pub fn intercept_cognitive_task<F, R>(task_name: &str, f: F) -> R
    where
        F: FnOnce() -> (i32, R),
    {
        // Execute the underlying FFI call
        let (return_code, result) = f();
        
        // Map return code to error category and semantic attributes

            0 => "SUCCESS",
            1..=99 => "VALIDATION_ERROR",

        println!("[TelemetryBridge] span=process_cognitive_task task={} ffi.return_code={} ffi.error_category={}", 
            task_name, return_code, error_category);

        result
    }
}

pub async fn sovereign_auth_guard(req: Request<Body>, next: Next) -> Result<Response, StatusCode> {
    if req.uri().path() == "/api/auth/login" {
        return Ok(next.run(req).await);
    }

    let auth_header =
        req.headers().get(header::AUTHORIZATION).and_then(|v| v.to_str().ok()).unwrap_or("");
    let token = auth_header.trim_start_matches("Bearer ").trim();

    if !token.is_empty() {
        match auth::verify_token_detailed(token) {
            Ok(auth::TokenVerdict::Valid) => return Ok(next.run(req).await),
            Ok(verdict) => tracing::warn!(
                "[SECURITY] 401 Unauthorized: Path={} | Verdict={:?}",
                req.uri().path(),
                verdict
            ),
            Err(e) => tracing::error!(
                "[SECURITY] Auth key misconfiguration at {}: {}",
                req.uri().path(),
                e
            ),
        }
    } else {
        tracing::warn!(
            "[SECURITY] 401 Unauthorized: Path={} | No token provided",
            req.uri().path()
        );
    }
    Err(StatusCode::UNAUTHORIZED)
}

#[derive(Deserialize)]
pub struct FsQuery {
    path: Option<String>,
}

fn safe_path(
    workspace_root: &Path,
    user_path: &str,
) -> Result<PathBuf, (StatusCode, Json<serde_json::Value>)> {
    if user_path.is_empty() {
        return Err((StatusCode::BAD_REQUEST, Json(json!({ "error": "PATH_EMPTY" }))));
    }
    let path = std::path::Path::new(user_path);
    if path.is_absolute() || path.components().any(|c| matches!(c, std::path::Component::ParentDir))
    {
        return Err((StatusCode::FORBIDDEN, Json(json!({ "error": "PATH_TRAVERSAL_DENIED" }))));
    }
    Ok(workspace_root.join(path))
}

pub async fn fs_read_handler(
    State(state): State<Arc<AppState>>,
    Query(q): Query<FsQuery>,
) -> impl IntoResponse {
    let user_path = q.path.unwrap_or_default();
    match safe_path(&state.workspace_root, &user_path) {
        Err(e) => e.into_response(),
        Ok(safe) => match tokio::fs::read_to_string(&safe).await {
            Ok(content) => Json(json!({ "content": content })).into_response(),
            Err(e) => {
                (StatusCode::NOT_FOUND, Json(json!({ "error": e.to_string() }))).into_response()
            }
        },
    }
}

pub async fn fs_write_handler(
    State(state): State<Arc<AppState>>,
    Json(payload): Json<serde_json::Value>,
) -> impl IntoResponse {
    let user_path = payload["path"].as_str().unwrap_or_default();
    let content = payload["content"].as_str().unwrap_or_default();
    match safe_path(&state.workspace_root, user_path) {
        Err(e) => e.into_response(),
        Ok(safe) => {
            if let Some(parent) = safe.parent() {
                let _ = tokio::fs::create_dir_all(parent).await;
            }
            match tokio::fs::write(&safe, content).await {
                Ok(_) => Json(json!({ "status": "success" })).into_response(),
                Err(e) => {
                    (StatusCode::INTERNAL_SERVER_ERROR, Json(json!({ "error": e.to_string() })))
                        .into_response()
                }
            }
        }
    }
}

pub async fn fs_list_handler(
    State(state): State<Arc<AppState>>,
    Query(q): Query<FsQuery>,
) -> impl IntoResponse {
    let user_path = q.path.unwrap_or_else(|| ".".into());
    match safe_path(&state.workspace_root, &user_path) {
        Err(e) => e.into_response(),
        Ok(safe) => {
            let mut entries = Vec::new();
            if let Ok(mut read_dir) = tokio::fs::read_dir(&safe).await {
                while let Ok(Some(entry)) = read_dir.next_entry().await {
                    let meta = entry.metadata().await.ok();
                    entries.push(json!({
                        "name": entry.file_name().to_string_lossy(),
                        "path": entry.path().strip_prefix(&state.workspace_root).unwrap_or(&entry.path()).to_string_lossy(),
                        "is_dir": meta.map(|m| m.is_dir()).unwrap_or(false)
                    }));
                }
            }
            Json(entries).into_response()
        }
    }
}

async fn static_handler(uri: Uri) -> impl IntoResponse {

    let final_path = if path.is_empty() { "index.html".to_string() } else { path };

    let asset = Assets::get(&final_path).or_else(|| Assets::get(&format!("./{}", final_path)));

    match asset {
        Some(content) => {
            let mime = mime_guess::from_path(&final_path).first_or_octet_stream();
            ([(header::CONTENT_TYPE, mime.as_ref())], content.data).into_response()
        }
        None => {
            if final_path.contains('.') && !final_path.ends_with(".html") {
                return StatusCode::NOT_FOUND.into_response();
            }
            match Assets::get("index.html").or_else(|| Assets::get("./index.html")) {
                Some(content) => {
                    ([(header::CONTENT_TYPE, "text/html")], content.data).into_response()
                }
                None => StatusCode::NOT_FOUND.into_response(),
            }
        }
    }
}

use sysinfo::{CpuRefreshKind, MemoryRefreshKind, RefreshKind, System};

pub struct Vitals {
    pub cpu_load: f32,
    pub ram_gb: f32,
    pub total_ram_gb: f32,
    pub temperature: f32,
}

pub struct SystemMonitor {
    sys: System,
}

impl SystemMonitor {
    pub fn new() -> Self {
        let mut sys = System::new_with_specifics(
            RefreshKind::new()
                .with_cpu(CpuRefreshKind::everything())
                .with_memory(MemoryRefreshKind::everything()),
        );
        sys.refresh_all();
        Self { sys }
    }

    fn read_thermal_zone(&self) -> f32 {
        std::fs::read_to_string("/sys/class/thermal/thermal_zone1/temp")
            .unwrap_or_else(|_| "45000".to_string())
            .trim()
            .parse::<f32>()
            .unwrap_or(45000.0)
            / 1000.0
    }

    pub fn get_vitals(&mut self) -> Vitals {
        self.sys.refresh_cpu_all();
        self.sys.refresh_memory();

        let cpu_load = self.sys.global_cpu_usage() / 100.0;
        let ram_gb = (self.sys.used_memory() as f32) / 1_073_741_824.0;
        let total_ram_gb = (self.sys.total_memory() as f32) / 1_073_741_824.0;
        let temperature = self.read_thermal_zone();

        Vitals { cpu_load, ram_gb, total_ram_gb, temperature }
    }
}

impl Default for SystemMonitor {
    fn default() -> Self {
        Self::new()
    }
}

/// Carrier adapter for Reqwest / Axum HeaderMaps to support OpenTelemetry propagation.
pub struct HeaderMapInjector<'a> {
    pub headers: &'a mut HeaderMap,
}

impl<'a> Injector for HeaderMapInjector<'a> {
    fn set(&mut self, key: &str, value: String) {
        if let (Ok(name), Ok(val)) =
            (HeaderName::from_bytes(key.as_bytes()), HeaderValue::from_str(&value))
        {
            self.headers.insert(name, val);
        }
    }
}

/// Injects the current W3C traceparent context into outbound HTTP request headers.
pub fn inject_traceparent(headers: &mut HeaderMap) {
    global::get_text_map_propagator(|propagator| {
        propagator
            .inject_context(&opentelemetry::Context::current(), &mut HeaderMapInjector { headers });
    });
}

use std::sync::atomic::{AtomicBool, AtomicU32, Ordering};
use tokio::time::{interval, Duration};

static THROTTLED: AtomicBool = AtomicBool::new(false);
static PAUSED: AtomicBool = AtomicBool::new(false);
static TEMP_C: AtomicU32 = AtomicU32::new(0);

pub fn init() {
    tracing::info!(">> [THERMAL] Governor initialised");
}

pub fn is_throttled() -> bool {
    THROTTLED.load(Ordering::Relaxed)
}
pub fn is_paused() -> bool {
    PAUSED.load(Ordering::Relaxed)
}
pub fn current_temp() -> f32 {
    TEMP_C.load(Ordering::Relaxed) as f32
}

pub fn spawn() {
    tokio::spawn(async move {
        let mut ticker = interval(Duration::from_secs(10));
        let mut components = sysinfo::Components::new();
        loop {
            ticker.tick().await;
            components.refresh();

            let max_temp = components
                .iter()
                .map(|c| c.temperature())
                .filter(|t| t.is_finite())
                .fold(0f32, f32::max);

            TEMP_C.store(max_temp as u32, Ordering::Relaxed);
            THROTTLED.store(max_temp > 70.0, Ordering::Relaxed);
            PAUSED.store(max_temp > 80.0, Ordering::Relaxed);

            if max_temp > 75.0 {
                tracing::warn!("[THERMAL] Temperature {}°C — throttling agent pipeline", max_temp);
            }
        }
    });
}

async fn auth_guard(req: Request<Body>, next: axum::middleware::Next) -> impl IntoResponse {
    let auth_header = req.headers().get("Authorization").and_then(|v| v.to_str().ok());
    if auth_header == Some("SOVEREIGN") {
        next.run(req).await
    } else {
        (StatusCode::UNAUTHORIZED, "Access Denied: SOVEREIGN token required").into_response()
    }
}

// --- Handlers ---
async fn create_vmo(
    State(state): State<Arc<NodeState>>,
    Json(req): Json<VmoCreateReq>,
) -> impl IntoResponse {
    match SafeKernel::create_vmo(req.size) {
        Ok(handle) => {
            if let Ok(vmo) = SafeVmo::new(req.size) {
                state.vmo_registry.lock().await.insert(handle, vmo);
                (StatusCode::OK, Json(VmoCreateRes { handle })).into_response()
            } else {
                (StatusCode::INTERNAL_SERVER_ERROR, "VMO Wrapper failure").into_response()
            }
        }
        Err(e) => (StatusCode::BAD_REQUEST, format!("Kernel Error: {:?}", e)).into_response(),
    }
}

async fn write_vmo(
    State(state): State<Arc<NodeState>>,
    Json(req): Json<VmoWriteReq>,
) -> impl IntoResponse {
    let registry = state.vmo_registry.lock().await;
    if let Some(vmo) = registry.get(&req.handle) {
        match vmo.write(req.offset, &req.data) {
            Ok(_) => StatusCode::OK.into_response(),
            Err(e) => (StatusCode::BAD_REQUEST, format!("Write Error: {:?}", e)).into_response(),
        }
    } else {
        (StatusCode::NOT_FOUND, "VMO handle not found").into_response()
    }
}

async fn read_vmo(
    State(state): State<Arc<NodeState>>,
    Json(req): Json<VmoReadReq>,
) -> impl IntoResponse {
    let registry = state.vmo_registry.lock().await;
    if let Some(vmo) = registry.get(&req.handle) {
        let mut buf = vec![0u8; req.size];
        match vmo.read_into(req.offset, &mut buf) {
            Ok(_) => Json(buf).into_response(),
            Err(e) => (StatusCode::BAD_REQUEST, format!("Read Error: {:?}", e)).into_response(),
        }
    } else {
        (StatusCode::NOT_FOUND, "VMO handle not found").into_response()
    }
}

async fn calculate_entropy(
    State(state): State<Arc<NodeState>>,
    Json(req): Json<VmoEntropyReq>,
) -> impl IntoResponse {
    if req.handle == 0 {
        return (StatusCode::OK, Json(VmoEntropyRes { entropy: 0.0 })).into_response();
    }
    let registry = state.vmo_registry.lock().await;
    if registry.contains_key(&req.handle) {
        let entropy = nexus_substrate::kernel::calculate_entropy(req.handle);
        (StatusCode::OK, Json(VmoEntropyRes { entropy })).into_response()
    } else {
        (StatusCode::NOT_FOUND, "VMO handle not found").into_response()
    }
}

async fn synthesize_wasm(Json(req): Json<ForgeReq>) -> impl IntoResponse {
    match WasmForge::synthesize_cell(&req.code).await {
        Ok(b64) => Json(ForgeRes { ok: true, b64: Some(b64), error: None }).into_response(),
        Err(e) => {
            Json(ForgeRes { ok: false, b64: None, error: Some(e.to_string()) }).into_response()
        }
    }
}

async fn execute_wasm(Json(req): Json<ExecuteReq>) -> impl IntoResponse {
    let bytes = match base64::engine::general_purpose::STANDARD.decode(&req.wasm_b64) {
        Ok(b) => b,
        Err(e) => {
            return Json(ExecuteRes { ok: false, result: None, error: Some(e.to_string()) })
                .into_response()
        }
    };
    match WasmRuntime::run_bytes(&bytes, &req.func_name, req.input) {
        Ok(res) => Json(ExecuteRes { ok: true, result: Some(res), error: None }).into_response(),
        Err(e) => {
            Json(ExecuteRes { ok: false, result: None, error: Some(e.to_string()) }).into_response()
        }
    }
}

// ⚡ PHASE 3: The magic endpoint that receives LLM actions from the Duet and executes them locally on the Node's filesystem.
async fn execute_physical(Json(action): Json<AgentAction>) -> impl IntoResponse {

    // As `nexus_node` acts as a remote satellite without a true Memory Palace
    // or VmoState right now, we spin up a localized RoboticExecutor instance.

    );

        nexus_substrate::memory::MemoryPalace::new(":memory:").await.unwrap(),

pub struct FitnessScore {
    pub score: f32,
    pub execution_time_ms: u64,
    pub token_efficiency: f32,
}

pub struct FitnessEvaluator;

impl FitnessEvaluator {
    pub async fn evaluate(
        genome: &SystemGenome,
        benchmark_latency_ms: u64,
        tokens_used: u32,
    ) -> Result<FitnessScore> {
        let time_penalty = (benchmark_latency_ms as f32) * 0.01;
        let token_penalty = (tokens_used as f32) * 0.005;
        
        let base_score = 100.0 + (genome.weights.heuristic_reward * 50.0);
        let final_score = (base_score - time_penalty - token_penalty).max(0.0);

        Ok(FitnessScore {
            score: final_score,
            execution_time_ms: benchmark_latency_ms,
            token_efficiency: if tokens_used > 0 { 1000.0 / (tokens_used as f32) } else { 0.0 },
        })
    }
}

impl Mutator {
    pub fn mutate(parent: &SystemGenome) -> SystemGenome {
        let mut rng = rand::thread_rng();
        let mut child = parent.clone();
        
        child.generation += 1;
        
        // Drift cognitive weights by +/- 10%
        let mut drift = || rng.gen_range(0.90..1.10);
        
        child.weights.meta_reasoning_threshold *= drift();
        child.weights.critic_penalty_weight *= drift();
        child.weights.heuristic_reward *= drift();

        // Enforce boundary constraints
        child.weights.meta_reasoning_threshold = child.weights.meta_reasoning_threshold.clamp(0.1, 0.9);
        child.weights.heuristic_reward = child.weights.heuristic_reward.clamp(0.1, 0.8);

        // Generate deterministic child hash
        child.hash = format!("gen_{}_{}", child.generation, hex::encode(rng.gen::<[u8; 4]>()));
        
        child
    }
}

fn main() {

    let args: Vec<String> = env::args().collect();

    // Accept objective from CLI or use a default test objective
    let objective = if args.len() > 1 {
        args[1..].join(" ")
    } else {
        "Audit the local file system structure, then analyze it for security flaws.".to_string()
    };

    println!("==================================================");
    println!(" AthenaFusionX: Executive Cortex & DAG Orchestrator");
    println!(" Target: ARM64 Shared DNA");

        Ok(tasks) => tasks,
        Err(e) => {
            eprintln!("[-] FATAL: Planner failed to generate DAG: {}", e);
            std::process::exit(1);
        }
    };

    println!("[+] Blueprint Acquired: {} tasks dynamically mapped.", planned_tasks.len());

    let mut manager = DagManager::new();
    for pt in planned_tasks {

        );
        manager.add_task(TaskNode {

            required_tier: pt.required_tier,
            dependencies: pt.dependencies,
            status: TaskStatus::Pending,
        });
    }

    let mut executor = DagExecutor::new(manager);
    let mut loop_count = 1;

    println!("\n[*] Executing Directed Acyclic Graph...\n");

    loop {
        let all_complete =
            executor.manager.nodes.values().all(|n| n.status == TaskStatus::Completed);
        if all_complete {
            println!("\n[+] DAG Execution Complete. All nodes synchronized.");
            break;
        }

        let failed = executor.manager.nodes.values().any(|n| n.status == TaskStatus::Failed);
        if failed {
            eprintln!("\n[-] DAG Execution Halted. One or more tasks failed.");
            break;
        }

        println!("--- [Tick {}] Evaluating DAG State ---", loop_count);
        match executor.run_step() {
            Ok(executed) => {
                if executed == 0 && loop_count == 1 {
                    println!("  -> No tasks ready. (Graph error or circular dependency)");
                    break;
                }
            }
            Err(e) => {
                eprintln!("[-] Executor Error: {}", e);
                break;
            }
        }

        loop_count += 1;
        thread::sleep(Duration::from_secs(2)); // Pause for API rate limits and stdout clarity
    }
}

use crate::graph::{DagManager, TaskStatus};
use std::env;
use std::path::PathBuf;
use std::process::{Command, Stdio};

pub struct DagExecutor {
    pub manager: DagManager,
}

impl DagExecutor {
    pub fn new(manager: DagManager) -> Self {
        Self { manager }
    }

    fn get_minion_binary_path() -> PathBuf {
        let mut path =
            env::current_exe().unwrap_or_else(|_| PathBuf::from("target/debug/nexus_minion"));
        path.pop();

        let bin_path = path.join("nexus_minion");
        if bin_path.exists() {
            bin_path
        } else {
            PathBuf::from("target/debug/nexus_minion")
        }
    }

    pub fn run_step(&mut self) -> Result<usize, String> {
        let ready_tasks = self.manager.get_ready_tasks();
        let mut executed_count = 0;

        if ready_tasks.is_empty() {
            return Ok(0);
        }

        let bin_path = Self::get_minion_binary_path();

        for task_id in ready_tasks {
            let (tier, description) = {
                let node = self.manager.nodes.get_mut(&task_id).unwrap();
                node.status = TaskStatus::InProgress;
                (node.required_tier.clone(), node.description.clone())
            };

            println!("\n==================================================");
            println!(
                "[Orchestrator] Dispatching Task: [{}] (Tier: {})",
                task_id,
                tier.to_uppercase()
            );
            println!("Objective: {}", description);
            println!("==================================================");

            // Spawn the minion with live stdout inheritance so ReAct sandbox logs stream in real-time
            let status = Command::new(&bin_path)
                .arg("--tier")
                .arg(&tier)
                .arg(&description)
                .stdout(Stdio::inherit())
                .stderr(Stdio::inherit())
                .status();

            match status {
                Ok(exit_status) if exit_status.success() => {
                    println!("\n[Orchestrator] Task [{}] Completed Successfully.", task_id);
                    self.manager.complete_task(&task_id);
                    executed_count += 1;
                }
                Ok(exit_status) => {
                    eprintln!(
                        "\n[Orchestrator] Task [{}] FAILED with code: {}",
                        task_id, exit_status
                    );
                    if let Some(node) = self.manager.nodes.get_mut(&task_id) {
                        node.status = TaskStatus::Failed;
                    }
                }
                Err(e) => {
                    eprintln!("\n[Orchestrator] FATAL: Failed to execute minion daemon: {}", e);
                    if let Some(node) = self.manager.nodes.get_mut(&task_id) {
                        node.status = TaskStatus::Failed;
                    }
                }
            }
        }

        Ok(executed_count)
    }
}

pub struct TaskNode {
    pub id: String,
    pub description: String,
    pub required_tier: String, // "light" or "heavy"
    pub dependencies: Vec<String>,
    pub status: TaskStatus,
}

pub struct DagManager {
    pub nodes: HashMap<String, TaskNode>,
}

impl DagManager {
    pub fn new() -> Self {
        Self { nodes: HashMap::new() }
    }

    /// Adds a new task to the execution graph.
    pub fn add_task(&mut self, node: TaskNode) {
        self.nodes.insert(node.id.clone(), node);
    }

    /// Returns a list of Task IDs that have all dependencies met and are ready to run.
    pub fn get_ready_tasks(&self) -> Vec<String> {
        let mut ready = Vec::new();

        for (id, node) in &self.nodes {
            if node.status != TaskStatus::Pending {
                continue;
            }

            let dependencies_met = node.dependencies.iter().all(|dep_id| {
                if let Some(dep_node) = self.nodes.get(dep_id) {
                    dep_node.status == TaskStatus::Completed
                } else {
                    false // Dependency doesn't exist
                }
            });

            if dependencies_met {
                ready.push(id.clone());
            }
        }

        ready
    }

    /// Marks a task as completed, which may unlock downstream dependencies.
    pub fn complete_task(&mut self, id: &str) {
        if let Some(node) = self.nodes.get_mut(id) {
            node.status = TaskStatus::Completed;
        }
    }
}

//! Micro-Agent API Gateway

use serde_json::json;
use std::env;
use std::error::Error;

#[derive(Debug, Clone)]
pub struct AgentRequest {
    pub prompt: String,
    pub temperature: f32,
    pub max_tokens: usize,
}

#[derive(Debug, Clone)]
pub struct AgentResponse {
    pub text: String,
    pub tokens_used: usize,
    pub model_used: String,
}

pub struct ApiGateway;

impl ApiGateway {
    pub fn new() -> Self {
        Self {}
    }

    /// Dynamically resolves the correct endpoint and API key based on the model string.
    fn resolve_provider(&self, model_name: &str) -> (String, String) {
        if model_name.contains("gemini") {
            (
                "https://generativelanguage.googleapis.com/v1beta/openai/chat/completions"
                    .to_string(),
                env::var("GEMINI_API_KEY").unwrap_or_default(),
            )
        } else if model_name.contains("nvidia")
            || model_name.contains("nemotron")
            || model_name.contains("llama")
            || model_name.contains("gemma")
        {
            (
                "https://generativelanguage.googleapis.com/v1beta/chat/completions".to_string(),
                env::var("NVIDIA_API_KEY").unwrap_or_default(),
            )
        } else {
            (
                env::var("NEXUS_API_ENDPOINT")
                    .unwrap_or_else(|_| "https://api.example.com/v1/chat/completions".to_string()),
                env::var("NEXUS_AUTH_KEY").unwrap_or_default(),
            )
        }
    }

    /// Executes the request, cascading through the provided model list on failure.
    pub fn dispatch_cascade(
        &self,
        request: AgentRequest,
        cascade: &[String],
    ) -> Result<AgentResponse, Box<dyn Error>> {
        if cascade.is_empty() {
            return Err("FATAL: Empty cascade list provided.".into());
        }

        let mut last_error: Option<Box<dyn Error>> = None;

        for model in cascade {
            println!("[Gateway] Attempting dispatch to: {}", model);
            let (endpoint, api_key) = self.resolve_provider(model);

            if api_key.is_empty() {
                println!("  -> [WARN] Skipping {} (Missing API Key)", model);
                continue;
            }

            let payload = json!({
                "model": model,
                "messages": [
                    {
                        "role": "system",
                        "content": "You are a Nexus Micro-Agent executing on an ARM64 edge node. Be concise, technical, and precise."
                    },
                    {
                        "role": "user",
                        "content": request.prompt
                    }
                ],
                "temperature": request.temperature,
                "max_tokens": request.max_tokens
            });

            match ureq::post(&endpoint)
                .set("Authorization", &format!("Bearer {}", api_key))
                .set("Content-Type", "application/json")
                .send_json(payload)
            {
                Ok(response) => {
                    let response_json: serde_json::Value = response.into_json()?;

                    let text = response_json["choices"][0]["message"]["content"]
                        .as_str()
                        .unwrap_or("[ERROR: Failed to parse model response]")
                        .to_string();

                    let tokens_used =
                        response_json["usage"]["total_tokens"].as_u64().unwrap_or(0) as usize;

                    return Ok(AgentResponse { text, tokens_used, model_used: model.to_string() });
                }
                Err(e) => {
                    println!("  -> [FAIL] {} unreachable or returned error: {}", model, e);
                    last_error = Some(Box::new(e));
                    // Loop continues to the next model in the cascade
                }
            }
        }

        Err(format!("All models in cascade failed. Last error: {:?}", last_error).into())
    }
}

    }

//! Self-attention projection + norm + RoPE + KV-cache pipeline for one layer.
//!
//! Produces per-head Q (RoPE'd) and pushes K/V into the StaticKVCache.
//! Attention scores (QK^T / softmax / *V) + o_proj are computed in
//! compute_attention_output.

use crate::kv_cache::StaticKVCache;
use crate::neon_math::matvec_bf16_to_f32;
use crate::rope::{apply_rope_scalar, AttentionMode, RoPEConfig};
use half::bf16;

/// RMSNorm using Gemma's (1 + weight) convention.
/// Gemma RMSNorm weights are zero-initialized and act as a delta from 1.0.
pub fn rms_norm_in_place(x: &mut [f32], weight: &[f32]) {
    debug_assert_eq!(x.len(), weight.len());
    const EPS: f32 = 1e-6;
    let mean_sq: f32 = x.iter().map(|v| v * v).sum::<f32>() / x.len() as f32;
    let scale = 1.0 / (mean_sq + EPS).sqrt();
    for (xi, wi) in x.iter_mut().zip(weight.iter()) {
        *xi = *xi * scale * (1.0 + wi);
    }
}

pub struct AttentionBlock {
    pub layer_idx: usize,
    pub mode: AttentionMode,
    pub rope_config: RoPEConfig,
    pub num_heads: usize,
    pub num_kv_heads: usize, // MQA: 1
    pub hidden_size: usize,  // 1536
}

impl AttentionBlock {
    pub fn new(layer_idx: usize, is_global: bool) -> Self {
        let mode = if is_global { AttentionMode::Global } else { AttentionMode::Sliding };
        Self {
            layer_idx,
            mode,
            rope_config: RoPEConfig::new(mode),
            num_heads: 8,
            num_kv_heads: 1,
            hidden_size: 1536,
        }
    }

    #[allow(clippy::too_many_arguments)]
    pub fn project_and_cache(
        &self,
        hidden_state_normed: &[f32],
        q_proj_w: &[bf16],
        k_proj_w: &[bf16],
        v_proj_w: &[bf16],
        q_norm_w: &[f32],
        k_norm_w: &[f32],
        position: usize,
        kv_cache: &mut StaticKVCache,
    ) -> Vec<f32> {
        let head_dim = self.rope_config.head_dim;
        let q_out_dim = self.num_heads * head_dim;
        let kv_out_dim = self.num_kv_heads * head_dim;

        assert_eq!(hidden_state_normed.len(), self.hidden_size);
        assert_eq!(q_proj_w.len(), q_out_dim * self.hidden_size);
        assert_eq!(k_proj_w.len(), kv_out_dim * self.hidden_size);
        assert_eq!(v_proj_w.len(), kv_out_dim * self.hidden_size);
        assert_eq!(q_norm_w.len(), head_dim);
        assert_eq!(k_norm_w.len(), head_dim);

        let x_bf16: Vec<bf16> = hidden_state_normed.iter().map(|&v| bf16::from_f32(v)).collect();

        let mut q = vec![0.0f32; q_out_dim];
        let mut k = vec![0.0f32; kv_out_dim];
        let mut v = vec![0.0f32; kv_out_dim];

        matvec_bf16_to_f32(q_proj_w, &x_bf16, &mut q, q_out_dim, self.hidden_size);
        matvec_bf16_to_f32(k_proj_w, &x_bf16, &mut k, kv_out_dim, self.hidden_size);
        matvec_bf16_to_f32(v_proj_w, &x_bf16, &mut v, kv_out_dim, self.hidden_size);

        // MQA: single KV head
        rms_norm_in_place(&mut k[..head_dim], k_norm_w);
        apply_rope_scalar(&mut k[..head_dim], position, &self.rope_config);

        let k_bf16: Vec<bf16> = k.iter().map(|&x| bf16::from_f32(x)).collect();
        let v_bf16: Vec<bf16> = v.iter().map(|&x| bf16::from_f32(x)).collect();
        kv_cache.push(&k_bf16[..head_dim], &v_bf16[..head_dim]);

        // Per-head Q
        for h in 0..self.num_heads {
            let start = h * head_dim;
            let end = start + head_dim;
            rms_norm_in_place(&mut q[start..end], q_norm_w);
            apply_rope_scalar(&mut q[start..end], position, &self.rope_config);
        }

        q
    }

    pub fn project_q_only(
        &self,
        hidden_state_normed: &[f32],
        q_proj_w: &[bf16],
        q_norm_w: &[f32],
        position: usize,
    ) -> Vec<f32> {
        let head_dim = self.rope_config.head_dim;
        let q_out_dim = self.num_heads * head_dim;

        assert_eq!(hidden_state_normed.len(), self.hidden_size);
        assert_eq!(q_proj_w.len(), q_out_dim * self.hidden_size);
        assert_eq!(q_norm_w.len(), head_dim);

        let x_bf16: Vec<bf16> = hidden_state_normed.iter().map(|&v| bf16::from_f32(v)).collect();
        let mut q = vec![0.0f32; q_out_dim];
        matvec_bf16_to_f32(q_proj_w, &x_bf16, &mut q, q_out_dim, self.hidden_size);

        for h in 0..self.num_heads {
            let start = h * head_dim;
            let end = start + head_dim;
            rms_norm_in_place(&mut q[start..end], q_norm_w);
            apply_rope_scalar(&mut q[start..end], position, &self.rope_config);
        }

        q
    }

    pub fn compute_attention_output(
        &self,
        q: &[f32],
        o_proj_w: &[bf16],
        kv_cache: &StaticKVCache,
    ) -> Vec<f32> {
        let head_dim = self.rope_config.head_dim;
        let q_out_dim = self.num_heads * head_dim;
        let seq_len = kv_cache.len();

        assert_eq!(q.len(), q_out_dim);
        assert_eq!(o_proj_w.len(), self.hidden_size * q_out_dim);

        if seq_len == 0 {
            return vec![0.0f32; self.hidden_size];
        }

        let scale = 1.0f32 / (head_dim as f32).sqrt();
        let mut combined_heads = vec![0.0f32; q_out_dim];

        // MQA: 8 Q heads attend to the same single K/V head
        for h in 0..self.num_heads {
            let q_start = h * head_dim;
            let q_h = &q[q_start..q_start + head_dim];

            let mut scores = vec![0.0f32; seq_len];
            let mut max_score = f32::NEG_INFINITY;

            // Scaled dot-product (NO attention softcapping for Gemma 4)
            for (pos, score_slot) in scores.iter_mut().enumerate().take(seq_len) {
                let k_bf16 = kv_cache.k_at(pos);
                let mut dot = 0.0f32;
                for i in 0..head_dim {
                    dot += q_h[i] * k_bf16[i].to_f32();
                }
                let score = dot * scale;
                *score_slot = score;
                if score > max_score {
                    max_score = score;
                }
            }

            // Numerically stable softmax
            let mut exp_sum = 0.0f32;
            for score in scores.iter_mut().take(seq_len) {
                let exp_val = (*score - max_score).exp();
                *score = exp_val;
                exp_sum += exp_val;
            }
            let inv_exp_sum = if exp_sum > 0.0 { 1.0 / exp_sum } else { 0.0 };
            for score in scores.iter_mut().take(seq_len) {
                *score *= inv_exp_sum;
            }

            // Weighted sum of V
            let out_start = h * head_dim;
            let head_out = &mut combined_heads[out_start..out_start + head_dim];
            for (pos, &weight) in scores.iter().enumerate().take(seq_len) {
                let v_bf16 = kv_cache.v_at(pos);
                for i in 0..head_dim {
                    head_out[i] += weight * v_bf16[i].to_f32();
                }
            }
        }

        // o_proj back to hidden_size
        let combined_bf16: Vec<bf16> = combined_heads.iter().map(|&x| bf16::from_f32(x)).collect();
        let mut attn_out = vec![0.0f32; self.hidden_size];
        matvec_bf16_to_f32(o_proj_w, &combined_bf16, &mut attn_out, self.hidden_size, q_out_dim);

        attn_out
    }
}

//! Nexus Edge Minion Binary Runner - Multi-Step Autoregressive Stress Test
//! Orchestrates local inference execution against the USB safetensors substrate.

use nexus_edge_minion::inference_engine::InferenceEngine;
use std::env;
use std::env::consts;
use std::time::Instant;

#[tokio::main]
async fn main() {
    let args: Vec<String> = env::args().collect();
    let prompt =
        if args.len() > 1 { args[1].clone() } else { "Substrate readiness verified.".to_string() };

    println!(">> [NEXUS MINION] Initializing execution pipeline...");
    println!(">> [NEXUS MINION] Task Prompt: \"{}\"", prompt);
    println!(">> [HOST ARCH] Target Architecture: {}-{}", consts::ARCH, consts::OS);

    if let Ok(cache_dir) = env::var("NEXUS_CACHE_DIR") {
        println!(">> [LOCAL EDGE] Detected NEXUS_CACHE_DIR: {}", cache_dir);
        println!(">> [LOCAL EDGE] Engaging local inference helper engine...");

        match InferenceEngine::new(&cache_dir) {
            Ok(mut engine) => {
                println!(">> [LOCAL EDGE] Safetensors weights mapped successfully.");
                let mut hidden_state = vec![0.0f32; 1536];

                let num_steps = 5;
                println!(
                    ">> [STRESS TEST] Running multi-step autoregressive loop ({} tokens)...",
                    num_steps
                );

                let start_time = Instant::now();
                for pos in 0..num_steps {
                    let token_id = (100 + pos) as u32;
                    let step_start = Instant::now();

                    match engine.step(token_id, pos, &mut hidden_state) {
                        Ok(_) => {
                            let duration = step_start.elapsed();
                            println!(
                                "  -> [STEP {}/{}]: Token ID {} processed in {:.2}ms (KV-cache active)",
                                pos + 1, num_steps, token_id, duration.as_secs_f64() * 1000.0
                            );
                        }
                        Err(e) => {
                            eprintln!("  -> [ERROR] Step {} failed: {}", pos + 1, e);
                            break;
                        }
                    }
                }

                let total_duration = start_time.elapsed();
                println!(
                    ">> [STRESS TEST] Completed {} steps in {:.2}s ({:.2} tokens/sec)",
                    num_steps,
                    total_duration.as_secs_f64(),
                    num_steps as f64 / total_duration.as_secs_f64()
                );
            }
            Err(e) => {
                eprintln!(">> [LOCAL EDGE ERROR] Failed to initialize local engine: {}", e);
            }
        }
    } else {
        println!(
            ">> [CLOUD ROUTE] NEXUS_CACHE_DIR not set. Handing off to API Gateway powerhouse..."
        );
    }

    println!(">> [NEXUS MINION] Execution cycle complete.");
}
