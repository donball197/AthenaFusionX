use crate::cluster::ClusterMesh;
use crate::wasm_runtime::WasmRuntime;
use anyhow::{anyhow, Result};
use base64::{engine::general_purpose, Engine as _};
use serde::{Deserialize, Serialize};
use std::sync::Arc;

#[derive(Clone)]
pub struct DistributedScheduler {
    pub mesh: Arc<ClusterMesh>,
}

impl DistributedScheduler {
    pub fn new(mesh: Arc<ClusterMesh>) -> Self {
        Self { mesh }
    }

    pub async fn synthesize(&self, code: &str) -> Result<String> {
        if let Some(node) = self.mesh.best_node() {
            if node.cpu < 85.0 && node.temp < 68.0 {
                println!(
                    "\x1b[35m>> [MESH_ROUTING] Offloading WASM Synthesis to {} (Temp: {:.1}°C Load: {:.1}%)\x1b[0m",
                    node.id, node.temp, node.cpu
                );
                return self.remote_synthesize(&node.host, node.port, code).await;
            }
        }

        println!(
            "\x1b[33m>> [LOCAL_EXEC] Substrate utilized for WASM synthesis (Heavy Load)\x1b[0m"
        );
        crate::wasm_forge::WasmForge::synthesize_cell(code).await
    }

    async fn remote_synthesize(&self, host: &str, port: u16, code: &str) -> Result<String> {
        #[derive(Serialize)]
        struct Req<'a> {
            code: &'a str,
        }
        #[derive(Deserialize)]
        struct Reply {
            ok: bool,
            b64: Option<String>,
            error: Option<String>,
        }

        let url = format!("http://{}:{}/api/cluster/forge", host, port);
        let res = reqwest::Client::new()
            .post(&url)
            .header("Authorization", "SOVEREIGN")
            .json(&Req { code })
            .send()
            .await?;

        let reply: Reply = res.json().await?;
        if reply.ok {
            Ok(reply.b64.unwrap_or_default())
        } else {
            Err(anyhow!(reply.error.unwrap_or_else(|| "Remote forge error".into())))
        }
    }

    pub async fn execute(&self, wasm: Vec<u8>, func_name: &str, input: f32) -> Result<f32> {
        if let Some(node) = self.mesh.best_node() {
            if node.cpu < 85.0 && node.temp < 68.0 {
                println!(">> [MESH_ROUTING] Offloading execution of {} to {}", func_name, node.id);
                return self.remote_call(&node.host, node.port, wasm, func_name, input).await;
            }
        }

        println!(">> [LOCAL_EXEC] Substrate utilized for {}", func_name);
        WasmRuntime::run_bytes(&wasm, func_name, input)
    }

    async fn remote_call(
        &self,
        host: &str,
        port: u16,
        wasm: Vec<u8>,
        func_name: &str,
        input: f32,
    ) -> Result<f32> {
        #[derive(Serialize)]
        struct Req<'a> {
            wasm_b64: String,
            func_name: &'a str,
            input: f32,
        }
        #[derive(Deserialize)]
        struct Reply {
            ok: bool,
            result: f32,
            error: Option<String>,
        }

        let url = format!("http://{}:{}/api/cluster/run", host, port);
        let res = reqwest::Client::new()
            .post(&url)
            .header("Authorization", "SOVEREIGN")
            .json(&Req { wasm_b64: general_purpose::STANDARD.encode(&wasm), func_name, input })
            .send()
            .await?;

        let reply: Reply = res.json().await?;
        if reply.ok {
            Ok(reply.result)
        } else {
            Err(anyhow!(reply.error.unwrap_or_else(|| "remote error".into())))
        }
    }
}

    pub async fn safe_api_call(
        client: &Client,
        api_key: &str,
        models: &[String],
        payload: &Value,
    ) -> Result<String> {
        let mut last_err = anyhow!("No models available in cascade.");
        for model_endpoint in models {
            // Fix: Route any model that doesn't contain "gemini" to NVIDIA NIM
            let is_gemini = model_endpoint.contains("gemini");
            let is_openai_format = !is_gemini;
            
            let short_name = model_endpoint.split('/').last().unwrap_or(model_endpoint);
            let mut retries = 0;

            loop {
                let (url, req_payload, use_auth) = if is_openai_format {
                    let nv_key = std::env::var("NVIDIA_API_KEY")
                        .unwrap_or_default()
                        .trim()
                        .trim_matches('"')
                        .to_string();
                    let text = payload
                        .pointer("/contents/0/parts/0/text")
                        .and_then(|v| v.as_str())
                        .unwrap_or("");

                    let mut p = serde_json::json!({
                        "model": model_endpoint,
                        "messages": [{"role": "user", "content": text}],
                        "max_tokens": 4096,
                        "temperature": 0.1
                    });

                    // ⚡ NATIVE TOOL CALLING: Translate Gemini tool schema to OpenAI/NVIDIA
                    if let Some(funcs) = payload.pointer("/tools/0/functionDeclarations").and_then(|v| v.as_array()) {
                        let oai_tools: Vec<Value> = funcs.iter().map(|f| {
                            serde_json::json!({
                                "type": "function",
                                "function": f
                            })
                        }).collect();
                        p.as_object_mut().unwrap().insert("tools".to_string(), Value::Array(oai_tools));
                    }

                    if let Some(mime) = payload
                        .pointer("/generationConfig/responseMimeType")
                        .and_then(|v| v.as_str())
                    {
                        if mime == "application/json" {
                            p.as_object_mut().unwrap().insert(
                                "response_format".to_string(),
                                serde_json::json!({"type": "json_object"}),
                            );
                            let msg = format!(
                                "{}\n\nCRITICAL: You must return a valid JSON object.",
                                text
                            );
                            p["messages"][0]["content"] = serde_json::json!(msg);
                        }
                    }

                    (
                        "https://integrate.api.nvidia.com/v1/chat/completions".to_string(),
                        p,
                        Some(nv_key),
                    )
                } else {
                    let key = if api_key.is_empty() {
                        std::env::var("GEMINI_API_KEY").unwrap_or_default().trim().to_string()
                    } else {
                        api_key.to_string()
                    };
                    let u = format!(
                        "https://generativelanguage.googleapis.com/v1beta/models/{}:generateContent?key={}",
                        model_endpoint.replace("models/", ""),
                        key
                    );
                    (u, payload.clone(), None)
                };

                let mut req = client.post(&url).json(&req_payload);
                if let Some(auth_key) = use_auth {
                    req = req.header("Authorization", format!("Bearer {}", auth_key));
                }

                let res = req.send().await?;
                let body: Value = match res.json().await {
                    Ok(b) => b,
                    Err(e) => {
                        last_err = anyhow::anyhow!("JSON Parse Error: {}", e);
                        retries += 1;
                        if retries > 2 {
                            break;
                        }
                        tokio::time::sleep(tokio::time::Duration::from_secs(2)).await;
                        continue;
                    }
                };

                // Error Handling
                if is_openai_format {
                    if body.get("error").is_some() {
                        last_err = anyhow!("NVIDIA/OpenAI API Error: {:?}", body);
                        break;
                    }
                } else {
                    if let Some(err) = body.pointer("/error/message").and_then(|v| v.as_str()) {
                        last_err = anyhow!("Gemini API Error: {}", err);
                        if err.contains("quota") || err.contains("rate limit") || err.contains("RESOURCE_EXHAUSTED") {
                            crate::model_config::report_rate_limit(model_endpoint);
                            break;
                        }
                        if err.contains("Internal error") || err.contains("503") || err.contains("500") || err.contains("Internal server error") {
                            retries += 1;
                            if retries > 12 { break; }
                            let backoff = std::cmp::min(2_u64.pow(retries as u32) * 2, 45);
                            tokio::time::sleep(Duration::from_secs(backoff)).await;
                            continue;
                        }
                        break;
                    }
                }

                // ⚡ NATIVE TOOL INTERCEPT ⚡
                // Stop raw string parsing if a native JSON Tool Call is detected!
                if let Ok(action) = extract_tool_call_from_response(&body) {
                    if let Ok(json_str) = serde_json::to_string(&vec![action]) {
                        return Ok(json_str);
                    }
                }

                // Standard Text Extraction Fallback
                let mut final_text = String::new();
                if is_openai_format {
                    if let Some(content) = body.pointer("/choices/0/message/content").and_then(|v| v.as_str()) {
                        final_text = content.to_string();
                    }
                    if final_text.trim().is_empty() {
                        if let Some(rc) = body.pointer("/choices/0/message/reasoning_content").and_then(|v| v.as_str()) {
                            final_text = rc.to_string();
                        }
                    }
                } else {
                    if let Some(parts) = body.pointer("/candidates/0/content/parts").and_then(|v| v.as_array()) {
                        for part in parts {
                            if part.get("thought").and_then(|t| t.as_bool()).unwrap_or(false) { continue; }
                            if let Some(text) = part.get("text").and_then(|t| t.as_str()) {
                                final_text.push_str(text);
                            }
                        }
                    }
                }

                if final_text.trim().is_empty() {
                    retries += 1;
                    if retries > 4 { break; }
                    tokio::time::sleep(Duration::from_secs(2_u64.pow(retries as u32) * 2)).await;
                    continue;
                }
                return Ok(final_text.replace("```json", "").replace("```", "").trim().to_string());
            }
        }
        Err(anyhow!("Model Cascade Exhausted. Final error: {}", last_err))
    }

        let prompt = format!("You are AthenaFusionX.         ).await?;
        let clean_text = sanitize_json_escapes(&text);

        if let Ok(hyps) = serde_json::from_str::<Vec<String>>(&clean_text) {
            if !hyps.is_empty() { return Ok(hyps); }

    }

use serde::{Deserialize, Serialize};
use tokio::sync::broadcast;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum EventKind {
    TaskCreated,
    TaskStarted,
    TaskCompleted,
    AgentCrashed,
    ArtifactCommitted,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct EventRef {
    pub seq: u64,
    pub kind: EventKind,
    pub object_id: String,
}

#[derive(Clone)]
pub struct EventBus {
    tx: broadcast::Sender<EventRef>,
}

impl EventBus {
    pub fn new(size: usize) -> Self {
        let (tx, _) = broadcast::channel(size);
        Self { tx }
    }

    pub fn emit(&self, evt: EventRef) {
        let _ = self.tx.send(evt);
    }

    pub fn subscribe(&self) -> broadcast::Receiver<EventRef> {
        self.tx.subscribe()
    }
}

use serde::{Deserialize, Serialize};
use std::ops::{Add, AddAssign, Mul};

/// Multidimensional belief state used by the Executive Cortex.
#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
pub struct BeliefVector {
    pub implementation_confidence: f32,
    pub evidence_quality: f32,
    pub architectural_risk: f32,
    pub security_confidence: f32,
    /// Absolute change from the previous update (used by residual scheduler).
    pub residual: f32,
    /// True once a concrete success marker has been observed.
    pub has_success_marker: bool,
    /// Bit-packed counter of verified artifacts (cache-friendly on ARM64).
    pub verified_artifacts: u8,
}

impl BeliefVector {
    pub const ZERO: Self = Self {
        implementation_confidence: 0.0,
        evidence_quality: 0.0,
        architectural_risk: 0.0,
        security_confidence: 0.0,
        residual: 0.0,
        has_success_marker: false,
        verified_artifacts: 0,
    };

    pub fn new(
        implementation_confidence: f32,
        evidence_quality: f32,
        architectural_risk: f32,
        security_confidence: f32,
    ) -> Self {
        Self {
            implementation_confidence: implementation_confidence.clamp(0.0, 1.0),
            evidence_quality: evidence_quality.clamp(0.0, 1.0),
            architectural_risk: architectural_risk.clamp(0.0, 1.0),
            security_confidence: security_confidence.clamp(0.0, 1.0),
            residual: 0.0,
            has_success_marker: false,
            verified_artifacts: 0,
        }
    }

    /// Hard penalty method when a Policy Denial / Security Violation occurs.
    pub fn penalize_security(mut self) -> Self {
        self.security_confidence = 0.0;
        self.implementation_confidence = 0.0;
        self.architectural_risk = 1.0;
        self.residual = 1.0;
        self
    }

    /// Scalar utility used by ranking logic.
    pub fn scalar_score(&self) -> f32 {
        let positive = (self.implementation_confidence * 0.40)
            + (self.evidence_quality * 0.35)
            + (self.security_confidence * 0.25);
        let penalty = self.architectural_risk * 0.45;
        (positive - penalty).clamp(0.0, 1.0)
    }

    pub fn update(&self, other: &Self, weight: f32) -> Self {
        let w = weight.clamp(0.0, 1.0);
        let mut next = Self::new(
            self.implementation_confidence * (1.0 - w) + other.implementation_confidence * w,
            self.evidence_quality * (1.0 - w) + other.evidence_quality * w,
            self.architectural_risk * (1.0 - w) + other.architectural_risk * w,
            self.security_confidence * (1.0 - w) + other.security_confidence * w,
        );
        next.residual = (self.scalar_score() - next.scalar_score()).abs();
        next.has_success_marker = self.has_success_marker || other.has_success_marker;
        next.verified_artifacts = self.verified_artifacts | other.verified_artifacts;
        next
    }

    /// Calibrated Bayesian update on the positive dimensions with damping.
    pub fn bayesian_update(&self, likelihood: f32, damping: f32) -> Self {
        let likelihood = likelihood.clamp(0.0, 1.0);
        let damping = damping.clamp(0.0, 1.0);

        fn bayes(prior: f32, like: f32) -> f32 {
            let num = prior * like;
            let den = num + (1.0 - prior) * (1.0 - like);
            if den == 0.0 {
                prior
            } else {
                num / den
            }
        }

        let impl_post = bayes(self.implementation_confidence, likelihood);
        let evid_post = bayes(self.evidence_quality, likelihood);
        let sec_post = bayes(self.security_confidence, likelihood);

        let impl_d = damping * impl_post + (1.0 - damping) * self.implementation_confidence;
        let evid_d = damping * evid_post + (1.0 - damping) * self.evidence_quality;
        let sec_d = damping * sec_post + (1.0 - damping) * self.security_confidence;

        let mut next = Self::new(impl_d, evid_d, self.architectural_risk, sec_d);
        next.residual = (self.scalar_score() - next.scalar_score()).abs();
        next.has_success_marker = self.has_success_marker;
        next.verified_artifacts = self.verified_artifacts;
        next
    }

    /// Hard evidence gate. Belief alone is never enough.
    /// required_artifacts is a bitmask -- ALL required bits must actually be
    /// verified (bitwise AND), not merely a numerically-larger bit pattern.
    pub fn may_prove(&self, required_artifacts: u8, threshold: f32) -> bool {
        self.scalar_score() >= threshold
            && self.has_success_marker
            && (self.verified_artifacts & required_artifacts) == required_artifacts
    }

    /// Mark that a concrete success artifact was observed.
    pub fn mark_success(&mut self, artifact_bit: u8) {
        self.has_success_marker = true;
        self.verified_artifacts |= artifact_bit;
    }

    /// Convenience for residual-driven schedulers.
    pub fn scheduling_priority(&self, impact: f32, mission_priority: f32) -> f32 {
        self.residual * impact * mission_priority
    }
}

impl Default for BeliefVector {
    fn default() -> Self {
        Self::new(0.45, 0.30, 0.40, 0.55)
    }
}

impl Add for BeliefVector {
    type Output = Self;
    fn add(self, rhs: Self) -> Self {
        Self::new(
            self.implementation_confidence + rhs.implementation_confidence,
            self.evidence_quality + rhs.evidence_quality,
            self.architectural_risk + rhs.architectural_risk,
            self.security_confidence + rhs.security_confidence,
        )
    }
}

impl AddAssign for BeliefVector {
    fn add_assign(&mut self, rhs: Self) {
        *self = *self + rhs;
    }
}

impl Mul<f32> for BeliefVector {
    type Output = Self;
    fn mul(self, rhs: f32) -> Self {
        Self::new(
            self.implementation_confidence * rhs,
            self.evidence_quality * rhs,
            self.architectural_risk * rhs,
            self.security_confidence * rhs,
        )
    }
}

use crate::evidence::{BeliefVector, Evidence};
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use uuid::Uuid;

/// EvidenceGraph – a lightweight, strategy-keyed store of Evidence nodes.
#[derive(Debug, Default, Serialize, Deserialize)]
pub struct EvidenceGraph {
    /// strategy_id → list of Evidence that support / refute it
    by_strategy: HashMap<Uuid, Vec<Evidence>>,

    /// Global chronological log (useful for later dream-cycle / audit)
    chronology: Vec<Uuid>,
}

impl EvidenceGraph {
    pub fn new() -> Self {
        Self::default()
    }

    /// Insert a fully materialised Evidence node.
    pub fn insert(&mut self, evidence: Evidence) {
        if let Some(strategy_id) = evidence.related_strategy_id {
            self.by_strategy.entry(strategy_id).or_default().push(evidence.clone());
        }
        self.chronology.push(evidence.id);
    }

    /// Aggregate all evidence belonging to a strategy into a single BeliefVector.
    pub fn aggregate_belief(&self, strategy_id: Uuid, baseline: BeliefVector) -> BeliefVector {
        let Some(evidences) = self.by_strategy.get(&strategy_id) else {
            return baseline;
        };

        let mut current = baseline;
        let mut max_residual = 0.0_f32;

        for ev in evidences {
            let weight = (0.35 + ev.belief_delta.evidence_quality * 0.55).clamp(0.15, 0.85);
            let next = current.update(&ev.belief_delta, weight);
            max_residual = max_residual.max(next.residual);
            current = next;
        }

        current.residual = max_residual;
        current
    }

    pub fn evidence_count(&self, strategy_id: Uuid) -> usize {
        self.by_strategy.get(&strategy_id).map(|v| v.len()).unwrap_or(0)
    }

    pub fn evidence_for(&self, strategy_id: Uuid) -> &[Evidence] {
        self.by_strategy.get(&strategy_id).map(|v| v.as_slice()).unwrap_or(&[])
    }

    pub fn chronology(&self) -> &[Uuid] {
        &self.chronology
    }
}

use crate::epistemic_engine::EpistemicEngine;
use crate::evidence::{BeliefVector, Evidence, EvidenceKind, Observation};
use reqwest::Client;
use uuid::Uuid;

/// Central translation layer between the RoboticExecutor and the EvidenceGraph.
///
/// - Fast path  : pure heuristic (microseconds)
/// - Slow path  : optional LLM escalation for truly ambiguous observations
#[derive(Debug, Default)]
pub struct EvidenceManager;

impl EvidenceManager {
    pub fn new() -> Self {
        Self
    }

    /// Synchronous, zero-network path (kept for compatibility).
    pub fn process_observation(&self, obs: Observation, strategy_id: Option<Uuid>) -> Evidence {
        let (kind, failure) = Evidence::classify(&obs.raw, obs.success);
        let belief_delta = self.compute_belief_delta(&obs, kind);
        let mut ev = Evidence::from_observation(obs, kind, belief_delta, strategy_id);
        ev.observation.failure_kind = failure;
        ev
    }

    /// Async path with smart escalation.
    pub async fn process_observation_with_escalation(
        &self,
        obs: Observation,
        strategy_id: Option<Uuid>,
        client: &Client,
        api_key: &str,
    ) -> Evidence {
        let (kind, failure) = Evidence::classify(&obs.raw, obs.success);

        // Fast path
        if !Self::should_escalate(&obs, kind) {
            let belief_delta = self.compute_belief_delta(&obs, kind);
            let mut ev = Evidence::from_observation(obs, kind, belief_delta, strategy_id);
            ev.observation.failure_kind = failure;
            return ev;
        }

        // Slow path
        println!(
            "\x1b[35m>> [EVIDENCE] Ambiguous observation ({} chars) — escalating to light critic\x1b[0m",
            obs.raw.len()
        );

        match EpistemicEngine::evaluate_evidence(
            client,
            api_key,
            "Analyze observation",
            &obs.source,
            &obs.raw,
        )
        .await
        {
            Ok(response) => {
                // Map the 1D float back into a multi-dimensional BeliefVector
                let val = response.confidence_modifier;
                let delta = BeliefVector::new(val * 0.5, val, val * -0.2, val * 0.3);
                let mut ev = Evidence::from_observation(obs, kind, delta, strategy_id);
                ev.observation.failure_kind = failure;
                ev
            }
            Err(e) => {
                println!(
                    "\x1b[33m>> [EVIDENCE] Escalation failed ({}) — falling back to Neutral/ZERO\x1b[0m",
                    e
                );
                let mut ev = Evidence::from_observation(
                    obs,
                    EvidenceKind::Neutral,
                    BeliefVector::ZERO,
                    strategy_id,
                );
                ev.observation.failure_kind = failure;
                ev
            }
        }
    }

    fn should_escalate(obs: &Observation, kind: EvidenceKind) -> bool {
        match kind {
            EvidenceKind::CommandSuccess
            | EvidenceKind::CommandFailure
            | EvidenceKind::SecurityViolation
            | EvidenceKind::FileRead
            | EvidenceKind::FileWrite
            | EvidenceKind::ArchitecturalSmell
            | EvidenceKind::PatternMatch
            | EvidenceKind::PositiveSignal => false,
            EvidenceKind::Neutral => obs.raw.len() > 500,
        }
    }

    fn compute_belief_delta(&self, obs: &Observation, kind: EvidenceKind) -> BeliefVector {
        match kind {
            EvidenceKind::SecurityViolation => BeliefVector::new(-0.15, -0.10, 0.55, -0.85),
            EvidenceKind::CommandSuccess => {
                if obs.source.contains("read") || obs.raw.contains("[READ_OK]") {
                    BeliefVector::new(0.18, 0.42, -0.08, 0.12)
                } else {
                    BeliefVector::new(0.22, 0.28, -0.05, 0.10)
                }
            }
            EvidenceKind::CommandFailure => BeliefVector::new(-0.25, 0.15, 0.20, -0.05),
            EvidenceKind::FileRead => BeliefVector::new(0.12, 0.45, -0.05, 0.08),
            EvidenceKind::FileWrite => BeliefVector::new(0.20, 0.30, 0.10, -0.05),
            EvidenceKind::PatternMatch => BeliefVector::new(0.15, 0.35, -0.05, 0.05),
            EvidenceKind::ArchitecturalSmell => BeliefVector::new(-0.18, 0.20, 0.40, -0.08),
            EvidenceKind::PositiveSignal => BeliefVector::new(0.18, 0.25, -0.08, 0.10),
            EvidenceKind::Neutral => BeliefVector::ZERO,
        }
    }
}

use crate::evidence::BeliefVector;
use serde::{Deserialize, Serialize};
use std::time::{SystemTime, UNIX_EPOCH};
use uuid::Uuid;

/// Explicit failure taxonomy to prevent infinite retry loops on security denials.
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum ExecutionFailureKind {
    /// Hard security / policy violation (e.g. PATH_TRAVERSAL_DENIED).
    /// MUST immediately zero out strategy belief and mark path as ineligible.
    PolicyDenial { path: String, reason: String },

    /// Target file/resource missing on disk (e.g., ENOENT os error 2).
    ResourceNotFound { target: String },

    /// Non-zero exit code or tool runtime failure.
    ExecutionError { exit_code: i32, stderr: String },

    /// Lock timeout or transient resource contention.
    TransientTimeout { resource: String },
}

/// Explicit provenance tracking for evidence claims.
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub enum ProvenanceLevel {
    /// Directly verified via tool execution or file contents read from disk.
    Observed { byte_count: usize },

    /// Derived logically from AST symbols or dependency graphs.
    Inferred { basis_symbol: String, confidence: f32 },

    /// Unverified assumption or heuristic generated by model reasoning.
    Hypothesized { rationale: String },
}

impl Default for ProvenanceLevel {
    fn default() -> Self {
        Self::Hypothesized { rationale: "Unverified initial assumption".to_string() }
    }
}

/// Raw observation string coming out of the RoboticExecutor.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Observation {
    pub raw: String,
    pub source: String, // "executor", "grep", "read_file", "podman", etc.
    pub success: bool,
    pub latency_ms: u64,
    pub failure_kind: Option<ExecutionFailureKind>,
}

/// Semantic classification of an observation.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum EvidenceKind {
    CommandSuccess,
    CommandFailure,
    FileRead,
    FileWrite,
    PatternMatch,
    SecurityViolation,
    ArchitecturalSmell,
    PositiveSignal,
    Neutral,
}

/// Fully materialised evidence node that can be inserted into the EvidenceGraph.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Evidence {
    pub id: Uuid,
    pub observation: Observation,
    pub kind: EvidenceKind,
    pub provenance: ProvenanceLevel,
    pub belief_delta: BeliefVector,
    pub timestamp: u64,
    pub related_strategy_id: Option<Uuid>,
}

impl Evidence {
    pub fn from_observation(
        obs: Observation,
        kind: EvidenceKind,
        belief_delta: BeliefVector,
        related_strategy_id: Option<Uuid>,
    ) -> Self {
        let timestamp = SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_secs();

        let provenance = if obs.raw.contains("[READ_OK]") || obs.raw.contains("[CMD_OK]") {
            ProvenanceLevel::Observed { byte_count: obs.raw.len() }
        } else {
            ProvenanceLevel::Hypothesized { rationale: "Derived from executor stdout".to_string() }
        };

        Self {
            id: Uuid::new_v4(),
            observation: obs,
            kind,
            provenance,
            belief_delta,
            timestamp,
            related_strategy_id,
        }
    }

    /// Heuristic classifier with explicit PolicyDenial extraction.
    pub fn classify(raw: &str, success: bool) -> (EvidenceKind, Option<ExecutionFailureKind>) {
        let lower = raw.to_lowercase();

        if lower.contains("sandbox_violation")
            || lower.contains("path_traversal")
            || lower.contains("threat pattern")
        {
            let path = raw.split(':').nth(1).unwrap_or("unknown_path").trim().to_string();
            return (
                EvidenceKind::SecurityViolation,
                Some(ExecutionFailureKind::PolicyDenial {
                    path,
                    reason: "PATH_TRAVERSAL_DENIED".to_string(),
                }),
            );
        }

        if lower.contains("no such file or directory") || lower.contains("os error 2") {
            return (
                EvidenceKind::CommandFailure,
                Some(ExecutionFailureKind::ResourceNotFound { target: raw.to_string() }),
            );
        }

        if lower.contains("pattern_not_found")
            || lower.contains("function_not_found")
            || lower.contains("action_failed")
            || lower.contains("[action failed]")
        {
            return (
                EvidenceKind::CommandFailure,
                Some(ExecutionFailureKind::ExecutionError {
                    exit_code: 1,
                    stderr: raw.to_string(),
                }),
            );
        }

        if lower.contains("[cmd_ok]") || lower.contains("[read_ok]") || lower.contains("[write_ok]")
        {
            return if success {
                (EvidenceKind::CommandSuccess, None)
            } else {
                (EvidenceKind::CommandFailure, None)
            };
        }

        if lower.contains("todo") || lower.contains("placeholder") {
            return (EvidenceKind::ArchitecturalSmell, None);
        }

        if success {
            (EvidenceKind::PositiveSignal, None)
        } else {
            (EvidenceKind::Neutral, None)
        }
    }
}

use crate::fitness_evaluator::FitnessEvaluator;
use crate::memory_cortex::MemoryCortex;
use crate::wasm_forge::WasmForge;
use anyhow::Result;
use base64::{engine::general_purpose, Engine as _};
use nexus_substrate::vector_bridge;
use reqwest::Client;
use std::sync::Arc;

pub struct EvolutionEngine;

impl EvolutionEngine {
    pub async fn evolve_and_memorize(
        _client: &Client,
        api_key: &str,
        memory: Arc<MemoryCortex>,
        objective: &str,
        candidate_code: &str,
        test_payload: &[u8],
        func_name: &str,
    ) -> Result<f32> {
        tracing::info!(
            "\x1b[35m>> [EVOLUTION] Synthesizing candidate genome for:\x1b[0m {}",
            objective
        );

        // 1. Synthesize the candidate code into a WASM cell
        let wasm_b64 = WasmForge::synthesize_cell(candidate_code).await?;
        let wasm_bytes = general_purpose::STANDARD.decode(&wasm_b64)?;

        tracing::info!("\x1b[36m>> [EVOLUTION] Candidate compiled. Initiating physical fitness benchmark...\x1b[0m");

        // 2. Evaluate physical fitness (Speed, Memory, Thermal)
        let report = FitnessEvaluator::evaluate_wasm_cell(&wasm_bytes, func_name, test_payload)?;

        tracing::info!(
            "\x1b[33m>> [EVOLUTION] Composite Fitness Score: {:.2}\x1b[0m ({}ms latency, {:.2}KB mem, {:.2}C temp delta)",
            report.composite_score, report.execution_time_ms, report.memory_delta_kb, report.thermal_impact
        );

        // 3. Selection: Only memorize if it meets a high fitness threshold
        if report.composite_score > 100.0 {
            tracing::info!("\x1b[1;32m>> [EVOLUTION] High-Fitness Pattern Detected! Extracting semantic embedding...\x1b[0m");

            // Embed the context
            let embedding_text = format!("OBJECTIVE: {}\nCODE:\n{}", objective, candidate_code);
            let embedding = vector_bridge::get_embedding(api_key, &embedding_text).await.ok();

            let insight = format!(
                "[HIGH_FITNESS_PATTERN] Score: {:.2} | Algorithm successfully optimized for ARM64 constraints. \nCODE_SNIPPET:\n{}",
                report.composite_score,
                &candidate_code[..candidate_code.len().min(500)]
            );

            // Store in Memory Palace
            let mem = memory;
            mem.store_memory("EvolutionEngine", objective, &insight, embedding).await?;

            tracing::info!("\x1b[1;36m>> [EVOLUTION] Genetic material successfully committed to VectorVault.\x1b[0m");
        } else {
            tracing::warn!("\x1b[90m>> [EVOLUTION] Candidate failed to meet fitness threshold. Discarding genome.\x1b[0m");
        }

        Ok(report.composite_score)
    }
}

use crate::memory_cortex::MemoryCortex;
use crate::orchestrator_v2::task_graph::TaskGraph;
use std::collections::HashMap;
use std::sync::Arc;
use tokio::sync::mpsc;
use uuid::Uuid;

#[derive(Debug, Clone, PartialEq, serde::Serialize, serde::Deserialize)]
pub enum ExecutiveGoalStatus {
    Active,
    Suspended,
    Completed,
    Abandoned,
}

#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
pub struct GoalNode {
    pub id: Uuid,
    pub objective: String,
    pub belief_score: f32,
    pub status: ExecutiveGoalStatus,
    pub children: Vec<Uuid>,
}

#[derive(Debug, Clone)]
pub enum ExecutiveCommand {
    InjectGoal { objective: String, parent_id: Option<Uuid> },
    EvaluateEnvironment,
    EmergencyHalt { reason: String },
}

pub struct ExecutiveCortex {
    goals: HashMap<Uuid, GoalNode>,
    active_graph: Option<TaskGraph>,
    memory: Arc<MemoryCortex>,
    command_rx: mpsc::Receiver<ExecutiveCommand>,
}

impl ExecutiveCortex {
    pub fn new(memory: Arc<MemoryCortex>) -> (Self, mpsc::Sender<ExecutiveCommand>) {
        let (tx, rx) = mpsc::channel(100);
        (Self { goals: HashMap::new(), active_graph: None, memory, command_rx: rx }, tx)
    }

    pub async fn run_actor_loop(mut self) {
        // Run a pulse every 10 seconds for persistent state checking
        let mut heartbeat = tokio::time::interval(std::time::Duration::from_secs(10));

        loop {
            tokio::select! {
                Some(cmd) = self.command_rx.recv() => {
                    self.handle_command(cmd).await;
                }
                _ = heartbeat.tick() => {
                    self.cognitive_tick().await;
                }
            }
        }
    }

    async fn cognitive_tick(&mut self) {
        let active_goals: Vec<&GoalNode> =
            self.goals.values().filter(|g| g.status == ExecutiveGoalStatus::Active).collect();

        if active_goals.is_empty() {
            return;
        }

        tracing::info!(
            "[EXECUTIVE CORTEX] Pulse check: Maintaining intent vector for {} active goal(s)...",
            active_goals.len()
        );

        for goal in active_goals {
            let graph_json = serde_json::to_string(&self.active_graph).unwrap_or_default();
            let status_str = format!("{:?}", goal.status);
            let created_at = std::time::SystemTime::now()
                .duration_since(std::time::UNIX_EPOCH)
                .unwrap_or_default()
                .as_secs() as i64;

            // Persist the active intention to SQLite, allowing it to survive reboots
            let _ = self
                .memory
                .upsert_plan_run(
                    &goal.id.to_string(),
                    &goal.objective,
                    &graph_json,
                    &status_str,
                    created_at,
                )
                .await;
        }
    }

    pub async fn evaluate_belief_state(
        &self,
        belief_id: uuid::Uuid,
        confidence: f32,
    ) -> anyhow::Result<crate::belief_policy::BeliefDecision> {
        let decision = crate::belief_policy::BeliefDecisionPolicy::evaluate(belief_id, confidence);

        self.memory.store_belief_decision(decision.clone()).await?;

        Ok(decision)
    }

    pub async fn apply_belief_decision(
        &mut self,
        goal_id: uuid::Uuid,
        decision: crate::belief_policy::BeliefDecision,
    ) -> anyhow::Result<crate::goal_adaptation::GoalAdaptationDecision> {
        let adaptation = crate::goal_adaptation::GoalAdaptationEngine::from_belief_action(
            goal_id,
            decision.action,
        );

        match adaptation.action {
            crate::goal_adaptation::GoalAdaptationAction::PauseGoal => {
                if let Some(goal) = self.goals.get_mut(&goal_id) {
                    goal.status = ExecutiveGoalStatus::Suspended;
                }
            }

            crate::goal_adaptation::GoalAdaptationAction::TerminateGoal => {
                if let Some(goal) = self.goals.get_mut(&goal_id) {
                    goal.status = ExecutiveGoalStatus::Abandoned;
                }
            }

            _ => {}
        }

        self.memory.store_goal_adaptation(adaptation.clone()).await?;

        Ok(adaptation)
    }

    async fn handle_command(&mut self, cmd: ExecutiveCommand) {
        match cmd {
            ExecutiveCommand::InjectGoal { objective, parent_id } => {
                let id = Uuid::new_v4();
                tracing::info!(
                    "\x1b[35m[EXECUTIVE CORTEX] Injecting new background goal: {}\x1b[0m",
                    objective
                );

                let goal = GoalNode {
                    id,
                    objective,
                    belief_score: 0.5, // Seed with standard prior
                    status: ExecutiveGoalStatus::Active,
                    children: vec![],
                };
                self.goals.insert(id, goal);

                if let Some(pid) = parent_id {
                    if let Some(parent) = self.goals.get_mut(&pid) {
                        parent.children.push(id);
                    }
                }
            }
            ExecutiveCommand::EmergencyHalt { reason } => {
                tracing::error!(
                    "\x1b[31m[EXECUTIVE CORTEX] GLOBAL HALT TRIGGERED: {}\x1b[0m",
                    reason
                );
                self.goals.values_mut().for_each(|g| g.status = ExecutiveGoalStatus::Suspended);
            }
            ExecutiveCommand::EvaluateEnvironment => {
                tracing::info!("[EXECUTIVE CORTEX] Environment evaluation requested.");
            }
        }
    }
}

fn canonicalize_or_parent(p: &Path) -> Result<PathBuf> {
    if let Ok(c) = std::fs::canonicalize(p) {
        return Ok(c);
    }
    let parent = p.parent().ok_or_else(|| anyhow!("no parent directory for {}", p.display()))?;
    let parent_canon = std::fs::canonicalize(parent)?;
    let file_name = p.file_name().ok_or_else(|| anyhow!("no file name for {}", p.display()))?;
    Ok(parent_canon.join(file_name))
}

pub fn sanitize(path: &str) -> Result<PathBuf> {
    if path.starts_with("/tmp/") {
        return Ok(Path::new(path).to_path_buf());
    }

    let workspace_root = std::env::current_dir()
        .and_then(std::fs::canonicalize)
        .map_err(|e| anyhow!("PATH_TRAVERSAL_DENIED: could not resolve workspace root ({})", e))?;

    let mut normalized_path = path.to_string();

    if let Some(ws_str) = workspace_root.to_str() {
        if normalized_path.starts_with(ws_str) {
            normalized_path =
                normalized_path.trim_start_matches(ws_str).trim_start_matches('/').to_string();
        }
    }

    if normalized_path.starts_with("src/") {
        normalized_path = format!("crates/runtime_core/{}", normalized_path);
    } else if normalized_path.starts_with("runtime_core/")
        || normalized_path.starts_with("nexus_substrate/")
        || normalized_path.starts_with("nexus_model_hub/")
        || normalized_path.starts_with("nexus_search/")
        || normalized_path.starts_with("memory_graph/")
        || normalized_path.starts_with("protocol/")
        || normalized_path.starts_with("executive_cortex/")
    {
        normalized_path = format!("crates/{}", normalized_path);
    } else if normalized_path.starts_with("ide_backend/")
        || normalized_path.starts_with("cli/")
        || normalized_path.starts_with("node/")
    {
        normalized_path = format!("bins/{}", normalized_path);
    }

    let p = Path::new(&normalized_path);
    let target = if p.is_absolute() { p.to_path_buf() } else { workspace_root.join(p) };

    if !target.exists() {
        let crate_fallback = workspace_root.join("crates/runtime_core/src").join(p);
        if crate_fallback.exists() {
            return Ok(crate_fallback);
        }

        if let Some(filename) = p.file_name().and_then(|f| f.to_str()) {
            let lower = filename.to_lowercase();
            if let Ok(entries) = std::fs::read_dir(&workspace_root) {
                for entry in entries.flatten() {
                    if entry.file_name().to_string_lossy().to_lowercase() == lower {
                        return Ok(entry.path());
                    }
                }
            }
        }

        // NOTE: deliberately no longer falling back to the bare `crates/`
        // directory here. A genuinely missing file (e.g. a new file the
        // agent hasn't written yet) used to silently resolve to that
        // directory instead of a clear not-found error -- WriteFile's
        // caller then tried to write file content INTO the crates/
        // directory itself, producing a confusing "Is a directory (os
        // error 21)" instead of a message that says the path is new.
        // Confirmed live: this exact fallback caused every write_file
        // attempt at a new top-level file to fail this way. Return a
        // clear not-found error instead; WriteFile has its own path
        // resolution for new files and does not call sanitize() at all.
        return Err(anyhow!(
            "PATH_NOT_FOUND: {} does not exist in the workspace (tried {}, crate-relative, and filename search)",
            path,
            target.display()
        ));
    }

    let canonical =
        canonicalize_or_parent(&target).map_err(|_| anyhow!("PATH_TRAVERSAL_DENIED: {}", path))?;

    // THE PHASE 2 FIX: Strict bounds checking against the workspace root
    if !canonical.starts_with(&workspace_root) {
        return Err(anyhow!("PATH_TRAVERSAL_DENIED: {} resolves outside the workspace root", path));
    }

    Ok(canonical)
}

use crate::wasm_runtime::WasmRuntime;
use anyhow::Result;
use std::time::Instant;
use sysinfo::{CpuRefreshKind, MemoryRefreshKind, RefreshKind, System};

#[derive(Debug, Clone)]
pub struct FitnessReport {
    pub execution_time_ms: u128,
    pub memory_delta_kb: f32,
    pub thermal_impact: f32,
    pub composite_score: f32,
}

pub struct FitnessEvaluator;

impl FitnessEvaluator {
    pub fn evaluate_wasm_cell(
        wasm_bytes: &[u8],
        func_name: &str,
        payload: &[u8],
    ) -> Result<FitnessReport> {
        let mut sys = System::new_with_specifics(
            RefreshKind::new()
                .with_cpu(CpuRefreshKind::everything())
                .with_memory(MemoryRefreshKind::everything()),
        );
        sys.refresh_all();

        // 1. Capture ARM64 Baseline Metrics
        let mem_before = sys.used_memory() as f32 / 1024.0;
        let temp_before = std::fs::read_to_string("/sys/class/thermal/thermal_zone0/temp")
            .unwrap_or_else(|_| "45000".to_string())
            .trim()
            .parse::<f32>()
            .unwrap_or(45000.0)
            / 1000.0;

        // 2. Execute the Generated WASM Payload
        let start = Instant::now();
        let _result = WasmRuntime::run_with_buffer(wasm_bytes, func_name, payload)?;
        let duration = start.elapsed().as_millis();

        // 3. Capture Post-Execution Metrics
        sys.refresh_all();
        let mem_after = sys.used_memory() as f32 / 1024.0;
        let temp_after = std::fs::read_to_string("/sys/class/thermal/thermal_zone0/temp")
            .unwrap_or_else(|_| "45000".to_string())
            .trim()
            .parse::<f32>()
            .unwrap_or(45000.0)
            / 1000.0;

        // Ensure we don't divide by zero
        let memory_delta = (mem_after - mem_before).max(0.1);
        let thermal_delta = (temp_after - temp_before).max(0.1);

        // 4. Calculate Mathematical Fitness Score (Higher is Better)
        // Inverse relations: Lower latency/memory/heat yields a higher score
        // Weights: 50% Speed, 30% Memory Efficiency, 20% Thermal Stability
        let speed_score = 1000.0 / (duration as f32 + 1.0);
        let mem_score = 1000.0 / memory_delta;
        let heat_score = 1000.0 / thermal_delta;

        let composite_score = (speed_score * 0.5) + (mem_score * 0.3) + (heat_score * 0.2);

        Ok(FitnessReport {
            execution_time_ms: duration,
            memory_delta_kb: memory_delta,
            thermal_impact: thermal_delta,
            composite_score,
        })
    }
}

impl MerkleLeaf {
    pub fn new(action: AgentAction, actor_id: &str) -> Self {
        let ts = SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_secs();
        let mut hasher = DefaultHasher::new();

        actor_id.hash(&mut hasher);
        ts.hash(&mut hasher);
        Self { action, actor_id: actor_id.to_string(), timestamp: ts, hash: hasher.finish() }
    }
}

#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct GovernanceLog {
    pub history: Vec<MerkleLeaf>,
    pub root_hash: u64,
}

impl GovernanceLog {
    pub fn init() -> Self {
        Self::default()
    }

    pub fn init_shared() -> Arc<Mutex<Self>> {
        Arc::new(Mutex::new(Self::init()))
    }

    /// record: Chains a new leaf into the Merkle root.
    /// Root = Hash(OldRoot ++ NewLeaf)
    pub fn record(&mut self, action: AgentAction, actor_id: &str) {
        let leaf = MerkleLeaf::new(action, actor_id);

        let mut hasher = DefaultHasher::new();
        self.root_hash.hash(&mut hasher);
        leaf.hash.hash(&mut hasher);
        self.root_hash = hasher.finish();

        self.history.push(leaf);

        if self.history.len().is_multiple_of(10) {
            info!(
                "[34m[GOVERNANCE] Checkpoint: Root=0x{:x} (Size: {})[0m",
                self.root_hash,
                self.history.len()
            );
        }
    }

    pub fn root(&self) -> u64 {
        self.root_hash
    }

    pub fn len(&self) -> usize {
        self.history.len()
    }

    pub fn is_empty(&self) -> bool {
        self.history.is_empty()
    }
}

// Process-wide persistent governance log. Callers should use this
// instead of constructing their own GovernanceLog, or records get
// discarded along with the throwaway instance.
pub static GOVERNANCE: LazyLock<Mutex<GovernanceLog>> =
    LazyLock::new(|| Mutex::new(GovernanceLog::init()));

/// Public shared JSON string escape sanitizer.
pub fn sanitize_json_escapes(json_str: &str) -> String {
    let mut sanitized = String::with_capacity(json_str.len() + 128);
    let mut in_string = false;
    let mut last_char = ' ';

    for c in json_str.chars() {
        match c {
            '"' if last_char != '\\' => {
                in_string = !in_string;
                sanitized.push('"');
            }
            '\n' if in_string => sanitized.push_str("\\n"),
            '\r' if in_string => sanitized.push_str("\\r"),
            '\t' if in_string => sanitized.push_str("\\t"),
            '\\' => {
                sanitized.push('\\');
            }
            _ => sanitized.push(c),
        }
        last_char = c;
    }
    sanitized
}

pub fn extract_actions(response: &str) -> Result<Vec<AgentAction>> {
    let text = response.trim();
    if text.is_empty() || text == "[]" { return Ok(vec![]); }

    // 1. Aggressively carve out code blocks
    let mut extracted = text;
    if let Some(start) = text.find("```json") {
        if let Some(end) = text[start + 7..].find("```") {
            extracted = &text[start + 7 .. start + 7 + end];
        }
    } else if let Some(start) = text.find("```") {
        if let Some(end) = text[start + 3..].find("```") {
            extracted = &text[start + 3 .. start + 3 + end];
        }
    }

    extracted = extracted.trim();

    // 2. Locate boundaries
    let json_str = if let (Some(s), Some(e)) = (extracted.find('['), extracted.rfind(']')) {
        &extracted[s..=e]
    } else if let (Some(s), Some(e)) = (extracted.find('{'), extracted.rfind('}')) {
        &extracted[s..=e]
    } else { return Err(anyhow::anyhow!("No JSON boundaries found in model output.")); };

    let clean_text = sanitize_json_escapes(json_str);

    // ANTI-DEGENERACY GATE
    if clean_text.contains("00000000000000000000") || clean_text.len() > 25_000 {
        tracing::error!("[CORTEX PARSE ABORT] Token Resonance / Runaway block. Returning safe halt.");
        return Ok(vec![]);
    }

    // 3. Serializer & Transparent Fallback Log
    
    // ⚡ ROBUST Auto-normalize arbitrary tool schemas to Athena AgentAction array
    let clean_text = if let Ok(mut val) = serde_json::from_str::<serde_json::Value>(&clean_text) {
        let process_obj = |obj: &mut serde_json::Map<String, serde_json::Value>| {
            // Map "tool" -> "type"
            if let Some(tool_val) = obj.remove("tool") {
                obj.insert("type".to_string(), tool_val);
            }
            // Flatten "args"
            if let Some(args_val) = obj.remove("args") {
                if let Some(args_obj) = args_val.as_object() {
                    for (k, v) in args_obj {
                        obj.insert(k.clone(), v.clone());
                    }
                }
            }
            // Flatten "arguments" (alias)
            if let Some(args_val) = obj.remove("arguments") {
                if let Some(args_obj) = args_val.as_object() {
                    for (k, v) in args_obj {
                        obj.insert(k.clone(), v.clone());
                    }
                }
            }
            // Map "bash_command" -> "command" (alias)
            if let Some(cmd) = obj.remove("bash_command") {
                obj.insert("command".to_string(), cmd);
            }
        };

        if let Some(arr) = val.as_array_mut() {
            for item in arr.iter_mut() {
                if let Some(obj) = item.as_object_mut() {
                    process_obj(obj);
                }
            }
            val.to_string()
        } else if let Some(obj) = val.as_object_mut() {
            process_obj(obj);
            serde_json::Value::Array(vec![val.clone()]).to_string()
        } else {
            clean_text
        }
    } else {
        clean_text
    };
    
    match serde_json::from_str::<Vec<AgentAction>>(&clean_text) {
        Ok(actions) => Ok(actions),
        Err(e1) => {
            match serde_json::from_str::<AgentAction>(&clean_text) {
                Ok(single) => Ok(vec![single]),
                Err(e2) => {
                    let snippet = if clean_text.len() > 400 { &clean_text[..400] } else { &clean_text };
                    tracing::warn!(
                        "[JSON PARSER] Serialization Failure!\nArray Err: {}\nSingle Err: {}\n--- OFFENDING SNIPPET ---\n{}\n-------------------------", 
                        e1, e2, snippet
                    );
                    Err(anyhow::anyhow!("JSON serialization failed. Invalid format."))
                }
            }
        }
    }
}

    pub path: String,

    pub directive: String,
    pub timestamp: String,
}

const CATALOG: &str = ".nexus_snapshots.json";

pub async fn list_snapshots() -> Result<Vec<Snapshot>> {
    if !Path::new(CATALOG).exists() {
        return Ok(vec![]);
    }
    let raw = fs::read_to_string(CATALOG).await?;
    let snapshots: Vec<Snapshot> = serde_json::from_str(&raw)?;
    Ok(snapshots)
}

pub async fn take_snapshot(path: &str, directive: &str) -> Result<Snapshot> {
    let content = fs::read(path).await?;
    let hash = fnv_hex(&content);
    let snap_path = format!("{}.{}.snap", path, &hash[..8]);
    fs::write(&snap_path, &content).await?;

    let snap = Snapshot {
        id: uuid_v4(),
        path: path.to_string(),
        hash: hash.clone(),
        directive: directive.to_string(),
        timestamp: epoch_secs().to_string(),
    };

    let mut catalog = list_snapshots().await.unwrap_or_default();
    catalog.push(snap.clone());
    if catalog.len() > 200 {
        catalog.remove(0);
    }
    fs::write(CATALOG, serde_json::to_string_pretty(&catalog)?).await?;
    Ok(snap)
}

fn fnv_hex(data: &[u8]) -> String {
    let mut h: u64 = 14695981039346656037;
    for &b in data {
        h ^= b as u64;
        h = h.wrapping_mul(1099511628211);
    }
    format!("{:016x}", h)
}

fn epoch_secs() -> u64 {
    std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).unwrap_or_default().as_secs()
}

fn uuid_v4() -> String {
    format!(
        "{:x}",
        std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).unwrap().as_nanos()
    )
}

use serde::de::DeserializeOwned;

pub fn safe_parse_json<T: DeserializeOwned>(raw: &str) -> Result<T, serde_json::Error> {
    // 1. Strip rogue Markdown formatting often hallucinated by models
    let mut s = raw.trim();
    if s.starts_with("```json") {
        s = s.trim_start_matches("```json").trim_end_matches("```").trim();
    } else if s.starts_with("```") {
        s = s.trim_start_matches("```").trim_end_matches("```").trim();
    }

    // 2. Scan and repair invalid escape sequences (e.g., regex inside JSON strings)
    let mut fixed = String::with_capacity(s.len() + 10);
    let mut chars = s.chars().peekable();
    
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(&next_c) = chars.peek() {
                match next_c {
                    'n' | 't' | 'r' | '"' | '\\' | '/' | 'u' | 'b' | 'f' => {
                        // Valid JSON escape sequence
                        fixed.push('\\');
                    },
                    _ => {
                        // Invalid escape sequence detected. Double-escape it to protect the parser.
                        fixed.push('\\');
                        fixed.push('\\'); 
                    }
                }
            } else {
                // Trailing backslash
                fixed.push('\\');
                fixed.push('\\');
            }
        } else {
            fixed.push(c);
        }
    }
    
    // 3. Attempt to parse the repaired string, fallback to original if structural failure
    serde_json::from_str(&fixed).or_else(|_| serde_json::from_str(s))
}

use tracing::warn;

pub struct LogicEngine;

impl LogicEngine {
    pub fn linearize_handoff(
        action: &crate::agent_protocol::AgentAction,
        result: crate::agent_protocol::SubagentReturn,
    ) -> String {
        format!(
            "[SYSTEM: DELEGATION REPORT]\nAction: {:?}\nStatus: {}\nOutput: {}\n[DIRECTIVE]: Matriarch, resume execution based on this report.",
            action, result.status, result.output
        )
    }

    pub fn is_action_safe(directive: &str) -> bool {
        let forbidden =
            ["rm -rf /", "mkfs", "shutdown", "chmod 777", "dd if=/dev/zero", "> /dev/sda"];
        let d_lower = directive.to_lowercase();
        !forbidden.iter().any(|&p| d_lower.contains(p))
    }

    pub fn detect_deadlock(observations: &[String]) -> bool {
        if observations.len() < 3 {
            return false;
        }

        let mut read_counts: HashMap<String, usize> = HashMap::new();

        for obs in observations {
            if let Some(after) = obs.strip_prefix("[READ_OK] ") {
                let path = after.split_whitespace().next().unwrap_or("").to_string();
                if !path.is_empty() {
                    *read_counts.entry(path).or_insert(0) += 1;
                }
            }
        }

        for (path, count) in read_counts {
            if count >= 3 {
                let file_progress = observations.iter().any(|obs| {
                    (obs.contains("[WRITE_OK]")
                        || obs.contains("[DIFF_PATCH_OK]")
                        || obs.contains("[PATCH_OK]"))
                        && obs.contains(&path)
                });

                if !file_progress {
                    warn!(
                        "[LOGIC] Localized Deadlock: {} read {} times with zero progress.",
                        path, count
                    );
                    return true;
                }
            }
        }

        let mut last_cmd_sig = String::new();
        let mut cmd_repeat_count = 0;

        for obs in observations {
            if let Some(start) = obs.find("[CMD:") {
                let sig: String = obs[start..]
                    .lines()
                    .filter(|line| !line.contains("Finished") && !line.contains("Blocking waiting"))
                    .collect::<Vec<&str>>()
                    .join("\n");

                if sig == last_cmd_sig && !sig.is_empty() {
                    cmd_repeat_count += 1;
                } else {
                    last_cmd_sig = sig;
                    cmd_repeat_count = 1;
                }
            }
        }

        if cmd_repeat_count >= 2 {
            warn!("[LOGIC] Deadlock: Time-invariant command repetition detected.");
            return true;
        }
        false
    }
}

/// Calculates the nth Fibonacci number using checked arithmetic to prevent overflow.
/// Returns None if the result exceeds the capacity of u128.
pub fn fibonacci(n: u32) -> Option<u128> {
    if n == 0 {
        return Some(0);
    }
    if n == 1 {
        return Some(1);
    }

    let mut a: u128 = 0;
    let mut b: u128 = 1;

    for _ in 2..=n {
        let next = a.checked_add(b)?;
        a = b;
        b = next;
    }

    Some(b)
}

                }
            }
        }
    }

pub struct MemoryConsolidator;

impl MemoryConsolidator {
    pub fn new() -> Self {
        Self
    }

    pub fn calculate_retention(memory: &CognitiveMemory) -> f32 {
        let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs() as i64;

        let age = (now - memory.created_at).max(0) as f32;

        let decay = (-age / 86400.0).exp();

        let score = memory.importance * memory.confidence * decay;

        score.clamp(0.0, 1.0)
    }

    pub fn should_consolidate(memory: &CognitiveMemory) -> bool {
        match memory.kind {
            MemoryKind::Failure | MemoryKind::Reflection | MemoryKind::Goal => true,

            _ => Self::calculate_retention(memory) > 0.5,
        }
    }
}

    fn default() -> Self {
        Self::new()
    }

#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
pub struct WorkingMemory {
    pub active_node: uuid::Uuid,
    pub required_objectives: Vec<String>,
    pub completed_objectives: Vec<String>,
}

impl WorkingMemory {
    pub fn new(active_node: uuid::Uuid, objectives: Vec<String>) -> Self {
        Self { active_node, required_objectives: objectives, completed_objectives: Vec::new() }
    }

    pub fn mark_completed(&mut self, objective: &str) {
        if let Some(pos) = self.required_objectives.iter().position(|x| x == objective) {
            let obj = self.required_objectives.remove(pos);
            self.completed_objectives.push(obj);
        }
    }

    pub fn get_prompt_context(&self) -> String {
        format!(
            "[WORKING MEMORY]\nPending Objectives: {:?}\nCompleted Objectives: {:?}",
            self.required_objectives, self.completed_objectives
        )
    }

    pub fn is_mission_accomplished(&self) -> bool {
        self.required_objectives.is_empty()
    }
}

#[derive(Debug, Clone)]
pub struct CachedAction {
    pub output: String,
    pub exit_status: i32,
    pub timestamp: u64,
}

#[derive(Debug, Default)]
pub struct ActionCache {
    cache: HashMap<String, CachedAction>,
}

impl ActionCache {
    pub fn new() -> Self {
        Self { cache: HashMap::new() }
    }

    pub fn generate_key(command: &str, ast_hash: &str) -> String {
        let mut s = DefaultHasher::new();
        command.hash(&mut s);
        ast_hash.hash(&mut s);
        s.finish().to_string()
    }

    pub fn get(&self, key: &str) -> Option<&CachedAction> {
        self.cache.get(key)
    }

    pub fn insert(&mut self, key: String, output: String, exit_status: i32) {
        let timestamp = SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_secs();
        self.cache.insert(key, CachedAction { output, exit_status, timestamp });
    }
}

            task_failures: AtomicU64::new(0),
            arbiter_blocks: AtomicU64::new(0),
            heal_successes: AtomicU64::new(0),

            api_latency_ms_sum: AtomicU64::new(0),
            api_calls: AtomicU64::new(0),
            circuit_opens: AtomicU64::new(0),
        }
    }

    pub fn inc_task(&self) {
        tracing::info!("Task counter incremented");
        self.tasks_total.fetch_add(1, Ordering::Relaxed);
    }

    pub fn inc_failure(&self) {
        self.task_failures.fetch_add(1, Ordering::Relaxed);
    }

    pub fn inc_arbiter_block(&self) {
        self.arbiter_blocks.fetch_add(1, Ordering::Relaxed);
    }

    pub fn inc_heal_success(&self) {
        self.heal_successes.fetch_add(1, Ordering::Relaxed);
    }

    pub fn inc_heal_failure(&self) {
        self.heal_failures.fetch_add(1, Ordering::Relaxed);
    }

    pub fn inc_circuit_open(&self) {
        self.circuit_opens.fetch_add(1, Ordering::Relaxed);
    }

    pub fn record_api_call(&self, latency_ms: u128) {
        self.api_calls.fetch_add(1, Ordering::Relaxed);
        self.api_latency_ms_sum.fetch_add(latency_ms as u64, Ordering::Relaxed);
    }

    pub fn avg_latency_ms(&self) -> u64 {
        let calls = self.api_calls.load(Ordering::Relaxed);
        self.api_latency_ms_sum.load(Ordering::Relaxed).checked_div(calls).unwrap_or(0)
    }
}

    let now = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .unwrap_or_default()
        .as_secs();
    let cooldown = 60;

    if model_name.contains("gemini") || model_name.contains("gemma") {
        GEMINI_DISABLED_UNTIL.store(now + cooldown, Ordering::Relaxed);
        tracing::warn!("[CIRCUIT BREAKER] Gemini rate-limited (429). Bypassing for 60s.");
    } else if model_name.contains("nvidia") || model_name.contains("nemotron") {
        NVIDIA_DISABLED_UNTIL.store(now + cooldown, Ordering::Relaxed);
        tracing::warn!("[CIRCUIT BREAKER] NVIDIA rate-limited (429). Bypassing for 60s.");
    }
}

use crate::agent_protocol::AgentAction;
use crate::intent_contract::IntentContract;
use crate::policy::Capability;
use serde::{Deserialize, Serialize};
use uuid::Uuid;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum MotorAction {
    ExecuteCommand { command: String },
    WriteFile { path: String, content: String },
    ReadFile { path: String },
    ReadFileChunk { path: String, start_line: usize, max_lines: usize },
    SearchCode { path: String, query: String },
    PatchFile {
        path: String,
        target: String,
        replacement: String,
    },
    SearchMemory { query: String },
    ReplaceLines { path: String, start_line: usize, end_line: usize, replacement: String },
    RunTests { test_target: String },
    DiffPatch { path: String, search: String, replace: String },
    PatchFunction { path: String, function_name: String, new_body: String },
    Halt { reasoning: String },
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct MotorCommand {
    pub command_id: Uuid,
    pub intent_id: Uuid,
    pub goal_id: Option<Uuid>,
    pub action: MotorAction,
    pub capability: Capability,
    pub expected_outcome: String,
    pub created_at: u64,
}

pub struct MotorCortex;

impl MotorCortex {
    pub fn translate(
        intent: &IntentContract,
        goal_id: Option<Uuid>,
        action: AgentAction,
        capability: Capability,
        expected_outcome: String,
    ) -> MotorCommand {
        let motor_action = match action {
            AgentAction::ExecuteCommand { command, .. } => MotorAction::ExecuteCommand { command },
            AgentAction::WriteFile { path, content, .. } => {
                MotorAction::WriteFile { path, content }
            }
            AgentAction::ReadFile { path, .. } => MotorAction::ReadFile { path },
            AgentAction::ReadFileChunk { path, start_line, max_lines, .. } => {
                MotorAction::ReadFileChunk { path, start_line, max_lines }
            }
            AgentAction::SearchCode { path, query, .. } => MotorAction::SearchCode { path, query },
            AgentAction::SearchMemory { query, .. } => MotorAction::SearchMemory { query },
            AgentAction::PatchFile { path, target, replacement, .. } => MotorAction::PatchFile { path, target, replacement },
            AgentAction::ReplaceLines { path, start_line, end_line, replacement, .. } => {
                MotorAction::ReplaceLines { path, start_line, end_line, replacement }
            }
            AgentAction::RunTests { test_target, .. } => MotorAction::RunTests { test_target },
            AgentAction::DiffPatch { path, search, replace, .. } => {
                MotorAction::DiffPatch { path, search, replace }
            }
            AgentAction::PatchFunction { path, function_name, new_body, .. } => {
                MotorAction::PatchFunction { path, function_name, new_body }
            }
            AgentAction::Halt { reasoning } => MotorAction::Halt { reasoning },
            _ => MotorAction::Halt { reasoning: "Fallback halt for unmapped action".to_string() },
        };

        let now = std::time::SystemTime::now()
            .duration_since(std::time::UNIX_EPOCH)
            .unwrap_or_default()
            .as_secs();

        MotorCommand {
            command_id: Uuid::new_v4(),
            intent_id: intent.id,
            goal_id,
            action: motor_action,
            capability,
            expected_outcome,
            created_at: now,
        }
    }
}

        Self::wrap(Observation::PlannerTimeout)
    }

    pub fn from_task_completion(task_id: &str, output: &str) -> ObservationEnvelope {

    pub fn new() -> Self {
        Self {}
    }

    pub async fn run(
        &self,
        prompt: String,
        ctx: ContextPacket,
        client: &Client,
        api_key: &str,
        tx: Option<broadcast::Sender<String>>,
    ) -> Result<String, String> {
        METRICS.api_calls.fetch_add(1, Ordering::SeqCst);

        let mut active_ctx = ctx;

        // Only classify and assign defaults if properties are not already defined
        if active_ctx.system_prompt.is_none() || active_ctx.model_override.is_none() {

        match agent.prepare_and_run(client, api_key, &prompt, active_ctx).await {

use crate::orchestrator_v2::task_graph::{TaskExecutionState, TaskGraph, TaskNode, TaskStatus};
use anyhow::Result;
use reqwest::Client;
use std::collections::HashMap;

pub struct PlannerAgent;

impl PlannerAgent {
    pub async fn decompose_task(
        client: &Client,
        api_key: &str,
        directive: &str,
    ) -> Result<TaskGraph> {

    }

    pub async fn generate_plan(
        client: &Client,
        api_key: &str,
        directive: &str,
    ) -> Result<TaskGraph> {
        let prompt = format!(
            "You are the Lead DAG Planner Agent for AthenaFusionX.\n\
             Decompose the user DIRECTIVE into a TaskGraph JSON object with a 'nodes' map.\n\
             CRITICAL RULE [GRANULARITY]: YOU MUST BREAK COMPLEX DIRECTIVES INTO AT LEAST 3 DISTINCT NODES. A single node 'task_1' is a FAILURE. If auditing or analyzing, create separate nodes for 'discovery', 'analysis', and 'reporting'. If modifying files, create one node per file.
             CRITICAL RULE [JSON ESCAPING]: Do NOT use raw newlines inside JSON strings. Use \n.\n\
             CRITICAL RULE [ROUTER LOCATION]: The Executive Cortex router is located at 'crates/runtime_core/src/router.rs'. Always use 'crates/runtime_core/src/router.rs' in task objectives when referring to the router.\n\
             CRITICAL RULE [DEPENDENCY INFERENCE]: If one task reads, parses, or consumes a file or resource that another task generates, writes, or produces, the consuming task's 'deps' array MUST include the producing task's id. Never schedule a read of a path before the task that creates it.\n\
             CRITICAL RULE [TERMINAL STEPS]: A directive step that is purely a control-flow or meta instruction (e.g. \"halt\", \"stop\", \"done\", \"finish execution\", \"end mission\") is NOT real work and MUST NOT become its own node. Omit it from the graph entirely -- mission completion is handled automatically once all real work nodes finish.\n\
             CRITICAL RULE [CONDITIONAL REMEDIATION]: Do not create a separate downstream node whose entire purpose is to \"fix\", \"remediate\", or \"correct\" something that a prior diagnostic node has not yet confirmed exists. When a directive says something like \"check X, then fix any issues found\" or \"run tests and correct all errors\", scope that as a SINGLE node (e.g. \"check X and fix any issues found\") rather than splitting diagnosis and remediation into two nodes -- the outcome depends on what the diagnostic step actually discovers, and a separate node presupposes there is necessarily something to fix.\n\
             DIRECTIVE: {}\n\
             OUTPUT ONLY VALID JSON with this exact structure:\n\
             {{\n  \"nodes\": {{\n    \"task_1\": {{\n      \"id\": \"task_1\",\n      \"deps\": [],\n      \"required_agent\": \"CodeAnalyzer\",\n      \"objective\": \"...\"\n    }}\n  }}\n}}",
            directive
        );

        let payload = serde_json::json!({
            "contents": [{"role": "user", "parts": [{"text": prompt}]}],
            "generationConfig": { "temperature": 0.1, "maxOutputTokens": 2048 }
        });

        let text_res = crate::epistemic_engine::EpistemicEngine::safe_api_call(
            client,
            api_key,
            &crate::model_config::get_heavy_models(),
            &payload,
        )
        .await;

        match text_res {
            Ok(text) => match Self::parse_task_graph(&text, directive) {
                Ok(graph) if !graph.nodes.is_empty() => Ok(graph),

            },
            Err(_) => Ok(Self::create_fallback_graph(directive)),
        }

    fn parse_task_graph(text: &str, directive: &str) -> Result<TaskGraph> {
        let clean = text.replace("```json", "").replace("```", "").trim().to_string();

        if let Ok(mut graph) = serde_json::from_str::<TaskGraph>(&clean) {
            Self::strip_terminal_nodes(&mut graph);
            if !graph.nodes.is_empty() {
                return Ok(graph);
            }
        }

        if let (Some(start), Some(end)) = (clean.find('{'), clean.rfind('}')) {
            if start <= end {
                let slice = &clean[start..=end];
                if let Ok(mut graph) = serde_json::from_str::<TaskGraph>(slice) {
                    Self::strip_terminal_nodes(&mut graph);
                    if !graph.nodes.is_empty() {
                        return Ok(graph);
                    }
                }
            }
        }

        if let Ok(nodes_arr) = serde_json::from_str::<Vec<serde_json::Value>>(&clean) {
            let mut map = HashMap::new();
            for (idx, item) in nodes_arr.into_iter().enumerate() {
                // FIX: Item #13 Unwrap Masking Error patched with proper tracing fallback
                let id = item
                    .get("id")
                    .and_then(|v| v.as_str())
                    .filter(|s| !s.is_empty())
                    .map(ToString::to_string)
                    .unwrap_or_else(|| {
                        let f_id = format!("task_fallback_{}", uuid::Uuid::new_v4().simple());
                        tracing::warn!(
                            "Missing task ID in graph, auto-generating unique fallback: {}",
                            f_id
                        );
                        f_id
                    });

                let obj = item
                    .get("objective")
                    .or_else(|| item.get("description"))
                    .and_then(|v| v.as_str())
                    .unwrap_or(directive)
                    .to_string();

                let lower_obj = obj.trim().to_lowercase();
                if lower_obj == "halt"
                    || lower_obj == "halt."
                    || lower_obj == "halt execution"
                    || lower_obj == "halt execution."
                    || lower_obj.starts_with("halt execution")
                {
                    continue;
                }

                let deps = item
                    .get("deps")
                    .or_else(|| item.get("dependencies"))
                    .and_then(|v| v.as_array())
                    .map(|arr| {
                        arr.iter().filter_map(|x| x.as_str().map(ToString::to_string)).collect()
                    })
                    .unwrap_or_default();
                let agent = item
                    .get("required_agent")
                    .or_else(|| item.get("agent"))
                    .and_then(|v| v.as_str())
                    .unwrap_or("CodeAnalyzer")
                    .to_string();

                map.insert(
                    id.clone(),
                    TaskNode {
                        id,
                        deps,
                        required_agent: agent,
                        status: TaskStatus::Pending,
                        objective: obj,
                        success_criteria: vec![],
                        execution_state: TaskExecutionState::default(),
                        output_evidence: None,
                        priority: idx + 1,
                    },
                );
            }
            if !map.is_empty() {
                return Ok(TaskGraph { nodes: map });
            }
        }

        Ok(Self::create_fallback_graph(directive))
    }

    fn strip_terminal_nodes(graph: &mut TaskGraph) {
        let terminal_ids: Vec<String> = graph
            .nodes
            .iter()
            .filter(|(_, node)| {
                let obj = node.objective.trim().to_lowercase();
                obj == "halt"
                    || obj == "halt."
                    || obj == "halt execution"
                    || obj == "halt execution."
                    || obj.starts_with("halt execution")
                    || obj.starts_with("stop execution")
                    || obj.starts_with("end mission")
            })
            .map(|(id, _)| id.clone())
            .collect();

        for id in terminal_ids {
            graph.nodes.remove(&id);
        }
    }

            },
        );

        TaskGraph { nodes }
    }
}

use crate::orchestrator_v2::task_graph::{TaskGraph, TaskStatus};
use crate::orchestrator_v2::verifier::VerificationResult;

pub struct Scheduler;

/// v1.9 residual-driven priority.
/// Higher residual → expand first. impact/mission_priority weight the mission value.
pub fn residual_priority(
    residual: f32,
    impact: f32,
    mission_priority: f32,
    dependency_weight: f32,
) -> f32 {
    residual * impact * mission_priority * (1.0 + dependency_weight)
}

impl Scheduler {
    const MAX_RETRIES: u8 = 2;

    /// Transitions viable tasks from Pending/Retrying to Ready.
    /// Returns the IDs of tasks that are newly marked as Ready.
    pub fn prepare_ready_tasks(graph: &mut TaskGraph) -> Vec<String> {
        let mut newly_ready = Vec::new();

        // We must clone the keys to avoid mutably borrowing the graph while iterating
        let node_ids: Vec<String> = graph.nodes.keys().cloned().collect();

        for id in node_ids {
            let mut is_ready = false;

            // Check dependency status immutably
            if let Some(node) = graph.nodes.get(&id) {
                if node.status == TaskStatus::Pending || node.status == TaskStatus::Retrying {
                    let all_deps_met = node.deps.iter().all(|dep_id| {
                        graph
                            .nodes
                            .get(dep_id)
                            .map(|n| n.status == TaskStatus::Completed)
                            .unwrap_or(false)
                    });
                    if all_deps_met {
                        is_ready = true;
                    }
                }
            }

            // Apply state transition
            if is_ready {
                if let Some(node) = graph.nodes.get_mut(&id) {
                    node.status = TaskStatus::Ready;
                    newly_ready.push(id.clone());
                }
            }
        }
        newly_ready
    }

    pub fn mark_running(graph: &mut TaskGraph, task_id: &str) {
        if let Some(node) = graph.nodes.get_mut(task_id) {
            if node.status == TaskStatus::Ready {
                node.status = TaskStatus::Running;
            }
        }
    }

    pub fn process_verification(graph: &mut TaskGraph, task_id: &str, result: VerificationResult) {
        if let Some(node) = graph.nodes.get_mut(task_id) {
            match result {
                VerificationResult::Success => {
                    node.status = TaskStatus::Completed;
                }
                VerificationResult::Retry(reason) => {
                    node.execution_state.attempts += 1;
                    node.execution_state.last_error = Some(reason.clone());

                    if node.execution_state.attempts >= Self::MAX_RETRIES {
                        node.status = TaskStatus::Escalated; // Budget exhausted
                    } else {
                        node.status = TaskStatus::Retrying;
                    }
                }
                VerificationResult::Escalate(reason) => {
                    node.execution_state.last_error = Some(reason);
                    node.status = TaskStatus::Escalated;
                }
                VerificationResult::Failure(reason) => {
                    node.execution_state.last_error = Some(reason);
                    node.status = TaskStatus::Failed;
                }
            }
        }

use anyhow::{anyhow, Result};
use serde::{Deserialize, Serialize};
#[allow(unused_imports)]
use std::collections::HashSet;
use std::collections::{HashMap, VecDeque};

#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub enum TaskStatus {
    Pending,
    Running,
    Success,
    Failed,
    Completed,
    Ready,
    Escalated,
    Retrying,
}

#[derive(Clone, Debug, Serialize, Deserialize, Default)]
pub struct TaskExecutionState {
    pub attempts: u8,
    pub last_error: Option<String>,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct TaskNode {
    pub id: String,

    #[serde(alias = "dependencies")]
    pub deps: Vec<String>,

    #[serde(alias = "assigned_agent")]
    pub required_agent: String,

    #[serde(default = "default_status")]
    pub status: TaskStatus,

    #[serde(default)]
    pub objective: String,

    #[serde(default)]
    pub success_criteria: Vec<String>,

    #[serde(default)]
    pub execution_state: TaskExecutionState,

    #[serde(default)]
    pub output_evidence: Option<String>,

    // HEFT Optimization field
    #[serde(default)]
    pub priority: usize,
}

fn default_status() -> TaskStatus {
    TaskStatus::Pending
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct TaskGraph {
    pub nodes: HashMap<String, TaskNode>,
}

impl TaskGraph {
    pub fn new() -> Self {
        Self { nodes: HashMap::new() }
    }

    pub fn add_task(&mut self, id: &str, deps: Vec<String>, required_agent: &str) {
        self.nodes.insert(
            id.to_string(),
            TaskNode {
                id: id.to_string(),
                deps,
                required_agent: required_agent.to_string(),
                status: TaskStatus::Pending,
                objective: String::new(),
                success_criteria: Vec::new(),
                execution_state: TaskExecutionState::default(),
                output_evidence: None,
                priority: 0,
            },
        );
    }

    pub fn compute_critical_path_priorities(&mut self) {
        let mut memo: HashMap<String, usize> = HashMap::new();

        fn b_level(node_id: &str, graph: &TaskGraph, memo: &mut HashMap<String, usize>) -> usize {
            if let Some(&cached) = memo.get(node_id) {
                return cached;
            }
            let mut max_child = 0;

            for (child_id, child) in &graph.nodes {
                if child.deps.contains(&node_id.to_string()) {
                    max_child = max_child.max(b_level(child_id, graph, memo));
                }
            }
            let level = max_child + 1;
            memo.insert(node_id.to_string(), level);
            level
        }

        let keys: Vec<String> = self.nodes.keys().cloned().collect();
        for id in keys {
            let prio = b_level(&id, self, &mut memo);
            if let Some(node) = self.nodes.get_mut(&id) {
                node.priority = prio;
            }
        }
    }

    pub fn resolve_execution_order(&self) -> Result<Vec<String>> {
        for node in self.nodes.values() {
            for dep in &node.deps {
                if !self.nodes.contains_key(dep.as_str()) {
                    return Err(anyhow!(
                        "DAG_ERROR: Task '{}' depends on unknown task '{}'.",
                        node.id,
                        dep
                    ));
                }
            }
        }

        let mut in_degree: HashMap<&str, usize> =
            self.nodes.keys().map(|k| (k.as_str(), 0)).collect();
        let mut dependents: HashMap<&str, Vec<&str>> = HashMap::new();

        for node in self.nodes.values() {
            for dep in &node.deps {
                dependents.entry(dep.as_str()).or_default().push(node.id.as_str());
                *in_degree.entry(node.id.as_str()).or_insert(0) += 1;
            }
        }

        let mut queue: VecDeque<&str> =
            in_degree.iter().filter(|(_, &d)| d == 0).map(|(&id, _)| id).collect();
        let mut order = Vec::with_capacity(self.nodes.len());

        while let Some(id) = queue.pop_front() {
            order.push(id.to_string());
            if let Some(kids) = dependents.get(id) {
                for &child in kids {
                    let deg = in_degree.get_mut(child).unwrap();
                    *deg -= 1;
                    if *deg == 0 {
                        queue.push_back(child);
                    }
                }
            }
        }

        if order.len() != self.nodes.len() {
            let resolved: HashSet<_> = order.iter().map(|s| s.as_str()).collect();
            let cycle_nodes: Vec<_> =
                self.nodes.keys().filter(|k| !resolved.contains(k.as_str())).cloned().collect();
            return Err(anyhow!(
                "DAG_DEADLOCK: Circular dependency detected among: {:?}",
                cycle_nodes
            ));
        }
        Ok(order)
    }
}

impl Default for TaskGraph {
    fn default() -> Self {
        Self::new()
    }
}

pub struct Verifier;

impl Verifier {
    /// LocalDiagnosis Layer: Cheap, deterministic heuristic matching.
    pub fn diagnose_output(stderr: &str, stdout: &str) -> VerificationResult {
        let combined = format!("{} {}", stderr, stdout).to_lowercase();

        // 1. Transient Faults -> Yields a localized Retry
        if combined.contains("timeout")
            || combined.contains("connection reset")
            || combined.contains("rate limit")
            || combined.contains("503 service unavailable")
            || combined.contains("429 too many requests")
        {
            return VerificationResult::Retry("Transient network/API fault detected.".into());
        }

        // 2. Deterministic Faults -> Immediate Escalation
        if combined.contains("error:")
            || combined.contains("cannot find crate")
            || combined.contains("borrowed value")
            || combined.contains("syntax error")
            || combined.contains("test failed")
            || combined.contains("panicked at")
        {
            return VerificationResult::Escalate(
                "Deterministic compilation/logic fault detected.".into(),
            );
        }

        // 3. Ambiguous Anomaly -> Escalate
        if !stderr.trim().is_empty() {
            return VerificationResult::Escalate("Unknown stderr anomaly detected.".into());
        }

        VerificationResult::Success
    }

    pub fn evaluate(expected: &str, observed: &str) -> OutcomeAssessment {
        let success = if expected == observed { 1.0 } else { 0.5 };
        OutcomeAssessment {
            success_probability: success,
            deviation_score: 1.0 - success,
            confidence_delta: success - 0.5,
            lessons: vec![format!("Expected: {} Observed: {}", expected, observed)],
        }
    }

    pub fn verify(_node_id: &str, local_evidence: &str) -> CompletionStatus {

        // Check if the executor captured and appended a serialized TelemetryArtifact
        if let Some(telemetry_json) = extract_telemetry(local_evidence) {
            if telemetry_json.exit_code == 0 && telemetry_json.test_digest.is_some() {
                return CompletionStatus::Approved;
            } else {
                return CompletionStatus::Rejected(
                    format!("Telemetry captured, but verification failed. Exit code: {}, Tests passed: {}", 
                        telemetry_json.exit_code, telemetry_json.test_digest.is_some())
                );
            }
        }

        // Fallback for non-mutation writes
        if local_evidence.contains("[WRITE_OK]") || local_evidence.contains("[CMD_OK]") {
            CompletionStatus::Approved
        } else {
            CompletionStatus::Rejected(
                "Missing verified TelemetryArtifact side-channel or execution marker.".to_string()
            )
        }
    }
}

    for line in evidence.lines() {
        if line.starts_with("[TELEMETRY_PAYLOAD]") {
            let raw_json = line.trim_start_matches("[TELEMETRY_PAYLOAD]").trim();
            if let Ok(artifact) = serde_json::from_str::<TelemetryArtifact>(raw_json) {
                return Some(artifact);
            }
        }
    }
    None
}

use std::io::{Error, ErrorKind, Read, Result};
use std::process::{Command, Output, Stdio};
use std::sync::mpsc;
use std::thread;
use std::time::{Duration, Instant};

pub struct PodmanShield;

// No command run through this shield may block indefinitely. Anything that
// hangs -- a stuck compile, an interactive prompt waiting on stdin, a network
// call with no timeout of its own -- previously had no circuit breaker at
// all, which surfaced as an indistinguishable-from-hung cortex session.
// Compiles on this workspace have been observed taking 3-4+ minutes clean,
// so this needs real headroom rather than a short default.
const DEFAULT_TIMEOUT: Duration = Duration::from_secs(420);

impl PodmanShield {
    pub fn execute(command: &str) -> Result<Output> {
        Self::execute_with_timeout(command, DEFAULT_TIMEOUT)
    }

    pub fn execute_with_timeout(command: &str, timeout: Duration) -> Result<Output> {
        if let Err(e) =
            crate::arbiter_agent::ArbiterAgent::validate_action("execute_command", command)
        {
            return Err(Error::new(ErrorKind::PermissionDenied, e));
        }
        // Use bash, not POSIX sh -- the Strategist routinely proposes bash-only
        // syntax (${PIPESTATUS[0]}, the `time` keyword, etc.) which fails under
        // sh with cryptic errors like "Bad substitution", wasting turns before
        // any working command is found.
        let mut child_cmd = Command::new("bash");
        child_cmd.arg("-c").arg(command);
        for (key, value) in std::env::vars() {
            if key != "CARGO_BUILD_TARGET" {
                child_cmd.env(key, value);
            }
        }
        child_cmd.env_remove("CARGO_BUILD_TARGET");
        child_cmd.stdout(Stdio::piped());
        child_cmd.stderr(Stdio::piped());

        let mut child = child_cmd.spawn()?;

        // Drain stdout/stderr on background threads concurrently with the
        // process running. Without this, a command producing more output
        // than the OS pipe buffer (~64KB on Linux) would block the child on
        // write() while nothing reads -- a deadlock indistinguishable from
        // the hang this timeout exists to prevent.
        let stdout_pipe = child.stdout.take();
        let stderr_pipe = child.stderr.take();

        let (stdout_tx, stdout_rx) = mpsc::channel();
        thread::spawn(move || {
            let mut buf = Vec::new();
            if let Some(mut h) = stdout_pipe {
                let _ = h.read_to_end(&mut buf);
            }
            let _ = stdout_tx.send(buf);
        });

        let (stderr_tx, stderr_rx) = mpsc::channel();
        thread::spawn(move || {
            let mut buf = Vec::new();
            if let Some(mut h) = stderr_pipe {
                let _ = h.read_to_end(&mut buf);
            }
            let _ = stderr_tx.send(buf);
        });

        let start = Instant::now();
        let status = loop {
            match child.try_wait()? {
                Some(status) => break status,
                None => {
                    if start.elapsed() >= timeout {
                        let _ = child.kill();
                        let _ = child.wait();
                        return Err(Error::new(
                            ErrorKind::TimedOut,
                            format!(
                                "Command exceeded {:?} timeout and was killed: {}",
                                timeout, command
                            ),
                        ));
                    }
                    thread::sleep(Duration::from_millis(200));
                }
            }
        };

        let stdout = stdout_rx.recv().unwrap_or_default();
        let stderr = stderr_rx.recv().unwrap_or_default();
        let out = Output { status, stdout, stderr };

        // REGRESSION FIX (reapplied): explicitly check command success and
        // expose output. Without this, a failing command (e.g. a broken
        // `cargo check`) is indistinguishable from a successful one -- both
        // get tagged [CMD_OK] by the executor, which the Cortical Loop's
        // is_terminal_cmd heuristic then treats as proof of success.
        //
        // Include BOTH stdout and stderr in the error, not stderr alone.
        // Callers that redirect 2>&1 in their own command string (e.g.
        // RunTests) merge all output into stdout at the shell level -- an
        // error message built from stderr only then reports a bare exit
        // code with no diagnostic content, hiding the actual compiler
        // error from the agent entirely and leaving it unable to diagnose
        // real failures (confirmed live: a genuine compile error was
        // completely invisible across 13 turns because of this gap).
        if !out.status.success() {
            let stdout_str = String::from_utf8(out.stdout.clone())
                .unwrap_or_else(|_| "[INVALID UTF-8 IN STDOUT]".to_string());
            let stderr_str = String::from_utf8(out.stderr.clone())
                .unwrap_or_else(|_| "[INVALID UTF-8 IN STDERR]".to_string());
            return Err(Error::other(format!(
                "Command failed with status: {}\nSTDOUT:\n{}\nSTDERR:\n{}",
                out.status, stdout_str, stderr_str
            )));
        }
        Ok(out)
    }
}

use std::collections::{BTreeMap, HashMap};

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct SymbolId(pub String);

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
    pub line: usize,
}

impl Ord for Span {
    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
        self.line
            .cmp(&other.line)
            .then_with(|| self.start.cmp(&other.start))
            .then_with(|| self.end.cmp(&other.end))
    }
}

impl PartialOrd for Span {
    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
        Some(self.cmp(other))
    }
}

#[derive(Debug, Clone)]
pub struct SymbolInfo {
    pub id: SymbolId,
    pub name: String,
    pub kind: String,
    pub span: Span,
    pub file_path: String,
}

#[derive(Debug, Clone)]
pub struct SymbolReference {
    pub symbol_id: SymbolId,
    pub span: Span,
    pub file_path: String,
}

#[derive(Debug, Default, Clone)]
pub struct SymbolIndex {
    pub symbols: HashMap<String, SymbolInfo>,
    pub references: Vec<SymbolReference>,
    pub span_map: BTreeMap<Span, SymbolId>,
}

impl SymbolIndex {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn insert_symbol(&mut self, info: SymbolInfo) {
        self.span_map.insert(info.span.clone(), info.id.clone());
        self.symbols.insert(info.id.0.clone(), info);
    }

    pub fn add_reference(&mut self, reference: SymbolReference) {
        self.references.push(reference);
    }

    pub fn get_symbol(&self, id: &SymbolId) -> Option<&SymbolInfo> {
        self.symbols.get(&id.0)
    }

    pub fn find_by_span(&self, span: &Span) -> Option<&SymbolId> {
        self.span_map.get(span)
    }
}

//! Temporal watchdog with cooperative cancellation.
//! WARNING: The inner closure MUST check the CancellationToken periodically.
//! Non-cooperative tasks may leak threads.

use tokio_util::sync::CancellationToken;
use std::time::Duration;
use thiserror::Error;
use tracing::error;

#[derive(Debug, Error)]
pub enum WatchdogError {
    #[error("Execution timed out after {0:?}")]
    Timeout(Duration),
    #[error("Task join error: {0}")]
    Join(#[from] tokio::task::JoinError),
}

#[derive(Clone)]
pub struct ExecutionWatchdog {
    timeout: Duration,
}

impl ExecutionWatchdog {
    pub fn new(timeout: Duration) -> Self {
        Self { timeout }
    }

    pub async fn monitor<F, R>(&self, f: F) -> Result<R, WatchdogError>
    where
        F: FnOnce(CancellationToken) -> R + Send + 'static,
        R: Send + 'static,
    {
        let token = CancellationToken::new();
        let token_clone = token.clone();

        let handle = tokio::task::spawn_blocking(move || {
            f(token_clone)
        });

        tokio::select! {
            res = handle => {
                token.cancel();
                res.map_err(WatchdogError::Join)
            }
            _ = tokio::time::sleep(self.timeout) => {
                token.cancel();
                error!(
                    "ExecutionWatchdog timeout after {:?}. Task may still be running – ensure cooperative cancellation.",
                    self.timeout
                );
                Err(WatchdogError::Timeout(self.timeout))
            }
        }
    }
}

//! Taint‑aware data flow – no arbitrary downgrade.

use thiserror::Error;

#[derive(Debug, Error)]
pub enum SecurityError {
    #[error("Taint violation: {0}")]
    TaintViolation(String),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub enum TaintLabel {
    Public = 0,
    Internal = 1,
    Secret = 2,
}

#[must_use = "Tainted values must be explicitly declassified via `flow_to`"]
#[derive(Debug, Clone)]
pub struct Tainted<T> {
    value: T,
    label: TaintLabel,
}

impl<T> Tainted<T> {
    pub fn new(value: T, label: TaintLabel) -> Self {
        Self { value, label }
    }

    pub fn flow_to(self, sink_level: TaintLabel) -> Result<T, SecurityError> {
        if self.label > sink_level {
            return Err(SecurityError::TaintViolation(format!(
                "Label {:?} cannot flow to sink level {:?}",
                self.label, sink_level
            )));
        }
        Ok(self.value)
    }

    pub fn map<U, F>(self, f: F) -> Tainted<U>
    where
        F: FnOnce(T) -> U,
    {
        Tainted {
            value: f(self.value),
            label: self.label,
        }
    }

    pub fn peek_label(&self) -> TaintLabel {
        self.label
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_flow_allowed() {
        let t = Tainted::new(42, TaintLabel::Internal);
        assert_eq!(t.flow_to(TaintLabel::Internal).unwrap(), 42);
    }

    #[test]
    fn test_flow_denied() {
        let t = Tainted::new("secret", TaintLabel::Secret);
        assert!(t.flow_to(TaintLabel::Public).is_err());
    }
}

        );

        let mut loop_cortex = CorticalLoop::new(
            ctx.api_key.clone(),
            world_model.clone(),
            ctx.memory.clone(),
            ctx.vmo_state.clone(),
            ctx.dist.clone(),
            ctx.tx.clone(),
        );

        let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[CORTEX] Igniting Epistemic Session for Node {}", task_id)}).to_string());

        let execution_evidence = match loop_cortex.ignite(&combined_mission).await {
            Ok(CortexOutcome::CompletionProposed { belief, evidence }) => {
                format!(
                    "[DAG NODE COMPLETE] Objective: {}\nBelief: {:.2}\nEvidence:\n{}",
                    node.objective, belief, evidence
                )
            }
            Ok(CortexOutcome::TurnsExhausted { best_belief, evidence }) => {
                mission_had_failures = true;
                let outcome = MissionOutcome {
                    mission_id: uuid::Uuid::parse_str(&task_id)
                        .unwrap_or_else(|_| uuid::Uuid::new_v4()),
                    success: false,
                    evidence_summary: "Turns exhausted without contract satisfaction.".to_string(),
                    failures: vec![evidence.clone()],
                    successful_strategies: vec![],
                    raw_telemetry: String::new(),
                    failed_strategy_id: None,
                };
                if let Ok(report) = ctx.reflection_engine.reflect(&outcome).await {
                    for lesson in report.lessons {
                        let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[REFLECTION ENGINE] {}", lesson)}).to_string());
                    }
                }
                let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[CORTEX INCOMPLETE] Node {} exhausted turns (best belief {:.2}).", task_id, best_belief)}).to_string());
                format!("[DAG NODE INCOMPLETE] Objective: {}\nBest belief: {:.2}\nPartial evidence:\n{}", node.objective, best_belief, evidence)
            }
            Err(e) => {
                mission_had_failures = true;
                let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[CORTEX FRACTURE] Node {} loop failed: {}", task_id, e)}).to_string());
                format!("[DAG NODE FAILED] Ancestor node '{}' failed: {}", node.objective, e)
            }
        };

        task_outputs.insert(task_id.clone(), execution_evidence.clone());

        // ⚡ FIX: Immediate Hard Abort if a dependency fails or runs out of turns
        if mission_had_failures
            || execution_evidence.contains("[CORTEX INCOMPLETE]")
            || execution_evidence.contains("[DAG NODE FAILED]")
            || execution_evidence.contains("[DAG NODE INCOMPLETE]")
        {
            let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[DAG HALT] Aborting downstream execution. Dependency '{}' failed to complete.", task_id)}).to_string());
            break;
        }
    }

    if mission_had_failures {
        format!("DAG/GoT Execution FINISHED WITH INCOMPLETE NODES. Plan ID: {}", plan_id)
    } else {
        format!("DAG/GoT Hybrid Execution Complete. Plan ID: {}", plan_id)
    }
}

/// Fully maps tool names and argument objects back into native AgentAction variants.
pub fn parse_tool_call(tool_name: &str, args: &serde_json::Value) -> Result<AgentAction, String> {
    let get_str = |field: &str| -> Result<String, String> {
        args.get(field)
            .and_then(|v| v.as_str())
            .map(|s| s.to_string())
            .ok_or_else(|| format!("Missing or invalid field '{}' in tool call args", field))
    };

    let get_int = |field: &str| -> Result<usize, String> {
        args.get(field)
            .and_then(|v| v.as_u64())
            .map(|n| n as usize)
            .ok_or_else(|| format!("Missing or invalid integer field '{}'", field))
    };

    match tool_name {
        "execute_command" => Ok(AgentAction::ExecuteCommand {
            command: get_str("command")?,
            reasoning: get_str("reasoning")?,
            confidence: 0.95,
        }),
        "read_file" => Ok(AgentAction::ReadFile {
            path: get_str("path")?,
            reasoning: get_str("reasoning")?,
        }),
        "read_file_chunk" => Ok(AgentAction::ReadFileChunk {
            path: get_str("path")?,
            start_line: get_int("start_line")?,
            max_lines: get_int("max_lines")?,
            reasoning: get_str("reasoning")?,
        }),
        "search_code" => Ok(AgentAction::SearchCode {
            path: get_str("path")?,
            query: get_str("query")?,
            reasoning: get_str("reasoning")?,
        }),
        "write_file" => Ok(AgentAction::WriteFile {
            path: get_str("path")?,
            content: get_str("content")?,
            reasoning: get_str("reasoning")?,
            confidence: 0.95,
        }),
        "diff_patch" => Ok(AgentAction::DiffPatch {
            path: get_str("path")?,
            search: get_str("search")?,
            replace: get_str("replace")?,
            reasoning: get_str("reasoning")?,
        }),
        "patch_function" => Ok(AgentAction::PatchFunction {
            path: get_str("path")?,
            function_name: get_str("function_name")?,
            new_body: get_str("new_body")?,
            reasoning: get_str("reasoning")?,
        }),
        "run_tests" => Ok(AgentAction::RunTests {
            test_target: get_str("test_target")?,
            reasoning: get_str("reasoning")?,
        }),
        "search_memory" => Ok(AgentAction::SearchMemory {
            query: get_str("query")?,
            reasoning: get_str("reasoning")?,
        }),
        "halt" => Ok(AgentAction::Halt {
            reasoning: get_str("reasoning")?,
        }),
        other => Err(format!("Unrecognized tool identifier received: {}", other)),
    }
}

/// Universal response inspector that extracts tool calls from either Gemini or OpenAI/NVIDIA NIM responses.
pub fn extract_tool_call_from_response(response_json: &serde_json::Value) -> Result<AgentAction, String> {
    // 1. Check Gemini format: candidates[0].content.parts[0].functionCall
    if let Some(fc) = response_json.pointer("/candidates/0/content/parts/0/functionCall") {
        let name = fc["name"].as_str().ok_or("Gemini tool call missing 'name'")?;
        let args = fc.get("args").cloned().unwrap_or_else(|| json!({}));
        return parse_tool_call(name, &args);
    }

    // 2. Check OpenAI / NVIDIA NIM format: choices[0].message.tool_calls[0]
    if let Some(tc) = response_json.pointer("/choices/0/message/tool_calls/0") {
        let name = tc.pointer("/function/name").and_then(|v| v.as_str()).ok_or("OpenAI tool call missing function name")?;
        let args_str = tc.pointer("/function/arguments").and_then(|v| v.as_str()).unwrap_or("{}");
        let args: serde_json::Value = serde_json::from_str(args_str).map_err(|e| format!("Failed to parse tool arguments JSON string: {}", e))?;
        return parse_tool_call(name, &args);
    }

    Err("No native tool call found in model response payload.".to_string())
}

#[derive(Debug, Clone)]
pub struct VerificationReport {
    pub evidence_quality: f32,
    pub mutation_detected: bool,
    pub command_completed: bool,
    pub security_violation_detected: bool,
    pub goal_verified: bool,
}

impl VerificationReport {
    pub fn from_evidence(evidence: &str, evaluator_confidence: f32) -> Self {
        let security_violation = evidence.contains("PATH_TRAVERSAL_DENIED")
            || evidence.contains("sandbox_violation")
            || evidence.contains("SECURITY_VIOLATION");

        let command_completed = (evidence.contains("[CMD_OK]")
            || evidence.contains("[READ_OK]")
            || evidence.contains("[WRITE_OK]"))
            && !security_violation;

        Self {
            evidence_quality: if security_violation { 0.0 } else { evaluator_confidence.max(0.0) },
            mutation_detected: evidence.contains("[WRITE_OK]")
                || evidence.contains("[PATCH_FUNCTION_OK]")
                || evidence.contains("[DIFF_PATCH_OK]"),
            command_completed,
            security_violation_detected: security_violation,
            goal_verified: evaluator_confidence > 0.5 && !security_violation,
        }
    }

    pub fn can_complete(&self, belief: f32) -> bool {
        !self.security_violation_detected
            && self.command_completed
            && belief >= 0.85
            && self.goal_verified
            && self.evidence_quality >= 0.5
    }
}

pub struct WasmForge;

/// Run wasm-opt -Oz when Binaryen is available. Returns original bytes on failure.
fn maybe_wasm_opt(wasm_bytes: &[u8]) -> Vec<u8> {
    use std::io::Write;
    use std::process::{Command, Stdio};
    use tempfile::NamedTempFile;

    // Fast path: wasm-opt not installed
    if Command::new("wasm-opt").arg("--version").output().is_err() {
        return wasm_bytes.to_vec();
    }

    let mut input = match NamedTempFile::new() {
        Ok(f) => f,
        Err(_) => return wasm_bytes.to_vec(),
    };
    if input.write_all(wasm_bytes).is_err() {
        return wasm_bytes.to_vec();
    }
    let output_path = input.path().with_extension("opt.wasm");

    let status = Command::new("wasm-opt")
        .args(["-Oz", "--enable-bulk-memory", "--strip-debug", "--strip-producers"])
        .arg(input.path())
        .arg("-o")
        .arg(&output_path)
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .status();

    match status {
        Ok(s) if s.success() => std::fs::read(&output_path).unwrap_or_else(|_| wasm_bytes.to_vec()),
        _ => wasm_bytes.to_vec(),
    }
}

impl WasmForge {
    pub async fn synthesize_cell(logic_code: &str) -> Result<String> {
        if logic_code.trim().starts_with("const std") || logic_code.contains("export fn") {
            return Self::synthesize_zig(logic_code).await;
        }
        Self::synthesize_rust(logic_code).await
    }

    async fn synthesize_zig(zig_code: &str) -> Result<String> {
        let ts = SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs();
        let tmp_path = format!("/tmp/nexus_zig_cell_{}.zig", ts);
        fs::write(&tmp_path, zig_code).await?;

        let output = tokio::process::Command::new("zig")
            .args([
                "build-exe",
                &tmp_path,
                "-target",
                "wasm32-freestanding",
                "-O",
                "ReleaseSmall",
                "--name",
                "cell",
            ])
            .current_dir("/tmp")
            .output()
            .await?;

        if !output.status.success() {
            let _ = fs::remove_file(&tmp_path).await;
            return Err(anyhow!("Zig compile error: {}", String::from_utf8_lossy(&output.stderr)));
        }

        let wasm_bytes = fs::read("/tmp/cell.wasm").await?;
        let wasm_bytes = maybe_wasm_opt(&wasm_bytes);
        let b64 = general_purpose::STANDARD.encode(&wasm_bytes);
        let _ = fs::remove_file(&tmp_path).await;
        Ok(b64)
    }

    // ⚡ REAL RUST TO WASM PIPELINE
    async fn synthesize_rust(logic_code: &str) -> Result<String> {
        let ts = SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs();
        let tmp_dir = format!("/tmp/nexus_rust_cell_{}", ts);
        fs::create_dir_all(&tmp_dir).await?;

        let cargo_toml = r#"[package]
name = "cell"
version = "0.1.0"
edition = "2021"
[lib]
crate-type = ["cdylib"]
"#;
        fs::write(format!("{}/Cargo.toml", tmp_dir), cargo_toml).await?;

        let src_dir = format!("{}/src", tmp_dir);
        fs::create_dir_all(&src_dir).await?;
        fs::write(format!("{}/lib.rs", src_dir), logic_code).await?;

        let output = tokio::process::Command::new("cargo")
            .args(["build", "--target", "wasm32-wasip1", "--release"])
            .current_dir(&tmp_dir)
            .output()
            .await?;

        if !output.status.success() {
            let stderr = String::from_utf8_lossy(&output.stderr);
            let _ = fs::remove_dir_all(&tmp_dir).await;
            return Err(anyhow!("Rust WASM compile error:\n{}", stderr));
        }

        let wasm_bytes =
            fs::read(format!("{}/target/wasm32-wasip1/release/cell.wasm", tmp_dir)).await?;
        let wasm_bytes = maybe_wasm_opt(&wasm_bytes);
        let b64 = general_purpose::STANDARD.encode(&wasm_bytes);
        let _ = fs::remove_dir_all(&tmp_dir).await;
        Ok(b64)
    }
}

use anyhow::{anyhow, Result};
use wasmtime::*;

pub struct WasmRuntime;

impl WasmRuntime {
    fn make_engine() -> Result<Engine> {
        let mut config = Config::new();
        // Size + speed balance for ARM64 / low-RAM nodes
        config.cranelift_opt_level(OptLevel::SpeedAndSize);
        // Hard limits so sidecars cannot starve the cortex
        config.consume_fuel(true);
        config.epoch_interruption(false);
        // Cheaper instantiation
        config.memory_init_cow(true);
        // Avoid memory spikes on constrained devices
        config.parallel_compilation(false);
        Ok(Engine::new(&config)?)
    }

    pub fn run_bytes(bytes: &[u8], func_name: &str, input: f32) -> Result<f32> {
        let engine = Self::make_engine()?;
        let module = Module::from_binary(&engine, bytes)?;
        let mut store = Store::new(&engine, ());
        store.set_fuel(50_000_000)?;
        let instance = Instance::new(&mut store, &module, &[])?;
        let func = instance.get_typed_func::<f32, f32>(&mut store, func_name)?;
        Ok(func.call(&mut store, input)?)
    }

    pub fn run_with_buffer(bytes: &[u8], func_name: &str, data: &[u8]) -> Result<f32> {
        let engine = Self::make_engine()?;
        let module =
            Module::from_binary(&engine, bytes).map_err(|e| anyhow!("WASM_LOAD_FAULT: {}", e))?;
        let mut store = Store::new(&engine, ());
        store.set_fuel(50_000_000)?;
        let instance = Instance::new(&mut store, &module, &[])
            .map_err(|e| anyhow!("WASM_INST_FAULT: {}", e))?;

        let memory = instance
            .get_memory(&mut store, "memory")
            .ok_or_else(|| anyhow!("WASM module must export memory"))?;

        let get_ptr = instance
            .get_typed_func::<(), i32>(&mut store, "get_buffer_ptr")
            .map_err(|e| anyhow!("Missing get_buffer_ptr export: {}", e))?;
        let ptr = get_ptr.call(&mut store, ())? as usize;

        if data.len() > 65_536 {
            return Err(anyhow!("VMO payload exceeds 64KB sandbox."));
        }

        // WASMtime 43+ API: Use .write() directly on the memory slice
        memory
            .write(&mut store, ptr, data)
            .map_err(|e| anyhow!("WASM_MEMORY_WRITE_FAULT: {}", e))?;

        let func = instance
            .get_typed_func::<i32, f32>(&mut store, func_name)
            .map_err(|e| anyhow!("WASM_FUNC_MISSING: {}", e))?;

        Ok(func.call(&mut store, data.len() as i32)?)
    }
}

    State(_state): State<()>,

    };

    let user_prompt = format!(
        "Extract data from this HTML based on the following prompt:\n{}\n\nHTML:\n{}",

    );

use anyhow::Result;
use clap::Parser;
use std::fs;
use std::path::PathBuf;

#[derive(Parser, Debug)]
#[command(author, version, about = "Nexus PR Reviewer Substrate Drone")]
struct Args {
    #[arg(short, long)]
    target_dir: PathBuf,
}

#[derive(serde::Serialize)]
struct ReviewReport {
    status: String,
    risk_score: u8,
    violations: Vec<String>,
}

fn main() -> Result<()> {
    let args = Args::parse();
    println!(">> [PR_REVIEWER_DRONE] Scanning target substrate at {}", args.target_dir.display());

    let mut violations = Vec::new();
    let mut risk_score = 0;

    // Scan files for substrate anti-patterns
    for entry in walkdir::WalkDir::new(&args.target_dir).into_iter().filter_map(|e| e.ok()) {
        if entry.path().extension().and_then(|s| s.to_str()) == Some("rs") {
            let content = fs::read_to_string(entry.path()).unwrap_or_default();

            if content.contains(".unwrap()") && !content.contains("unwrap_or") {
                violations.push(format!("Unwrapped panic vector in {:?}", entry.path()));
                risk_score += 10;
            }
            if content.contains("std::process::Command") && !content.contains("ActionGate") {
                violations
                    .push(format!("Unsanitized system command invocation in {:?}", entry.path()));
                risk_score += 40;
            }
        }
    }

    let report = ReviewReport {
        status: if risk_score > 30 { "REJECTED" } else { "APPROVED" }.to_string(),
        risk_score,
        violations,
    };

    println!("{}", serde_json::to_string_pretty(&report)?);
    Ok(())
}

const std = @import("std");
const core = @import("safetensors.zig");

const EngineContext = struct {
    gpa: std.heap.GeneralPurposeAllocator(.{}),
    mapped: []const u8,
    st: core.SafeTensors,
};

pub const CTensorView = extern struct {
    data_ptr: [*]const u8,
    byte_size: u64,
    dtype_code: u8,
    shape_ptr: [*]const i64,
    shape_len: usize,
    internal_view: ?*anyopaque,
};

export fn nexus_dtype_element_size(dtype_code: u8) callconv(.c) u8 {
    if (dtype_code > @intFromEnum(core.DType.BOOL)) return 0;
    const dt: core.DType = @enumFromInt(dtype_code);
    return dt.elementSize();
}

export fn nexus_st_init(file_path: [*:0]const u8) callconv(.c) ?*anyopaque {
    const path = std.mem.span(file_path);

    const ctx = std.heap.c_allocator.create(EngineContext) catch return null;
    ctx.gpa = .{};
    const allocator = ctx.gpa.allocator();

    const file = std.fs.cwd().openFile(path, .{ .mode = .read_only }) catch {
        _ = ctx.gpa.deinit();
        std.heap.c_allocator.destroy(ctx);
        return null;
    };
    defer file.close();

    const stat = file.stat() catch {
        _ = ctx.gpa.deinit();
        std.heap.c_allocator.destroy(ctx);
        return null;
    };
    const file_size = stat.size;
    if (file_size == 0) {
        _ = ctx.gpa.deinit();
        std.heap.c_allocator.destroy(ctx);
        return null;
    }

    const mapped = std.posix.mmap(
        null,
        file_size,
        std.posix.PROT.READ,
        .{ .TYPE = .SHARED },
        file.handle,
        0,
    ) catch {
        _ = ctx.gpa.deinit();
        std.heap.c_allocator.destroy(ctx);
        return null;
    };

    ctx.mapped = mapped;

    ctx.st = core.SafeTensors.init(ctx.mapped, allocator) catch {
        std.posix.munmap(@alignCast(ctx.mapped));
        _ = ctx.gpa.deinit();
        std.heap.c_allocator.destroy(ctx);
        return null;
    };

    return ctx;
}

export fn nexus_st_get(
    handle: ?*anyopaque,
    name: [*:0]const u8,
    out_view: *CTensorView,
) callconv(.c) bool {
    const ctx: *EngineContext = @ptrCast(@alignCast(handle orelse return false));
    const name_slice = std.mem.span(name);

    const view = ctx.st.get(name_slice) catch return false;

    const view_ptr = ctx.st.allocator.create(core.TensorView) catch {
        ctx.st.freeView(view);
        return false;
    };
    view_ptr.* = view;

    out_view.* = .{
        .data_ptr = view.data.ptr,
        .byte_size = view.byteSize(),
        .dtype_code = @intFromEnum(view.dtype),
        .shape_ptr = view.shape.ptr,
        .shape_len = view.shape.len,
        .internal_view = view_ptr,
    };

    return true;
}

export fn nexus_st_free_view(handle: ?*anyopaque, view: *CTensorView) callconv(.c) void {
    const ctx: *EngineContext = @ptrCast(@alignCast(handle orelse return));
    if (view.internal_view) |ptr| {
        const view_ptr: *core.TensorView = @ptrCast(@alignCast(ptr));
        ctx.st.freeView(view_ptr.*);
        ctx.st.allocator.destroy(view_ptr);
        view.internal_view = null;
    }
}

export fn nexus_st_deinit(handle: ?*anyopaque) callconv(.c) void {
    const ctx: *EngineContext = @ptrCast(@alignCast(handle orelse return));
    ctx.st.deinit();
    std.posix.munmap(@alignCast(ctx.mapped));
    _ = ctx.gpa.deinit();
    std.heap.c_allocator.destroy(ctx);
}

const std = @import("std");

pub const DType = enum(u8) {
    F64 = 0,
    F32 = 1,
    F16 = 2,
    BF16 = 3,
    I64 = 4,
    I32 = 5,
    I16 = 6,
    I8 = 7,
    U8 = 8,
    BOOL = 9,

    pub fn fromString(s: []const u8) !DType {
        if (std.mem.eql(u8, s, "F64")) return .F64;
        if (std.mem.eql(u8, s, "F32")) return .F32;
        if (std.mem.eql(u8, s, "F16")) return .F16;
        if (std.mem.eql(u8, s, "BF16")) return .BF16;
        if (std.mem.eql(u8, s, "I64")) return .I64;
        if (std.mem.eql(u8, s, "I32")) return .I32;
        if (std.mem.eql(u8, s, "I16")) return .I16;
        if (std.mem.eql(u8, s, "I8")) return .I8;
        if (std.mem.eql(u8, s, "U8")) return .U8;
        if (std.mem.eql(u8, s, "BOOL")) return .BOOL;
        return error.UnknownDType;
    }

    pub fn elementSize(self: DType) u8 {
        return switch (self) {
            .F64, .I64 => 8,
            .F32, .I32 => 4,
            .F16, .BF16, .I16 => 2,
            .I8, .U8, .BOOL => 1,
        };
    }
};

pub const TensorView = struct {
    name: []const u8,
    dtype: DType,
    shape: []const i64,
    data: []const u8,
    data_begin: u64,
    data_end: u64,

    pub fn byteSize(self: TensorView) u64 {
        return self.data_end - self.data_begin;
    }

    pub fn numel(self: TensorView) u64 {
        var n: u64 = 1;
        for (self.shape) |d| n *= @intCast(d);
        return n;
    }
};

pub const SafeTensors = struct {
    mapped: []const u8,
    header_size: u64,
    data_offset: u64,
    parsed: std.json.Parsed(std.json.Value),
    allocator: std.mem.Allocator,

    pub fn init(mapped: []const u8, allocator: std.mem.Allocator) !SafeTensors {
        if (mapped.len < 8) return error.FileTooSmall;

        const header_size = std.mem.readInt(u64, mapped[0..8], .little);
        if (header_size == 0 or header_size > 100_000_000) return error.InvalidHeaderSize;
        if (8 + header_size > mapped.len) return error.HeaderOverflow;

        const header_json = mapped[8 .. 8 + header_size];
        if (header_json[0] != '{') return error.InvalidHeader;

        var parsed = try std.json.parseFromSlice(std.json.Value, allocator, header_json, .{});
        errdefer parsed.deinit();

        if (parsed.value != .object) {
            parsed.deinit();
            return error.ExpectedObject;
        }

        return .{
            .mapped = mapped,
            .header_size = header_size,
            .data_offset = 8 + header_size,
            .parsed = parsed,
            .allocator = allocator,
        };
    }

    pub fn deinit(self: *SafeTensors) void {
        self.parsed.deinit();
    }

    pub fn get(self: *const SafeTensors, name: []const u8) !TensorView {
        const root = self.parsed.value.object;
        const entry = root.get(name) orelse return error.TensorNotFound;
        if (entry != .object) return error.InvalidTensorEntry;

        const obj = entry.object;

        const dtype_str = (obj.get("dtype") orelse return error.MissingDtype).string;
        const dtype = try DType.fromString(dtype_str);

        const shape_val = obj.get("shape") orelse return error.MissingShape;
        if (shape_val != .array) return error.InvalidShape;

        var shape = try self.allocator.alloc(i64, shape_val.array.items.len);
        errdefer self.allocator.free(shape);
        for (shape_val.array.items, 0..) |v, i| {
            shape[i] = v.integer;
        }

        const offsets_val = obj.get("data_offsets") orelse return error.MissingOffsets;
        if (offsets_val != .array or offsets_val.array.items.len != 2)
            return error.InvalidOffsets;

        const begin: u64 = @intCast(offsets_val.array.items[0].integer);
        const end: u64 = @intCast(offsets_val.array.items[1].integer);

        if (end < begin or self.data_offset + end > self.mapped.len) {
            self.allocator.free(shape);
            return error.InvalidOffsetsRange;
        }

        const data = self.mapped[self.data_offset + begin .. self.data_offset + end];

        return TensorView{
            .name = name,
            .dtype = dtype,
            .shape = shape,
            .data = data,
            .data_begin = begin,
            .data_end = end,
        };
    }

    pub fn freeView(self: *const SafeTensors, view: TensorView) void {
        self.allocator.free(view.shape);
    }
};

var buffer: [65536]u8 = undefined;

export fn get_buffer_ptr() [*]u8 {
    return &buffer;
}

export fn sumBytes(len: i32) f32 {
    var total: u32 = 0;
    var i: usize = 0;
    const ulen = @as(usize, @intCast(len));
    while (i < ulen) : (i += 1) {
        total += buffer[i];
    }
    return @as(f32, @floatFromInt(total));
}

const std = @import("std");

// Compact Swarm Unit Definition
pub const SwarmUnit = struct {
    id: u32,
    status: u8,   // 0: Offline, 1: Idle, 2: Active
    load: u8,     // 0-100 percentage
};

// Fixed memory allocation for 68 units (The Hive)
var hive_memory: [68]SwarmUnit = undefined;

export fn init_hive() void {
    for (&hive_memory, 0..) |*unit, i| {
        unit.* = SwarmUnit{
            .id = @intCast(i),
            .status = 1, // Start in Idle
            .load = 0,
        };
    }
}

export fn get_hive_ptr() [*]SwarmUnit {
    return &hive_memory;
}

export fn update_unit(id: u32, status: u8, load: u8) void {
    if (id < 68) {
        hive_memory[id].status = status;
        hive_memory[id].load = load;
    }
}
