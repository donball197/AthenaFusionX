const std = @import("std");
const core = @import("safetensors.zig");

const EngineContext = struct {
    gpa: std.heap.GeneralPurposeAllocator(.{}),
    mapped: []const u8,
    st: core.SafeTensors,
};

pub const CTensorView = extern struct {
    data_ptr: [*]const u8,
    byte_size: u64,
    dtype_code: u8,
    shape_ptr: [*]const i64,
    shape_len: usize,
    internal_view: ?*anyopaque,
};

export fn nexus_dtype_element_size(dtype_code: u8) callconv(.c) u8 {
    if (dtype_code > @intFromEnum(core.DType.BOOL)) return 0;
    const dt: core.DType = @enumFromInt(dtype_code);
    return dt.elementSize();
}

export fn nexus_st_init(file_path: [*:0]const u8) callconv(.c) ?*anyopaque {
    const path = std.mem.span(file_path);

    const ctx = std.heap.c_allocator.create(EngineContext) catch return null;
    ctx.gpa = .{};
    const allocator = ctx.gpa.allocator();

    const file = std.fs.cwd().openFile(path, .{ .mode = .read_only }) catch {
        _ = ctx.gpa.deinit();
        std.heap.c_allocator.destroy(ctx);
        return null;
    };
    defer file.close();

    const stat = file.stat() catch {
        _ = ctx.gpa.deinit();
        std.heap.c_allocator.destroy(ctx);
        return null;
    };
    const file_size = stat.size;
    if (file_size == 0) {
        _ = ctx.gpa.deinit();
        std.heap.c_allocator.destroy(ctx);
        return null;
    }

    const mapped = std.posix.mmap(
        null,
        file_size,
        std.posix.PROT.READ,
        .{ .TYPE = .SHARED },
        file.handle,
        0,
    ) catch {
        _ = ctx.gpa.deinit();
        std.heap.c_allocator.destroy(ctx);
        return null;
    };

    ctx.mapped = mapped;

    ctx.st = core.SafeTensors.init(ctx.mapped, allocator) catch {
        std.posix.munmap(@alignCast(ctx.mapped));
        _ = ctx.gpa.deinit();
        std.heap.c_allocator.destroy(ctx);
        return null;
    };

    return ctx;
}

export fn nexus_st_get(
    handle: ?*anyopaque,
    name: [*:0]const u8,
    out_view: *CTensorView,
) callconv(.c) bool {
    const ctx: *EngineContext = @ptrCast(@alignCast(handle orelse return false));
    const name_slice = std.mem.span(name);

    const view = ctx.st.get(name_slice) catch return false;

    const view_ptr = ctx.st.allocator.create(core.TensorView) catch {
        ctx.st.freeView(view);
        return false;
    };
    view_ptr.* = view;

    out_view.* = .{
        .data_ptr = view.data.ptr,
        .byte_size = view.byteSize(),
        .dtype_code = @intFromEnum(view.dtype),
        .shape_ptr = view.shape.ptr,
        .shape_len = view.shape.len,
        .internal_view = view_ptr,
    };

    return true;
}

export fn nexus_st_free_view(handle: ?*anyopaque, view: *CTensorView) callconv(.c) void {
    const ctx: *EngineContext = @ptrCast(@alignCast(handle orelse return));
    if (view.internal_view) |ptr| {
        const view_ptr: *core.TensorView = @ptrCast(@alignCast(ptr));
        ctx.st.freeView(view_ptr.*);
        ctx.st.allocator.destroy(view_ptr);
        view.internal_view = null;
    }
}

export fn nexus_st_deinit(handle: ?*anyopaque) callconv(.c) void {
    const ctx: *EngineContext = @ptrCast(@alignCast(handle orelse return));
    ctx.st.deinit();
    std.posix.munmap(@alignCast(ctx.mapped));
    _ = ctx.gpa.deinit();
    std.heap.c_allocator.destroy(ctx);
}

const std = @import("std");

pub const DType = enum(u8) {
    F64 = 0,
    F32 = 1,
    F16 = 2,
    BF16 = 3,
    I64 = 4,
    I32 = 5,
    I16 = 6,
    I8 = 7,
    U8 = 8,
    BOOL = 9,

    pub fn fromString(s: []const u8) !DType {
        if (std.mem.eql(u8, s, "F64")) return .F64;
        if (std.mem.eql(u8, s, "F32")) return .F32;
        if (std.mem.eql(u8, s, "F16")) return .F16;
        if (std.mem.eql(u8, s, "BF16")) return .BF16;
        if (std.mem.eql(u8, s, "I64")) return .I64;
        if (std.mem.eql(u8, s, "I32")) return .I32;
        if (std.mem.eql(u8, s, "I16")) return .I16;
        if (std.mem.eql(u8, s, "I8")) return .I8;
        if (std.mem.eql(u8, s, "U8")) return .U8;
        if (std.mem.eql(u8, s, "BOOL")) return .BOOL;
        return error.UnknownDType;
    }

    pub fn elementSize(self: DType) u8 {
        return switch (self) {
            .F64, .I64 => 8,
            .F32, .I32 => 4,
            .F16, .BF16, .I16 => 2,
            .I8, .U8, .BOOL => 1,
        };
    }
};

pub const TensorView = struct {
    name: []const u8,
    dtype: DType,
    shape: []const i64,
    data: []const u8,
    data_begin: u64,
    data_end: u64,

    pub fn byteSize(self: TensorView) u64 {
        return self.data_end - self.data_begin;
    }

    pub fn numel(self: TensorView) u64 {
        var n: u64 = 1;
        for (self.shape) |d| n *= @intCast(d);
        return n;
    }
};

pub const SafeTensors = struct {
    mapped: []const u8,
    header_size: u64,
    data_offset: u64,
    parsed: std.json.Parsed(std.json.Value),
    allocator: std.mem.Allocator,

    pub fn init(mapped: []const u8, allocator: std.mem.Allocator) !SafeTensors {
        if (mapped.len < 8) return error.FileTooSmall;

        const header_size = std.mem.readInt(u64, mapped[0..8], .little);
        if (header_size == 0 or header_size > 100_000_000) return error.InvalidHeaderSize;
        if (8 + header_size > mapped.len) return error.HeaderOverflow;

        const header_json = mapped[8 .. 8 + header_size];
        if (header_json[0] != '{') return error.InvalidHeader;

        var parsed = try std.json.parseFromSlice(std.json.Value, allocator, header_json, .{});
        errdefer parsed.deinit();

        if (parsed.value != .object) {
            parsed.deinit();
            return error.ExpectedObject;
        }

        return .{
            .mapped = mapped,
            .header_size = header_size,
            .data_offset = 8 + header_size,
            .parsed = parsed,
            .allocator = allocator,
        };
    }

    pub fn deinit(self: *SafeTensors) void {
        self.parsed.deinit();
    }

    pub fn get(self: *const SafeTensors, name: []const u8) !TensorView {
        const root = self.parsed.value.object;
        const entry = root.get(name) orelse return error.TensorNotFound;
        if (entry != .object) return error.InvalidTensorEntry;

        const obj = entry.object;

        const dtype_str = (obj.get("dtype") orelse return error.MissingDtype).string;
        const dtype = try DType.fromString(dtype_str);

        const shape_val = obj.get("shape") orelse return error.MissingShape;
        if (shape_val != .array) return error.InvalidShape;

        var shape = try self.allocator.alloc(i64, shape_val.array.items.len);
        errdefer self.allocator.free(shape);
        for (shape_val.array.items, 0..) |v, i| {
            shape[i] = v.integer;
        }

        const offsets_val = obj.get("data_offsets") orelse return error.MissingOffsets;
        if (offsets_val != .array or offsets_val.array.items.len != 2)
            return error.InvalidOffsets;

        const begin: u64 = @intCast(offsets_val.array.items[0].integer);
        const end: u64 = @intCast(offsets_val.array.items[1].integer);

        if (end < begin or self.data_offset + end > self.mapped.len) {
            self.allocator.free(shape);
            return error.InvalidOffsetsRange;
        }

        const data = self.mapped[self.data_offset + begin .. self.data_offset + end];

        return TensorView{
            .name = name,
            .dtype = dtype,
            .shape = shape,
            .data = data,
            .data_begin = begin,
            .data_end = end,
        };
    }

    pub fn freeView(self: *const SafeTensors, view: TensorView) void {
        self.allocator.free(view.shape);
    }
};

const posix = std.posix;

var active_vmos: u32 = 0;
var next_handle: u32 = 0xc8;
const MAX_SLOTS: usize = 4096;
const SLOT_MASK: u32 = 4095;
const PAGE_SIZE: usize = 4096;
const SENTINEL_MAGIC: u32 = 0xDEADC0DE;
const GENERATION_SHIFT: u5 = 12;

var current_load: f32 = 0.0;

const VmoMetadata = struct {
    ptr: [*]align(PAGE_SIZE) u8,
    size: usize,
    magic: u32,
    generation: u32,
    refcount: u32,
    is_cow_child: bool,
    parent_handle: u32 = 0,
};

var handle_table: [MAX_SLOTS]?VmoMetadata = [_]?VmoMetadata{null} ** MAX_SLOTS;

export fn set_substrate_load(load: f32) void { current_load = load; }
export fn extract_thermal_pulse_raw() i32 {
    const temp = (current_load * 40.0) + 30.0;
    return @intFromFloat(temp * 1000.0);
}

export fn zx_vmo_create_native(size: usize) u32 {
    const aligned_size = (size + PAGE_SIZE - 1) & ~(PAGE_SIZE - 1);
    const ptr = posix.mmap(null, aligned_size, posix.PROT.READ | posix.PROT.WRITE, .{ .TYPE = .PRIVATE, .ANONYMOUS = true }, -1, 0) catch return 0;
    const slot = next_handle & SLOT_MASK;
    handle_table[slot] = VmoMetadata{
        .ptr = @alignCast(ptr.ptr), .size = aligned_size, .magic = SENTINEL_MAGIC,
        .generation = next_handle >> GENERATION_SHIFT, .refcount = 1, .is_cow_child = false,
    };
    const handle = next_handle;
    next_handle += 1;
    active_vmos += 1;
    return handle;
}

export fn zx_vmo_clone_native(parent_handle: u32, _: usize) u32 {
    const p_slot = parent_handle & SLOT_MASK;
    if (handle_table[p_slot]) |*p| {
        const slot = next_handle & SLOT_MASK;
        handle_table[slot] = VmoMetadata{
            .ptr = p.ptr, .size = p.size, .magic = SENTINEL_MAGIC,
            .generation = next_handle >> GENERATION_SHIFT, .refcount = 1,
            .is_cow_child = true, .parent_handle = parent_handle,
        };
        const handle = next_handle;
        next_handle += 1;
        active_vmos += 1;
        return handle;
    }
    return 0;
}

export fn zx_vmo_write_native(h: u32, o: usize, b: [*]const u8, s: usize) i32 {
    const slot = h & SLOT_MASK;
    if (handle_table[slot]) |*meta| {
        if (o + s > meta.size) return -1;
        @memcpy(meta.ptr[o .. o + s], b[0..s]);
        return 0;
    }
    return -1;
}

export fn zx_vmo_read_native(h: u32, o: usize, b: [*]u8, s: usize) i32 {
    const slot = h & SLOT_MASK;
    if (handle_table[slot]) |meta| {
        if (o + s > meta.size) return -1;
        @memcpy(b[0..s], meta.ptr[o .. o + s]);
        return 0;
    }
    return -1;
}

export fn zx_handle_close_native(h: u32) void {
    const slot = h & SLOT_MASK;
    if (handle_table[slot]) |*meta| {
        meta.refcount -= 1;
        if (meta.refcount == 0) {
            posix.munmap(meta.ptr[0..meta.size]);
            handle_table[slot] = null;
            active_vmos -= 1;
        }
    }
}

export fn muscle_get_active_vmos() u32 { return active_vmos; }
export fn muscle_get_next_handle() u32 { return next_handle; }

const VmoInfo = extern struct {
    handle: u32, size: usize, committed_pages: usize, is_cow: u8, parent_handle: u32,
};

export fn zx_vmo_get_info(h: u32, out_info: *VmoInfo) i32 {
    const slot = h & SLOT_MASK;
    if (handle_table[slot]) |meta| {
        out_info.* = VmoInfo{
            .handle = h, .size = meta.size, .committed_pages = meta.size / PAGE_SIZE,
            .is_cow = if (meta.is_cow_child) 1 else 0, .parent_handle = meta.parent_handle,
        };
        return 0;
    }
    return -1;
}

export fn zx_vmo_get_all_handles(out_handles: [*]u32, max_count: usize) usize {
    var count: usize = 0;
    for (handle_table, 0..) |m, i| {
        if (m != null and count < max_count) {
            out_handles[count] = @intCast(i);
            count += 1;
        }
    }
    return count;
}

    var sum: f32 = 0.0;
    // Calculate a safety floor based on the first 8 dimensions
    for (latent_intent[0..8]) |val| {
        sum += val;
    }
    return sum / 8.0;
}

/// Original Resonance Logic for Hardware Monitoring
pub export fn calculate_resonance_e2b(mmap_ptr: [*]const u8, length: usize) f32 {
    var total_energy: f32 = 0.0;
    var i: usize = 0;
    while (i < length) : (i += 64) {
        const byte = mmap_ptr[i];
        const val = (@as(f32, @floatFromInt(byte)) / 255.0);
        total_energy += (val * val);
    }
    const count = @as(f32, @floatFromInt(length / 64));
    return if (count > 0) total_energy / count else 0.0;
}

var free_list: [MAX_SLOTS]u16 = undefined;
var free_count: usize = 0;
var high_water: u16 = 0;
var generation_table: [MAX_SLOTS]u20 = [_]u20{1} ** MAX_SLOTS;

    ptr: [*]align(PAGE_SIZE) u8,

inline fn shard_for(h: u32) usize { return @intCast(h & SHARD_MASK); }
inline fn decode_slot(h: u32) usize { return @intCast(h & SLOT_MASK); }
inline fn decode_generation(h: u32) u32 { return h >> GENERATION_SHIFT; }

    defer alloc_mutex.unlock();
