use nexus_edge_minion::inference_engine::InferenceEngine;
use std::path::Path;

#[test]
fn test_local_engine_safetensors_smoke() {
    let weights_path = Path::new("gemma-4-E2B-it.safetensors");

    // If the weights file isn't present yet, skip gracefully instead of failing CI/test suites
    if !weights_path.exists() {
        println!(">> [SKIP] gemma-4-E2B-it.safetensors not found in root. Skipping local weights smoke test.");
        return;
    }

    println!(">> [TEST] Initializing InferenceEngine with local safetensors...");
    let mut engine = match InferenceEngine::new(weights_path) {
        Ok(eng) => eng,
        Err(e) => {
            panic!("FATAL: Failed to initialize InferenceEngine: {}", e);
        }
    };

    println!(">> [TEST] Engine online. Executing single token warmup step at position 0...");
    let mut hidden_state = vec![0.0f32; 1536];
    let result = engine.step(42, 0, &mut hidden_state);

    assert!(result.is_ok(), "Token execution step failed.");

    // Verify the hidden state was mutated by the layers and contains valid finite numbers
    let has_finite = hidden_state.iter().any(|&v| v.is_finite() && v != 0.0);
    assert!(has_finite, "Hidden state remained entirely zero or non-finite after forward step.");

    println!(">> [TEST] Computing logits via tied LM head + softcap...");
    let logits = engine.compute_logits(&hidden_state).expect("compute_logits failed");
    assert_eq!(logits.len(), 262144, "Unexpected vocab size in logits.");
    assert!(logits.iter().all(|v| v.is_finite()), "Non-finite value in logits.");

    let mut indexed: Vec<(usize, f32)> = logits.iter().copied().enumerate().collect();
    indexed.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap());
    println!(">> [TEST] Top-5 tokens by logit:");
    for (rank, (tok_id, val)) in indexed.iter().take(5).enumerate() {
        println!("    #{} token_id={} logit={:.4}", rank + 1, tok_id, val);
    }

    let (argmax_id, argmax_val) = indexed[0];
    assert!(argmax_id < 262144, "Argmax token id out of vocab range.");
    println!(">> [TEST] Argmax token_id={} logit={:.4}", argmax_id, argmax_val);

    println!(">> [PASS] Local inference engine smoke test passed successfully!");
}

use nexus_edge_minion::ple::{Stage1PLE, Stage2PLE};

#[test]
fn test_35_layer_pipeline_validation() {
    let num_layers = 35;
    let layer_dim = 256;
    let total_dim = num_layers * layer_dim;

    let ple1 = Stage1PLE::new(num_layers, layer_dim);
    let raw_ple = vec![0.1f32; total_dim];
    let proj = vec![0.2f32; total_dim];
    let mut blended = vec![0.0f32; total_dim];

    // Stage 1 blend
    ple1.blend_stage1(&raw_ple, &proj, &mut blended);
    assert!((blended[0] - 0.212132).abs() < 1e-4);

    // Stage 2 layer pass across all 35 layers
    let residual = vec![0.5f32; layer_dim];
    let mut layer_states = vec![0.3f32; layer_dim];

    for _layer in 0..num_layers {
        Stage2PLE::forward_layer(&mut layer_states, &residual, 1.0);
    }

    assert_eq!(layer_states.len(), layer_dim);
}

use nexus_edge_minion::rope::{apply_rope_scalar, AttentionMode, RoPEConfig};

#[test]
fn test_rope_position_zero_invariant() {
    let config = RoPEConfig::new(AttentionMode::Global);
    let mut vec = vec![1.0f32; 256];
    let original = vec.clone();

    apply_rope_scalar(&mut vec, 0, &config);

    // At position 0, angle is 0 -> cos(0)=1, sin(0)=0 -> vector must remain unchanged
    for i in 0..256 {
        assert!(
            (vec[i] - original[i]).abs() < 1e-6,
            "Position 0 failed at index {}: got {}, expected {}",
            i,
            vec[i],
            original[i]
        );
    }
}

#[test]
fn test_rope_global_partial_rotary_bounds() {
    let config = RoPEConfig::new(AttentionMode::Global); // theta = 1e6, rotary_dim = 64
    config.validate();

    let mut vec: Vec<f32> = (0..256).map(|i| i as f32 + 1.0).collect();
    let original = vec.clone();

    apply_rope_scalar(&mut vec, 5, &config);

    // First 64 dimensions must be modified (rotated)
    let mut modified_count = 0;
    for i in 0..64 {
        if (vec[i] - original[i]).abs() > 1e-5 {
            modified_count += 1;
        }
    }
    assert!(
        modified_count > 0,
        "Global attention failed: expected rotation in the first 64 dimensions."
    );

    // Dimensions 64..256 must remain strictly invariant (identity rotation)
    for i in 64..256 {
        assert!(
            (vec[i] - original[i]).abs() < 1e-6,
            "Global attention invariant failure at index {}: got {}, expected {}",
            i,
            vec[i],
            original[i]
        );
    }
}

fn test_rope_sliding_full_rotary() {

fn find_tokenizer_json() -> Option<PathBuf> {
    let manifest_dir = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
    let candidates = [
        manifest_dir.join("tokenizer.json"),
        manifest_dir.join("../../tokenizer.json"),
        manifest_dir.join("../tokenizer.json"),
    ];
    candidates.into_iter().find(|p| p.exists())
}

#[test]
fn test_tokenizer_roundtrip() {
    let path = match find_tokenizer_json() {
        Some(p) => p,
        None => {
            println!(">> [SKIP] tokenizer.json not found in any candidate location. Skipping tokenizer smoke test.");
            return;
        }
    };
    println!(">> [TEST] Using tokenizer.json at: {}", path.display());

    let tok =
        NexusTokenizer::from_file(path.to_str().unwrap()).expect("Failed to load tokenizer.json");

    let text = "The capital of France is";
    let ids = tok.encode(text, true).expect("encode failed");
    println!(">> [TEST] \"{}\" -> {:?}", text, ids);
    assert!(!ids.is_empty(), "Encoding produced no tokens.");

    let decoded = tok.decode(&ids, true).expect("decode failed");
    println!(">> [TEST] decoded back -> \"{}\"", decoded);

    // Loose round-trip check -- tokenizers can add/normalize whitespace,
    // so we check substring containment rather than exact equality.
    assert!(
        decoded.to_lowercase().contains("capital") && decoded.to_lowercase().contains("france"),
        "Decoded text lost key content: got \"{}\"",
        decoded
    );

    println!(">> [PASS] Tokenizer round-trip smoke test passed successfully!");
}

const MAX_HISTORY_TURNS: usize = 12;
const SOFT_TOKEN_LIMIT: usize = 110_000;
const HARD_TOKEN_LIMIT: usize = 140_000;
const RESERVED_TOKENS: usize = 8192;

#[derive(Default, Serialize, Deserialize, Clone)]
pub struct ContextPacket {

#[derive(Debug, thiserror::Error)]
pub enum BuilderError {

#[derive(Default)]
pub struct MicroAgentBuilder {
    system_prompt: Option<String>,
    model: Option<String>,

impl MicroAgentBuilder {
    pub fn new() -> Self {
        Self { system_prompt: None, model: None }
    }

    pub fn system_prompt(mut self, prompt: &str) -> Self {
        self.system_prompt = Some(prompt.to_string());

    }

    pub fn model(mut self, model: &str) -> Self {

        }
        if self.model.is_none() {
            return Err(BuilderError::MissingField("model".to_string()));
        }

        Ok(MicroAgent { memory: Mutex::new(VecDeque::with_capacity(MAX_HISTORY_TURNS * 2)), tx })
    }

pub struct MicroAgent {
    pub memory: Mutex<VecDeque<Value>>,
    pub tx: broadcast::Sender<String>,
}

impl MicroAgent {
    pub fn builder() -> MicroAgentBuilder {
        MicroAgentBuilder::new()
    }

    pub fn init(tx: broadcast::Sender<String>) -> Result<Self> {
        Ok(MicroAgent { memory: Mutex::new(VecDeque::with_capacity(MAX_HISTORY_TURNS * 2)), tx })
    }

    pub async fn prepare_and_run(
        &self,
        client: &Client,
        api_key: &str,
        user_text: &str,
        context: ContextPacket,
    ) -> Result<(String, u128)> {
        let start = Instant::now();
        let default_model = std::env::var("NEXUS_HEAVY_MODEL").unwrap_or_else(|_| "nvidia/nemotron-3-ultra-550b-a55b".to_string());
    let model = context.model_override.as_deref().unwrap_or(&default_model);
        let url = format!(
            "https://generativelanguage.googleapis.com/v1beta/models/{}:streamGenerateContent?alt=sse&key={}",
            model,
            api_key.trim()
        );

        let system_instruction =
            context.system_prompt.unwrap_or_else(|| "You are NEXUS.".to_string());

        let mut enriched = format!(
            "SYSTEM: {}

USER REQUEST:
{}",
            system_instruction, user_text
        );
        if let Some(code) = &context.file_content {
            enriched += &format!(
                "

ACTIVE FILE CONTENT:
```
{}
```",
                code
            );
        }
        if let Some(logs) = &context.terminal_log {
            enriched += &format!(
                "

RECENT TERMINAL LOGS:
```
{}
```",
                logs
            );
        }

        let mut history: Vec<Value> = self.memory.lock().unwrap().iter().cloned().collect();
        trim_history_token_aware(&mut history, &enriched);

        history.push(json!({
            "role": "user",
            "parts": [{"text": enriched}]
        }));

        let payload = json!({
            "contents": history,
            "generationConfig": {
                "maxOutputTokens": 8192,
                "temperature": 0.15
            }
        });

        let response =
            client.post(&url).json(&payload).send().await.context("API request failed")?;
        let mut full_response = String::new();
        let mut byte_buffer = Vec::new();
        let mut stream = response.bytes_stream();

        while let Some(chunk_result) = stream.next().await {
            if let Ok(chunk) = chunk_result {
                byte_buffer.extend_from_slice(&chunk);
                while let Some(pos) = byte_buffer.iter().position(|&b| b == b'\n') {
                    let line_bytes: Vec<u8> = byte_buffer.drain(..=pos).collect();
                    let line = String::from_utf8_lossy(&line_bytes);
                    let clean = line.trim_start_matches("data: ").trim();
                    if clean.is_empty() || clean == "[DONE]" {
                        continue;
                    }

                    if let Ok(val) = serde_json::from_str::<Value>(clean) {
                        if let Some(parts) =
                            val.pointer("/candidates/0/content/parts").and_then(|v| v.as_array())
                        {
                            for part in parts {
                                if part.get("thought").and_then(|t| t.as_bool()).unwrap_or(false) {
                                    if let Some(token) = part.get("text").and_then(|t| t.as_str()) {
                                        let _ = self.tx.send(
                                            json!({"type": "NEURAL_THOUGHT", "payload": token})
                                                .to_string(),
                                        );
                                    }
                                    continue;
                                }

                                if let Some(token) = part.get("text").and_then(|t| t.as_str()) {
                                    full_response.push_str(token);
                                    let _ = self.tx.send(
                                        json!({"type": "NEURAL_STREAM", "payload": token})
                                            .to_string(),
                                    );
                                }
                            }
                        }
                    }
                }
            }
        }

        {
            let mut mem = self.memory.lock().unwrap();
            mem.push_back(json!({"role": "user", "parts": [{"text": user_text}]}));
            if !full_response.is_empty() {
                mem.push_back(json!({"role": "model", "parts": [{"text": &full_response}]}));
            }
            while mem.len() > MAX_HISTORY_TURNS * 2 {
                mem.pop_front();
            }
        }

        Ok((full_response, start.elapsed().as_millis()))
    }
}

fn rough_token_count(text: &str) -> usize {
    let chars = text.chars().count();
    let words = text.split_whitespace().count();
    let is_code_heavy = text.contains("```") || text.contains("{") || text.contains("fn ");
    let factor = if is_code_heavy { 3.6 } else { 4.1 };
    ((chars as f64 / factor) + (words as f64 * 0.3)).ceil() as usize
}

fn estimate_message_tokens(msg: &Value) -> usize {
    let empty_vec = Vec::new();
    let parts = msg.pointer("/parts").and_then(|v| v.as_array()).unwrap_or(&empty_vec);
    parts
        .iter()
        .filter_map(|p| {
            p.get("text")
                .and_then(|t| t.as_str())
                .or_else(|| p.get("functionCall").and_then(|fc| fc.as_str()))
                .or_else(|| p.get("functionResponse").and_then(|fr| fr.as_str()))
        })
        .map(rough_token_count)
        .sum::<usize>()
        + 10
}

fn trim_history_token_aware(history: &mut Vec<Value>, current_user_msg: &str) {
    if history.is_empty() {
        return;
    }

    let current_tokens = rough_token_count(current_user_msg) + RESERVED_TOKENS;
    let mut total_tokens: usize = history.iter().map(estimate_message_tokens).sum();
    total_tokens += current_tokens;

    if total_tokens <= SOFT_TOKEN_LIMIT {
        return;
    }

    while total_tokens > SOFT_TOKEN_LIMIT && history.len() >= 4 {
        if let (Some(old_user), Some(old_model)) = (history.first(), history.get(1)) {
            let removed = estimate_message_tokens(old_user) + estimate_message_tokens(old_model);
            total_tokens = total_tokens.saturating_sub(removed);
        }
        history.drain(0..2);
    }

    while total_tokens > HARD_TOKEN_LIMIT && !history.is_empty() {
        let removed = estimate_message_tokens(&history[0]);
        total_tokens = total_tokens.saturating_sub(removed);
        history.remove(0);
    }
}

use anyhow::{anyhow, Result};
use reqwest::Client;
use serde_json::{json, Value};
use std::env;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ProviderKind {
    Google,
    OpenAiCompatible, // Covers NVIDIA NIM, Ollama, vLLM, Groq, Together
}

impl ProviderKind {
    pub fn detect(model_or_provider: &str) -> Self {
        let env_provider = env::var("NEXUS_PROVIDER").unwrap_or_default().to_lowercase();
        if env_provider == "nvidia" || env_provider == "openai" || env_provider == "ollama" {
            return ProviderKind::OpenAiCompatible;
        }

        if model_or_provider.contains('/')
            || model_or_provider.contains("llama")
            || model_or_provider.contains("deepseek")
            || model_or_provider.contains("nvidia")
            || model_or_provider.contains("mistral")
        {
            ProviderKind::OpenAiCompatible
        } else {
            ProviderKind::Google
        }
    }
}

pub struct UniversalProvider;

impl UniversalProvider {
    pub async fn generate(
        client: &Client,
        api_key: &str,
        model: &str,
        system_prompt: Option<&str>,
        user_prompt: &str,
        temperature: f32,
        max_tokens: u32,
    ) -> Result<String> {
        let kind = ProviderKind::detect(model);

        match kind {
            ProviderKind::Google => {
                let base_url = env::var("NEXUS_API_GATEWAY")
                    .unwrap_or_else(|_| "https://generativelanguage.googleapis.com".to_string());
                let url = format!(
                    "{}/v1beta/models/{}:generateContent?key={}",
                    base_url.trim_end_matches('/'),
                    model.replace("models/", ""),
                    api_key.trim()
                );

                let mut contents = Vec::new();
                if let Some(sys) = system_prompt {
                    contents.push(json!({
                        "role": "user",
                        "parts": [{"text": format!("SYSTEM INSTRUCTION: {}", sys)}]
                    }));
                }
                contents.push(json!({
                    "role": "user",
                    "parts": [{"text": user_prompt}]
                }));

                let payload = json!({
                    "contents": contents,
                    "generationConfig": {
                        "temperature": temperature,
                        "maxOutputTokens": max_tokens
                    }
                });

                let res = client.post(&url).json(&payload).send().await?;
                let body: Value = res.json().await.unwrap_or_default();

                if let Some(err) = body.pointer("/error/message").and_then(|v| v.as_str()) {
                    return Err(anyhow!("Google API Error: {}", err));
                }

                let text = body
                    .pointer("/candidates/0/content/parts/0/text")
                    .and_then(|v| v.as_str())
                    .unwrap_or_default();

                Ok(text.replace("```json", "").replace("```", "").trim().to_string())
            }

            ProviderKind::OpenAiCompatible => {
                let base_url = env::var("NVIDIA_API_GATEWAY")
                    .or_else(|_| env::var("NEXUS_OPENAI_GATEWAY"))
                    .unwrap_or_else(|_| "https://generativelanguage.googleapis.com/v1beta".to_string());

                let url = format!("{}/chat/completions", base_url.trim_end_matches('/'));

                let mut messages = Vec::new();
                if let Some(sys) = system_prompt {
                    messages.push(json!({"role": "system", "content": sys}));
                }
                messages.push(json!({"role": "user", "content": user_prompt}));

                let payload = json!({
                    "model": model,
                    "messages": messages,
                    "temperature": temperature,
                    "max_tokens": max_tokens
                });

                let auth_key = env::var("NVIDIA_API_KEY").unwrap_or_else(|_| api_key.to_string());

                let res = client
                    .post(&url)
                    .header("Authorization", format!("Bearer {}", auth_key.trim()))
                    .header("Content-Type", "application/json")
                    .json(&payload)
                    .send()
                    .await?;

                let body: Value = res.json().await.unwrap_or_default();

                if let Some(err) = body.pointer("/error/message").and_then(|v| v.as_str()) {
                    return Err(anyhow!("OpenAI/NIM API Error: {}", err));
                }

                let text = body
                    .pointer("/choices/0/message/content")
                    .and_then(|v| v.as_str())
                    .unwrap_or_default();

                Ok(text.replace("```json", "").replace("```", "").trim().to_string())
            }
        }
    }
}

impl VectorVault {
    pub fn new(dimension: usize, bit_width: usize) -> Self {

            index: TurboQuantIndex::new(dimension, bit_width),
            id_map: HashMap::new(),
            current_idx: 0,

    pub fn commit_state(&mut self, id: u64, embedding: &[f32]) -> Result<(), String> {

                "Dimension mismatch: expected {}, got {}",
                self.dimension,
                embedding.len()
            ));
        }
        self.index.add(embedding);
        self.id_map.insert(self.current_idx, id);
        self.current_idx += 1;
        Ok(())
    }

    pub fn recall_context(
        &self,
        query_embedding: &[f32],
        k: usize,
    ) -> Result<(Vec<f32>, Vec<u64>), String> {
        if query_embedding.len() != self.dimension {
            return Err("Query embedding mismatch".into());
        }
        let results = self.index.search(query_embedding, k);
        let mut resolved_ids = Vec::new();
        for raw_idx in results.indices {
            if let Some(&real_id) = self.id_map.get(&(raw_idx as usize)) {
                resolved_ids.push(real_id);
            }
        }
        Ok((results.scores, resolved_ids))
    }

    pub fn persist_vault<P: AsRef<Path>>(&self, path: P) -> Result<(), String> {
        let path_str = path.as_ref().to_string_lossy().into_owned();
        self.index.write(&path_str).map_err(|e| e.to_string())?;
        Ok(())
    }
}

//! Async bridge for SQLite using parking_lot::Mutex (no poisoning).

use parking_lot::Mutex;
use std::sync::Arc;
use thiserror::Error;

#[derive(Debug, Error)]
pub enum DbError {
    #[error("SQLite error: {0}")]
    Sqlite(#[from] rusqlite::Error),
    #[error("Join error: {0}")]
    Join(#[from] tokio::task::JoinError),
}

pub async fn db_exec<F, R>(
    conn: Arc<Mutex<rusqlite::Connection>>,
    f: F,
) -> Result<R, DbError>
where
    F: FnOnce(&rusqlite::Connection) -> Result<R, rusqlite::Error> + Send + 'static,
    R: Send + 'static,
{
    tokio::task::spawn_blocking(move || {
        let guard = conn.lock();
        f(&guard).map_err(DbError::Sqlite)
    })
    .await
    .map_err(DbError::Join)?
}

pub async fn db_exec_mut<F, R>(
    conn: Arc<Mutex<rusqlite::Connection>>,
    f: F,
) -> Result<R, DbError>
where
    F: FnOnce(&mut rusqlite::Connection) -> Result<R, rusqlite::Error> + Send + 'static,
    R: Send + 'static,
{
    tokio::task::spawn_blocking(move || {
        let mut guard = conn.lock();
        f(&mut guard).map_err(DbError::Sqlite)
    })
    .await
    .map_err(DbError::Join)?
}

pub struct GenomeVault {
    genomes: Arc<Mutex<HashMap<String, GenomeRecord>>>,
    max_retain: usize,
}

impl GenomeVault {
    pub fn new(max_retain: usize) -> Self {
        Self {
            genomes: Arc::new(Mutex::new(HashMap::new())),
            max_retain,
        }
    }

    pub async fn store_high_fitness(&self, objective: &str, record: GenomeRecord) {
        let mut guard = self.genomes.lock().await;
        guard.insert(objective.to_string(), record);

        if guard.len() > self.max_retain {
            let mut sorted: Vec<_> = guard.drain().collect();
            sorted.sort_by(|a, b| b.1.composite_score.partial_cmp(&a.1.composite_score).unwrap());
            sorted.truncate(self.max_retain);
            *guard = sorted.into_iter().collect();
        }
    }

        let guard = self.genomes.lock().await;
        guard.get(objective).cloned()
    }

    pub async fn get_global_best(&self) -> Option<GenomeRecord> {
        let guard = self.genomes.lock().await;
        guard.values()
            .max_by(|a, b| a.composite_score.partial_cmp(&b.composite_score).unwrap())
            .cloned()
    }
}

        Self::new(50)
    }

pub fn calculate_entropy(handle: u32) -> f32 {
    if let Some(vmo) = VMO_REGISTRY.get(&handle) {
        let data = vmo.value();
        if data.is_empty() {
            return 0.0;
        }

        let mut counts = [0u32; 256];
        for &b in data {
            counts[b as usize] += 1;
        }

        let mut entropy = 0.0f32;
        let len = data.len() as f32;
        for &c in &counts {
            if c > 0 {
                let p = c as f32 / len;
                entropy -= p * p.log2();
            }
        }
        entropy / 8.0
    } else {
        0.0
    }
}

pub struct SafeKernel;

impl SafeKernel {
    pub fn create_vmo(size: usize) -> Result<u32, KernelError> {
        if size == 0 || size > 16_777_216 {
            return Err(KernelError::InvalidArgs);
        }
        let handle = NEXT_HANDLE.fetch_add(1, Ordering::Relaxed);
        VMO_REGISTRY.insert(handle, vec![0u8; size]);
        Ok(handle)
    }

    pub fn close_handle(handle: u32) {
        VMO_REGISTRY.remove(&handle);
    }
}

use crate::kernel::error::KernelError;
use crate::kernel::safe_kernel::SafeKernel;
use crate::kernel::VMO_REGISTRY;

pub struct SafeVmo {
    handle: u32,
    pub size: usize,
}

impl SafeVmo {
    pub fn new(size: usize) -> Result<Self, KernelError> {
        let handle = SafeKernel::create_vmo(size)?;
        Ok(Self { handle, size })
    }

    pub fn write(&self, offset: usize, data: &[u8]) -> Result<(), KernelError> {
        if self.size == 0 {
            return Err(KernelError::InvalidArgs);
        }
        let end = offset.checked_add(data.len()).ok_or(KernelError::InvalidArgs)?;
        if end > self.size {
            return Err(KernelError::InvalidArgs);
        }

        if let Some(mut vmo) = VMO_REGISTRY.get_mut(&self.handle) {
            vmo[offset..end].copy_from_slice(data);
            Ok(())
        } else {
            Err(KernelError::InvalidArgs)
        }
    }

    pub fn read_into(&self, offset: usize, buffer: &mut [u8]) -> Result<(), KernelError> {
        let end = offset.checked_add(buffer.len()).ok_or(KernelError::InvalidArgs)?;
        if end > self.size {
            return Err(KernelError::InvalidArgs);
        }

        if let Some(vmo) = VMO_REGISTRY.get(&self.handle) {
            buffer.copy_from_slice(&vmo[offset..end]);
            Ok(())
        } else {
            Err(KernelError::InvalidArgs)
        }
    }

    pub fn handle(&self) -> u32 {
        self.handle
    }
}

impl Drop for SafeVmo {
    fn drop(&mut self) {
        SafeKernel::close_handle(self.handle);
    }
}

#[derive(Debug, Clone)]
pub struct ModelQuota {
    pub rpm_limit: u32,
    pub tpm_limit: u32,
}

impl ModelQuota {
    fn for_model(model: &str) -> Self {
        match model {
            m if m.contains("flash-lite") => Self { rpm_limit: 10, tpm_limit: 100_000 },
            m if m.contains("flash") => Self { rpm_limit: 5, tpm_limit: 60_000 },
            m if m.contains("31b") => Self { rpm_limit: 3, tpm_limit: 50_000 },
            m if m.contains("pro") => Self { rpm_limit: 2, tpm_limit: 30_000 },
            m if m.contains("research") => Self { rpm_limit: 1, tpm_limit: 10_000 },
            _ => Self { rpm_limit: 3, tpm_limit: 30_000 },
        }
    }
}

#[derive(Debug)]
struct WindowEntry {
    timestamp: Instant,
    tokens: u32,
}

#[derive(Debug, Default)]
struct ModelWindow {
    entries: Vec<WindowEntry>,
}

impl ModelWindow {
    fn purge_stale(&mut self) {
        let cutoff = Instant::now() - Duration::from_secs(60);
        self.entries.retain(|e| e.timestamp > cutoff);
    }
    fn token_sum(&self) -> u32 {
        self.entries.iter().map(|e| e.tokens).sum()
    }
    fn ms_until_slot_free(&self) -> u64 {
        if self.entries.is_empty() {
            return 0;
        }
        let oldest = self.entries.iter().map(|e| e.timestamp).min().unwrap();
        (Duration::from_secs(60).saturating_sub(oldest.elapsed())).as_millis() as u64 + 100
    }
}

pub struct TokenLedger {
    windows: Mutex<HashMap<String, ModelWindow>>,
}

impl TokenLedger {
    pub fn new() -> Self {
        Self { windows: Mutex::new(HashMap::new()) }
    }

    pub fn check_and_reserve(&self, model: &str, estimated_tokens: u32) -> Result<(), u64> {
        let quota = ModelQuota::for_model(model);
        let mut map = self.windows.lock().unwrap();
        let win = map.entry(model.to_string()).or_default();
        win.purge_stale();
        if win.entries.len() as u32 >= quota.rpm_limit {
            return Err(win.ms_until_slot_free());
        }
        if win.token_sum() + estimated_tokens > quota.tpm_limit {
            return Err(win.ms_until_slot_free());
        }
        win.entries.push(WindowEntry { timestamp: Instant::now(), tokens: estimated_tokens });
        Ok(())
    }
}

pub static LEDGER: Lazy<TokenLedger> = Lazy::new(TokenLedger::new);

/// Returns Ok(()) when a quota slot is reserved.
/// Returns Err(message) after MAX_ATTEMPTS — caller MUST NOT fire the API on Err.
pub async fn wait_for_quota(model: &str, estimated_tokens: u32) -> Result<(), String> {
    const MAX_ATTEMPTS: usize = 10;
    for attempt in 0..MAX_ATTEMPTS {
        match LEDGER.check_and_reserve(model, estimated_tokens) {
            Ok(()) => return Ok(()),
            Err(wait_ms) => {
                let jitter = generate_seed() % 500;
                tracing::warn!(
                    "[LEDGER] {} ceiling hit — waiting {}ms (attempt {}/{}).",
                    model,
                    wait_ms + jitter,
                    attempt + 1,
                    MAX_ATTEMPTS
                );
                tokio::time::sleep(Duration::from_millis(wait_ms + jitter)).await;
            }
        }
    }
    Err(format!("[LEDGER] Quota for '{}' unavailable after {} attempts.", model, MAX_ATTEMPTS))
}

fn http_rpm_limit() -> u32 {
    std::env::var("NEXUS_HTTP_RPM").ok().and_then(|v| v.parse().ok()).unwrap_or(60)
}

static HTTP_WINDOWS: Lazy<Mutex<HashMap<String, Vec<Instant>>>> =
    Lazy::new(|| Mutex::new(HashMap::new()));

pub fn http_check(ip: &str) -> Result<(), u64> {
    let limit = http_rpm_limit();
    let cutoff = Instant::now() - Duration::from_secs(60);
    let mut map = HTTP_WINDOWS.lock().unwrap();
    let win = map.entry(ip.to_string()).or_default();
    win.retain(|&ts| ts > cutoff);
    if win.len() as u32 >= limit {
        let oldest = *win.first().unwrap();
        return Err(
            (Duration::from_secs(60).saturating_sub(oldest.elapsed())).as_millis() as u64 + 100
        );
    }
    win.push(Instant::now());
    Ok(())
}

fn generate_seed() -> u64 {
    use std::hash::{BuildHasher, Hasher};
    let t = SystemTime::now().duration_since(SystemTime::UNIX_EPOCH).unwrap_or_default().as_nanos()
        as u64;
    let mut h = std::collections::hash_map::RandomState::new().build_hasher();
    h.write_u64(t);
    h.finish()
}

pub fn cosine_similarity(a: &[f32], b: &[f32]) -> f32 {
    if a.len() != b.len() || a.is_empty() {
        return 0.0;
    }
    let dot: f32 = a.iter().zip(b.iter()).map(|(x, y)| x * y).sum();
    let norm_a: f32 = a.iter().map(|x| x * x).sum::<f32>().sqrt();
    let norm_b: f32 = b.iter().map(|x| x * x).sum::<f32>().sqrt();
    if norm_a == 0.0 || norm_b == 0.0 {
        return 0.0;
    }
    dot / (norm_a * norm_b)
}

//! Lock-free VMO registry using DashMap.
//!
//! Previously used `RwLock<HashMap<VmoHandle, Arc<SafeVmo>>>` which serialised
//! every write (allocate/release) behind a single global lock. Under concurrent
//! agent workloads this became a hot contention point.
//!
//! DashMap shards the map into N buckets (default: 2× logical CPUs). Each
//! shard has its own RwLock, so concurrent operations on different handles
//! proceed in parallel with no cross-shard blocking.
//!
//! API is fully backward-compatible — callers need zero changes.

use dashmap::DashMap;
use std::sync::{
    atomic::{AtomicU32, Ordering},
    Arc,
};

pub type VmoHandle = u32;

pub struct SafeVmo {
    pub handle: VmoHandle,
    pub data: Vec<u8>,
    pub timestamp: u64,
}

pub struct VmoHandler {
    /// Sharded concurrent map — no global lock on reads or cross-shard writes.
    registry: DashMap<VmoHandle, Arc<SafeVmo>>,
    /// Atomic counter replaces the second RwLock entirely.
    next_handle: AtomicU32,
}

impl VmoHandler {
    pub fn new() -> Self {
        Self { registry: DashMap::new(), next_handle: AtomicU32::new(1) }
    }

    pub fn allocate_cognitive_window(&self, raw_data: &[u8]) -> Result<VmoHandle, String> {
        // fetch_add is lock-free — no mutex involved at all for handle allocation.
        let handle = self.next_handle.fetch_add(1, Ordering::Relaxed);
        let vmo = Arc::new(SafeVmo {
            handle,
            data: raw_data.to_vec(),
            timestamp: std::time::SystemTime::now()
                .duration_since(std::time::UNIX_EPOCH)
                .unwrap_or_default()
                .as_secs(),
        });
        self.registry.insert(handle, vmo);
        Ok(handle)
    }

    pub fn read_cognitive_window(&self, handle: VmoHandle) -> Result<Vec<u8>, String> {
        self.registry
            .get(&handle)
            .map(|v| v.data.clone())
            .ok_or_else(|| format!("Invalid handle: {}", handle))
    }

    pub fn export_provenance_checkpoint(
        &self,
        handle: VmoHandle,
    ) -> Result<(VmoHandle, Vec<u8>, u64), String> {
        self.registry
            .get(&handle)
            .map(|v| (v.handle, v.data.clone(), v.timestamp))
            .ok_or_else(|| format!("Handle not found: {}", handle))
    }

    pub fn release_cognitive_window(&self, handle: VmoHandle) -> Result<(), String> {
        self.registry
            .remove(&handle)
            .map(|_| ())
            .ok_or_else(|| format!("Handle not found on release: {}", handle))
    }

    /// Returns the number of live VMO allocations.
    pub fn active_count(&self) -> usize {
        self.registry.len()
    }
}

impl Default for VmoHandler {
    fn default() -> Self {
        Self::new()
    }
}

/// When the Translator LLM produces an empty action list, this parses the
/// Strategist's raw proposed-action text directly as a fallback. Pure string
/// parsing -- no side effects, no borrows of CorticalLoop state -- which is
/// exactly why it's the first piece pulled out of the ~500-line
/// ignite_with_evidence function in cortical_loop.rs.
///
/// Returns None if the text doesn't match any known tool prefix or labeled
/// multi-line form; the caller falls back further to raw ExecuteCommand
/// (see cortical_loop.rs).
///
/// NOTE: deliberately does NOT include a write_file branch. That branch was
/// removed from cortical_loop.rs earlier this session (see [CORTEX SAFETY]
/// log line) because raw, un-JSON-escaped Strategist text written straight
/// to disk caused real file corruption. Do not re-add it here.
pub fn intercept_fallback_action(action_str: &str) -> Option<AgentAction> {
    let action_str = action_str.trim();

    // --- Prefix form (translator / single-line) ---
    if let Some(cmd) = action_str.strip_prefix("execute_command") {
        let clean_cmd = cmd.trim().trim_matches('\'').trim_matches('"').to_string();
        return Some(AgentAction::ExecuteCommand {
            command: clean_cmd,
            reasoning: "Intercepted fallback execute_command".to_string(),
            confidence: 0.9,
        });
    }
    if action_str.starts_with("search_code") {
        let parts: Vec<&str> = action_str.split_whitespace().collect();
        if parts.len() >= 3 {
            let path = parts[1].to_string();
            let query = parts[2..].join(" ").trim_matches('\'').trim_matches('"').to_string();
            return Some(AgentAction::SearchCode {
                path,
                query,
                reasoning: "Intercepted fallback search_code".to_string(),
            });
        }
        return None;
    }
    if action_str.starts_with("read_file_chunk") {
        let parts: Vec<&str> = action_str.split_whitespace().collect();
        if parts.len() >= 2 {
            let path = parts[1].to_string();
            let start_line = parts.get(2).and_then(|s| s.parse().ok()).unwrap_or(0);
            let max_lines = parts.get(3).and_then(|s| s.parse().ok()).unwrap_or(500);
            return Some(AgentAction::ReadFileChunk {
                path,
                start_line,
                max_lines,
                reasoning: "Intercepted fallback read_file_chunk".to_string(),
            });
        }
        return None;
    }
    if let Some(stripped) = action_str.strip_prefix("diff_patch ") {
        let parts: Vec<&str> = stripped.split_whitespace().collect();
        if parts.len() >= 3 {
            let path = parts[0].to_string();
            let search = parts[1].to_string();
            let replace = parts[2..].join(" ");
            return Some(AgentAction::DiffPatch {
                path,
                search,
                replace,
                reasoning: "Intercepted fallback diff_patch".to_string(),
            });
        }
        return None;
    }
    if let Some(stripped) = action_str.strip_prefix("patch_function ") {
        let parts: Vec<&str> = stripped.split_whitespace().collect();
        if parts.len() >= 3 {
            let path = parts[0].to_string();
            let function_name = parts[1].to_string();
            let new_body = parts[2..].join(" ");
            return Some(AgentAction::PatchFunction {
                path,
                function_name,
                new_body,
                reasoning: "Intercepted fallback patch_function".to_string(),
            });
        }
        return None;
    }
    if let Some(path) = action_str.strip_prefix("read_file ") {
        return Some(AgentAction::ReadFile {
            path: path.trim().to_string(),
            reasoning: "Intercepted fallback read_file".to_string(),
        });
    }
    if let Some(raw_query) = action_str.strip_prefix("search_memory") {
        let query = raw_query.trim().trim_matches('\'').trim_matches('"').to_string();
        return Some(AgentAction::SearchMemory {
            query,
            reasoning: "Intercepted fallback search_memory".to_string(),
        });
    }

    // --- Labeled multi-line form (Strategist NL) ---
    // Matches proposals like:
    //   Tool: read_file
    //   Target Path: /tmp/openfang/SECURITY.md
    // Does not handle write_file (same safety contract as above).
    if let Some(action) = intercept_labeled_proposal(action_str) {
        return Some(action);
    }

    None
}

/// Strategist-style labeled proposals (Tool: / Target Path: / Command: / Query:).
fn intercept_labeled_proposal(text: &str) -> Option<AgentAction> {
    let lower = text.to_lowercase();

    // read_file
    if lower.contains("tool: read_file")
        || lower.contains("use the `read_file`")
        || lower.contains("use the read_file")
    {
        if let Some(path) =
            labeled_value(text, &["Target Path:", "target path:", "Path:", "path:"])
        {
            if looks_like_path(&path) {
                return Some(AgentAction::ReadFile {
                    path,
                    reasoning: "Intercepted labeled read_file".to_string(),
                });
            }
        }
    }

    // search_code
    if lower.contains("tool: search_code")
        || lower.contains("use the `search_code`")
        || lower.contains("use the search_code")
    {
        let path = labeled_value(text, &["Path:", "path:"]);
        let query = labeled_value(text, &["Query:", "query:"]);
        if let (Some(path), Some(query)) = (path, query) {
            if looks_like_path(&path) && !query.is_empty() {
                return Some(AgentAction::SearchCode {
                    path,
                    query,
                    reasoning: "Intercepted labeled search_code".to_string(),
                });
            }
        }
    }

    // execute_command — require Tool/Command labels, not bare "git clone" anywhere
    if lower.contains("tool: execute_command")
        || lower.contains("use the execute_command")
        || lower.contains("use the `execute_command`")
    {
        if let Some(cmd) = labeled_value(text, &["Command:", "command:"]) {
            let cmd = cmd.trim().trim_matches('`').to_string();
            if !cmd.is_empty() && !cmd.to_lowercase().starts_with("tool") {
                return Some(AgentAction::ExecuteCommand {
                    command: cmd,
                    reasoning: "Intercepted labeled execute_command".to_string(),
                    confidence: 0.85,
                });
            }
        }
    }

    None
}

fn labeled_value(text: &str, markers: &[&str]) -> Option<String> {
    for marker in markers {
        for line in text.lines() {
            let trimmed = line.trim();
            if trimmed.len() >= marker.len()
                && trimmed[..marker.len()].eq_ignore_ascii_case(marker)
            {
                let rest = trimmed[marker.len()..]
                    .trim()
                    .trim_matches('`')
                    .trim_matches('"');
                if !rest.is_empty() {
                    return Some(rest.to_string());
                }
            }
        }
    }
    None
}

fn looks_like_path(p: &str) -> bool {
    p.starts_with('/')
        || p.starts_with("./")
        || p.ends_with(".rs")
        || p.ends_with(".toml")
        || p.ends_with(".md")
        || p.ends_with(".json")
        || p.contains('/')
}

#[derive(Clone)]
pub struct AgenticRuntime {
    cmd_sender: mpsc::Sender<AgentCommand>,
}

impl AgenticRuntime {
    pub fn new(execution_budget_ms: u64) -> Self {
        let (cmd_sender, mut cmd_receiver) = mpsc::channel::<AgentCommand>(32);
        tokio::spawn(async move {
            while let Some(cmd) = cmd_receiver.recv().await {
                match cmd {
                    AgentCommand::DispatchTask { prompt, respond_to } => {
                        let result =
                            Self::process_execution_loop(&prompt, execution_budget_ms).await;
                        let _ = respond_to.send(result);
                    }
                }
            }
        });
        Self { cmd_sender }
    }

    pub async fn dispatch(&self, prompt: &str) -> RuntimeResult<()> {
        let (tx, rx) = oneshot::channel();
        self.cmd_sender
            .send(AgentCommand::DispatchTask { prompt: prompt.to_string(), respond_to: tx })
            .await
            .map_err(|e| format!("Failed to queue task: {}", e))?;
        rx.await.map_err(|e| format!("Runtime channel dropped: {}", e))?
    }

    async fn process_execution_loop(prompt: &str, _budget: u64) -> RuntimeResult<()> {
        let raw_source = "<html><body><div id='editor' contenteditable='true'></div></body></html>";
        let _compressed_context = DomBridge::minify_dom(raw_source)?;
        tracing::info!("[COGNITIVE FSM] Step evaluation processed: {}", prompt);

    }

    Store(String),
    Recall(String),
    Shutdown,
}

    tx: mpsc::Sender<MemoryEvent>,
}

impl MemoryActor {
    pub fn start() -> (Self, JoinHandle<()>) {
        let (tx, mut rx) = mpsc::channel::<MemoryEvent>(64);

        let handle = tokio::spawn(async move {
            while let Some(event) = rx.recv().await {
                match event {
                    MemoryEvent::Store(data) => {
                        println!("[MEMORY] storing: {}", data);
                    }

                    MemoryEvent::Recall(query) => {
                        println!("[MEMORY] recalling: {}", query);
                    }

                    MemoryEvent::Shutdown => {
                        println!("[MEMORY] shutdown");
                        break;
                    }
                }
            }
        });

        (Self { tx }, handle)
    }

    pub async fn send(
        &self,
        event: MemoryEvent,
    ) -> Result<(), mpsc::error::SendError<MemoryEvent>> {
        self.tx.send(event).await
    }
}

pub struct ArbiterAgent {
    pub agent: Arc<MicroAgent>,
}

impl ArbiterAgent {
    pub fn new(agent: Arc<MicroAgent>) -> Self {
        Self { agent }
    }

    pub fn validate_action(action_type: &str, command: &str) -> Result<(), String> {
        let dangerous_patterns = [
            "rm -rf /",
            "rm -rf ~",
            "mkfs",
            "> /dev/sda",
            "> /dev/nvme",
            "chmod 777 /",
            "/etc/shadow",
            "mv /etc/",
            "mv /usr/",
            ":(){ :|:& };:",
            "dd if=/dev/zero",
            "fork bomb",
            "sudo su",
            "sudo -i",
            "sudo bash",
            "sudo sh",
            "passwd root",
            "chown root",
            "$'\\x",
            "\\u0072\\u006d",
            "base64 -d |",
            "eval $(",
            "exec(",
            "curl | bash",
            "wget | bash",
            "curl | sh",
            "history -c",
            "unset HISTFILE",
        ];

        let normalized = command.to_lowercase();

        // Narrow, explicit exception: allow `rm -rf /tmp/...` for legitimate
        // local temp-folder cleanup without loosening the general root-path
        // rm -rf block (which the broad "rm -rf /" pattern below restores).
        let is_scoped_tmp_cleanup =
            normalized.contains("rm -rf /tmp/") && !normalized.contains("rm -rf /tmp/../");

        for pattern in dangerous_patterns.iter() {
            if normalized.contains(pattern) {
                if *pattern == "rm -rf /" && is_scoped_tmp_cleanup {
                    continue;
                }
                return Err(format!(
                    "SANDBOX_VIOLATION: Threat pattern '{}' detected in {}.",
                    pattern, action_type
                ));
            }
        }

        if normalized.contains("../../../") || normalized.contains("/etc/passwd") {
            return Err("SANDBOX_VIOLATION: Directory traversal attempt blocked.".to_string());
        }

        if (normalized.contains("| bash")
            || normalized.contains("| sh")
            || normalized.contains("|bash")
            || normalized.contains("|sh"))
            && !normalized.starts_with("echo")
        {
            return Err("SANDBOX_VIOLATION: Pipe-to-shell pattern blocked.".to_string());
        }

        Ok(())
    }

    pub fn evaluate_entropy_delta(base_entropy: f32, new_entropy: f32) -> Result<(), String> {
        let delta = (new_entropy - base_entropy).abs();
        if delta > 0.5 {
            return Err(format!("ENTROPY_ANOMALY: Delta of {:.4} exceeds 0.5 threshold.", delta));
        }
        Ok(())
    }

    pub fn resolve_consensus(alpha_vector: &[f32], beta_vector: &[f32]) -> bool {
        if alpha_vector.len() != beta_vector.len() {
            return false;
        }
        let diff: f32 =
            alpha_vector.iter().zip(beta_vector.iter()).map(|(a, b)| (a - b).abs()).sum();
        diff < 0.5
    }

    pub fn bft_consensus(vectors: &[Vec<f32>], epsilon: f32) -> Option<Vec<f32>> {
        if vectors.is_empty() {
            return None;
        }
        let n = vectors.len();
        let threshold = (2 * n / 3) + 1;
        let dim = vectors[0].len();
        if vectors.iter().any(|v| v.len() != dim) {
            return None;
        }

        let (best_idx, neighbor_count) = (0..n)
            .map(|i| {
                let neighbors = (0..n)
                    .filter(|&j| {
                        vectors[i]
                            .iter()
                            .zip(vectors[j].iter())
                            .map(|(a, b)| (a - b).abs())
                            .sum::<f32>()
                            < epsilon
                    })
                    .count();
                (i, neighbors)
            })
            .max_by_key(|&(_, count)| count)?;

        if neighbor_count < threshold {
            return None;
        }

        let agreeing: Vec<&Vec<f32>> = (0..n)
            .filter(|&j| {
                vectors[best_idx]
                    .iter()
                    .zip(vectors[j].iter())
                    .map(|(a, b)| (a - b).abs())
                    .sum::<f32>()
                    < epsilon
            })
            .map(|j| &vectors[j])
            .collect();

        let centroid: Vec<f32> = (0..dim)
            .map(|k| agreeing.iter().map(|v| v[k]).sum::<f32>() / agreeing.len() as f32)
            .collect();

        Some(centroid)
    }
}

use anyhow::{anyhow, Result};
use std::fs;
use tree_sitter::{Parser, Query, QueryCursor};
pub struct AstEditor;
impl AstEditor {
    /// update_function: Finds a function by name and replaces its entire body.
    /// This is the "Semantic Precision" upgrade to replace simple string diffs.
    pub fn read_function(path: &str, function_name: &str) -> Result<String> {
        let source_code = fs::read_to_string(path)?;
        let mut parser = Parser::new();
        parser.set_language(&tree_sitter_rust::LANGUAGE.into())?;
        let tree = parser
            .parse(&source_code, None)
            .ok_or_else(|| anyhow!("AST_PARSE_FAIL: Could not parse {}", path))?;
        let query_str = format!(
            "(function_item name: (identifier) @name (#eq? @name \"{}\")) @func",
            function_name
        );
        let query = Query::new(&tree_sitter_rust::LANGUAGE.into(), &query_str)?;
        let mut cursor = QueryCursor::new();
        let matches = cursor.matches(&query, tree.root_node(), source_code.as_bytes());
        for m in matches {
            for capture in m.captures {
                if capture.index == 1 {
                    let node = capture.node;
                    let start = node.start_byte();
                    let end = node.end_byte();
                    return Ok(source_code[start..end].to_string());
                }
            }
        }
        Err(anyhow!("FUNCTION_NOT_FOUND: Could not find function '{}' in {}", function_name, path))
    }

    /// Validates that `new_body` is exactly one function definition named
    /// `function_name` and nothing else -- no top-level `use` statements, no
    /// extra functions, no `mod` blocks alongside it. Without this, a
    /// Strategist proposal whose `new_body` accidentally (or confidently)
    /// contains a whole re-typed file gets spliced in verbatim: real
    /// incident -- new_body containing `use axum::{...}`, a duplicate
    /// `#[tokio::main]`, and a `mod tests` block all landed inside what was
    /// meant to be just the function body, producing duplicate-import
    /// compile errors. A naive "does it parse" check does not catch this,
    /// since each individual piece (a use statement, a second `fn main`) is
    /// syntactically valid Rust on its own -- duplicate-definition is a
    /// semantic error tree-sitter's grammar-only parser never sees. This
    /// check instead requires new_body to parse as a standalone fragment
    /// containing exactly one named top-level item, and that item must be a
    /// function_item matching function_name.
    fn validate_single_function_body(new_body: &str, function_name: &str) -> Result<()> {
        let mut parser = Parser::new();
        parser.set_language(&tree_sitter_rust::LANGUAGE.into())?;
        let fragment_tree = parser.parse(new_body, None).ok_or_else(|| {
            anyhow!("PATCH_VALIDATION_FAILED: new_body for '{}' could not be parsed as Rust", function_name)
        })?;

        let root = fragment_tree.root_node();
        let mut walker = root.walk();
        let mut named_children = root.named_children(&mut walker);
        let first = named_children.next().ok_or_else(|| {
            anyhow!(
                "PATCH_VALIDATION_FAILED: new_body for '{}' is empty or contains no recognizable Rust item",
                function_name
            )
        })?;

        if named_children.next().is_some() {
            return Err(anyhow!(
                "PATCH_VALIDATION_FAILED: new_body for '{}' contains more than one top-level item \
                 (e.g. extra use statements, functions, or a mod block splice into the middle of a \
                 single-function patch) -- new_body must contain exactly one function definition",
                function_name
            ));
        }

        if first.kind() != "function_item" {
            return Err(anyhow!(
                "PATCH_VALIDATION_FAILED: new_body for '{}' is a '{}', not a function_item -- \
                 patch_function only replaces function bodies, not other item kinds",
                function_name,
                first.kind()
            ));
        }

        let name_node = first
            .child_by_field_name("name")
            .ok_or_else(|| anyhow!("PATCH_VALIDATION_FAILED: could not read the name of the function in new_body"))?;
        let actual_name = &new_body[name_node.start_byte()..name_node.end_byte()];
        if actual_name != function_name {
            return Err(anyhow!(
                "PATCH_VALIDATION_FAILED: new_body defines function '{}' but the patch target is '{}' -- \
                 refusing to rename a function via patch_function",
                actual_name,
                function_name
            ));
        }

        Ok(())
    }

    pub fn update_function(path: &str, function_name: &str, new_body: &str) -> Result<String> {
        Self::validate_single_function_body(new_body, function_name)?;

        let source_code = fs::read_to_string(path)?;
        let mut parser = Parser::new();
        parser.set_language(&tree_sitter_rust::LANGUAGE.into())?;
        let tree = parser
            .parse(&source_code, None)
            .ok_or_else(|| anyhow!("AST_PARSE_FAIL: Could not parse {}", path))?;
        // Query to find the specific function node
        let query_str = format!(
            "(function_item name: (identifier) @name (#eq? @name \"{}\")) @func",
            function_name
        );
        let query = Query::new(&tree_sitter_rust::LANGUAGE.into(), &query_str)?;
        let mut cursor = QueryCursor::new();
        let matches = cursor.matches(&query, tree.root_node(), source_code.as_bytes());
        for m in matches {
            for capture in m.captures {
                if capture.index == 1 {
                    // The @func capture
                    let node = capture.node;
                    let start = node.start_byte();
                    let end = node.end_byte();
                    let mut new_code = source_code.clone();
                    new_code.replace_range(start..end, new_body);

                    // Safety net beyond the standalone-fragment check above:
                    // confirm the *spliced* result still parses cleanly.
                    // Catches cases the isolated fragment check can't see,
                    // e.g. new_body being syntactically fine alone but
                    // producing a parse error once merged with surrounding
                    // context (mismatched braces at the splice boundary).
                    let mut verify_parser = Parser::new();
                    verify_parser.set_language(&tree_sitter_rust::LANGUAGE.into())?;
                    if let Some(verify_tree) = verify_parser.parse(&new_code, None) {
                        if verify_tree.root_node().has_error() {
                            return Err(anyhow!(
                                "PATCH_VALIDATION_FAILED: applying new_body to '{}' would leave {} with a Rust syntax error",
                                function_name,
                                path
                            ));
                        }
                    }

                    return Ok(new_code);
                }
            }
        }
        Err(anyhow!("FUNCTION_NOT_FOUND: Could not find function '{}' in {}", function_name, path))
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BeliefDelta {
    pub previous_confidence: f32,

    pub evidence_strength: f32,

    pub failure_penalty: f32,

    pub reflection_gain: f32,

    pub new_confidence: f32,

    pub explanation: String,
}

impl BeliefDelta {
    pub fn calculate(
        previous: f32,
        evidence: f32,
        failure: f32,
        reflection: f32,
        explanation: String,
    ) -> Self {
        let updated = (previous + evidence - failure + reflection).clamp(0.0, 1.0);

        Self {
            previous_confidence: previous,
            evidence_strength: evidence,
            failure_penalty: failure,
            reflection_gain: reflection,
            new_confidence: updated,
            explanation,
        }
    }
}

use crate::evidence::BeliefVector;
use serde::{Deserialize, Serialize};
use uuid::Uuid;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BeliefState {
    pub id: Uuid,
    pub proposition: String,
    pub confidence: f32,
    pub evidence_count: usize,
    pub stability: f32,
    /// Rich multidimensional view (canonical path going forward).
    pub vector: BeliefVector,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BeliefEvidence {
    pub source: String,
    pub observation: String,
    pub reliability: f32,
}

pub struct BeliefEngine;

impl BeliefEngine {
    pub fn new() -> Self {
        Self
    }

    /// Compatibility entry point – routes through BeliefVector::bayesian_update.
    pub fn update(&self, current: Option<BeliefState>, evidence: BeliefEvidence) -> BeliefState {
        let likelihood = evidence.reliability.clamp(0.0, 1.0);
        match current {
            Some(mut belief) => {
                // Canonical update
                belief.vector = belief.vector.bayesian_update(likelihood, 0.85);
                // Keep scalar fields in sync for legacy callers
                belief.confidence = belief.vector.scalar_score();
                belief.evidence_count += 1;
                belief.stability = (belief.stability + 0.05).min(1.0);
                belief
            }
            None => {
                let vector = BeliefVector::default().bayesian_update(likelihood, 0.85);
                BeliefState {
                    id: Uuid::new_v4(),
                    proposition: evidence.observation,
                    confidence: vector.scalar_score(),
                    evidence_count: 1,
                    stability: 0.5,
                    vector,
                }
            }
        }
    }
}

impl Default for BeliefEngine {
    fn default() -> Self {
        Self::new()
    }
}

impl BeliefState {
    pub fn update_confidence(&mut self, delta: f32) {
        self.confidence = (self.confidence + delta).clamp(0.0, 1.0);
    }

    pub fn apply_delta(&mut self, delta: crate::belief_delta::BeliefDelta) {
        self.confidence = delta.new_confidence;
        // Keep vector roughly in sync
        self.vector.implementation_confidence = delta.new_confidence;
    }

    /// Hard gate – prefer this over raw confidence checks.
    pub fn may_prove(&self, required_artifacts: u8, threshold: f32) -> bool {
        self.vector.may_prove(required_artifacts, threshold)
    }
}

use serde::{Deserialize, Serialize};
use uuid::Uuid;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum BeliefAction {
    Continue,
    Adapt,
    Suspend,
    Abandon,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BeliefDecision {
    pub id: Uuid,

    pub belief_id: Uuid,

    pub confidence: f32,

    pub action: BeliefAction,

    pub reason: String,

    pub timestamp: i64,
}

pub struct BeliefDecisionPolicy;

impl BeliefDecisionPolicy {
    pub fn evaluate(belief_id: Uuid, confidence: f32) -> BeliefDecision {
        let (action, reason) = if confidence >= 0.75 {
            (BeliefAction::Continue, "Confidence sufficient to continue objective".to_string())
        } else if confidence >= 0.45 {
            (
                BeliefAction::Adapt,
                "Confidence degraded, strategy adjustment recommended".to_string(),
            )
        } else if confidence >= 0.20 {
            (BeliefAction::Suspend, "Confidence too low, suspend objective".to_string())
        } else {
            (BeliefAction::Abandon, "Evidence no longer supports objective".to_string())
        };

        Self::decision(belief_id, confidence, action, reason)
    }

    fn decision(
        belief_id: Uuid,
        confidence: f32,
        action: BeliefAction,
        reason: String,
    ) -> BeliefDecision {
        BeliefDecision {
            id: Uuid::new_v4(),

            belief_id,

            confidence,

            action,

            reason,

            timestamp: chrono::Utc::now().timestamp(),
        }
    }
}

        let error_score = if prediction == observed { 0.0 } else { 1.0 };

        let confidence_delta = if error_score == 0.0 { 0.05 } else { -0.25 };

        let lesson = if error_score == 0.0 {
            "Prediction confirmed".to_string()
        } else {
            format!("Prediction failed. Expected '{}', observed '{}'", prediction, observed)
        };

        Self { prediction, observed, error_score, confidence_delta, lesson }
    }
}

    pub fn validate(prediction: String, observed: String) -> BeliefValidationEvent {
        BeliefValidationEvent::new(prediction, observed)
    }

// FIX: duration_since().unwrap() → unwrap_or_default() to survive NTP jumps.
use parking_lot::RwLock;
use serde::{Deserialize, Serialize};
use std::sync::Arc;
use std::time::{SystemTime, UNIX_EPOCH};

#[derive(Clone, Serialize, Deserialize, Debug)]
pub struct ClusterNode {
    pub id: String,
    pub host: String,
    pub port: u16,
    pub cpu: f32,
    pub ram: f32,
    pub temp: f32,
    pub last_seen: u64,
}

#[derive(Clone)]
pub struct ClusterMesh {
    nodes: Arc<RwLock<Vec<ClusterNode>>>,
}

fn now_secs() -> u64 {
    SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_secs()
}

impl ClusterMesh {
    pub fn new() -> Self {
        Self { nodes: Arc::new(RwLock::new(vec![])) }
    }

    pub fn register(&self, mut node: ClusterNode) {
        node.last_seen = now_secs();
        let mut guard = self.nodes.write();
        if let Some(existing) = guard.iter_mut().find(|n| n.id == node.id) {
            *existing = node;
        } else {
            guard.push(node);
        }
    }

    pub fn get_all_nodes(&self) -> Vec<ClusterNode> {
        self.nodes.read().clone()
    }

    pub fn best_node(&self) -> Option<ClusterNode> {
        let now = now_secs();
        self.nodes.read().iter().filter(|n| now.saturating_sub(n.last_seen) < 40).cloned().min_by(
            |a, b| {
                (a.cpu + a.ram).partial_cmp(&(b.cpu + b.ram)).unwrap_or(std::cmp::Ordering::Equal)
            },
        )
    }
}

impl Default for ClusterMesh {
    fn default() -> Self {
        Self::new()
    }
}

use parking_lot::RwLock;

#[derive(Clone)]
pub struct AttentionSystem {
    /// Weighted salience map (future). Single focus for now.
    salience: Arc<RwLock<HashMap<String, f32>>>,
    focus: Arc<RwLock<Option<String>>>,
    bus: EventBus,
}

impl AttentionSystem {
    pub fn new(bus: EventBus) -> Self {
        Self {
            salience: Arc::new(RwLock::new(HashMap::new())),
            focus: Arc::new(RwLock::new(None)),
            bus,
        }
    }

    pub fn current_focus(&self) -> Option<String> {
        self.focus.read().clone()
    }

    pub async fn shift(&self, new_focus: &str) {
        *self.focus.write() = Some(new_focus.to_string());
        self.salience.write().insert(new_focus.to_string(), 1.0);
        self.bus
            .publish(CognitiveEvent {
                meta: EventMeta::new("attention"),
                mission_id: None,
                kind: CognitiveEventKind::AttentionShift {
                    focus: new_focus.to_string(),
                },
            })
            .await;
    }

    pub fn set_salience(&self, topic: &str, weight: f32) {
        self.salience.write().insert(topic.to_string(), weight.clamp(0.0, 1.0));
    }

    pub fn top_salient(&self, k: usize) -> Vec<(String, f32)> {
        let mut items: Vec<_> = self.salience.read().iter().map(|(k, v)| (k.clone(), *v)).collect();
        items.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
        items.into_iter().take(k).collect()
    }

    pub fn clear(&self) {
        *self.focus.write() = None;
    }
}

use parking_lot::RwLock;
use std::collections::HashMap;
use std::sync::Arc;
use std::time::{SystemTime, UNIX_EPOCH};

#[derive(Clone, Debug)]
pub struct WeightedFact {
    pub confidence: f32,
    pub last_updated: u64,
}

/// V5 BeliefEngine: Maintains persistent environment facts (e.g., "gemini_api_healthy")
/// with automatic temporal decay for staleness.
#[derive(Clone, Default)]
pub struct BeliefEngine {
    facts: Arc<RwLock<HashMap<String, WeightedFact>>>,
}

impl BeliefEngine {
    pub fn new() -> Self {
        Self { facts: Arc::new(RwLock::new(HashMap::new())) }
    }

    /// Modifies a belief based on real-world evidence.
    /// Applies decay calculation before adjustment.
    pub fn update_belief(&self, key: &str, delta: f32) -> f32 {
        let mut lock = self.facts.write();
        let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs();
        let fact = lock
            .entry(key.to_string())
            .or_insert(WeightedFact { confidence: 0.5, last_updated: now });

        let elapsed_hours = now.saturating_sub(fact.last_updated) as f32 / 3600.0;
        let decay = elapsed_hours * 0.05; // 5% decay per hour of staleness

        fact.confidence = (fact.confidence - decay + delta).clamp(0.0, 1.0);
        fact.last_updated = now;
        fact.confidence
    }

    /// Retrieves current confidence of a belief, factoring in decay since last update.
    pub fn get_belief(&self, key: &str) -> f32 {
        let lock = self.facts.read();
        if let Some(fact) = lock.get(key) {
            let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs();
            let elapsed_hours = now.saturating_sub(fact.last_updated) as f32 / 3600.0;
            let decay = elapsed_hours * 0.05;
            (fact.confidence - decay).clamp(0.0, 1.0)
        } else {
            0.5 // Unknown facts rest at neutral parity
        }
    }
}

use anyhow::Result;
use serde::{Deserialize, Serialize};
use sqlx::{SqlitePool, Row};
use uuid::Uuid;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct StateCheckpoint {
    pub checkpoint_id: Uuid,
    pub parent_checkpoint_id: Option<Uuid>,
    pub plan_id: String,
    pub task_id: String,
    pub vmo_handles: Vec<u32>,
    pub channel_values: serde_json::Value,
    pub timestamp: i64,
}

#[derive(Clone)]
pub struct SqliteCheckpointer {
    pool: SqlitePool,
}

impl SqliteCheckpointer {
    pub fn new(pool: SqlitePool) -> Self {
        Self { pool }
    }

    pub async fn ensure_checkpoints_table(&self) -> Result<()> {
        sqlx::query(
            r#"
            CREATE TABLE IF NOT EXISTS state_checkpoints (
                checkpoint_id TEXT PRIMARY KEY,
                parent_checkpoint_id TEXT,
                plan_id TEXT NOT NULL,
                task_id TEXT NOT NULL,
                vmo_handles TEXT NOT NULL,
                channel_values TEXT NOT NULL,
                timestamp INTEGER NOT NULL
            );
            CREATE INDEX IF NOT EXISTS idx_state_checkpoints_plan_id ON state_checkpoints(plan_id);
            CREATE INDEX IF NOT EXISTS idx_state_checkpoints_checkpoint_id ON state_checkpoints(checkpoint_id);
            "#,
        )
        .execute(&self.pool)
        .await?;
        Ok(())
    }

    pub async fn put(&self, checkpoint: StateCheckpoint) -> Result<()> {
        let vmo_json = serde_json::to_string(&checkpoint.vmo_handles)?;
        let channel_json = serde_json::to_string(&checkpoint.channel_values)?;
        let parent_id_str = checkpoint.parent_checkpoint_id.map(|id| id.to_string());

        sqlx::query(
            r#"
            INSERT INTO state_checkpoints (
                checkpoint_id, parent_checkpoint_id, plan_id, task_id, vmo_handles, channel_values, timestamp
            ) VALUES (?, ?, ?, ?, ?, ?, ?)
            ON CONFLICT(checkpoint_id) DO UPDATE SET
                parent_checkpoint_id = excluded.parent_checkpoint_id,
                plan_id = excluded.plan_id,
                task_id = excluded.task_id,
                vmo_handles = excluded.vmo_handles,
                channel_values = excluded.channel_values,
                timestamp = excluded.timestamp;
            "#,
        )
        .bind(checkpoint.checkpoint_id.to_string())
        .bind(parent_id_str)
        .bind(&checkpoint.plan_id)
        .bind(&checkpoint.task_id)
        .bind(vmo_json)
        .bind(channel_json)
        .bind(checkpoint.timestamp)
        .execute(&self.pool)
        .await?;

        Ok(())
    }

    pub async fn get(&self, checkpoint_id: Uuid) -> Result<Option<StateCheckpoint>> {
        let row = sqlx::query(
            r#"
            SELECT checkpoint_id, parent_checkpoint_id, plan_id, task_id, vmo_handles, channel_values, timestamp
            FROM state_checkpoints
            WHERE checkpoint_id = ?
            "#,
        )
        .bind(checkpoint_id.to_string())
        .fetch_optional(&self.pool)
        .await?;

        if let Some(r) = row {
            let id_str: String = r.get("checkpoint_id");
            let parent_str: Option<String> = r.get("parent_checkpoint_id");
            let plan_id: String = r.get("plan_id");
            let task_id: String = r.get("task_id");
            let vmo_json: String = r.get("vmo_handles");
            let channel_json: String = r.get("channel_values");
            let timestamp: i64 = r.get("timestamp");

            let checkpoint_id = Uuid::parse_str(&id_str)?;
            let parent_checkpoint_id = match parent_str {
                Some(s) if !s.is_empty() => Some(Uuid::parse_str(&s)?),
                _ => None,
            };
            let vmo_handles: Vec<u32> = serde_json::from_str(&vmo_json)?;
            let channel_values: serde_json::Value = serde_json::from_str(&channel_json)?;

            Ok(Some(StateCheckpoint {
                checkpoint_id,
                parent_checkpoint_id,
                plan_id,
                task_id,
                vmo_handles,
                channel_values,
                timestamp,
            }))
        } else {
            Ok(None)
        }
    }
}

impl CriticAgent {
    pub fn evaluate(graph: &TaskGraph) -> f32 {
        if graph.nodes.is_empty() { return 0.0; }
        let mut score = 1.0_f32;
        for node in graph.nodes.values() {
            if node.deps.is_empty() && graph.nodes.len() > 1 { score *= 0.9; }
            if node.required_agent.trim().is_empty() { score *= 0.6; }
            if node.priority == 0 { score *= 0.85; }
        }
        score.clamp(0.0, 1.0)
    }
}

use crate::cognition_graph::{CognitiveGraph, EdgeRelation, NodeKind};
use crate::cognitive_events::CognitiveEvent;
use crate::completion_engine::{CognitiveBudget, SubgoalCompletionEngine, SubgoalContract};
use crate::epistemic_engine::CriticResponse;
use std::collections::HashSet;
use uuid::Uuid;

#[derive(Debug, Clone)]
pub struct EpistemicSession {
    pub session_id: Uuid,
    pub mission: String,
    pub contract: SubgoalContract,
    pub budget: CognitiveBudget,
    pub graph: CognitiveGraph,
    pub local_evidence: String,
    pub executed_actions: HashSet<String>,
    pub max_observed_belief: f32,
    pub best_hypothesis_id: Option<Uuid>,
    pub consecutive_mutation_vetoes: u32,
}

impl EpistemicSession {
    pub fn new(mission: &str) -> Self {
        Self {
            session_id: Uuid::new_v4(),
            mission: mission.to_string(),
            contract: SubgoalCompletionEngine::infer_contract("session_node", mission),
            budget: CognitiveBudget::default_for_node(),
            graph: CognitiveGraph::new(),
            local_evidence: String::new(),
            executed_actions: HashSet::new(),
            max_observed_belief: 0.0,
            best_hypothesis_id: None,
            consecutive_mutation_vetoes: 0,
        }
    }

    pub fn apply_event(&mut self, event: &CognitiveEvent) {
        match event {
            CognitiveEvent::HypothesisCreated { hypothesis_id, hypothesis, .. } => {
                self.graph.add_node(NodeKind::Hypothesis, hypothesis, 0.5, 100);
                if self.best_hypothesis_id.is_none() {
                    self.best_hypothesis_id = Some(*hypothesis_id);
                }
            }
            CognitiveEvent::ObservationRecorded { evidence, modifier, .. } => {
                self.local_evidence.push_str(evidence);
                self.local_evidence.push('\n');

                if let Some(h_id) = self.best_hypothesis_id {
                    let obs_id = self.graph.add_node(NodeKind::Observation, evidence, 1.0, 50);
                    let relation = if *modifier > 0.0 {
                        EdgeRelation::Supports
                    } else {
                        EdgeRelation::Contradicts
                    };
                    self.graph.add_edge(obs_id, h_id, relation, modifier.abs());
                    self.graph.propagate_beliefs();

                    if let Some(node) = self.graph.nodes.get(&h_id) {
                        if node.belief_score > self.max_observed_belief {
                            self.max_observed_belief = node.belief_score;
                        }
                    }
                }
            }
            CognitiveEvent::BeliefUpdated { hypothesis_id, new_belief, .. } => {
                if let Some(node) = self.graph.nodes.get_mut(hypothesis_id) {
                    node.belief_score = *new_belief;
                    if *new_belief > self.max_observed_belief {
                        self.max_observed_belief = *new_belief;
                    }
                }
            }
            _ => {}
        }
    }

    pub fn record_action_attempt(&mut self, action_key: &str) {
        self.executed_actions.insert(action_key.to_string());
    }

    pub fn is_action_attempted(&self, action_key: &str) -> bool {
        self.executed_actions.contains(action_key)
    }

    pub fn record_evidence(&mut self, evidence: &str, eval: &CriticResponse, h_id: Uuid) {
        let event = CognitiveEvent::ObservationRecorded {
            session_id: self.session_id,
            evidence: evidence.to_string(),
            modifier: eval.confidence_modifier,
        };
        self.best_hypothesis_id = Some(h_id);
        self.apply_event(&event);
    }

    pub fn is_subgoal_satisfied(&self, current_belief: f32) -> bool {
        if current_belief < 0.85 { return false; } SubgoalCompletionEngine::evaluate(&self.contract, &self.local_evidence, current_belief)
    }

    pub fn collect_evidence(&self, h_id: Uuid) -> String {
        let mut history = String::new();
        for edge in &self.graph.edges {
            if edge.target == h_id
                && (edge.relation == EdgeRelation::Supports
                    || edge.relation == EdgeRelation::Contradicts)
            {
                if let Some(obs) = self.graph.nodes.get(&edge.source) {
                    history.push_str(&format!("- [{:?}] {}\n", edge.relation, obs.content));
                }
            }
        }
        if history.is_empty() {
            "No evidence recorded.".to_string()
        } else {
            history
        }
    }
}

//! Cognitive Event Bus – Phase 8B v2
//! Async broadcast + pluggable EventStore.

use async_trait::async_trait;
use parking_lot::RwLock;
use serde::{Deserialize, Serialize};
use std::sync::Arc;
use std::time::{SystemTime, UNIX_EPOCH};
use tokio::sync::broadcast;
use uuid::Uuid;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct EventMeta {
    pub event_id: Uuid,
    pub timestamp_ms: u64,
    pub source: String,
    pub correlation_id: Option<Uuid>,
    pub causation_id: Option<Uuid>,
}

impl EventMeta {
    pub fn new(source: &str) -> Self {
        let ts =
            SystemTime::now().duration_since(UNIX_EPOCH).map(|d| d.as_millis() as u64).unwrap_or(0);
        Self {
            event_id: Uuid::new_v4(),
            timestamp_ms: ts,
            source: source.to_string(),
            correlation_id: None,
            causation_id: None,
        }
    }

    pub fn with_correlation(mut self, id: Uuid) -> Self {
        self.correlation_id = Some(id);
        self
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum CognitiveEventKind {
    Observation { content: String },
    BeliefUpdated { hypothesis_id: Uuid, new_score: f32 },
    GoalActivated { goal_id: Uuid },
    GoalCompleted { goal_id: Uuid },
    ReflectionCompleted { lessons: Vec<String> },
    MissionFinished { success: bool },
    StrategyChanged { strategy: String },
    Interrupt { reason: String },
    AttentionShift { focus: String },
    ResourceBudgetLow { remaining: f32 },
    DecisionMade { decision_summary: String },
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CognitiveEvent {
    pub meta: EventMeta,
    pub mission_id: Option<Uuid>,
    pub kind: CognitiveEventKind,
}

/// Pluggable history backend (Vec today, SQLite/RocksDB later).
#[async_trait]
pub trait EventStore: Send + Sync {
    async fn append(&self, event: &CognitiveEvent);
    async fn recent(&self, n: usize) -> Vec<CognitiveEvent>;
}

#[derive(Default)]
pub struct InMemoryEventStore {
    events: RwLock<Vec<CognitiveEvent>>,
    max: usize,
}

impl InMemoryEventStore {
    pub fn new(max: usize) -> Self {
        Self { events: RwLock::new(Vec::new()), max }
    }
}

#[async_trait]
impl EventStore for InMemoryEventStore {
    async fn append(&self, event: &CognitiveEvent) {
        let mut guard = self.events.write();
        guard.push(event.clone());
        if guard.len() > self.max {
            let excess = guard.len() - self.max;
            guard.drain(0..excess);
        }
    }

    async fn recent(&self, n: usize) -> Vec<CognitiveEvent> {
        let guard = self.events.read();
        guard.iter().rev().take(n).cloned().collect()
    }
}

#[derive(Clone)]
pub struct EventBus {
    tx: broadcast::Sender<CognitiveEvent>,
    store: Arc<dyn EventStore>,
}

impl EventBus {
    pub fn new() -> Self {
        let (tx, _) = broadcast::channel(256);
        Self { tx, store: Arc::new(InMemoryEventStore::new(256)) }
    }

    pub fn with_store(store: Arc<dyn EventStore>) -> Self {
        let (tx, _) = broadcast::channel(256);
        Self { tx, store }
    }

    pub fn subscribe(&self) -> broadcast::Receiver<CognitiveEvent> {
        self.tx.subscribe()
    }

    pub async fn publish(&self, event: CognitiveEvent) {
        self.store.append(&event).await;
        // Ignore lagging receivers – cognitive runtime must not block on slow listeners.
        let _ = self.tx.send(event);
    }

    pub async fn recent(&self, n: usize) -> Vec<CognitiveEvent> {

//! Executive Cortex – Phase 8B v2
//! Staged decision pipeline:
//!   observe → evaluate_policy → evaluate_goals → evaluate_risk → choose → emit

use super::event_bus::{CognitiveEvent, CognitiveEventKind, EventBus, EventMeta};
use super::policy_engine::PolicyEngine;
use super::traits::*;
use std::time::{SystemTime, UNIX_EPOCH};
use uuid::Uuid;

#[derive(Clone)]
pub struct ExecutiveCortex {
    bus: EventBus,
    policy: PolicyEngine,
}

#[derive(Debug, Default)]
struct Evaluation {
    policy_ok: bool,
    goal_pressure: f32,   // 0 = none, 1 = urgent
    risk: f32,            // 0 = safe, 1 = critical
    resource_pressure: f32,
}

impl ExecutiveCortex {
    pub fn new(bus: EventBus) -> Self {
        Self {
            bus,
            policy: PolicyEngine::new(),
        }
    }

    pub fn with_policy(bus: EventBus, policy: PolicyEngine) -> Self {
        Self { bus, policy }
    }

    // ── Pipeline stages ────────────────────────────────────────────────────

    fn observe(&self, state: &MissionState) -> Evaluation {
        let mut ev = Evaluation::default();
        ev.policy_ok = true; // policy checked against actions, not mission state
        ev.risk = 1.0 - state.health.clamp(0.0, 1.0);
        ev.resource_pressure = 0.0; // filled by scheduler later
        ev.goal_pressure = state.priority.clamp(0.0, 1.0);
        ev
    }

    fn evaluate_policy(&self, ev: &mut Evaluation, _state: &MissionState) {
        // Placeholder: mission-level policy always passes in 8B.
        // Action-level checks happen via PolicyEngine::check_action.
        ev.policy_ok = true;
    }

    fn evaluate_goals(&self, ev: &mut Evaluation, state: &MissionState) {
        // Higher priority + active status → higher pressure to continue.
        if state.status == MissionStatus::Active {
            ev.goal_pressure = state.priority.clamp(0.0, 1.0);
        } else {
            ev.goal_pressure = 0.0;
        }
    }

    fn evaluate_risk(&self, ev: &mut Evaluation, state: &MissionState) {
        ev.risk = 1.0 - state.health.clamp(0.0, 1.0);
    }

    fn choose(&self, ev: &Evaluation, state: &MissionState) -> (ExecutiveDecisionKind, String, f32) {
        match state.status {
            MissionStatus::Completed | MissionStatus::Failed | MissionStatus::Aborted => {
                (ExecutiveDecisionKind::Reflect, "mission terminal".into(), 0.95)
            }
            MissionStatus::Paused => {
                (ExecutiveDecisionKind::Continue, "resume paused mission".into(), 0.7)
            }
            MissionStatus::Active => {
                if !ev.policy_ok {
                    (ExecutiveDecisionKind::Abort, "policy violation".into(), 0.9)
                } else if ev.risk > 0.75 {
                    (
                        ExecutiveDecisionKind::Abort,
                        format!("risk critical ({:.2})", ev.risk),
                        0.9,
                    )
                } else if ev.risk > 0.5 {
                    (
                        ExecutiveDecisionKind::Replan,
                        format!("risk elevated ({:.2})", ev.risk),
                        0.8,
                    )
                } else if ev.resource_pressure > 0.8 {
                    (
                        ExecutiveDecisionKind::Pause,
                        "resource pressure high".into(),
                        0.75,
                    )
                } else {
                    (ExecutiveDecisionKind::Continue, "mission healthy".into(), 0.85)
                }
            }
        }
    }

    // ── Public API ─────────────────────────────────────────────────────────

    pub async fn decide(&self, state: &MissionState) -> ExecutiveDecision {
        let mut ev = self.observe(state);
        self.evaluate_policy(&mut ev, state);
        self.evaluate_goals(&mut ev, state);
        self.evaluate_risk(&mut ev, state);
        let (kind, rationale, confidence) = self.choose(&ev, state);

        let decision = ExecutiveDecision {
            kind,
            reason: DecisionReason {
                confidence,
                supporting_evidence: vec![
                    format!("health={:.2}", state.health),
                    format!("risk={:.2}", ev.risk),
                    format!("goal_pressure={:.2}", ev.goal_pressure),
                ],
                policy_trigger: Some("phase8b_v2_pipeline".into()),
                expected_cost: 0.0,
                rationale,
            },
        };

        self.bus
            .publish(CognitiveEvent {
                meta: EventMeta::new("executive_cortex"),
                mission_id: Some(state.id),
                kind: CognitiveEventKind::DecisionMade {
                    decision_summary: format!("{:?}", decision.kind),
                },
            })
            .await;

        decision
    }

    pub fn start_mission(&self, directive: &str, priority: f32) -> MissionState {
        MissionState {
            id: Uuid::new_v4(),
            directive: directive.to_string(),
            priority,
            health: 1.0,
            status: MissionStatus::Active,
            intent: Some(directive.to_string()),
        }
    }

    pub fn policy(&self) -> &PolicyEngine {
        &self.policy
    }
}

use serde::{Deserialize, Serialize};
use std::collections::{HashMap, HashSet};
use uuid::Uuid;

#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub enum NodeKind {
    Observation,
    Verification,
    Reflection,
    Hypothesis,
    ProposedAction { action_json: String },
    Strategy,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct VerificationReport {
    pub goal_satisfied: bool,
    pub evidence_confidence: f32,
    pub independent_check_passed: bool,
    pub rationale: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CognitiveNode {
    pub id: Uuid,
    pub kind: NodeKind,
    pub content: String,
    pub belief_score: f32,
    pub computation_cost: u32,
    pub is_active: bool,
}

#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub enum EdgeRelation {
    Supports,
    Contradicts,
    Prerequisite,
    DerivesFrom,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GraphEdge {
    pub source: Uuid,
    pub target: Uuid,
    pub relation: EdgeRelation,
    pub weight: f32,
}

#[derive(Debug, Clone, Default)]
pub struct CognitiveGraph {
    pub nodes: HashMap<Uuid, CognitiveNode>,
    pub edges: Vec<GraphEdge>,
    pub frontiers: HashSet<Uuid>,
}

impl CognitiveGraph {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn add_node(
        &mut self,
        kind: NodeKind,
        content: &str,
        initial_belief: f32,
        cost: u32,
    ) -> Uuid {
        let id = Uuid::new_v4();
        let node = CognitiveNode {
            id,
            kind,
            content: content.to_string(),
            belief_score: initial_belief,
            computation_cost: cost,
            is_active: false,
        };
        self.nodes.insert(id, node);
        self.frontiers.insert(id);
        id
    }

    pub fn add_edge(&mut self, source: Uuid, target: Uuid, relation: EdgeRelation, weight: f32) {
        self.edges.push(GraphEdge { source, target, relation, weight });
    }

    pub fn propagate_beliefs(&mut self) {
        // Accumulate signed impacts first so multiple edges targeting the same
        // hypothesis are combined instead of overwriting each other.
        let mut delta: HashMap<Uuid, f32> = HashMap::new();
        const DAMPING: f32 = 0.85;

        for edge in &self.edges {
            let Some(source_node) = self.nodes.get(&edge.source) else {
                continue;
            };
            let impact = source_node.belief_score * edge.weight;
            let signed = match edge.relation {
                EdgeRelation::Supports => impact,
                EdgeRelation::Contradicts => -impact,
                // Prerequisite / DerivesFrom intentionally ignored for score updates
                _ => continue,
            };
            *delta.entry(edge.target).or_insert(0.0) += signed;
        }

        for (id, d) in delta {
            if let Some(node) = self.nodes.get_mut(&id) {
                let candidate = (node.belief_score + d).clamp(0.0, 1.0);
                // Keep a fraction of the previous score (PageRank-style damping)
                node.belief_score = DAMPING * candidate + (1.0 - DAMPING) * node.belief_score;
            }
        }
    }

    pub fn select_best_hypothesis(&self) -> Option<&CognitiveNode> {
        self.nodes
            .values()
            // ⚡ FIX: Filter out dead hypotheses (belief < 0.15).
            // v1.9: When BeliefVector is available on the node, prefer may_prove() over raw score.
            .filter(|n| matches!(n.kind, NodeKind::Hypothesis) && n.belief_score >= 0.15)
            .max_by(|a, b| {
                let score_a = a.belief_score / (a.computation_cost as f32).max(1.0);
                let score_b = b.belief_score / (b.computation_cost as f32).max(1.0);
                score_a.partial_cmp(&score_b).unwrap_or(std::cmp::Ordering::Equal)
            })
    }
}

//! Policy Engine – Phase 8B v2
//! Capability-based authorization.

use std::collections::HashSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum Capability {
    FilesystemRead,
    FilesystemWrite,
    Network,
    Shell,
    Patch,
    Delete,
    MemoryWrite,
    SpawnSubagent,
}

#[derive(Debug, Clone)]
pub struct PolicyEngine {
    allowed: HashSet<Capability>,
}

impl Default for PolicyEngine {
    fn default() -> Self {
        let mut allowed = HashSet::new();
        allowed.insert(Capability::FilesystemRead);
        allowed.insert(Capability::FilesystemWrite);
        allowed.insert(Capability::Shell);
        allowed.insert(Capability::Patch);
        allowed.insert(Capability::MemoryWrite);
        // Delete and unrestricted Network denied by default
        Self { allowed }
    }
}

impl PolicyEngine {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn allows(&self, cap: Capability) -> bool {
        self.allowed.contains(&cap)
    }

    pub fn grant(&mut self, cap: Capability) {
        self.allowed.insert(cap);
    }

    pub fn revoke(&mut self, cap: Capability) {
        self.allowed.remove(&cap);
    }

    /// Map a raw action string to required capabilities (best-effort).
    pub fn required_capabilities(action: &str) -> Vec<Capability> {
        let lower = action.to_lowercase();
        let mut caps = Vec::new();
        if lower.contains("write_file") || lower.contains("diff_patch") || lower.contains("patch_function") {
            caps.push(Capability::FilesystemWrite);
            caps.push(Capability::Patch);
        }
        if lower.contains("read_file") || lower.contains("search_code") || lower.contains("read_file_chunk") {
            caps.push(Capability::FilesystemRead);
        }
        if lower.contains("execute_command") || lower.starts_with("rg ") || lower.starts_with("find ") {
            caps.push(Capability::Shell);
        }
        if lower.contains("http") || lower.contains("curl") || lower.contains("wget") {
            caps.push(Capability::Network);
        }
        if lower.contains("rm ") || lower.contains("delete") {
            caps.push(Capability::Delete);
        }
        if lower.contains("spawn") || lower.contains("delegate") {
            caps.push(Capability::SpawnSubagent);
        }
        if caps.is_empty() {
            caps.push(Capability::Shell); // conservative default
        }
        caps
    }

    pub fn check_action(&self, action: &str) -> bool {
        Self::required_capabilities(action)
            .into_iter()
            .all(|c| self.allows(c))
    }
}

use super::traits::{Prediction, PredictionEvaluation};
use parking_lot::RwLock;
use std::collections::HashMap;
use std::sync::Arc;
use std::time::{SystemTime, UNIX_EPOCH};
use uuid::Uuid;

#[derive(Clone, Debug, Default)]
pub struct ActionHistory {
    pub success_count: u32,
    pub total_count: u32,
    pub avg_cost: f32,
}

/// V5 PredictionEngine: Operates a fast nearest-neighbor lookup on historical logs
/// using action-verb + target-extension signatures.
#[derive(Clone, Default)]
pub struct PredictionEngine {
    active_predictions: Arc<RwLock<HashMap<Uuid, Prediction>>>,
    history: Arc<RwLock<HashMap<String, ActionHistory>>>,
}

impl PredictionEngine {
    pub fn new() -> Self {
        Self {
            active_predictions: Arc::new(RwLock::new(HashMap::new())),
            history: Arc::new(RwLock::new(HashMap::new())),
        }
    }

    /// Generates a heuristic signature: e.g. "read_file src/main.rs" -> "read_file_rs"
    fn get_signature(action_str: &str) -> String {
        let parts: Vec<&str> = action_str.split_whitespace().collect();
        let verb = parts.first().unwrap_or(&"unknown");
        let ext = parts.get(1).and_then(|p| p.split('.').next_back()).unwrap_or("none");
        format!("{}_{}", verb, ext)
    }

    pub fn generate_prediction(&self, action_id: Uuid, action_str: &str) -> Prediction {
        let sig = Self::get_signature(action_str);
        let hist_lock = self.history.read();

        let (confidence, cost) = if let Some(h) = hist_lock.get(&sig) {
            let conf =
                if h.total_count > 0 { h.success_count as f32 / h.total_count as f32 } else { 0.5 };
            (conf, h.avg_cost)
        } else {
            (0.5, 0.1) // Baseline default
        };

        let ts = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs();
        let pred = Prediction {
            action_id,
            expected_state: sig, // We store the signature here to match against upon evaluation
            confidence,
            cost_estimate: cost,
            timestamp: ts,
        };

        self.active_predictions.write().insert(action_id, pred.clone());
        pred
    }

    pub fn evaluate_prediction(
        &self,
        action_id: Uuid,
        actual_cost: f32,
        success: bool,
    ) -> PredictionEvaluation {
        let mut active = self.active_predictions.write();
        if let Some(pred) = active.remove(&action_id) {
            let mut hist_lock = self.history.write();
            let h = hist_lock.entry(pred.expected_state.clone()).or_default();

            h.total_count += 1;
            if success {
                h.success_count += 1;
            }

            // Rolling average update
            h.avg_cost =
                (h.avg_cost * (h.total_count - 1) as f32 + actual_cost) / h.total_count as f32;

            let accuracy = if success == (pred.confidence >= 0.5) { 1.0 } else { 0.0 };
            let error = (actual_cost - pred.cost_estimate).abs();

            PredictionEvaluation { accuracy, error }
        } else {
            PredictionEvaluation { accuracy: 0.0, error: 0.0 }
        }
    }
}

//! Working Memory – Phase 8B v2
//! Partitioned: facts / observations / reasoning / hypotheses.

use parking_lot::RwLock;
use std::collections::HashMap;
use std::sync::Arc;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum MemoryPartition {
    Facts,
    Observations,
    Reasoning,
    Hypotheses,
}

#[derive(Clone, Default)]
pub struct WorkingMemory {
    partitions: Arc<RwLock<HashMap<MemoryPartition, HashMap<String, String>>>>,
}

impl WorkingMemory {
    pub fn new() -> Self {
        let mut map = HashMap::new();
        for p in [
            MemoryPartition::Facts,
            MemoryPartition::Observations,
            MemoryPartition::Reasoning,
            MemoryPartition::Hypotheses,
        ] {
            map.insert(p, HashMap::new());
        }
        Self {
            partitions: Arc::new(RwLock::new(map)),
        }
    }

    pub fn store(&self, partition: MemoryPartition, key: &str, value: &str) {
        if let Some(slot) = self.partitions.write().get_mut(&partition) {
            slot.insert(key.to_string(), value.to_string());
        }
    }

    pub fn recall(&self, partition: MemoryPartition, key: &str) -> Option<String> {
        self.partitions
            .read()
            .get(&partition)
            .and_then(|s| s.get(key).cloned())
    }

    pub fn clear_partition(&self, partition: MemoryPartition) {
        if let Some(slot) = self.partitions.write().get_mut(&partition) {
            slot.clear();
        }
    }

    pub fn clear_all(&self) {
        for slot in self.partitions.write().values_mut() {
            slot.clear();
        }
    }

    pub fn snapshot(&self, partition: MemoryPartition) -> HashMap<String, String> {
        self.partitions
            .read()
            .get(&partition)
            .cloned()
            .unwrap_or_default()
    }
}

fn has_word(text: &str, word: &str) -> bool {
    text.split_whitespace()
        .any(|w| w.trim_matches(|c: char| !c.is_alphanumeric()) == word)
}

fn any_word(text: &str, words: &[&str]) -> bool {
    words.iter().any(|w| has_word(text, w))
}

// Negation-aware variant of any_word. A directive that says "Do NOT write
// implementation code" or "No patches" contains the literal words "write"/
// "patch" just as much as one that actually requests them -- any_word alone
// can't tell the two apart, which is exactly what misclassified a read-only
// architectural-audit directive as a mutation task (its own prohibition
// clauses repeatedly used "write"/"patch"). This checks each occurrence of
// the target word against the 3 preceding tokens for a negation marker and
// only counts it as a real trigger if no such marker is present.
fn has_word_unnegated(text: &str, word: &str) -> bool {
    let negations = ["not", "never", "no", "cannot", "n't"];
    let tokens: Vec<&str> = text.split_whitespace().collect();
    for (i, tok) in tokens.iter().enumerate() {
        let clean = tok.trim_matches(|c: char| !c.is_alphanumeric());
        if clean == word {
            let start = i.saturating_sub(3);
            let negated = tokens[start..i].iter().any(|w| {
                let wl = w.trim_matches(|c: char| !c.is_alphanumeric()).to_lowercase();
                negations.contains(&wl.as_str()) || wl.ends_with("n't")
            });
            if !negated {
                return true;
            }
        }
    }
    false
}

fn any_word_unnegated(text: &str, words: &[&str]) -> bool {
    words.iter().any(|w| has_word_unnegated(text, w))
}

#[allow(dead_code)]
const NOTHING_TO_FIX_BELIEF_THRESHOLD: f32 = 0.85;

#[allow(dead_code)]
fn has_source_file_evidence(local_evidence: &str) -> bool {
    local_evidence.lines().any(|line| {
        (line.starts_with("[READ_OK]") || line.starts_with("[SEARCH_CODE_OK]"))
            && line.contains(".rs")
    }) || local_evidence.contains(".rs\n")
        || local_evidence.contains(".rs:")
}

impl SubgoalCompletionEngine {
    pub fn infer_contract(node_id: &str, objective: &str) -> SubgoalContract {
        let obj_lower = objective.trim().to_lowercase();
        let classify_text = obj_lower
            .strip_prefix("objective:")
            .or_else(|| obj_lower.strip_prefix("objective"))
            .unwrap_or(&obj_lower)
            .trim();

        let mut conditions = Vec::new();
        let mut is_mutation_task = false;

        let first_word = classify_text.split_whitespace().next().unwrap_or("");
        let is_recon_intent = [
            "read", "analyze", "inspect", "search", "examine", "locate", "profile", "benchmark",
        ]
        .contains(&first_word);

        if is_recon_intent {
            conditions.push(CompletionCondition::ReadMarkerAcquired);
            conditions.push(CompletionCondition::MinBeliefScore(0.65));
        } else if any_word_unnegated(
            classify_text,
            &["write", "export", "save", "implement", "integrate", "add", "create"],
        ) {
            conditions.push(CompletionCondition::WriteMarkerAcquired);
            is_mutation_task = true;

            if any_word(classify_text, &["roadmap", "audit", "comparison"]) {
                conditions.push(CompletionCondition::SourceCodeEvidenceRequired);
            }
        } else if any_word_unnegated(classify_text, &["patch", "apply", "fix", "remediate"]) {
            conditions.push(CompletionCondition::PatchMarkerAcquired);
            is_mutation_task = true;
        } else if any_word(classify_text, &["search", "find", "locate"]) {
            conditions.push(CompletionCondition::SearchMarkerAcquired);
            conditions.push(CompletionCondition::MinBeliefScore(0.65));
        } else if any_word(classify_text, &["read", "inspect", "audit"]) {
            conditions.push(CompletionCondition::ReadMarkerAcquired);
            conditions.push(CompletionCondition::MinBeliefScore(0.65));
        } else {
            conditions.push(CompletionCondition::CommandMarkerAcquired);
            conditions.push(CompletionCondition::MinBeliefScore(0.85));
        }

        SubgoalContract {
            node_id: node_id.to_string(),
            objective: objective.to_string(),
            conditions,
            is_mutation_task,
        }
    }

    pub fn evaluate(contract: &SubgoalContract, local_evidence: &str, _belief_score: f32) -> bool {
        // ⚡ DETERMINISTIC VERIFIER INTERFACE
        let verifier_result = crate::outcome_verifier::OutcomeVerifier::verify(&contract.node_id, local_evidence);
        matches!(verifier_result, crate::outcome_verifier::CompletionStatus::Approved)
    }
}

use crate::metrics::METRICS;
use crate::orchestrator::Orchestrator;
use crate::router::AgentProfile;
use nexus_model_hub::factory::AgentFactory;
use nexus_model_hub::neural_agent::ContextPacket;
use nexus_substrate::token_ledger::wait_for_quota;
use reqwest::Client;
use serde_json::json;
use std::sync::atomic::Ordering;
use tokio::sync::broadcast;

const ESTIMATED_TOKENS: u32 = 2_000;

pub struct Conductor;

pub struct TurnContext<'a> {
    pub orch: &'a Orchestrator,
    pub client: &'a Client,
    pub api_key: &'a str,
    pub tx: Option<broadcast::Sender<String>>,
}

impl Conductor {
    pub async fn run_turn(
        profile: &AgentProfile,
        prompt: String,
        turn: usize,
        ctx: TurnContext<'_>,
        errors: usize,
    ) -> Result<String, String> {
        METRICS.api_calls.fetch_add(1, Ordering::Relaxed);

        let model = AgentFactory::get_model_for_turn(turn, errors);
        let _ = wait_for_quota(&model, ESTIMATED_TOKENS).await;

        let neural_ctx = ContextPacket {
            system_prompt: Some(profile.persona.clone()),
            model_override: Some(model.to_string()),
            ..Default::default()
        };

        if let Some(ref sender) = ctx.tx {
            let _ = sender.send(
                json!({"type": "NEURAL_THOUGHT", "payload": format!("[CONDUCTOR] turn {} → {}", turn, model)}).to_string()
            );
        }

        let result =
            ctx.orch.run(prompt.clone(), neural_ctx, ctx.client, ctx.api_key, ctx.tx.clone()).await;

        if let Err(ref e) = result {
            if e.contains("STALE")
                || e.contains("500")
                || e.contains("RESOURCE_EXHAUSTED")
                || e.contains("INTERNAL")
                || e.contains("CONTEXT_OVERFLOW")
            {
                METRICS.task_failures.fetch_add(1, Ordering::Relaxed);

                let failover = AgentFactory::failover_model(&model);
                tracing::warn!("[CONDUCTOR] {} failed → failing over to {}", model, failover);
                let _ = wait_for_quota(&failover, ESTIMATED_TOKENS).await;

                let fail_ctx = ContextPacket {
                    system_prompt: Some(profile.persona.clone()),
                    model_override: Some(failover.to_string()),
                    ..Default::default()
                };
                return ctx.orch.run(prompt, fail_ctx, ctx.client, ctx.api_key, ctx.tx).await;
            }
        }
        result
    }
}

                    })
                } else {
                    session.local_evidence.lines().any(|l| l.starts_with("[READ_CHUNK_OK]") || l.starts_with("[SEARCH_CODE_OK]") || l.starts_with("[READ_OK]") || l.starts_with("[WRITE_OK]") || l.starts_with("[CMD_OK]"))
                };

                let compound_satisfied = if is_mutation_task {
                    session.is_subgoal_satisfied(current_belief)
                } else if is_write_task {
                    true
                } else {
                    let mission_lower = mission.to_lowercase();
                    let objective_part = mission_lower.lines().find(|line| line.starts_with("objective:")).unwrap_or(mission_lower.as_str());
                    let evidence_lower = format!("{} {}", evidence_history.to_lowercase(), session.local_evidence.to_lowercase());
                    let and_segments: Vec<&str> = objective_part.split(" and ").map(|s| s.trim()).filter(|s| !s.is_empty()).collect();
                    and_segments.iter().all(|segment| {
                        let or_parts: Vec<&str> = segment.split(" or ").map(|s| s.trim()).filter(|s| !s.is_empty()).collect();
                        if or_parts.len() <= 1 {
                            segment.split_whitespace().filter(|w| w.len() > 3).any(|w| evidence_lower.contains(w))
                        } else {
                            or_parts.iter().any(|part| part.split_whitespace().filter(|w| w.len() > 3).any(|w| evidence_lower.contains(w)))
                        }
                    })
                };

                // If the state is mathematically satisfied and belief is extremely high, force the node to halt.
                if has_prior_evidence && compound_satisfied && current_belief >= 0.95 {
                    self.log("\x1b[1;36m>> [CORTEX OMNI-GUARD] Contract mathematically satisfied. Stripping LLM authority and forcing deterministic node completion.\x1b[0m".to_string());
                    return Ok(CortexOutcome::CompletionProposed {
                        belief: current_belief,
                        evidence: session.collect_evidence(h_id),
                    });
                }

                self.log("\x1b[36m>> [STRATEGIST] Planning next maneuver...\x1b[0m".to_string());
                session.budget.record_call();
                let proposed_action = match EpistemicEngine::propose_action(
                    &self.client,
                    &self.api_key,
                    mission,
                    &h_content,
                    &world_context,
                    &evidence_history,
                )
                .await
                {
                    Ok(action) => action,
                    Err(e) => {
                        self.log(format!("\x1b[31m>> [STRATEGIST FAULT]\x1b[0m {}", e));
                        continue;
                    }
                };
                self.log(format!("\x1b[35m>> [STRATEGIST PROPOSAL]:\x1b[0m {}", proposed_action));

                // ⚡ DETERMINISTIC HALT SHORT-CIRCUIT
                let prop_lower = proposed_action.trim().to_lowercase();
                if prop_lower == "halt" || prop_lower.starts_with("halt ") {
                    let has_prior_evidence = if is_write_task {
                        session
                            .local_evidence
                            .lines()
                            .any(|l| l.starts_with("[WRITE_OK]") || l.starts_with("[CMD_OK]"))
                    } else if is_mutation_task {
                        session.local_evidence.lines().any(|l| {
                            l.starts_with("[DIFF_PATCH_OK]") || l.starts_with("[REPLACE_LINES_OK]")
                        }) || session
                            .local_evidence
                            .lines()
                            .any(|l| l.starts_with("[PATCH_FUNCTION_OK]"))
                            || session.local_evidence.lines().any(|l| l.starts_with("[WRITE_OK]"))
                    } else {
                        session.local_evidence.lines().any(|l| l.starts_with("[READ_CHUNK_OK]"))
                            || session
                                .local_evidence
                                .lines()
                                .any(|l| l.starts_with("[SEARCH_CODE_OK]"))
                            || session.local_evidence.lines().any(|l| l.starts_with("[READ_OK]"))
                            || session.local_evidence.lines().any(|l| l.starts_with("[WRITE_OK]"))
                            || session.local_evidence.lines().any(|l| l.starts_with("[CMD_OK]"))
                    };

                    let compound_satisfied = if is_mutation_task {
                        session.is_subgoal_satisfied(current_belief)
                    } else if is_write_task {
                        true
                    } else {
                        let mission_lower = mission.to_lowercase();
                        let objective_part = mission_lower
                            .lines()
                            .find(|line| line.starts_with("objective:"))
                            .unwrap_or(mission_lower.as_str());
                        let evidence_lower = format!(
                            "{} {}",
                            evidence_history.to_lowercase(),
                            session.local_evidence.to_lowercase()
                        );
                        let and_segments: Vec<&str> = objective_part
                            .split(" and ")
                            .map(|s| s.trim())
                            .filter(|s| !s.is_empty())
                            .collect();
                        and_segments.iter().all(|segment| {
                            let or_parts: Vec<&str> = segment
                                .split(" or ")
                                .map(|s| s.trim())
                                .filter(|s| !s.is_empty())
                                .collect();
                            if or_parts.len() <= 1 {
                                segment
                                    .split_whitespace()
                                    .filter(|w| w.len() > 3)
                                    .any(|w| evidence_lower.contains(w))
                            } else {
                                or_parts.iter().any(|part| {
                                    part.split_whitespace()
                                        .filter(|w| w.len() > 3)
                                        .any(|w| evidence_lower.contains(w))
                                })
                            }
                        })
                    };

                    if has_prior_evidence && compound_satisfied {
                        self.log("\x1b[1;36m>> [CORTEX] DETERMINISTIC GUARD: Halt proposed following successful local evidence. Subtask complete.\x1b[0m".to_string());
                        return Ok(CortexOutcome::CompletionProposed {
                            belief: 1.0,
                            evidence: session.collect_evidence(h_id),
                        });
                    } else if has_prior_evidence && !compound_satisfied {
                        self.log("\x1b[1;33m>> [CORTEX] DETERMINISTIC GUARD SUPPRESSED: Halt proposed but evidence covers only part of a compound objective; continuing to explore.\x1b[0m".to_string());
                    } else if (is_mutation_task || is_write_task) && !has_prior_evidence {
                        if current_belief > 0.90 {
                            self.log("\x1b[1;33m>> [CORTEX GUARD OVERRIDE] High belief (>.90) halt proposed without mutation. Allowing Evaluator to assess validity.\x1b[0m".to_string());
                        } else {
                        self.log("\x1b[1;31m>> [CORTEX FAULT] Halt rejected: Objective requires patching/writing, but no mutation success marker found.\x1b[0m".to_string());
                        let obs_id = session.graph.add_node(NodeKind::Observation, "[HALT_REJECTED] The directive requires modifying files. You proposed halt without successfully patching anything. You MUST use diff_patch, patch_function, or write_file.", 1.0, 50);
                        session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.6);
                        session.graph.propagate_beliefs();
                        continue;
                    }
                    }
                }

                // --- ANTI-LOOP KINETIC GUARD (PATCHED) ---
                let action_key = proposed_action.trim().to_string();
                if session.is_action_attempted(&action_key) {
                    if session.is_subgoal_satisfied(current_belief)
                        || (!session.local_evidence.trim().is_empty() && current_belief >= 0.60)
                    {
                        self.log("\x1b[1;36m>> [SUBGOAL_COMPLETE] Loop detected but evidence is sufficient. Halting.\x1b[0m".to_string());
                        return Ok(CortexOutcome::CompletionProposed {
                            belief: current_belief,
                            evidence: session.collect_evidence(h_id),
                        });
                    } else {
                        self.log(format!("\x1b[1;31m>> [ENTERPRISE CORTEX] Anti-Loop Guard Triggered: Action [{}] already attempted.\x1b[0m", action_key));
                        let obs_id = session.graph.add_node(
                            NodeKind::Observation,
                            "[CRITICAL VETO] You are repeating a failed action! YOU MUST MUTATE YOUR STRATEGY, USE A DIFFERENT TOOL, OR HALT.",
                            1.0, 95
                        );
                        session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.99);
                        session.graph.propagate_beliefs();
                        continue;
                    }
                }
                session.record_action_attempt(&action_key);

                self.log("\x1b[31m>> [RED TEAM] Criticizing proposal...\x1b[0m".to_string());
                session.budget.record_call();
                match EpistemicEngine::criticize_proposal(
                    &self.client,
                    &self.api_key,
                    mission,
                    &h_content,
                    &proposed_action,
                    &world_context,
                    &evidence_history,
                )
                .await
                {
                    Err(e) => {
                        self.log(format!(
                            "\x1b[31m>> [CORTEX FATAL] Red Team critic parser panicked: {}\x1b[0m",
                            e
                        ));
                        let evidence = format!(
                            "[CRITIC_PARSE_FAILURE] Critic response could not be parsed: {}",
                            e
                        );
                        let obs_id =
                            session.graph.add_node(NodeKind::Observation, &evidence, 1.0, 50);
                        session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.15);
                        session.graph.propagate_beliefs();
                    }
                    Ok(mut critic) => {
                        self.log(format!(
                            "\x1b[31m>> [CRITIC] Verdict:\x1b[0m {} (Modifier: {:.2})",
                            critic.critique, critic.confidence_modifier
                        ));

                        // VETO CIRCUIT BREAKER: the Critic can get stuck repeatedly
                        // rejecting the exact mutation action that would satisfy an
                        // unmet write/patch contract, reasoning e.g. "the file hasn't
                        // been written yet" against the very write_file proposal that
                        // would write it. A prompt fix reduces how often this occurs
                        // but can't guarantee it never recurs under a different model
                        // or phrasing. This is the deterministic backstop: after enough
                        // consecutive vetoes of a mutation-shaped proposal while the
                        // contract's mutation marker is still missing, force the
                        // proposal through instead of trusting the Critic indefinitely.
                        let is_mutation_shaped_proposal = prop_lower.contains("write_file")
                            || prop_lower.contains("patch_function")
                            || prop_lower.contains("diff_patch")
                            || prop_lower.contains("replace_lines");
                        let contract_still_unmet = (is_mutation_task || is_write_task)
                            && !session.local_evidence.lines().any(|l| {
                                l.starts_with("[WRITE_OK]")
                                    || l.starts_with("[PATCH_FUNCTION_OK]")
                                    || l.starts_with("[REPLACE_LINES_OK]")
                                    || l.starts_with("[DIFF_PATCH_OK]")
                            });

                        if critic.confidence_modifier <= -0.2
                            && is_mutation_shaped_proposal
                            && contract_still_unmet
                        {
                            session.consecutive_mutation_vetoes += 1;
                            if session.consecutive_mutation_vetoes >= 3 {
                                self.log(format!(
                                    "\x1b[1;35m>> [CIRCUIT_BREAKER_OVERRIDE] Critic vetoed {} consecutive mutation proposals while the contract's mutation marker is still missing. Overriding veto and executing.\x1b[0m",
                                    session.consecutive_mutation_vetoes
                                ));
                                critic.confidence_modifier = 0.1;
                                session.consecutive_mutation_vetoes = 0;
                            }
                        } else {
                            session.consecutive_mutation_vetoes = 0;
                        }

                        if critic.confidence_modifier > -0.2 {
                            self.log("\x1b[36m>> [CORTEX] Translating intent to physical action...\x1b[0m".to_string());
                            session.budget.record_call();

                            let parsed_actions = if let Ok(direct_actions) =
                                crate::json_parser::extract_actions(&proposed_action)
                            {
                                if !direct_actions.is_empty() {
                                    self.log("\x1b[32m>> [CORTEX INTERCEPT] Direct JSON Proposal Intercept: Bypassing Translator LLM.\x1b[0m".to_string());
                                    Ok(direct_actions)
                                } else {
                                    if let Ok(raw_json) = EpistemicEngine::translate_to_action(
                                        &self.client,
                                        &self.api_key,
                                        &h_content,
                                        &proposed_action,
                                    )
                                    .await
                                    {
                                        crate::json_parser::extract_actions(&raw_json)
                                    } else {
                                        Ok(vec![])
                                    }
                                }
                            } else {
                                if let Ok(raw_json) = EpistemicEngine::translate_to_action(
                                    &self.client,
                                    &self.api_key,
                                    &h_content,
                                    &proposed_action,
                                )
                                .await
                                {
                                    crate::json_parser::extract_actions(&raw_json)
                                } else {
                                    Ok(vec![])
                                }
                            };

                            if let Ok(mut actions) = parsed_actions {
                                if actions.is_empty() && !proposed_action.trim().is_empty() {
                                    let action_str = proposed_action.trim();
                                    if let Some(intercepted) =
                                        crate::action_intercept::intercept_fallback_action(
                                            action_str,
                                        )
                                    {
                                        self.log("\x1b[32m>> [CORTEX INTERCEPT] Native Tool Intercept: Handling action directly.\x1b[0m".to_string());
                                        actions.push(intercepted);
                                    } else {
                                        // Deliberately NOT falling back to raw
                                        // ExecuteCommand here. This branch used to
                                        // hand the Strategist's unparsed proposal
                                        // text (which is frequently a raw JSON blob,
                                        // not shell syntax) straight to bash -c --
                                        // same corruption class as the write_file
                                        // fallback removed earlier this session,
                                        // just luckier so far because malformed
                                        // JSON happens to fail as bash syntax. A
                                        // proposal whose text is coincidentally
                                        // valid (and destructive) shell would not
                                        // be so lucky. Treat unrecognized text as a
                                        // genuine translation failure instead.
                                        self.log("\x1b[31m>> [CORTEX FAULT] Translator produced empty JSON and proposal text matched no known tool. Refusing raw fallback execution.\x1b[0m".to_string());
                                        let obs_id = session.graph.add_node(
                                            NodeKind::Observation,
                                            "[TRANSLATION_UNRECOGNIZED] Proposal text did not match any known action type and was not executed as a raw command.",
                                            1.0,
                                            50,
                                        );
                                        session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.3);
                                        session.graph.propagate_beliefs();
                                    }
                                }

                                if actions.is_empty() {
                                    self.log("\x1b[31m>> [CORTEX FAULT] Translator produced an empty action list.\x1b[0m".to_string());
                                    let obs_id = session.graph.add_node(
                                        NodeKind::Observation,
                                        "[TRANSLATION_EMPTY]",
                                        1.0,
                                        50,
                                    );
                                    session.graph.add_edge(
                                        obs_id,
                                        h_id,
                                        EdgeRelation::Contradicts,
                                        0.15,
                                    );
                                    session.graph.propagate_beliefs();
                                } else {
                                    let mut executor =
                                        RoboticExecutor::new(self.api_key.clone(), self.tx.clone());

                                    for action in actions {
                                        let intent =
                                            IntentContract::new(&h_content, current_belief);
                                        let capability = match &action {
                                            AgentAction::ExecuteCommand { .. } => {
                                                Capability::ExecuteSystemCall
                                            }
                                            AgentAction::WriteFile { .. }
                                            | AgentAction::DiffPatch { .. }
                                            | AgentAction::PatchFunction { .. } => {
                                                Capability::WriteSubstrate
                                            }
                                            AgentAction::Halt { .. } => Capability::HaltCognition,
                                            _ => Capability::ReadSubstrate,
                                        };

                                        // Capture the mutation target path now -- `action` is moved
                                        // into MotorCortex::translate() below and won't be readable
                                        // after this point, but we need the path later to auto-revert
                                        // if the Evaluator scores this mutation negative.
                                        let mutation_path: Option<String> = match &action {
                                            AgentAction::WriteFile { path, .. }
                                            | AgentAction::DiffPatch { path, .. }
                                            | AgentAction::PatchFunction { path, .. } => {
                                                Some(path.clone())
                                            }
                                            _ => None,
                                        };

                                        let is_halt_action_captured = matches!(action, AgentAction::Halt { .. }) || proposed_action.trim().to_lowercase().starts_with("halt");
                                        let motor_cmd = MotorCortex::translate(
                                            &intent,
                                            None,
                                            action,
                                            capability,
                                            h_content.clone(),
                                        );

                                        if let Err(gate_err) = self.action_gate.approve(&motor_cmd)
                                        {
                                            self.log(format!(
                                                "\x1b[1;31m>> [ACTION GATE REJECTED]: {}\x1b[0m",
                                                gate_err
                                            ));
                                            let obs_id = session.graph.add_node(
                                                NodeKind::Observation,
                                                &format!("[ACTION_GATE_BLOCKED] {}", gate_err),
                                                1.0,
                                                50,
                                            );
                                            session.graph.add_edge(
                                                obs_id,
                                                h_id,
                                                EdgeRelation::Contradicts,
                                                0.50,
                                            );
                                            session.graph.propagate_beliefs();
                                            continue;
                                        }

                                        self.log("\x1b[35m>> [MOTOR CORTEX] Emitting authorized MotorCommand to RoboticExecutor...\x1b[0m".to_string());
                                        let evidence = match executor
                                            .execute_motor_command(
                                                motor_cmd,
                                                self.memory.clone(),
                                                self.vmo_state.clone(),
                                                self.dist.clone(),
                                            )
                                            .await
                                        {
                                            Ok(ev) => ev,
                                            Err(e) => {
                                                format!("[ACTION FAILED] OS rejected action: {}", e)
                                            }
                                        };

                                        session.local_evidence.push_str(&evidence);
                                        session.local_evidence.push('\n');

                                        let preview =
                                            evidence.lines().take(5).collect::<Vec<_>>().join("\n");
                                        self.log(format!(
                                            "\x1b[32m>> [EVIDENCE ACQUIRED]\x1b[0m\n{}",
                                            preview
                                        ));

                                        self.log("\x1b[33m>> [CORTEX] Evaluating physical evidence against strategy...\x1b[0m".to_string());
                                        let eval_evidence_context = format!("Current Action Output:\n{}\n\nPrior Evidence in Task:\n{}", evidence, evidence_history);
                                        session.budget.record_call();
                                        match EpistemicEngine::evaluate_evidence(
                                            &self.client,
                                            &self.api_key,
                                            &h_content,
                                            &proposed_action,
                                            &eval_evidence_context,
                                        )
                                        .await
                                        {
                                            Err(e) => {
                                                self.log(format!("\x1b[31m>> [CORTEX FATAL] Evaluator panicked: {}\x1b[0m", e));
                                            }
                                            Ok(eval) => {
                                                self.log(format!(
                                                    "\x1b[33m>> [EVALUATION]\x1b[0m {} (Modifier: {:.2})",
                                                    eval.critique, eval.confidence_modifier
                                                ));

                                                session.record_evidence(&evidence, &eval, h_id);

                                                // TERMINAL HALT INTERCEPT
                                                // If the action was 'halt' and the Evaluator didn't penalize it,
                                                // force-exit the node to prevent infinite halt loops.
                                                if is_halt_action_captured && eval.confidence_modifier >= 0.0 && current_belief > 0.85 {
                                                    self.log("\x1b[1;36m>> [CORTEX TERMINAL] Valid halt executed and verified. Submitting evidence to ContractVerifier.\x1b[0m".to_string());
                                                    return Ok(CortexOutcome::CompletionProposed {
                                                        belief: current_belief.max(0.95),
                                                        evidence: session.collect_evidence(h_id),
                                                    });
                                                }

                                                // AUTO-REVERT: a mutation the Evaluator scored negative must
                                                // not be allowed to stand as satisfied evidence. Revert the
                                                // file on disk and strip the raw marker from local_evidence so
                                                // completion contracts correctly see this as unresolved instead
                                                // of "patch succeeded technically" being treated as "objective
                                                // accomplished."
                                                if eval.confidence_modifier < 0.0 {
                                                    if let Some(path) = &mutation_path {
                                                        self.log(format!("\x1b[1;31m>> [CORTEX SAFETY] Mutation to '{}' scored negative ({:.2}) -- reverting via git checkout.\x1b[0m", path, eval.confidence_modifier));
                                                        let _ =
                                                            crate::podman::PodmanShield::execute(
                                                                &format!(
                                                                    "git checkout -- {}",
                                                                    path
                                                                ),
                                                            );
                                                        session.local_evidence = session
                                                            .local_evidence
                                                            .replace(&evidence, "");
                                                        let reverted_tag = format!(
                                                            "[MUTATION_REVERTED] {} -- Evaluator rejected this mutation ({:.2}): {}\n",
                                                            path, eval.confidence_modifier, eval.critique
                                                        );
                                                        session
                                                            .local_evidence
                                                            .push_str(&reverted_tag);
                                                    }
                                                }

                                                if let Some(updated_node) =
                                                    session.graph.nodes.get(&h_id)
                                                {
                                                    self.log(format!("\x1b[1;32m>> [CORTEX] Strategy Belief updated to {:.2}\x1b[0m", updated_node.belief_score));

                                                    let _evidence_check = if is_mutation_task || is_write_task {
                                                        session.local_evidence.lines().any(|l| l.starts_with("[WRITE_OK]") || l.starts_with("[PATCH_FUNCTION_OK]") || l.starts_with("[REPLACE_LINES_OK]") || l.starts_with("[DIFF_PATCH_OK]"))
                                                    } else {
                                                        true
                                                    };
                                                    
                                                    // Deterministic early exits disabled. 
                                                    // The LLM must explicitly issue a HALT action.
                                                    if session.is_subgoal_satisfied(updated_node.belief_score) {
                                                        self.log("\x1b[1;32m>> [CORTEX] Contract criteria met. Awaiting explicit Agent HALT.\x1b[0m".to_string());
                                                    }
                                                }
                                            }
                                        }
                                    }
                                }
                            }
                        } else {
                            self.log(
                                "\x1b[31m>> [RED TEAM] VETO. Proposal rejected.\x1b[0m".to_string(),
                            );
                            let veto_msg = format!(
                                "[RED TEAM VETO] Action blocked. Critic feedback: {}",
                                critic.critique
                            );
                            let veto_obs_id =
                                session.graph.add_node(NodeKind::Observation, &veto_msg, 1.0, 50);
                            session.graph.add_edge(
                                veto_obs_id,
                                h_id,
                                EdgeRelation::Contradicts,
                                critic.confidence_modifier.abs(),
                            );
                            session.graph.propagate_beliefs();
                        }
                    }
                }
            } else {
                self.log(
                    "\x1b[1;35m>> [CORTEX] All strategies exhausted. Re-orienting graph...\x1b[0m"
                        .to_string(),
                );
                let mut reorient_context = world_context.clone();
                if !session.executed_actions.is_empty() {
                    reorient_context.push_str("\n[ALREADY ATTEMPTED ACTIONS -- DO NOT REPEAT]\n");
                    for a in &session.executed_actions {
                        reorient_context.push_str(&format!("- {}\n", a));
                    }
                }
                session.budget.record_call();
                let new_hypotheses = EpistemicEngine::generate_hypotheses(
                    &self.client,
                    &self.api_key,
                    mission,
                    &reorient_context,
                )
                .await;
                for h in new_hypotheses.unwrap_or_default() {
                    session.graph.add_node(NodeKind::Hypothesis, &h, 0.5, 100);
                    self.log(format!("\x1b[90m  -> Strategy: {}\x1b[0m", h));
                }
            }
            sleep(Duration::from_secs(2)).await;
        }
        self.log("\x1b[1;32m>> [CORTEX] Epistemic Session Complete.\x1b[0m".to_string());
        let final_evidence = session
            .best_hypothesis_id
            .map(|id| session.collect_evidence(id))
            .unwrap_or_else(|| "No evidence gathered.".to_string());

        if is_mutation_task
            && !session.local_evidence.contains("[DIFF_PATCH_OK]")
            && !session.local_evidence.contains("[PATCH_FUNCTION_OK]")
            && !session.local_evidence.contains("[WRITE_OK]")
        {
            self.log("\x1b[1;31m>> [CORTEX FAULT] Task exhausted budget without satisfying mutation constraints.\x1b[0m".to_string());
            return Ok(CortexOutcome::TurnsExhausted {
                best_belief: 0.0,
                evidence: final_evidence,
            });
        }

        Ok(CortexOutcome::CompletionProposed {
            belief: session.max_observed_belief.max(0.70),
            evidence: final_evidence,
        })
    }
}

use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct CriticVerdict {
    pub approved: bool,
    pub confidence: f32,
    pub reason: String,
    pub suggested_recovery: Option<String>,
}

pub struct Critic;

impl Critic {
    pub async fn evaluate(
        _objective: &str,
        observations: &str,
        _history: &str,
    ) -> Result<CriticVerdict, Box<dyn std::error::Error>> {
        // If there is no observation history yet, we can't be done
        if observations.trim().is_empty() {
            return Ok(CriticVerdict {
                approved: false,
                confidence: 0.1,
                reason: "No actions have been executed yet. Cannot halt.".into(),
                suggested_recovery: Some("Execute an action first.".into()),
            });
        }

        // We do NOT check for [HALT] here, because the Critic is invoked specifically
        // to validate the Halt intent *before* it gets added to the log.
        let has_write = observations.contains("[WRITE_OK]")
            || observations.contains("[DIFF_PATCH_OK]")
            || observations.contains("[PATCH_FUNCTION_OK]")
            || observations.contains("success");
        let has_read = observations.contains("[READ_OK]");
        let has_cmd = observations.contains("[CMD_OK]");
        let has_search = observations.contains("[SEARCH_OK]");

        let approved = has_write || has_read || has_cmd || has_search;
        let confidence = if approved { 0.95 } else { 0.4 };

        Ok(CriticVerdict {
            approved,
            confidence,
            reason: if approved {
                "Concrete success indicators found in action history.".into()
            } else {
                "Missing concrete success indicators (Read/Write/Cmd).".into()
            },
            suggested_recovery: if !approved {
                Some("Perform a read, write, or command execution to gather evidence.".into())
            } else {
                None
            },
        })
    }
}

/**
 * NEXUS Diff Engine — v2.0 [SURGICAL]
 * Copyright 2026 Donald Ball
 * Handles precise edits even when LLM mangles whitespace or line order.
 */
use anyhow::{anyhow, Result};

pub struct DiffEngine;

impl DiffEngine {
    /// Primary entry point: tries exact match first, then indentation-agnostic.
    pub fn apply(original: &str, search: &str, replace: &str) -> Result<String> {
        if search.trim().is_empty() {
            return Ok(original.to_string());
        }

        // ⚡ IDEMPOTENCY GUARD: Prevent the agent from injecting the same code multiple times
        if replace.trim().len() > 15 && Self::apply_agnostic(original, replace, "").is_some() {
            return Err(anyhow!(
                "PATCH_REJECTED_IDEMPOTENT: The replacement text is already fully present in the file. Do not duplicate it."
            ));
        }

        // Tier 1: Exact match (fastest, most precise)
        if let Some(pos) = original.find(search) {
            return Ok(replace_first_occurrence(original, pos, search.len(), replace));
        }

        // Tier 2: Indentation-agnostic matching (handles LLM whitespace issues)
        if let Some(result) = Self::apply_agnostic(original, search, replace) {
            return Ok(result);
        }

        Err(anyhow!(
            "PATTERN_NOT_FOUND: Could not locate search block.\n\
            Search preview (first 120 chars):\n{}\n\n\
            Hint: Try making the search string more unique or shorter.",
            search.chars().take(120).collect::<String>()
        ))
    }

    fn apply_agnostic(original: &str, search: &str, replace: &str) -> Option<String> {
        let search_lines: Vec<&str> = search.lines().map(|l| l.trim_start()).collect();
        let orig_lines: Vec<&str> = original.lines().collect();
        let search_len = search_lines.len();

        if search_len == 0 || orig_lines.len() < search_len {
            return None;
        }

        for i in 0..=(orig_lines.len() - search_len) {
            let window = &orig_lines[i..i + search_len];
            let window_trimmed: Vec<&str> = window.iter().map(|l| l.trim_start()).collect();

            if window_trimmed == search_lines {
                let mut result = orig_lines[..i].join("\n");
                if !result.is_empty() {
                    result.push('\n');
                }
                result.push_str(replace);
                if i + search_len < orig_lines.len() {
                    result.push('\n');
                    result.push_str(&orig_lines[i + search_len..].join("\n"));
                }
                return Some(result);
            }
        }
        None
    }

    pub fn unified_diff(_old: &str, _new: &str, path: &str) -> String {
        format!("--- a/{}\n+++ b/{}\n[MODIFICATION APPLIED]", path, path)
    }
}

fn replace_first_occurrence(s: &str, pos: usize, old_len: usize, new: &str) -> String {
    let mut result = String::with_capacity(s.len() - old_len + new.len());
    result.push_str(&s[..pos]);
    result.push_str(new);
    result.push_str(&s[pos + old_len..]);
    result
}

        assert_eq!(result, "fn new() {}\nfn keep() {}");
    }

        let search = "fn foo() {\n    println!(\"hi\");\n}";
        let replace = "fn bar() {\n    println!(\"hi\");\n}";
