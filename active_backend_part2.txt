// cluster.rs — v1.1
// FIX: duration_since().unwrap() → unwrap_or_default() to survive NTP jumps.
use parking_lot::RwLock;
use serde::{Deserialize, Serialize};
use std::sync::Arc;
use std::time::{SystemTime, UNIX_EPOCH};

#[derive(Clone, Serialize, Deserialize, Debug)]
pub struct ClusterNode {
    pub id: String,
    pub host: String,
    pub port: u16,
    pub cpu: f32,
    pub ram: f32,
    pub temp: f32,
    pub last_seen: u64,
}

#[derive(Clone)]
pub struct ClusterMesh {
    nodes: Arc<RwLock<Vec<ClusterNode>>>,
}

fn now_secs() -> u64 {
    SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_secs()
}

impl ClusterMesh {
    pub fn new() -> Self {
        Self { nodes: Arc::new(RwLock::new(vec![])) }
    }

    pub fn register(&self, mut node: ClusterNode) {
        node.last_seen = now_secs();
        let mut guard = self.nodes.write();
        if let Some(existing) = guard.iter_mut().find(|n| n.id == node.id) {
            *existing = node;
        } else {
            guard.push(node);
        }
    }

    pub fn get_all_nodes(&self) -> Vec<ClusterNode> {
        self.nodes.read().clone()
    }

    pub fn best_node(&self) -> Option<ClusterNode> {
        let now = now_secs();
        self.nodes.read().iter().filter(|n| now.saturating_sub(n.last_seen) < 40).cloned().min_by(
            |a, b| {
                (a.cpu + a.ram).partial_cmp(&(b.cpu + b.ram)).unwrap_or(std::cmp::Ordering::Equal)
            },
        )
    }
}

impl Default for ClusterMesh {
    fn default() -> Self {
        Self::new()
    }
}

//! Attention System – Phase 8B v2

use super::event_bus::{CognitiveEvent, CognitiveEventKind, EventBus, EventMeta};
use parking_lot::RwLock;
use std::collections::HashMap;
use std::sync::Arc;

#[derive(Clone)]
pub struct AttentionSystem {
    /// Weighted salience map (future). Single focus for now.
    salience: Arc<RwLock<HashMap<String, f32>>>,
    focus: Arc<RwLock<Option<String>>>,
    bus: EventBus,
}

impl AttentionSystem {
    pub fn new(bus: EventBus) -> Self {
        Self {
            salience: Arc::new(RwLock::new(HashMap::new())),
            focus: Arc::new(RwLock::new(None)),
            bus,
        }
    }

    pub fn current_focus(&self) -> Option<String> {
        self.focus.read().clone()
    }

    pub async fn shift(&self, new_focus: &str) {
        *self.focus.write() = Some(new_focus.to_string());
        self.salience.write().insert(new_focus.to_string(), 1.0);
        self.bus
            .publish(CognitiveEvent {
                meta: EventMeta::new("attention"),
                mission_id: None,
                kind: CognitiveEventKind::AttentionShift {
                    focus: new_focus.to_string(),
                },
            })
            .await;
    }

    pub fn set_salience(&self, topic: &str, weight: f32) {
        self.salience.write().insert(topic.to_string(), weight.clamp(0.0, 1.0));
    }

    pub fn top_salient(&self, k: usize) -> Vec<(String, f32)> {
        let mut items: Vec<_> = self.salience.read().iter().map(|(k, v)| (k.clone(), *v)).collect();
        items.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
        items.into_iter().take(k).collect()
    }

    pub fn clear(&self) {
        *self.focus.write() = None;
    }
}

use std::collections::HashMap;
use parking_lot::RwLock;
use std::sync::Arc;
use uuid::Uuid;

#[derive(Clone, Default)]
pub struct BeliefEngine {
    beliefs: Arc<RwLock<HashMap<Uuid, f32>>>,
}

impl BeliefEngine {
    pub fn new() -> Self {
        Self {
            beliefs: Arc::new(RwLock::new(HashMap::new())),
        }
    }

    pub fn update_belief(&self, hypothesis_id: Uuid, evidence_weight: f32) -> f32 {
        let mut lock = self.beliefs.write();
        let current = lock.entry(hypothesis_id).or_insert(0.5);
        let adjustment = evidence_weight * 0.1;
        *current = (*current + adjustment).clamp(0.0, 1.0);
        *current
    }

    pub fn get_belief(&self, hypothesis_id: Uuid) -> f32 {
        *self.beliefs.read().get(&hypothesis_id).unwrap_or(&0.5)
    }
    
    pub fn cull_low_confidence(&self, threshold: f32) {
        self.beliefs.write().retain(|_, v| *v >= threshold);
    }
}

use crate::orchestrator_v2::task_graph::TaskGraph;

pub struct CriticAgent;

impl CriticAgent {
    pub fn evaluate(graph: &TaskGraph) -> f32 {
        if graph.nodes.is_empty() { return 0.0; }
        let mut score = 1.0_f32;
        for node in graph.nodes.values() {
            if node.deps.is_empty() && graph.nodes.len() > 1 { score *= 0.9; }
            if node.required_agent.trim().is_empty() { score *= 0.6; }
            if node.priority == 0 { score *= 0.85; }
        }
        score.clamp(0.0, 1.0)
    }
}

use crate::cognitive_events::CognitiveEvent;
use crate::cognition_graph::{CognitiveGraph, EdgeRelation, NodeKind};
use crate::completion_engine::{CognitiveBudget, SubgoalCompletionEngine, SubgoalContract};
use crate::epistemic_engine::CriticResponse;
use std::collections::HashSet;
use uuid::Uuid;

#[derive(Debug, Clone)]
pub struct EpistemicSession {
    pub session_id: Uuid,
    pub mission: String,
    pub contract: SubgoalContract,
    pub budget: CognitiveBudget,
    pub graph: CognitiveGraph,
    pub local_evidence: String,
    pub executed_actions: HashSet<String>,
    pub max_observed_belief: f32,
    pub best_hypothesis_id: Option<Uuid>,
}

impl EpistemicSession {
    pub fn new(mission: &str) -> Self {
        Self {
            session_id: Uuid::new_v4(),
            mission: mission.to_string(),
            contract: SubgoalCompletionEngine::infer_contract("session_node", mission),
            budget: CognitiveBudget::default_for_node(),
            graph: CognitiveGraph::new(),
            local_evidence: String::new(),
            executed_actions: HashSet::new(),
            max_observed_belief: 0.0,
            best_hypothesis_id: None,
        }
    }

    pub fn apply_event(&mut self, event: &CognitiveEvent) {
        match event {
            CognitiveEvent::HypothesisCreated { hypothesis_id, hypothesis, .. } => {
                self.graph.add_node(NodeKind::Hypothesis, hypothesis, 0.5, 100);
                if self.best_hypothesis_id.is_none() {
                    self.best_hypothesis_id = Some(*hypothesis_id);
                }
            }
            CognitiveEvent::ObservationRecorded { evidence, modifier, .. } => {
                self.local_evidence.push_str(evidence);
                self.local_evidence.push('\n');

                if let Some(h_id) = self.best_hypothesis_id {
                    let obs_id = self.graph.add_node(NodeKind::Observation, evidence, 1.0, 50);
                    let relation = if *modifier > 0.0 {
                        EdgeRelation::Supports
                    } else {
                        EdgeRelation::Contradicts
                    };
                    self.graph.add_edge(obs_id, h_id, relation, modifier.abs());
                    self.graph.propagate_beliefs();

                    if let Some(node) = self.graph.nodes.get(&h_id) {
                        if node.belief_score > self.max_observed_belief {
                            self.max_observed_belief = node.belief_score;
                        }
                    }
                }
            }
            CognitiveEvent::BeliefUpdated { hypothesis_id, new_belief, .. } => {
                if let Some(node) = self.graph.nodes.get_mut(hypothesis_id) {
                    node.belief_score = *new_belief;
                    if *new_belief > self.max_observed_belief {
                        self.max_observed_belief = *new_belief;
                    }
                }
            }
            _ => {}
        }
    }

    pub fn record_action_attempt(&mut self, action_key: &str) {
        self.executed_actions.insert(action_key.to_string());
    }

    pub fn is_action_attempted(&self, action_key: &str) -> bool {
        self.executed_actions.contains(action_key)
    }

    pub fn record_evidence(&mut self, evidence: &str, eval: &CriticResponse, h_id: Uuid) {
        let event = CognitiveEvent::ObservationRecorded {
            session_id: self.session_id,
            evidence: evidence.to_string(),
            modifier: eval.confidence_modifier,
        };
        self.best_hypothesis_id = Some(h_id);
        self.apply_event(&event);
    }

    pub fn is_subgoal_satisfied(&self, current_belief: f32) -> bool {
        SubgoalCompletionEngine::evaluate(&self.contract, &self.local_evidence, current_belief)
    }

    pub fn collect_evidence(&self, h_id: Uuid) -> String {
        let mut history = String::new();
        for edge in &self.graph.edges {
            if edge.target == h_id
                && (edge.relation == EdgeRelation::Supports || edge.relation == EdgeRelation::Contradicts)
            {
                if let Some(obs) = self.graph.nodes.get(&edge.source) {
                    history.push_str(&format!("- [{:?}] {}\n", edge.relation, obs.content));
                }
            }
        }
        if history.is_empty() {
            "No evidence recorded.".to_string()
        } else {
            history
        }
    }
}

//! Cognitive Event Bus – Phase 8B v2
//! Async broadcast + pluggable EventStore.

use async_trait::async_trait;
use parking_lot::RwLock;
use serde::{Deserialize, Serialize};
use std::sync::Arc;
use std::time::{SystemTime, UNIX_EPOCH};
use tokio::sync::broadcast;
use uuid::Uuid;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct EventMeta {
    pub event_id: Uuid,
    pub timestamp_ms: u64,
    pub source: String,
    pub correlation_id: Option<Uuid>,
    pub causation_id: Option<Uuid>,
}

impl EventMeta {
    pub fn new(source: &str) -> Self {
        let ts =
            SystemTime::now().duration_since(UNIX_EPOCH).map(|d| d.as_millis() as u64).unwrap_or(0);
        Self {
            event_id: Uuid::new_v4(),
            timestamp_ms: ts,
            source: source.to_string(),
            correlation_id: None,
            causation_id: None,
        }
    }

    pub fn with_correlation(mut self, id: Uuid) -> Self {
        self.correlation_id = Some(id);
        self
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum CognitiveEventKind {
    Observation { content: String },
    BeliefUpdated { hypothesis_id: Uuid, new_score: f32 },
    GoalActivated { goal_id: Uuid },
    GoalCompleted { goal_id: Uuid },
    ReflectionCompleted { lessons: Vec<String> },
    MissionFinished { success: bool },
    StrategyChanged { strategy: String },
    Interrupt { reason: String },
    AttentionShift { focus: String },
    ResourceBudgetLow { remaining: f32 },
    DecisionMade { decision_summary: String },
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CognitiveEvent {
    pub meta: EventMeta,
    pub mission_id: Option<Uuid>,
    pub kind: CognitiveEventKind,
}

/// Pluggable history backend (Vec today, SQLite/RocksDB later).
#[async_trait]
pub trait EventStore: Send + Sync {
    async fn append(&self, event: &CognitiveEvent);
    async fn recent(&self, n: usize) -> Vec<CognitiveEvent>;
}

#[derive(Default)]
pub struct InMemoryEventStore {
    events: RwLock<Vec<CognitiveEvent>>,
    max: usize,
}

impl InMemoryEventStore {
    pub fn new(max: usize) -> Self {
        Self { events: RwLock::new(Vec::new()), max }
    }
}

#[async_trait]
impl EventStore for InMemoryEventStore {
    async fn append(&self, event: &CognitiveEvent) {
        let mut guard = self.events.write();
        guard.push(event.clone());
        if guard.len() > self.max {
            let excess = guard.len() - self.max;
            guard.drain(0..excess);
        }
    }

    async fn recent(&self, n: usize) -> Vec<CognitiveEvent> {
        let guard = self.events.read();
        guard.iter().rev().take(n).cloned().collect()
    }
}

#[derive(Clone)]
pub struct EventBus {
    tx: broadcast::Sender<CognitiveEvent>,
    store: Arc<dyn EventStore>,
}

impl EventBus {
    pub fn new() -> Self {
        let (tx, _) = broadcast::channel(256);
        Self { tx, store: Arc::new(InMemoryEventStore::new(256)) }
    }

    pub fn with_store(store: Arc<dyn EventStore>) -> Self {
        let (tx, _) = broadcast::channel(256);
        Self { tx, store }
    }

    pub fn subscribe(&self) -> broadcast::Receiver<CognitiveEvent> {
        self.tx.subscribe()
    }

    pub async fn publish(&self, event: CognitiveEvent) {
        self.store.append(&event).await;
        // Ignore lagging receivers – cognitive runtime must not block on slow listeners.
        let _ = self.tx.send(event);
    }

    pub async fn recent(&self, n: usize) -> Vec<CognitiveEvent> {
        self.store.recent(n).await
    }
}

impl Default for EventBus {
    fn default() -> Self {
        Self::new()
    }
}

//! Executive Cortex – Phase 8B v2
//! Staged decision pipeline:
//!   observe → evaluate_policy → evaluate_goals → evaluate_risk → choose → emit

use super::event_bus::{CognitiveEvent, CognitiveEventKind, EventBus, EventMeta};
use super::policy_engine::PolicyEngine;
use super::traits::*;
use std::time::{SystemTime, UNIX_EPOCH};
use uuid::Uuid;

#[derive(Clone)]
pub struct ExecutiveCortex {
    bus: EventBus,
    policy: PolicyEngine,
}

#[derive(Debug, Default)]
struct Evaluation {
    policy_ok: bool,
    goal_pressure: f32,   // 0 = none, 1 = urgent
    risk: f32,            // 0 = safe, 1 = critical
    resource_pressure: f32,
}

impl ExecutiveCortex {
    pub fn new(bus: EventBus) -> Self {
        Self {
            bus,
            policy: PolicyEngine::new(),
        }
    }

    pub fn with_policy(bus: EventBus, policy: PolicyEngine) -> Self {
        Self { bus, policy }
    }

    // ── Pipeline stages ────────────────────────────────────────────────────

    fn observe(&self, state: &MissionState) -> Evaluation {
        let mut ev = Evaluation::default();
        ev.policy_ok = true; // policy checked against actions, not mission state
        ev.risk = 1.0 - state.health.clamp(0.0, 1.0);
        ev.resource_pressure = 0.0; // filled by scheduler later
        ev.goal_pressure = state.priority.clamp(0.0, 1.0);
        ev
    }

    fn evaluate_policy(&self, ev: &mut Evaluation, _state: &MissionState) {
        // Placeholder: mission-level policy always passes in 8B.
        // Action-level checks happen via PolicyEngine::check_action.
        ev.policy_ok = true;
    }

    fn evaluate_goals(&self, ev: &mut Evaluation, state: &MissionState) {
        // Higher priority + active status → higher pressure to continue.
        if state.status == MissionStatus::Active {
            ev.goal_pressure = state.priority.clamp(0.0, 1.0);
        } else {
            ev.goal_pressure = 0.0;
        }
    }

    fn evaluate_risk(&self, ev: &mut Evaluation, state: &MissionState) {
        ev.risk = 1.0 - state.health.clamp(0.0, 1.0);
    }

    fn choose(&self, ev: &Evaluation, state: &MissionState) -> (ExecutiveDecisionKind, String, f32) {
        match state.status {
            MissionStatus::Completed | MissionStatus::Failed | MissionStatus::Aborted => {
                (ExecutiveDecisionKind::Reflect, "mission terminal".into(), 0.95)
            }
            MissionStatus::Paused => {
                (ExecutiveDecisionKind::Continue, "resume paused mission".into(), 0.7)
            }
            MissionStatus::Active => {
                if !ev.policy_ok {
                    (ExecutiveDecisionKind::Abort, "policy violation".into(), 0.9)
                } else if ev.risk > 0.75 {
                    (
                        ExecutiveDecisionKind::Abort,
                        format!("risk critical ({:.2})", ev.risk),
                        0.9,
                    )
                } else if ev.risk > 0.5 {
                    (
                        ExecutiveDecisionKind::Replan,
                        format!("risk elevated ({:.2})", ev.risk),
                        0.8,
                    )
                } else if ev.resource_pressure > 0.8 {
                    (
                        ExecutiveDecisionKind::Pause,
                        "resource pressure high".into(),
                        0.75,
                    )
                } else {
                    (ExecutiveDecisionKind::Continue, "mission healthy".into(), 0.85)
                }
            }
        }
    }

    // ── Public API ─────────────────────────────────────────────────────────

    pub async fn decide(&self, state: &MissionState) -> ExecutiveDecision {
        let mut ev = self.observe(state);
        self.evaluate_policy(&mut ev, state);
        self.evaluate_goals(&mut ev, state);
        self.evaluate_risk(&mut ev, state);
        let (kind, rationale, confidence) = self.choose(&ev, state);

        let decision = ExecutiveDecision {
            kind,
            reason: DecisionReason {
                confidence,
                supporting_evidence: vec![
                    format!("health={:.2}", state.health),
                    format!("risk={:.2}", ev.risk),
                    format!("goal_pressure={:.2}", ev.goal_pressure),
                ],
                policy_trigger: Some("phase8b_v2_pipeline".into()),
                expected_cost: 0.0,
                rationale,
            },
        };

        self.bus
            .publish(CognitiveEvent {
                meta: EventMeta::new("executive_cortex"),
                mission_id: Some(state.id),
                kind: CognitiveEventKind::DecisionMade {
                    decision_summary: format!("{:?}", decision.kind),
                },
            })
            .await;

        decision
    }

    pub fn start_mission(&self, directive: &str, priority: f32) -> MissionState {
        MissionState {
            id: Uuid::new_v4(),
            directive: directive.to_string(),
            priority,
            health: 1.0,
            status: MissionStatus::Active,
            intent: Some(directive.to_string()),
        }
    }

    pub fn policy(&self) -> &PolicyEngine {
        &self.policy
    }
}

use std::collections::{HashMap, HashSet};
use uuid::Uuid;

#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub enum NodeKind {
    Observation,
    Verification,
    Reflection,
    Hypothesis,
    ProposedAction { action_json: String },
    Strategy,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct VerificationReport {
    pub goal_satisfied: bool,
    pub evidence_confidence: f32,
    pub independent_check_passed: bool,
    pub rationale: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CognitiveNode {
    pub id: Uuid,
    pub kind: NodeKind,
    pub content: String,
    pub belief_score: f32,
    pub computation_cost: u32,
    pub is_active: bool,
}

#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub enum EdgeRelation {
    Supports,
    Contradicts,
    Prerequisite,
    DerivesFrom,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GraphEdge {
    pub source: Uuid,
    pub target: Uuid,
    pub relation: EdgeRelation,
    pub weight: f32,
}

#[derive(Debug, Clone, Default)]
pub struct CognitiveGraph {
    pub nodes: HashMap<Uuid, CognitiveNode>,
    pub edges: Vec<GraphEdge>,
    pub frontiers: HashSet<Uuid>,
}

impl CognitiveGraph {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn add_node(
        &mut self,
        kind: NodeKind,
        content: &str,
        initial_belief: f32,
        cost: u32,
    ) -> Uuid {
        let id = Uuid::new_v4();
        let node = CognitiveNode {
            id,
            kind,
            content: content.to_string(),
            belief_score: initial_belief,
            computation_cost: cost,
            is_active: false,
        };
        self.nodes.insert(id, node);
        self.frontiers.insert(id);
        id
    }

    pub fn add_edge(&mut self, source: Uuid, target: Uuid, relation: EdgeRelation, weight: f32) {
        self.edges.push(GraphEdge { source, target, relation, weight });
    }

    pub fn propagate_beliefs(&mut self) {
        // Accumulate signed impacts first so multiple edges targeting the same
        // hypothesis are combined instead of overwriting each other.
        let mut delta: HashMap<Uuid, f32> = HashMap::new();
        const DAMPING: f32 = 0.85;

        for edge in &self.edges {
            let Some(source_node) = self.nodes.get(&edge.source) else {
                continue;
            };
            let impact = source_node.belief_score * edge.weight;
            let signed = match edge.relation {
                EdgeRelation::Supports => impact,
                EdgeRelation::Contradicts => -impact,
                // Prerequisite / DerivesFrom intentionally ignored for score updates
                _ => continue,
            };
            *delta.entry(edge.target).or_insert(0.0) += signed;
        }

        for (id, d) in delta {
            if let Some(node) = self.nodes.get_mut(&id) {
                let candidate = (node.belief_score + d).clamp(0.0, 1.0);
                // Keep a fraction of the previous score (PageRank-style damping)
                node.belief_score = DAMPING * candidate + (1.0 - DAMPING) * node.belief_score;
            }
        }
    }

    pub fn select_best_hypothesis(&self) -> Option<&CognitiveNode> {
        self.nodes
            .values()
            // ⚡ FIX: Filter out dead hypotheses (belief < 0.15).
            // v1.9: When BeliefVector is available on the node, prefer may_prove() over raw score.
            .filter(|n| matches!(n.kind, NodeKind::Hypothesis) && n.belief_score >= 0.15)
            .max_by(|a, b| {
                let score_a = a.belief_score / (a.computation_cost as f32).max(1.0);
                let score_b = b.belief_score / (b.computation_cost as f32).max(1.0);
                score_a.partial_cmp(&score_b).unwrap_or(std::cmp::Ordering::Equal)
            })
    }
}

//! Policy Engine – Phase 8B v2
//! Capability-based authorization.

use std::collections::HashSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum Capability {
    FilesystemRead,
    FilesystemWrite,
    Network,
    Shell,
    Patch,
    Delete,
    MemoryWrite,
    SpawnSubagent,
}

#[derive(Debug, Clone)]
pub struct PolicyEngine {
    allowed: HashSet<Capability>,
}

impl Default for PolicyEngine {
    fn default() -> Self {
        let mut allowed = HashSet::new();
        allowed.insert(Capability::FilesystemRead);
        allowed.insert(Capability::FilesystemWrite);
        allowed.insert(Capability::Shell);
        allowed.insert(Capability::Patch);
        allowed.insert(Capability::MemoryWrite);
        // Delete and unrestricted Network denied by default
        Self { allowed }
    }
}

impl PolicyEngine {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn allows(&self, cap: Capability) -> bool {
        self.allowed.contains(&cap)
    }

    pub fn grant(&mut self, cap: Capability) {
        self.allowed.insert(cap);
    }

    pub fn revoke(&mut self, cap: Capability) {
        self.allowed.remove(&cap);
    }

    /// Map a raw action string to required capabilities (best-effort).
    pub fn required_capabilities(action: &str) -> Vec<Capability> {
        let lower = action.to_lowercase();
        let mut caps = Vec::new();
        if lower.contains("write_file") || lower.contains("diff_patch") || lower.contains("patch_function") {
            caps.push(Capability::FilesystemWrite);
            caps.push(Capability::Patch);
        }
        if lower.contains("read_file") || lower.contains("search_code") || lower.contains("read_file_chunk") {
            caps.push(Capability::FilesystemRead);
        }
        if lower.contains("execute_command") || lower.starts_with("rg ") || lower.starts_with("find ") {
            caps.push(Capability::Shell);
        }
        if lower.contains("http") || lower.contains("curl") || lower.contains("wget") {
            caps.push(Capability::Network);
        }
        if lower.contains("rm ") || lower.contains("delete") {
            caps.push(Capability::Delete);
        }
        if lower.contains("spawn") || lower.contains("delegate") {
            caps.push(Capability::SpawnSubagent);
        }
        if caps.is_empty() {
            caps.push(Capability::Shell); // conservative default
        }
        caps
    }

    pub fn check_action(&self, action: &str) -> bool {
        Self::required_capabilities(action)
            .into_iter()
            .all(|c| self.allows(c))
    }
}

            }
        } else {
            PredictionEvaluation { accuracy: 0.0, error: 0.0 }

        let mut lessons = Vec::new();
        let mut metrics = HashMap::new();
        let mut memory_updates = Vec::new();
        let mut belief_updates = Vec::new();
        
        let confidence = if outcome.success { 0.9 } else { 0.4 };
        
        if outcome.success {
            lessons.push("Mission successful. Core strategy validated.".to_string());

                lessons.push(format!("Failure vector identified: {}", failure));
            }
            metrics.insert("success_rate".to_string(), 0.0);
            
            if let Some(strategy_id) = outcome.failed_strategy_id {
                belief_updates.push(BeliefUpdate {
                    target: strategy_id,
                    delta: -0.2, 
                });
            }
        }

        ReflectionReport {
            mission_id: outcome.mission_id,
            lessons,
            confidence,
            recommended_strategy: outcome.successful_strategies.first().cloned(),
            memory_updates,
            belief_updates,
            metrics,
        }
    }
}

        Ok(report)
    }

//! Working Memory – Phase 8B v2
//! Partitioned: facts / observations / reasoning / hypotheses.

use parking_lot::RwLock;
use std::collections::HashMap;
use std::sync::Arc;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum MemoryPartition {
    Facts,
    Observations,
    Reasoning,
    Hypotheses,
}

#[derive(Clone, Default)]
pub struct WorkingMemory {
    partitions: Arc<RwLock<HashMap<MemoryPartition, HashMap<String, String>>>>,
}

impl WorkingMemory {
    pub fn new() -> Self {
        let mut map = HashMap::new();
        for p in [
            MemoryPartition::Facts,
            MemoryPartition::Observations,
            MemoryPartition::Reasoning,
            MemoryPartition::Hypotheses,
        ] {
            map.insert(p, HashMap::new());
        }
        Self {
            partitions: Arc::new(RwLock::new(map)),
        }
    }

    pub fn store(&self, partition: MemoryPartition, key: &str, value: &str) {
        if let Some(slot) = self.partitions.write().get_mut(&partition) {
            slot.insert(key.to_string(), value.to_string());
        }
    }

    pub fn recall(&self, partition: MemoryPartition, key: &str) -> Option<String> {
        self.partitions
            .read()
            .get(&partition)
            .and_then(|s| s.get(key).cloned())
    }

    pub fn clear_partition(&self, partition: MemoryPartition) {
        if let Some(slot) = self.partitions.write().get_mut(&partition) {
            slot.clear();
        }
    }

    pub fn clear_all(&self) {
        for slot in self.partitions.write().values_mut() {
            slot.clear();
        }
    }

    pub fn snapshot(&self, partition: MemoryPartition) -> HashMap<String, String> {
        self.partitions
            .read()
            .get(&partition)
            .cloned()
            .unwrap_or_default()
    }
}

use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub enum CompletionCondition {
    ReadMarkerAcquired,
    SearchMarkerAcquired,
    WriteMarkerAcquired,
    PatchMarkerAcquired,
    CommandMarkerAcquired,
    MinBeliefScore(f32),
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SubgoalContract {
    pub node_id: String,
    pub objective: String,
    pub conditions: Vec<CompletionCondition>,
    pub is_mutation_task: bool,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CognitiveBudget {
    pub max_turns: u32,
    pub max_model_calls: u32,
    pub model_calls_made: u32,
}

impl CognitiveBudget {
    pub fn default_for_node() -> Self {
        Self {
            max_turns: 8,
            max_model_calls: 15,
            model_calls_made: 0,
        }
    }

    pub fn record_call(&mut self) {
        self.model_calls_made += 1;
    }

    pub fn is_exhausted(&self, current_turn: u32) -> bool {
        current_turn >= self.max_turns || self.model_calls_made >= self.max_model_calls
    }
}

pub struct SubgoalCompletionEngine;

impl SubgoalCompletionEngine {
    pub fn infer_contract(node_id: &str, objective: &str) -> SubgoalContract {
        let obj_lower = objective.to_lowercase();
        let mut conditions = Vec::new();
        let mut is_mutation_task = false;

        if obj_lower.contains("write") || obj_lower.contains("export") || obj_lower.contains("save") || obj_lower.contains("output") {
            conditions.push(CompletionCondition::WriteMarkerAcquired);
            is_mutation_task = true;
        } else if obj_lower.contains("patch") || obj_lower.contains("apply") || obj_lower.contains("fix") || obj_lower.contains("remediate") {
            conditions.push(CompletionCondition::PatchMarkerAcquired);
            is_mutation_task = true;
        } else if obj_lower.contains("search") || obj_lower.contains("find") || obj_lower.contains("locate") {
            conditions.push(CompletionCondition::SearchMarkerAcquired);
            conditions.push(CompletionCondition::MinBeliefScore(0.65));
        } else if obj_lower.contains("read") || obj_lower.contains("inspect") || obj_lower.contains("audit") {
            conditions.push(CompletionCondition::ReadMarkerAcquired);
            conditions.push(CompletionCondition::MinBeliefScore(0.65));
        } else {
            conditions.push(CompletionCondition::CommandMarkerAcquired);
            conditions.push(CompletionCondition::MinBeliefScore(0.65));
        }

        SubgoalContract {
            node_id: node_id.to_string(),
            objective: objective.to_string(),
            conditions,
            is_mutation_task,
        }
    }

    pub fn evaluate(contract: &SubgoalContract, local_evidence: &str, belief_score: f32) -> bool {
        for cond in &contract.conditions {
            match cond {
                CompletionCondition::WriteMarkerAcquired => {
                    if !local_evidence.lines().any(|l| l.starts_with("[WRITE_OK]")) {
                        return false;
                    }
                }
                CompletionCondition::PatchMarkerAcquired => {
                    let has_patch = local_evidence.lines().any(|l| l.starts_with("[DIFF_PATCH_OK]"))
                        || local_evidence.lines().any(|l| l.starts_with("[PATCH_FUNCTION_OK]"))
                        || local_evidence.lines().any(|l| l.starts_with("[WRITE_OK]"));
                    if !has_patch {
                        return false;
                    }
                }
                CompletionCondition::SearchMarkerAcquired => {
                    let has_search = local_evidence.lines().any(|l| l.starts_with("[SEARCH_CODE_OK]"))
                        || local_evidence.lines().any(|l| l.starts_with("[SEARCH_OK]"))
                        || local_evidence.lines().any(|l| l.starts_with("[READ_CHUNK_OK]"))
                        || local_evidence.lines().any(|l| l.starts_with("[READ_OK]"));
                    if !has_search {
                        return false;
                    }
                }
                CompletionCondition::ReadMarkerAcquired => {
                    let has_read = local_evidence.lines().any(|l| l.starts_with("[READ_CHUNK_OK]"))
                        || local_evidence.lines().any(|l| l.starts_with("[READ_OK]"))
                        || local_evidence.lines().any(|l| l.starts_with("[SEARCH_CODE_OK]"));
                    if !has_read {
                        return false;
                    }
                }
                CompletionCondition::CommandMarkerAcquired => {
                    let has_cmd = local_evidence.lines().any(|l| l.starts_with("[CMD_OK]"))
                        || local_evidence.lines().any(|l| l.starts_with("[READ_CHUNK_OK]"))
                        || local_evidence.lines().any(|l| l.starts_with("[SEARCH_CODE_OK]"))
                        || local_evidence.lines().any(|l| l.starts_with("[READ_OK]"));
                    if !has_cmd {
                        return false;
                    }
                }
                CompletionCondition::MinBeliefScore(min) => {
                    if belief_score < *min {
                        return false;
                    }
                }
            }
        }
        true
    }
}

use crate::metrics::METRICS;
use crate::orchestrator::Orchestrator;
use crate::router::AgentProfile;
use nexus_model_hub::factory::AgentFactory;
use nexus_model_hub::neural_agent::ContextPacket;
use nexus_substrate::token_ledger::wait_for_quota;
use reqwest::Client;
use serde_json::json;
use std::sync::atomic::Ordering;
use tokio::sync::broadcast;

const ESTIMATED_TOKENS: u32 = 2_000;

pub struct Conductor;

pub struct TurnContext<'a> {
    pub orch: &'a Orchestrator,
    pub client: &'a Client,
    pub api_key: &'a str,
    pub tx: Option<broadcast::Sender<String>>,
}

impl Conductor {
    pub async fn run_turn(
        profile: &AgentProfile,
        prompt: String,
        turn: usize,
        ctx: TurnContext<'_>,
        errors: usize,
    ) -> Result<String, String> {
        METRICS.api_calls.fetch_add(1, Ordering::Relaxed);

        let model = AgentFactory::get_model_for_turn(turn, errors);
        let _ = wait_for_quota(model, ESTIMATED_TOKENS).await;

        let neural_ctx = ContextPacket {
            system_prompt: Some(profile.persona.clone()),
            model_override: Some(model.to_string()),
            ..Default::default()
        };

        if let Some(ref sender) = ctx.tx {
            let _ = sender.send(
                json!({"type": "NEURAL_THOUGHT", "payload": format!("[CONDUCTOR] turn {} → {}", turn, model)}).to_string()
            );
        }

        let result =
            ctx.orch.run(prompt.clone(), neural_ctx, ctx.client, ctx.api_key, ctx.tx.clone()).await;

        if let Err(ref e) = result {
            if e.contains("STALE")
                || e.contains("500")
                || e.contains("RESOURCE_EXHAUSTED")
                || e.contains("INTERNAL")
                || e.contains("CONTEXT_OVERFLOW")
            {
                METRICS.task_failures.fetch_add(1, Ordering::Relaxed);

                let failover = AgentFactory::failover_model(model);
                tracing::warn!("[CONDUCTOR] {} failed → failing over to {}", model, failover);
                let _ = wait_for_quota(failover, ESTIMATED_TOKENS).await;

                let fail_ctx = ContextPacket {
                    system_prompt: Some(profile.persona.clone()),
                    model_override: Some(failover.to_string()),
                    ..Default::default()
                };
                return ctx.orch.run(prompt, fail_ctx, ctx.client, ctx.api_key, ctx.tx).await;
            }
        }
        result
    }
}

use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct CriticVerdict {
    pub approved: bool,
    pub confidence: f32,
    pub reason: String,
    pub suggested_recovery: Option<String>,
}

pub struct Critic;

impl Critic {
    pub async fn evaluate(
        _objective: &str,
        observations: &str,
        _history: &str,
    ) -> Result<CriticVerdict, Box<dyn std::error::Error>> {
        // If there is no observation history yet, we can't be done
        if observations.trim().is_empty() {
            return Ok(CriticVerdict {
                approved: false,
                confidence: 0.1,
                reason: "No actions have been executed yet. Cannot halt.".into(),
                suggested_recovery: Some("Execute an action first.".into()),
            });
        }

        // We do NOT check for [HALT] here, because the Critic is invoked specifically
        // to validate the Halt intent *before* it gets added to the log.
        let has_write = observations.contains("[WRITE_OK]")
            || observations.contains("[DIFF_PATCH_OK]")
            || observations.contains("[PATCH_FUNCTION_OK]")
            || observations.contains("success");
        let has_read = observations.contains("[READ_OK]");
        let has_cmd = observations.contains("[CMD_OK]");
        let has_search = observations.contains("[SEARCH_OK]");

        let approved = has_write || has_read || has_cmd || has_search;
        let confidence = if approved { 0.95 } else { 0.4 };

        Ok(CriticVerdict {
            approved,
            confidence,
            reason: if approved {
                "Concrete success indicators found in action history.".into()
            } else {
                "Missing concrete success indicators (Read/Write/Cmd).".into()
            },
            suggested_recovery: if !approved {
                Some("Perform a read, write, or command execution to gather evidence.".into())
            } else {
                None
            },
        })
    }
}

/**
 * NEXUS Diff Engine — v2.0 [SURGICAL + FUZZY]
 * Copyright 2026 Donald Ball
 * Handles precise edits even when LLM mangles whitespace or line order.
 */
use anyhow::{anyhow, Result};

pub struct DiffEngine;

impl DiffEngine {
    /// Primary entry point: tries exact match first, then fuzzy/indentation-agnostic.
    pub fn apply(original: &str, search: &str, replace: &str) -> Result<String> {
        if search.trim().is_empty() {
            return Ok(original.to_string());
        }

        // ⚡ IDEMPOTENCY GUARD: Prevent the agent from injecting the same code multiple times
        // If the replacement block is substantial and already exists in the file (ignoring indentation), reject it.
        if replace.trim().len() > 15 && Self::apply_agnostic(original, replace, "").is_some() {
            return Err(anyhow!(
                "PATCH_REJECTED_IDEMPOTENT: The replacement text is already fully present in the file. Do not duplicate it."
            ));
        }

        // Tier 1: Exact match (fastest, most precise)
        if let Some(pos) = original.find(search) {
            return Ok(replace_first_occurrence(original, pos, search.len(), replace));
        }

        // Tier 2: Indentation-agnostic matching (handles LLM whitespace issues)
        if let Some(result) = Self::apply_agnostic(original, search, replace) {
            return Ok(result);
        }

        // Tier 3: Fuzzy line-based matching (for larger blocks)
        if let Some(result) = Self::apply_fuzzy_lines(original, search, replace) {
            return Ok(result);
        }

        Err(anyhow!(
            "PATTERN_NOT_FOUND: Could not locate search block.\n\
            Search preview (first 120 chars):\n{}\n\n\
            Hint: Try making the search string more unique or shorter.",
            search.chars().take(120).collect::<String>()
        ))
    }

    fn apply_agnostic(original: &str, search: &str, replace: &str) -> Option<String> {
        let search_lines: Vec<&str> = search.lines().map(|l| l.trim_start()).collect();
        let orig_lines: Vec<&str> = original.lines().collect();
        let search_len = search_lines.len();

        if search_len == 0 || orig_lines.len() < search_len {
            return None;
        }

        for i in 0..=(orig_lines.len() - search_len) {
            let window = &orig_lines[i..i + search_len];
            let window_trimmed: Vec<&str> = window.iter().map(|l| l.trim_start()).collect();

            if window_trimmed == search_lines {
                let mut result = orig_lines[..i].join("\n");
                if !result.is_empty() {
                    result.push('\n');
                }
                result.push_str(replace);
                if i + search_len < orig_lines.len() {
                    result.push('\n');
                    result.push_str(&orig_lines[i + search_len..].join("\n"));
                }
                return Some(result);
            }
        }
        None
    }

    fn apply_fuzzy_lines(original: &str, search: &str, replace: &str) -> Option<String> {
        // Very coarse fallback: if search is long, try to match by first and last line
        let search_lines: Vec<&str> = search.lines().collect();
        if search_lines.len() < 3 {
            return None;
        }

        let first = search_lines[0].trim();
        let last = search_lines.last().unwrap().trim();

        let orig_lines: Vec<&str> = original.lines().collect();

        for i in 0..orig_lines.len() {
            if orig_lines[i].trim() == first {
                for j in (i + 2)..orig_lines.len() {
                    if orig_lines[j].trim() == last {
                        let mut result = orig_lines[..i].join("\n");
                        if !result.is_empty() {
                            result.push('\n');
                        }
                        result.push_str(replace);
                        if j + 1 < orig_lines.len() {
                            result.push('\n');
                            result.push_str(&orig_lines[j + 1..].join("\n"));
                        }
                        return Some(result);
                    }
                }
            }
        }
        None
    }

    pub fn unified_diff(_old: &str, _new: &str, path: &str) -> String {
        format!("--- a/{}\n+++ b/{}\n[MODIFICATION APPLIED]", path, path)
    }
}

fn replace_first_occurrence(s: &str, pos: usize, old_len: usize, new: &str) -> String {
    let mut result = String::with_capacity(s.len() - old_len + new.len());
    result.push_str(&s[..pos]);
    result.push_str(new);
    result.push_str(&s[pos + old_len..]);
    result
}

pub enum EventKind {

    TaskStarted,
    TaskCompleted,
    AgentCrashed,
    ArtifactCommitted,
}

pub struct EventRef {
    pub seq: u64,
    pub kind: EventKind,
    pub object_id: String,
}

#[derive(Clone)]
pub struct EventBus {
    tx: broadcast::Sender<EventRef>,
}

impl EventBus {
    pub fn new(size: usize) -> Self {
        let (tx, _) = broadcast::channel(size);
        Self { tx }
    }

    pub fn emit(&self, evt: EventRef) {
        let _ = self.tx.send(evt);
    }

    pub fn subscribe(&self) -> broadcast::Receiver<EventRef> {
        self.tx.subscribe()
    }
}

        self.implementation_confidence = 0.0;
        self.architectural_risk = 1.0;
        self.residual = 1.0;
        self
    }

    /// Scalar utility used by ranking logic.
    pub fn scalar_score(&self) -> f32 {
        let positive = (self.implementation_confidence * 0.40)
            + (self.evidence_quality * 0.35)
            + (self.security_confidence * 0.25);
        let penalty = self.architectural_risk * 0.45;
        (positive - penalty).clamp(0.0, 1.0)
    }

    pub fn update(&self, other: &Self, weight: f32) -> Self {
        let w = weight.clamp(0.0, 1.0);
        let mut next = Self::new(
            self.implementation_confidence * (1.0 - w) + other.implementation_confidence * w,
            self.evidence_quality * (1.0 - w) + other.evidence_quality * w,
            self.architectural_risk * (1.0 - w) + other.architectural_risk * w,
            self.security_confidence * (1.0 - w) + other.security_confidence * w,
        );
        next.residual = (self.scalar_score() - next.scalar_score()).abs();
        next.has_success_marker = self.has_success_marker || other.has_success_marker;
        next.verified_artifacts = self.verified_artifacts | other.verified_artifacts;
        next
    }

    /// Calibrated Bayesian update on the positive dimensions with damping.
    pub fn bayesian_update(&self, likelihood: f32, damping: f32) -> Self {
        let likelihood = likelihood.clamp(0.0, 1.0);
        let damping = damping.clamp(0.0, 1.0);

        fn bayes(prior: f32, like: f32) -> f32 {
            let num = prior * like;
            let den = num + (1.0 - prior) * (1.0 - like);
            if den == 0.0 {
                prior
            } else {
                num / den
            }
        }

        let impl_post = bayes(self.implementation_confidence, likelihood);
        let evid_post = bayes(self.evidence_quality, likelihood);
        let sec_post = bayes(self.security_confidence, likelihood);

        let impl_d = damping * impl_post + (1.0 - damping) * self.implementation_confidence;
        let evid_d = damping * evid_post + (1.0 - damping) * self.evidence_quality;
        let sec_d = damping * sec_post + (1.0 - damping) * self.security_confidence;

        let mut next = Self::new(impl_d, evid_d, self.architectural_risk, sec_d);
        next.residual = (self.scalar_score() - next.scalar_score()).abs();
        next.has_success_marker = self.has_success_marker;
        next.verified_artifacts = self.verified_artifacts;
        next
    }

    /// Hard evidence gate. Belief alone is never enough.
    /// required_artifacts is a bitmask -- ALL required bits must actually be
    /// verified (bitwise AND), not merely a numerically-larger bit pattern.
    pub fn may_prove(&self, required_artifacts: u8, threshold: f32) -> bool {
        self.scalar_score() >= threshold
            && self.has_success_marker
            && (self.verified_artifacts & required_artifacts) == required_artifacts
    }

    /// Mark that a concrete success artifact was observed.
    pub fn mark_success(&mut self, artifact_bit: u8) {
        self.has_success_marker = true;
        self.verified_artifacts |= artifact_bit;
    }

    /// Convenience for residual-driven schedulers.
    pub fn scheduling_priority(&self, impact: f32, mission_priority: f32) -> f32 {
        self.residual * impact * mission_priority
    }
}

impl Default for BeliefVector {
    fn default() -> Self {
        Self::new(0.45, 0.30, 0.40, 0.55)
    }
}

impl Add for BeliefVector {
    type Output = Self;
    fn add(self, rhs: Self) -> Self {
        Self::new(
            self.implementation_confidence + rhs.implementation_confidence,
            self.evidence_quality + rhs.evidence_quality,
            self.architectural_risk + rhs.architectural_risk,
            self.security_confidence + rhs.security_confidence,
        )
    }
}

impl AddAssign for BeliefVector {
    fn add_assign(&mut self, rhs: Self) {
        *self = *self + rhs;
    }
}

impl Mul<f32> for BeliefVector {
    type Output = Self;
    fn mul(self, rhs: f32) -> Self {
        Self::new(
            self.implementation_confidence * rhs,
            self.evidence_quality * rhs,
            self.architectural_risk * rhs,
            self.security_confidence * rhs,
        )
    }
}

/// EvidenceGraph – a lightweight, strategy-keyed store of Evidence nodes.
#[derive(Debug, Default, Serialize, Deserialize)]
pub struct EvidenceGraph {
    /// strategy_id → list of Evidence that support / refute it
    by_strategy: HashMap<Uuid, Vec<Evidence>>,

    /// Global chronological log (useful for later dream-cycle / audit)
    chronology: Vec<Uuid>,
}

impl EvidenceGraph {
    pub fn new() -> Self {
        Self::default()
    }

    /// Insert a fully materialised Evidence node.
    pub fn insert(&mut self, evidence: Evidence) {
        if let Some(strategy_id) = evidence.related_strategy_id {
            self.by_strategy.entry(strategy_id).or_default().push(evidence.clone());
        }
        self.chronology.push(evidence.id);
    }

    /// Aggregate all evidence belonging to a strategy into a single BeliefVector.
    pub fn aggregate_belief(&self, strategy_id: Uuid, baseline: BeliefVector) -> BeliefVector {
        let Some(evidences) = self.by_strategy.get(&strategy_id) else {
            return baseline;
        };

        let mut current = baseline;
        let mut max_residual = 0.0_f32;

        for ev in evidences {
            let weight = (0.35 + ev.belief_delta.evidence_quality * 0.55).clamp(0.15, 0.85);
            let next = current.update(&ev.belief_delta, weight);
            max_residual = max_residual.max(next.residual);
            current = next;
        }

        current.residual = max_residual;
        current
    }

    pub fn evidence_count(&self, strategy_id: Uuid) -> usize {
        self.by_strategy.get(&strategy_id).map(|v| v.len()).unwrap_or(0)
    }

    pub fn evidence_for(&self, strategy_id: Uuid) -> &[Evidence] {
        self.by_strategy.get(&strategy_id).map(|v| v.as_slice()).unwrap_or(&[])
    }

    pub fn chronology(&self) -> &[Uuid] {
        &self.chronology
    }
}

/// Central translation layer between the RoboticExecutor and the EvidenceGraph.

pub struct EvidenceManager;

impl EvidenceManager {
    pub fn new() -> Self {
        Self
    }

    /// Synchronous, zero-network path (kept for compatibility).
    pub fn process_observation(&self, obs: Observation, strategy_id: Option<Uuid>) -> Evidence {
        let (kind, failure) = Evidence::classify(&obs.raw, obs.success);
        let belief_delta = self.compute_belief_delta(&obs, kind);
        let mut ev = Evidence::from_observation(obs, kind, belief_delta, strategy_id);
        ev.observation.failure_kind = failure;
        ev
    }

    /// Async path with smart escalation.
    pub async fn process_observation_with_escalation(
        &self,
        obs: Observation,
        strategy_id: Option<Uuid>,
        client: &Client,
        api_key: &str,
    ) -> Evidence {
        let (kind, failure) = Evidence::classify(&obs.raw, obs.success);

        // Fast path
        if !Self::should_escalate(&obs, kind) {
            let belief_delta = self.compute_belief_delta(&obs, kind);
            let mut ev = Evidence::from_observation(obs, kind, belief_delta, strategy_id);
            ev.observation.failure_kind = failure;
            return ev;
        }

        // Slow path
        println!(
            "\x1b[35m>> [EVIDENCE] Ambiguous observation ({} chars) — escalating to light critic\x1b[0m",
            obs.raw.len()
        );

        match EpistemicEngine::evaluate_evidence(
            client,
            api_key,
            "Analyze observation",
            &obs.source,
            &obs.raw,
        )
        .await
        {
            Ok(response) => {
                // Map the 1D float back into a multi-dimensional BeliefVector
                let val = response.confidence_modifier;
                let delta = BeliefVector::new(val * 0.5, val, val * -0.2, val * 0.3);
                let mut ev = Evidence::from_observation(obs, kind, delta, strategy_id);
                ev.observation.failure_kind = failure;
                ev
            }
            Err(e) => {
                println!(
                    "\x1b[33m>> [EVIDENCE] Escalation failed ({}) — falling back to Neutral/ZERO\x1b[0m",
                    e
                );
                let mut ev = Evidence::from_observation(
                    obs,
                    EvidenceKind::Neutral,
                    BeliefVector::ZERO,
                    strategy_id,
                );
                ev.observation.failure_kind = failure;
                ev
            }
        }
    }

    fn should_escalate(obs: &Observation, kind: EvidenceKind) -> bool {
        match kind {
            EvidenceKind::CommandSuccess
            | EvidenceKind::CommandFailure
            | EvidenceKind::SecurityViolation
            | EvidenceKind::FileRead
            | EvidenceKind::FileWrite
            | EvidenceKind::ArchitecturalSmell
            | EvidenceKind::PatternMatch
            | EvidenceKind::PositiveSignal => false,
            EvidenceKind::Neutral => obs.raw.len() > 500,
        }
    }

    fn compute_belief_delta(&self, obs: &Observation, kind: EvidenceKind) -> BeliefVector {
        match kind {
            EvidenceKind::SecurityViolation => BeliefVector::new(-0.15, -0.10, 0.55, -0.85),
            EvidenceKind::CommandSuccess => {
                if obs.source.contains("read") || obs.raw.contains("[READ_OK]") {
                    BeliefVector::new(0.18, 0.42, -0.08, 0.12)
                } else {
                    BeliefVector::new(0.22, 0.28, -0.05, 0.10)
                }
            }
            EvidenceKind::CommandFailure => BeliefVector::new(-0.25, 0.15, 0.20, -0.05),
            EvidenceKind::FileRead => BeliefVector::new(0.12, 0.45, -0.05, 0.08),
            EvidenceKind::FileWrite => BeliefVector::new(0.20, 0.30, 0.10, -0.05),
            EvidenceKind::PatternMatch => BeliefVector::new(0.15, 0.35, -0.05, 0.05),
            EvidenceKind::ArchitecturalSmell => BeliefVector::new(-0.18, 0.20, 0.40, -0.08),
            EvidenceKind::PositiveSignal => BeliefVector::new(0.18, 0.25, -0.08, 0.10),
            EvidenceKind::Neutral => BeliefVector::ZERO,
        }
    }

/// Explicit failure taxonomy to prevent infinite retry loops on security denials.
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum ExecutionFailureKind {
    /// Hard security / policy violation (e.g. PATH_TRAVERSAL_DENIED).
    /// MUST immediately zero out strategy belief and mark path as ineligible.
    PolicyDenial { path: String, reason: String },

    /// Target file/resource missing on disk (e.g., ENOENT os error 2).
    ResourceNotFound { target: String },

    /// Non-zero exit code or tool runtime failure.
    ExecutionError { exit_code: i32, stderr: String },

    /// Lock timeout or transient resource contention.
    TransientTimeout { resource: String },
}

/// Explicit provenance tracking for evidence claims.
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub enum ProvenanceLevel {
    /// Directly verified via tool execution or file contents read from disk.
    Observed { byte_count: usize },

    /// Derived logically from AST symbols or dependency graphs.
    Inferred { basis_symbol: String, confidence: f32 },

    /// Unverified assumption or heuristic generated by model reasoning.
    Hypothesized { rationale: String },
}

impl Default for ProvenanceLevel {
    fn default() -> Self {
        Self::Hypothesized { rationale: "Unverified initial assumption".to_string() }
    }
}

/// Raw observation string coming out of the RoboticExecutor.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Observation {
    pub raw: String,
    pub source: String, // "executor", "grep", "read_file", "podman", etc.
    pub success: bool,
    pub latency_ms: u64,
    pub failure_kind: Option<ExecutionFailureKind>,
}

/// Semantic classification of an observation.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum EvidenceKind {
    CommandSuccess,
    CommandFailure,
    FileRead,
    FileWrite,
    PatternMatch,
    SecurityViolation,
    ArchitecturalSmell,
    PositiveSignal,
    Neutral,
}

/// Fully materialised evidence node that can be inserted into the EvidenceGraph.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Evidence {
    pub id: Uuid,
    pub observation: Observation,
    pub kind: EvidenceKind,
    pub provenance: ProvenanceLevel,
    pub belief_delta: BeliefVector,
    pub timestamp: u64,
    pub related_strategy_id: Option<Uuid>,
}

impl Evidence {
    pub fn from_observation(
        obs: Observation,
        kind: EvidenceKind,
        belief_delta: BeliefVector,
        related_strategy_id: Option<Uuid>,
    ) -> Self {
        let timestamp = SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_secs();

        let provenance = if obs.raw.contains("[READ_OK]") || obs.raw.contains("[CMD_OK]") {
            ProvenanceLevel::Observed { byte_count: obs.raw.len() }
        } else {
            ProvenanceLevel::Hypothesized { rationale: "Derived from executor stdout".to_string() }
        };

        Self {
            id: Uuid::new_v4(),
            observation: obs,
            kind,
            provenance,
            belief_delta,
            timestamp,
            related_strategy_id,
        }
    }

    /// Heuristic classifier with explicit PolicyDenial extraction.
    pub fn classify(raw: &str, success: bool) -> (EvidenceKind, Option<ExecutionFailureKind>) {
        let lower = raw.to_lowercase();

        if lower.contains("sandbox_violation")
            || lower.contains("path_traversal")
            || lower.contains("threat pattern")
        {
            let path = raw.split(':').nth(1).unwrap_or("unknown_path").trim().to_string();
            return (
                EvidenceKind::SecurityViolation,
                Some(ExecutionFailureKind::PolicyDenial {
                    path,
                    reason: "PATH_TRAVERSAL_DENIED".to_string(),
                }),
            );
        }

        if lower.contains("no such file or directory") || lower.contains("os error 2") {
            return (
                EvidenceKind::CommandFailure,
                Some(ExecutionFailureKind::ResourceNotFound { target: raw.to_string() }),
            );
        }

        if lower.contains("pattern_not_found")
            || lower.contains("function_not_found")
            || lower.contains("action_failed")
            || lower.contains("[action failed]")
        {
            return (
                EvidenceKind::CommandFailure,
                Some(ExecutionFailureKind::ExecutionError {
                    exit_code: 1,
                    stderr: raw.to_string(),
                }),
            );
        }

        if lower.contains("[cmd_ok]") || lower.contains("[read_ok]") || lower.contains("[write_ok]")
        {
            return if success {
                (EvidenceKind::CommandSuccess, None)
            } else {
                (EvidenceKind::CommandFailure, None)
            };
        }

        if lower.contains("todo") || lower.contains("placeholder") {
            return (EvidenceKind::ArchitecturalSmell, None);
        }

        if success {
            (EvidenceKind::PositiveSignal, None)
        } else {
            (EvidenceKind::Neutral, None)
        }
    }
}

#[derive(Debug, Clone, PartialEq, serde::Serialize, serde::Deserialize)]
pub enum ExecutiveGoalStatus {

    Suspended,
    Completed,
    Abandoned,
}

#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
pub struct GoalNode {
    pub id: Uuid,
    pub objective: String,
    pub belief_score: f32,
    pub status: ExecutiveGoalStatus,
    pub children: Vec<Uuid>,
}

#[derive(Debug, Clone)]
pub enum ExecutiveCommand {
    InjectGoal { objective: String, parent_id: Option<Uuid> },
    EvaluateEnvironment,
    EmergencyHalt { reason: String },
}

pub struct ExecutiveCortex {
    goals: HashMap<Uuid, GoalNode>,
    active_graph: Option<TaskGraph>,
    memory: Arc<MemoryCortex>,
    command_rx: mpsc::Receiver<ExecutiveCommand>,
}

impl ExecutiveCortex {
    pub fn new(memory: Arc<MemoryCortex>) -> (Self, mpsc::Sender<ExecutiveCommand>) {
        let (tx, rx) = mpsc::channel(100);
        (Self { goals: HashMap::new(), active_graph: None, memory, command_rx: rx }, tx)
    }

    pub async fn run_actor_loop(mut self) {
        // Run a pulse every 10 seconds for persistent state checking
        let mut heartbeat = tokio::time::interval(std::time::Duration::from_secs(10));

        loop {
            tokio::select! {
                Some(cmd) = self.command_rx.recv() => {
                    self.handle_command(cmd).await;
                }
                _ = heartbeat.tick() => {
                    self.cognitive_tick().await;
                }
            }
        }
    }

    async fn cognitive_tick(&mut self) {
        let active_goals: Vec<&GoalNode> =
            self.goals.values().filter(|g| g.status == ExecutiveGoalStatus::Active).collect();

        if active_goals.is_empty() {
            return;
        }

        tracing::info!(
            "[EXECUTIVE CORTEX] Pulse check: Maintaining intent vector for {} active goal(s)...",
            active_goals.len()
        );

        for goal in active_goals {
            let graph_json = serde_json::to_string(&self.active_graph).unwrap_or_default();
            let status_str = format!("{:?}", goal.status);
            let created_at = std::time::SystemTime::now()
                .duration_since(std::time::UNIX_EPOCH)
                .unwrap_or_default()
                .as_secs() as i64;

            // Persist the active intention to SQLite, allowing it to survive reboots
            let _ = self
                .memory
                .upsert_plan_run(
                    &goal.id.to_string(),
                    &goal.objective,
                    &graph_json,
                    &status_str,
                    created_at,
                )
                .await;
        }
    }

    pub async fn evaluate_belief_state(
        &self,
        belief_id: uuid::Uuid,
        confidence: f32,
    ) -> anyhow::Result<crate::belief_policy::BeliefDecision> {
        let decision = crate::belief_policy::BeliefDecisionPolicy::evaluate(belief_id, confidence);

        self.memory.store_belief_decision(decision.clone()).await?;

        Ok(decision)
    }

    pub async fn apply_belief_decision(
        &mut self,
        goal_id: uuid::Uuid,
        decision: crate::belief_policy::BeliefDecision,
    ) -> anyhow::Result<crate::goal_adaptation::GoalAdaptationDecision> {
        let adaptation = crate::goal_adaptation::GoalAdaptationEngine::from_belief_action(
            goal_id,
            decision.action,
        );

        match adaptation.action {
            crate::goal_adaptation::GoalAdaptationAction::PauseGoal => {
                if let Some(goal) = self.goals.get_mut(&goal_id) {
                    goal.status = ExecutiveGoalStatus::Suspended;
                }
            }

            crate::goal_adaptation::GoalAdaptationAction::TerminateGoal => {
                if let Some(goal) = self.goals.get_mut(&goal_id) {
                    goal.status = ExecutiveGoalStatus::Abandoned;
                }
            }

            _ => {}
        }

        self.memory.store_goal_adaptation(adaptation.clone()).await?;

        Ok(adaptation)
    }

    async fn handle_command(&mut self, cmd: ExecutiveCommand) {
        match cmd {
            ExecutiveCommand::InjectGoal { objective, parent_id } => {
                let id = Uuid::new_v4();
                tracing::info!(
                    "\x1b[35m[EXECUTIVE CORTEX] Injecting new background goal: {}\x1b[0m",
                    objective
                );

                let goal = GoalNode {
                    id,
                    objective,
                    belief_score: 0.5, // Seed with standard prior
                    status: ExecutiveGoalStatus::Active,
                    children: vec![],
                };
                self.goals.insert(id, goal);

                if let Some(pid) = parent_id {
                    if let Some(parent) = self.goals.get_mut(&pid) {
                        parent.children.push(id);
                    }
                }
            }
            ExecutiveCommand::EmergencyHalt { reason } => {
                tracing::error!(
                    "\x1b[31m[EXECUTIVE CORTEX] GLOBAL HALT TRIGGERED: {}\x1b[0m",
                    reason
                );
                self.goals.values_mut().for_each(|g| g.status = ExecutiveGoalStatus::Suspended);
            }
            ExecutiveCommand::EvaluateEnvironment => {
                tracing::info!("[EXECUTIVE CORTEX] Environment evaluation requested.");
            }
        }
    }
}

fn canonicalize_or_parent(p: &Path) -> Result<PathBuf> {
    if let Ok(c) = std::fs::canonicalize(p) {
        return Ok(c);
    }
    let parent = p.parent().ok_or_else(|| anyhow!("no parent directory for {}", p.display()))?;
    let parent_canon = std::fs::canonicalize(parent)?;
    let file_name = p.file_name().ok_or_else(|| anyhow!("no file name for {}", p.display()))?;
    Ok(parent_canon.join(file_name))
}

pub fn sanitize(path: &str) -> Result<PathBuf> {
    if path.starts_with("/tmp/") {
        return Ok(Path::new(path).to_path_buf());
    }

    let workspace_root = std::env::current_dir()
        .and_then(std::fs::canonicalize)
        .map_err(|e| anyhow!("PATH_TRAVERSAL_DENIED: could not resolve workspace root ({})", e))?;

    let mut normalized_path = path.to_string();

    if let Some(ws_str) = workspace_root.to_str() {
        if normalized_path.starts_with(ws_str) {
            normalized_path =
                normalized_path.trim_start_matches(ws_str).trim_start_matches('/').to_string();
        }
    }

    if normalized_path.starts_with("src/") {
        normalized_path = format!("crates/runtime_core/{}", normalized_path);
    } else if normalized_path.starts_with("runtime_core/")
        || normalized_path.starts_with("nexus_substrate/")
        || normalized_path.starts_with("nexus_model_hub/")
        || normalized_path.starts_with("nexus_search/")
        || normalized_path.starts_with("memory_graph/")
        || normalized_path.starts_with("protocol/")
        || normalized_path.starts_with("executive_cortex/")
    {
        normalized_path = format!("crates/{}", normalized_path);
    } else if normalized_path.starts_with("ide_backend/")
        || normalized_path.starts_with("cli/")
        || normalized_path.starts_with("node/")
    {
        normalized_path = format!("bins/{}", normalized_path);
    }

    let p = Path::new(&normalized_path);
    let target = if p.is_absolute() { p.to_path_buf() } else { workspace_root.join(p) };

    if !target.exists() {
        let crate_fallback = workspace_root.join("crates/runtime_core/src").join(p);
        if crate_fallback.exists() {
            return Ok(crate_fallback);
        }

        if let Some(filename) = p.file_name().and_then(|f| f.to_str()) {
            let lower = filename.to_lowercase();
            if let Ok(entries) = std::fs::read_dir(&workspace_root) {
                for entry in entries.flatten() {
                    if entry.file_name().to_string_lossy().to_lowercase() == lower {
                        return Ok(entry.path());
                    }
                }
            }
        }

        let crates_fallback = workspace_root.join("crates");
        if crates_fallback.exists() {
            return Ok(crates_fallback);
        }
    }

    let canonical =
        canonicalize_or_parent(&target).map_err(|_| anyhow!("PATH_TRAVERSAL_DENIED: {}", path))?;

    if !canonical.starts_with(&workspace_root) {
        return Err(anyhow!("PATH_TRAVERSAL_DENIED: {} resolves outside the workspace root", path));
    }

    Ok(canonical)
}

impl RoboticExecutor {
    pub fn new(api_key: String, tx: tokio::sync::broadcast::Sender<String>) -> Self {
        Self {
            current_state: FsmState::Idle,
            api_key,
            tx,
            action_cache: Arc::new(tokio::sync::Mutex::new(
                crate::memory_cortex::ActionCache::new(),
            )),
        }
    }

    pub async fn execute_motor_command(
        &mut self,
        command: MotorCommand,
        memory: Arc<MemoryCortex>,
        vmo_state: Arc<Mutex<VmoState>>,
        dist: Arc<DistributedScheduler>,
    ) -> Result<String> {
        self.execute_single_action(command.action, memory, vmo_state, dist).await
    }

    async fn execute_single_action(
        &mut self,
        action: MotorAction,
        _memory: Arc<MemoryCortex>,
        _vmo_state: Arc<Mutex<VmoState>>,
        _dist: Arc<DistributedScheduler>,
    ) -> Result<String> {
        let result: Result<String> = async {
            match action {
                MotorAction::ExecuteCommand { command } => {
                    let cache_key = crate::memory_cortex::ActionCache::generate_key(&command, "v7_workspace_ast");
                    {
                        let cache = self.action_cache.lock().await;
                        if let Some(cached) = cache.get(&cache_key) {
                            return Ok(format!("[CACHE_HIT] (Cached execution)\n{}", cached.output));
                        }
                    }

                    let out = crate::podman::PodmanShield::execute(&command)?;
                    let stdout = String::from_utf8_lossy(&out.stdout).into_owned();
                    let stderr = String::from_utf8_lossy(&out.stderr).into_owned();

                    let mut stdout_trunc = stdout;
                    let mut stderr_trunc = stderr;
                    if stdout_trunc.len() > 8000 {
                        stdout_trunc.truncate(8000);
                        stdout_trunc.push_str("\n...[STDOUT TRUNCATED: OUTPUT TOO LARGE]");
                    }
                    if stderr_trunc.len() > 8000 {
                        stderr_trunc.truncate(8000);
                        stderr_trunc.push_str("\n...[STDERR TRUNCATED: OUTPUT TOO LARGE]");
                    }

                    let mut res = format!("[CMD_OK] Status: {}\n", out.status);
                    if !stdout_trunc.trim().is_empty() { res.push_str(&format!("STDOUT:\n{}\n", stdout_trunc)); }
                    if !stderr_trunc.trim().is_empty() { res.push_str(&format!("STDERR:\n{}\n", stderr_trunc)); }
                    if stdout_trunc.trim().is_empty() && stderr_trunc.trim().is_empty() { res.push_str("(Command produced no output)\n"); }

                    {
                        let mut cache = self.action_cache.lock().await;
                        cache.insert(cache_key, res.clone(), out.status.code().unwrap_or(0));
                    }

                    Ok(res)
                }
                MotorAction::WriteFile { path, content } => {
                    let safe_path = sanitize(&path)?;
                    if let Ok(existing) = tokio::fs::read_to_string(&safe_path).await {
                        if existing == content {
                            return Ok(format!("[WRITE_NOOP] {} -- content identical to existing file", path));
                        }
                    }
                    if let Some(parent) = safe_path.parent() {
                        let _ = tokio::fs::create_dir_all(parent).await;
                    }
                    tokio::fs::write(&safe_path, content).await?;
                    Ok(format!("[WRITE_OK] {}", path))
                }
                MotorAction::ReadFile { path } => {
                    let safe_path = sanitize(&path)?;
                    let content = tokio::fs::read_to_string(&safe_path).await?;
                    let snippet = if content.len() > 8000 { &content[..8000] } else { &content };
                    let mem = _memory;
                    if let Ok(emb) = nexus_substrate::vector_bridge::get_embedding(&self.api_key, snippet).await {
                        let _ = mem.store_memory("ReadFile", &path, &content, Some(emb)).await;
                    } else {
                        let _ = mem.store_memory("ReadFile", &path, &content, None).await;
                    }

                    let return_content = if content.len() > 12000 {
                        format!("{}...\n\n[CONTENT TRUNCATED. FULL DATA SAVED TO VECTOR VAULT.]", &content[..12000])
                    } else {
                        content.clone()
                    };

                    Ok(format!("[READ_OK] {}\nCONTENT PREVIEW:\n{}", path, return_content))
                }
                MotorAction::ReadFileChunk { path, start_line, max_lines } => {
                    let safe_path = sanitize(&path)?;
                    let content = tokio::fs::read_to_string(&safe_path).await?;
                    let lines: Vec<&str> = content.lines().collect();
                    let total_lines = lines.len();

                    if start_line >= total_lines {
                        return Ok(format!("[READ_CHUNK_EOF] Reached end of file {}. Total lines: {}", path, total_lines));
                    }

                    let end_line = std::cmp::min(start_line + max_lines, total_lines);
                    let chunk = lines[start_line..end_line].join("\n");

                    Ok(format!(
                        "[READ_CHUNK_OK] File: {} | Lines {}-{} of {}\nCONTENT:\n{}",
                        path, start_line, end_line, total_lines, chunk
                    ))
                }
                MotorAction::SearchCode { path, query } => {
                    let safe_path = sanitize(&path).unwrap_or_else(|_| PathBuf::from("crates"));
                    let p_str = safe_path.to_str().unwrap_or(&path);

                    let cmd = format!(
                        "(rg -n -C 2 -i '{}' {} || grep -rn -C 2 --exclude-dir={{.git,target,.surgical_backup,.nexus,.lazarus,node_modules}} -iE '{}' {}) 2>/dev/null || true",
                        query, p_str, query, p_str
                    );
                    let out = crate::podman::PodmanShield::execute(&cmd)?;
                    let mut stdout = String::from_utf8_lossy(&out.stdout).to_string();

                    // ⚡ AUTOMATIC WORKSPACE FALLBACK: If subpath returns 0 matches and query was for cortex/router, search crates/runtime_core/src/
                    if stdout.trim().is_empty() && (p_str.contains("executive_cortex") || query.contains("router")) {
                        let fallback_cmd = format!(
                            "(rg -n -C 2 -i '{}' crates/runtime_core/src/ || grep -rn -C 2 --exclude-dir={{.git,target,.surgical_backup,.nexus,.lazarus,node_modules}} -iE '{}' crates/runtime_core/src/) 2>/dev/null || true",
                            query, query
                        );
                        if let Ok(fb_out) = crate::podman::PodmanShield::execute(&fallback_cmd) {
                            let fb_stdout = String::from_utf8_lossy(&fb_out.stdout);
                            if !fb_stdout.trim().is_empty() {
                                stdout = format!("[FALLBACK MATCHES FROM crates/runtime_core/src/]:\n{}", fb_stdout);
                            }
                        }
                    }

                    let trunc = if stdout.len() > 8000 {
                        format!("{}...\n[MATCHES TRUNCATED: OUTPUT TOO LARGE]", &stdout[..8000])
                    } else {
                        stdout
                    };

                    if trunc.trim().is_empty() {
                        Ok(format!("[SEARCH_CODE_OK] Path: {} | Query: '{}'\nNo code matches found in specified path.", path, query))
                    } else {
                        Ok(format!("[SEARCH_CODE_OK] Path: {} | Query: '{}'\nMATCHES:\n{}", path, query, trunc))
                    }
                }
                MotorAction::DiffPatch { path, search, replace } => {
                    let safe_path = sanitize(&path)?;
                    let original = tokio::fs::read_to_string(&safe_path).await?;
                    let patched = crate::diff::DiffEngine::apply(&original, &search, &replace)?;
                    tokio::fs::write(&safe_path, patched).await?;
                    Ok(format!("[DIFF_PATCH_OK] Applied changes to {}", path))
                }
                MotorAction::PatchFunction { path, function_name, new_body } => {
                    let safe_path = sanitize(&path)?;
                    let patched = crate::ast_editor::AstEditor::update_function(safe_path.to_str().unwrap_or(&path), &function_name, &new_body)?;
                    tokio::fs::write(&safe_path, patched).await?;
                    Ok(format!("[PATCH_FUNCTION_OK] Updated function '{}' in {}", function_name, path))
                }
                MotorAction::SearchMemory { query } => {
                    let mem = _memory;
                    let mut results_text = String::new();

                    if let Ok(emb) = nexus_substrate::vector_bridge::get_embedding(&self.api_key, &query).await {
                        if let Ok(results) = mem.semantic_search(&emb).await {
                            for r in results {
                                let snippet = if r.result.len() > 3000 { &r.result[..3000] } else { &r.result };
                                results_text.push_str(&format!("Source: {}\nData: {}...\n---\n", r.directive, snippet));
                            }
                        }
                    }

                    if results_text.is_empty() {
                        Ok(format!("[SEARCH_OK] Native FastEmbed vector recall queried for: '{}'. VectorVault active.", query))
                    } else {
                        Ok(format!("[SEARCH_OK] Found semantic context:\n{}", results_text))
                    }
                }
                MotorAction::Halt { reasoning } => Ok(format!("[HALT] {}", reasoning)),
            }
        }.await;

        result
    }
}

use crate::wasm_runtime::WasmRuntime;
use anyhow::Result;
use std::time::Instant;
use sysinfo::{CpuRefreshKind, MemoryRefreshKind, RefreshKind, System};

#[derive(Debug, Clone)]
pub struct FitnessReport {
    pub execution_time_ms: u128,
    pub memory_delta_kb: f32,
    pub thermal_impact: f32,
    pub composite_score: f32,
}

pub struct FitnessEvaluator;

impl FitnessEvaluator {
    pub fn evaluate_wasm_cell(
        wasm_bytes: &[u8],
        func_name: &str,
        payload: &[u8],
    ) -> Result<FitnessReport> {
        let mut sys = System::new_with_specifics(
            RefreshKind::new()
                .with_cpu(CpuRefreshKind::everything())
                .with_memory(MemoryRefreshKind::everything()),
        );
        sys.refresh_all();

        // 1. Capture ARM64 Baseline Metrics
        let mem_before = sys.used_memory() as f32 / 1024.0;
        let temp_before = std::fs::read_to_string("/sys/class/thermal/thermal_zone0/temp")
            .unwrap_or_else(|_| "45000".to_string())
            .trim()
            .parse::<f32>()
            .unwrap_or(45000.0)
            / 1000.0;

        // 2. Execute the Generated WASM Payload
        let start = Instant::now();
        let _result = WasmRuntime::run_with_buffer(wasm_bytes, func_name, payload)?;
        let duration = start.elapsed().as_millis();

        // 3. Capture Post-Execution Metrics
        sys.refresh_all();
        let mem_after = sys.used_memory() as f32 / 1024.0;
        let temp_after = std::fs::read_to_string("/sys/class/thermal/thermal_zone0/temp")
            .unwrap_or_else(|_| "45000".to_string())
            .trim()
            .parse::<f32>()
            .unwrap_or(45000.0)
            / 1000.0;

        // Ensure we don't divide by zero
        let memory_delta = (mem_after - mem_before).max(0.1);
        let thermal_delta = (temp_after - temp_before).max(0.1);

        // 4. Calculate Mathematical Fitness Score (Higher is Better)
        // Inverse relations: Lower latency/memory/heat yields a higher score
        // Weights: 50% Speed, 30% Memory Efficiency, 20% Thermal Stability
        let speed_score = 1000.0 / (duration as f32 + 1.0);
        let mem_score = 1000.0 / memory_delta;
        let heat_score = 1000.0 / thermal_delta;

        let composite_score = (speed_score * 0.5) + (mem_score * 0.3) + (heat_score * 0.2);

        Ok(FitnessReport {
            execution_time_ms: duration,
            memory_delta_kb: memory_delta,
            thermal_impact: thermal_delta,
            composite_score,
        })
    }
}

    pub goal_id: Uuid,

    pub action: GoalAdaptationAction,

    pub reason: String,

    pub timestamp: i64,

        goal_id: Uuid,
        action: crate::belief_policy::BeliefAction,
    ) -> GoalAdaptationDecision {

            crate::belief_policy::BeliefAction::Continue => {

                GoalAdaptationAction::ModifyStrategy,

                (GoalAdaptationAction::TerminateGoal, "Evidence no longer supports objective")
            }
        };

        GoalAdaptationDecision {
            id: Uuid::new_v4(),

            goal_id,

            action: adaptation,

            reason: reason.to_string(),

            timestamp: chrono::Utc::now().timestamp(),
        }
    }
}

use crate::agent_protocol::AgentAction;
use serde::{Deserialize, Serialize};
use std::collections::hash_map::DefaultHasher;
use std::hash::{Hash, Hasher};
use std::sync::{Arc, LazyLock};
use std::time::{SystemTime, UNIX_EPOCH};
use tokio::sync::Mutex;
use tracing::info;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct MerkleLeaf {
    pub action: AgentAction,
    pub actor_id: String,
    pub timestamp: u64,
    pub hash: u64,
}

impl MerkleLeaf {
    pub fn new(action: AgentAction, actor_id: &str) -> Self {
        let ts = SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_secs();
        let mut hasher = DefaultHasher::new();
        serde_json::to_vec(&action).unwrap_or_default().hash(&mut hasher);
        actor_id.hash(&mut hasher);
        ts.hash(&mut hasher);
        Self { action, actor_id: actor_id.to_string(), timestamp: ts, hash: hasher.finish() }
    }
}

#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct GovernanceLog {
    pub history: Vec<MerkleLeaf>,
    pub root_hash: u64,
}

impl GovernanceLog {
    pub fn init() -> Self {
        Self::default()
    }

    pub fn init_shared() -> Arc<Mutex<Self>> {
        Arc::new(Mutex::new(Self::init()))
    }

    /// record: Chains a new leaf into the Merkle root.
    /// Root = Hash(OldRoot ++ NewLeaf)
    pub fn record(&mut self, action: AgentAction, actor_id: &str) {
        let leaf = MerkleLeaf::new(action, actor_id);

        let mut hasher = DefaultHasher::new();
        self.root_hash.hash(&mut hasher);
        leaf.hash.hash(&mut hasher);
        self.root_hash = hasher.finish();

        self.history.push(leaf);

        if self.history.len().is_multiple_of(10) {
            info!(
                "[34m[GOVERNANCE] Checkpoint: Root=0x{:x} (Size: {})[0m",
                self.root_hash,
                self.history.len()
            );
        }
    }

    pub fn root(&self) -> u64 {
        self.root_hash
    }

    pub fn len(&self) -> usize {
        self.history.len()
    }

    pub fn is_empty(&self) -> bool {
        self.history.is_empty()
    }
}

// Process-wide persistent governance log. Callers should use this
// instead of constructing their own GovernanceLog, or records get
// discarded along with the throwaway instance.
pub static GOVERNANCE: LazyLock<Mutex<GovernanceLog>> =
    LazyLock::new(|| Mutex::new(GovernanceLog::init()));

             1. Analyze the fault.\n\
             2. Generate exactly 3 distinct hypotheses for WHY it failed.\n\

             4. Select the highest probability hypothesis as the 'selected_path'.\n\
             Output ONLY valid JSON matching this schema:\n\

                {
                    "role": "user",
                    "parts": [{ "text": format!("OBJECTIVE:\n{}\n\nFAULT TRACE:\n{}", objective, fault_log) }]
                }
            ],
            "generationConfig": {
                "temperature": 0.3
            }
        });

        let base_url = std::env::var("NEXUS_API_GATEWAY")
            .unwrap_or_else(|_| "https://generativelanguage.googleapis.com".to_string());

        let url =
            format!("{}/v1beta/models/gemma-4-31b-it:generateContent?key={}", base_url, api_key);

        let res = client.post(&url).json(&payload).send().await?;
        let body: serde_json::Value = res.json().await?;

        if let Some(err) = body.pointer("/error/message") {
            return Err(anyhow!("GoT API Error: {}", err));
        }

        let text = body
            .pointer("/candidates/0/content/parts/0/text")
            .and_then(|v| v.as_str())
            .unwrap_or("{}")
            .replace("```json", "")
            .replace("```", "");

        let resolution: GoTResolution =
            serde_json::from_str(text.trim()).map_err(|e| anyhow!("GoT Schema Fracture: {}", e))?;

        Ok(resolution)
    }
}

use crate::agent_protocol::AgentAction;
use anyhow::Result;

/// Public shared JSON string escape sanitizer.
pub fn sanitize_json_escapes(raw: &str) -> String {
    let mut out = String::with_capacity(raw.len());
    let mut in_string = false;
    let mut chars = raw.chars().peekable();

    while let Some(c) = chars.next() {
        match c {
            '"' => {
                in_string = !in_string;
                out.push('"');
            }
            '\n' if in_string => {
                out.push_str("\\n");
            }
            '\r' if in_string => {
                out.push_str("\\r");
            }
            '\\' => {
                if let Some(&next) = chars.peek() {
                    if next == '"'
                        || next == '\\'
                        || next == '/'
                        || next == 'b'
                        || next == 'f'
                        || next == 'n'
                        || next == 'r'
                        || next == 't'
                        || next == 'u'
                    {
                        out.push('\\');
                    } else {
                        out.push_str("\\\\");
                    }
                } else {
                    out.push_str("\\\\");
                }
            }
            _ => out.push(c),
        }
    }
    out
}

pub fn extract_actions(response: &str) -> Result<Vec<AgentAction>> {
    let text = response.trim();

    if text.is_empty() || text == "[]" {
        return Ok(vec![]);
    }

    let trimmed = text.replace("```json", "").replace("```", "");
    let clean_text = sanitize_json_escapes(trimmed.trim());

    if let Ok(actions) = serde_json::from_str::<Vec<AgentAction>>(&clean_text) {
        return Ok(actions);
    }

    if let Ok(single) = serde_json::from_str::<AgentAction>(&clean_text) {
        return Ok(vec![single]);
    }

    if let (Some(start), Some(end)) = (clean_text.find('['), clean_text.rfind(']')) {
        if start < end {
            let slice = &clean_text[start..=end];
            if let Ok(actions) = serde_json::from_str::<Vec<AgentAction>>(slice) {
                return Ok(actions);
            }
        }
    }

    if let (Some(start), Some(end)) = (clean_text.find('{'), clean_text.rfind('}')) {
        if start < end {
            let slice = &clean_text[start..=end];
            if let Ok(single) = serde_json::from_str::<AgentAction>(slice) {
                return Ok(vec![single]);
            }
        }
    }

    tracing::warn!("[JSON PARSER] Could not parse action payload. Falling back safely.");
    Ok(vec![])
}

use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Snapshot {
    pub id: String,
    pub path: String,
    pub hash: String,
    pub directive: String,
    pub timestamp: String,
}

const CATALOG: &str = ".nexus_snapshots.json";

pub async fn list_snapshots() -> Result<Vec<Snapshot>> {
    if !Path::new(CATALOG).exists() {
        return Ok(vec![]);
    }
    let raw = fs::read_to_string(CATALOG).await?;
    let snapshots: Vec<Snapshot> = serde_json::from_str(&raw)?;
    Ok(snapshots)
}

pub async fn take_snapshot(path: &str, directive: &str) -> Result<Snapshot> {
    let content = fs::read(path).await?;
    let hash = fnv_hex(&content);
    let snap_path = format!("{}.{}.snap", path, &hash[..8]);
    fs::write(&snap_path, &content).await?;

    let snap = Snapshot {
        id: uuid_v4(),
        path: path.to_string(),
        hash: hash.clone(),
        directive: directive.to_string(),
        timestamp: epoch_secs().to_string(),
    };

    let mut catalog = list_snapshots().await.unwrap_or_default();
    catalog.push(snap.clone());
    if catalog.len() > 200 {
        catalog.remove(0);
    }
    fs::write(CATALOG, serde_json::to_string_pretty(&catalog)?).await?;
    Ok(snap)
}

fn fnv_hex(data: &[u8]) -> String {
    let mut h: u64 = 14695981039346656037;
    for &b in data {
        h ^= b as u64;
        h = h.wrapping_mul(1099511628211);
    }
    format!("{:016x}", h)
}

fn epoch_secs() -> u64 {
    std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).unwrap_or_default().as_secs()
}

fn uuid_v4() -> String {
    format!(
        "{:x}",
        std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).unwrap().as_nanos()
    )
}

use serde::de::DeserializeOwned;

pub fn safe_parse_json<T: DeserializeOwned>(raw: &str) -> Result<T, serde_json::Error> {
    // 1. Strip rogue Markdown formatting often hallucinated by models
    let mut s = raw.trim();
    if s.starts_with("```json") {
        s = s.trim_start_matches("```json").trim_end_matches("```").trim();
    } else if s.starts_with("```") {
        s = s.trim_start_matches("```").trim_end_matches("```").trim();
    }

    // 2. Scan and repair invalid escape sequences (e.g., regex inside JSON strings)
    let mut fixed = String::with_capacity(s.len() + 10);
    let mut chars = s.chars().peekable();
    
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(&next_c) = chars.peek() {
                match next_c {
                    'n' | 't' | 'r' | '"' | '\\' | '/' | 'u' | 'b' | 'f' => {
                        // Valid JSON escape sequence
                        fixed.push('\\');
                    },
                    _ => {
                        // Invalid escape sequence detected. Double-escape it to protect the parser.
                        fixed.push('\\');
                        fixed.push('\\'); 
                    }
                }
            } else {
                // Trailing backslash
                fixed.push('\\');
                fixed.push('\\');
            }
        } else {
            fixed.push(c);
        }
    }
    
    // 3. Attempt to parse the repaired string, fallback to original if structural failure
    serde_json::from_str(&fixed).or_else(|_| serde_json::from_str(s))
}

use std::collections::HashMap;
use tracing::warn;

pub struct LogicEngine;

impl LogicEngine {
    pub fn linearize_handoff(
        action: &crate::agent_protocol::AgentAction,
        result: crate::agent_protocol::SubagentReturn,
    ) -> String {
        format!(
            "[SYSTEM: DELEGATION REPORT]\nAction: {:?}\nStatus: {}\nOutput: {}\n[DIRECTIVE]: Matriarch, resume execution based on this report.",
            action, result.status, result.output
        )
    }

    pub fn is_action_safe(directive: &str) -> bool {
        let forbidden =
            ["rm -rf /", "mkfs", "shutdown", "chmod 777", "dd if=/dev/zero", "> /dev/sda"];
        let d_lower = directive.to_lowercase();
        !forbidden.iter().any(|&p| d_lower.contains(p))
    }

    pub fn detect_deadlock(observations: &[String]) -> bool {
        if observations.len() < 3 {
            return false;
        }

        let mut read_counts: HashMap<String, usize> = HashMap::new();

        for obs in observations {
            if let Some(after) = obs.strip_prefix("[READ_OK] ") {
                let path = after.split_whitespace().next().unwrap_or("").to_string();
                if !path.is_empty() {
                    *read_counts.entry(path).or_insert(0) += 1;
                }
            }
        }

        for (path, count) in read_counts {
            if count >= 3 {
                let file_progress = observations.iter().any(|obs| {
                    (obs.contains("[WRITE_OK]")
                        || obs.contains("[DIFF_PATCH_OK]")
                        || obs.contains("[PATCH_OK]"))
                        && obs.contains(&path)
                });

                if !file_progress {
                    warn!(
                        "[LOGIC] Localized Deadlock: {} read {} times with zero progress.",
                        path, count
                    );
                    return true;
                }
            }
        }

        let mut last_cmd_sig = String::new();
        let mut cmd_repeat_count = 0;

        for obs in observations {
            if let Some(start) = obs.find("[CMD:") {
                let sig: String = obs[start..]
                    .lines()
                    .filter(|line| !line.contains("Finished") && !line.contains("Blocking waiting"))
                    .collect::<Vec<&str>>()
                    .join("\n");

                if sig == last_cmd_sig && !sig.is_empty() {
                    cmd_repeat_count += 1;
                } else {
                    last_cmd_sig = sig;
                    cmd_repeat_count = 1;
                }
            }
        }

        if cmd_repeat_count >= 2 {
            warn!("[LOGIC] Deadlock: Time-invariant command repetition detected.");
            return true;
        }
        false
    }
}

/// Calculates the nth Fibonacci number using checked arithmetic to prevent overflow.
/// Returns None if the result exceeds the capacity of u128.
pub fn fibonacci(n: u32) -> Option<u128> {
    if n == 0 {
        return Some(0);
    }
    if n == 1 {
        return Some(1);
    }

    let mut a: u128 = 0;
    let mut b: u128 = 1;

    for _ in 2..=n {
        let next = a.checked_add(b)?;
        a = b;
        b = next;
    }

    Some(b)
}

pub enum MemoryCommand {

    Load { key: String, respond_to: oneshot::Sender<Option<Vec<u8>>> },
}

pub struct MemoryActor {
    rx: mpsc::Receiver<MemoryCommand>,
}

impl MemoryActor {
    pub fn new(rx: mpsc::Receiver<MemoryCommand>) -> Self {
        Self { rx }
    }

    pub async fn run(mut self) {
        let mut db = HashMap::new();
        while let Some(cmd) = self.rx.recv().await {
            match cmd {
                MemoryCommand::Store { key, value } => {
                    db.insert(key, value);
                }
                MemoryCommand::Load { key, respond_to } => {
                    let _ = respond_to.send(db.get(&key).cloned());
                }
            }
        }
    }
}

pub struct MemoryConsolidator;

impl MemoryConsolidator {
    pub fn new() -> Self {
        Self
    }

    pub fn calculate_retention(memory: &CognitiveMemory) -> f32 {
        let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs() as i64;

        let age = (now - memory.created_at).max(0) as f32;

        let decay = (-age / 86400.0).exp();

        let score = memory.importance * memory.confidence * decay;

        score.clamp(0.0, 1.0)
    }

    pub fn should_consolidate(memory: &CognitiveMemory) -> bool {
        match memory.kind {
            MemoryKind::Failure | MemoryKind::Reflection | MemoryKind::Goal => true,

            _ => Self::calculate_retention(memory) > 0.5,
        }
    }
}

impl Default for MemoryConsolidator {
    fn default() -> Self {
        Self::new()
    }
}

    pub const fn new() -> Self {
        Self {
            tasks_total: AtomicU64::new(0),
            task_failures: AtomicU64::new(0),
            arbiter_blocks: AtomicU64::new(0),
            heal_successes: AtomicU64::new(0),
            heal_failures: AtomicU64::new(0),
            api_latency_ms_sum: AtomicU64::new(0),
            api_calls: AtomicU64::new(0),
            circuit_opens: AtomicU64::new(0),
        }
    }

    pub fn inc_task(&self) {
        tracing::info!("Task counter incremented");
        self.tasks_total.fetch_add(1, Ordering::Relaxed);
    }

    pub fn inc_failure(&self) {
        self.task_failures.fetch_add(1, Ordering::Relaxed);
    }

    pub fn inc_arbiter_block(&self) {
        self.arbiter_blocks.fetch_add(1, Ordering::Relaxed);
    }

    pub fn inc_heal_success(&self) {
        self.heal_successes.fetch_add(1, Ordering::Relaxed);
    }

    pub fn inc_heal_failure(&self) {
        self.heal_failures.fetch_add(1, Ordering::Relaxed);
    }

    pub fn inc_circuit_open(&self) {
        self.circuit_opens.fetch_add(1, Ordering::Relaxed);
    }

    pub fn record_api_call(&self, latency_ms: u128) {
        self.api_calls.fetch_add(1, Ordering::Relaxed);
        self.api_latency_ms_sum.fetch_add(latency_ms as u64, Ordering::Relaxed);
    }

    pub fn avg_latency_ms(&self) -> u64 {
        let calls = self.api_calls.load(Ordering::Relaxed);
        self.api_latency_ms_sum.load(Ordering::Relaxed).checked_div(calls).unwrap_or(0)
    }
}

        .duration_since(std::time::UNIX_EPOCH)

        GEMINI_DISABLED_UNTIL.store(now + cooldown, Ordering::Relaxed);
        tracing::warn!("[CIRCUIT BREAKER] Gemini/Gemma rate-limited (429). Bypassing for 60s.");
    } else if model_name.contains("nvidia") || model_name.contains("nemotron") {
        NVIDIA_DISABLED_UNTIL.store(now + cooldown, Ordering::Relaxed);
        tracing::warn!("[CIRCUIT BREAKER] NVIDIA rate-limited (429). Bypassing for 60s.");
    }
}

use crate::agent_protocol::AgentAction;
use crate::intent_contract::IntentContract;
use crate::policy::Capability;
use serde::{Deserialize, Serialize};
use uuid::Uuid;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum MotorAction {
    ExecuteCommand { command: String },
    WriteFile { path: String, content: String },
    ReadFile { path: String },
    ReadFileChunk { path: String, start_line: usize, max_lines: usize },
    SearchCode { path: String, query: String },
    SearchMemory { query: String },
    DiffPatch { path: String, search: String, replace: String },
    PatchFunction { path: String, function_name: String, new_body: String },
    Halt { reasoning: String },
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct MotorCommand {
    pub command_id: Uuid,
    pub intent_id: Uuid,
    pub goal_id: Option<Uuid>,
    pub action: MotorAction,
    pub capability: Capability,
    pub expected_outcome: String,
    pub created_at: u64,
}

pub struct MotorCortex;

impl MotorCortex {
    pub fn translate(
        intent: &IntentContract,
        goal_id: Option<Uuid>,
        action: AgentAction,
        capability: Capability,
        expected_outcome: String,
    ) -> MotorCommand {
        let motor_action = match action {
            AgentAction::ExecuteCommand { command, .. } => MotorAction::ExecuteCommand { command },
            AgentAction::WriteFile { path, content, .. } => MotorAction::WriteFile { path, content },
            AgentAction::ReadFile { path, .. } => MotorAction::ReadFile { path },
            AgentAction::ReadFileChunk { path, start_line, max_lines, .. } => MotorAction::ReadFileChunk { path, start_line, max_lines },
            AgentAction::SearchCode { path, query, .. } => MotorAction::SearchCode { path, query },
            AgentAction::SearchMemory { query, .. } => MotorAction::SearchMemory { query },
            AgentAction::DiffPatch { path, search, replace, .. } => MotorAction::DiffPatch { path, search, replace },
            AgentAction::PatchFunction { path, function_name, new_body, .. } => MotorAction::PatchFunction { path, function_name, new_body },
            AgentAction::Halt { reasoning } => MotorAction::Halt { reasoning },
            _ => MotorAction::Halt { reasoning: "Fallback halt for unmapped action".to_string() },
        };

        let now = std::time::SystemTime::now()
            .duration_since(std::time::UNIX_EPOCH)
            .unwrap_or_default()
            .as_secs();

        MotorCommand {
            command_id: Uuid::new_v4(),
            intent_id: intent.id,
            goal_id,
            action: motor_action,
            capability,
            expected_outcome,
            created_at: now,
        }
    }
}

use nexus_executive_cortex::schema::{Observation, ObservationEnvelope};
use std::time::{SystemTime, UNIX_EPOCH};
use uuid::Uuid;

/// Translates raw, ad-hoc runtime events into pure, timestamped Observations.
pub struct ObservationAdapter;

impl ObservationAdapter {
    /// Core wrapping function to ensure all observations are uniformly timestamped and ID'd.
    pub fn wrap(observation: Observation) -> ObservationEnvelope {
        let timestamp = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .unwrap_or_default()
            .as_secs();

        ObservationEnvelope {
            id: Uuid::new_v4(),
            timestamp,
            observation,
        }
    }

    /// Translation: User inputs a directive via CLI or Web UI
    pub fn from_human_directive(directive: &str) -> ObservationEnvelope {
        Self::wrap(Observation::HumanDirective {
            directive: directive.to_string(),
        })
    }

    /// Translation: The LLM planner exhausts its retry quota
    pub fn from_planner_timeout() -> ObservationEnvelope {
        Self::wrap(Observation::PlannerTimeout)
    }

    /// Translation: A MicroAgent finishes its assigned task
    pub fn from_task_completion(task_id: &str, output: &str) -> ObservationEnvelope {
        Self::wrap(Observation::TaskCompleted {
            task_id: task_id.to_string(),
            output: output.to_string(),
        })
    }

    /// Translation: An agent hits a fatal error
    pub fn from_goal_failure(goal_id: Uuid, reason: &str) -> ObservationEnvelope {
        Self::wrap(Observation::GoalFailed {
            goal_id,
            reason: reason.to_string(),
        })
    }

    /// Translation: Filtered health spikes from the thermal/system monitor
    pub fn from_health_spike(cpu: f32, ram: f32) -> ObservationEnvelope {
        Self::wrap(Observation::HealthChanged { cpu, ram })
    }
}

            let (dummy_tx, _) = broadcast::channel(1);

        });

        match agent.prepare_and_run(client, api_key, &prompt, active_ctx).await {

            client,
            api_key,

            &payload,
        )
        .await;

        match text_res {
            Ok(text) => match Self::parse_task_graph(&text, directive) {
                Ok(graph) if !graph.nodes.is_empty() => Ok(graph),

            },
            Err(_) => Ok(Self::create_fallback_graph(directive)),

    fn parse_task_graph(text: &str, directive: &str) -> Result<TaskGraph> {
        let clean = text.replace("```json", "").replace("```", "").trim().to_string();

        if let Ok(mut graph) = serde_json::from_str::<TaskGraph>(&clean) {
            Self::strip_terminal_nodes(&mut graph);
            if !graph.nodes.is_empty() {
                return Ok(graph);
            }
        }

        if let (Some(start), Some(end)) = (clean.find('{'), clean.rfind('}')) {
            if start <= end {
                let slice = &clean[start..=end];
                if let Ok(mut graph) = serde_json::from_str::<TaskGraph>(slice) {
                    Self::strip_terminal_nodes(&mut graph);
                    if !graph.nodes.is_empty() {
                        return Ok(graph);
                    }
                }
            }
        }

        if let Ok(nodes_arr) = serde_json::from_str::<Vec<serde_json::Value>>(&clean) {
            let mut map = HashMap::new();
            for (idx, item) in nodes_arr.into_iter().enumerate() {
                let id = item.get("id").and_then(|v| v.as_str()).map(ToString::to_string)
                    .unwrap_or_else(|| format!("task_{}", idx + 1));
                let obj = item.get("objective").or_else(|| item.get("description"))
                    .and_then(|v| v.as_str()).unwrap_or(directive).to_string();

                let lower_obj = obj.trim().to_lowercase();
                if lower_obj == "halt" || lower_obj == "halt." || lower_obj == "halt execution" || lower_obj == "halt execution." || lower_obj.starts_with("halt execution") {
                    continue;
                }

                let deps = item.get("deps").or_else(|| item.get("dependencies"))
                    .and_then(|v| v.as_array())
                    .map(|arr| arr.iter().filter_map(|x| x.as_str().map(ToString::to_string)).collect())
                    .unwrap_or_default();
                let agent = item.get("required_agent").or_else(|| item.get("agent"))
                    .and_then(|v| v.as_str()).unwrap_or("CodeAnalyzer").to_string();

                map.insert(id.clone(), TaskNode {
                    id,
                    deps,
                    required_agent: agent,
                    status: TaskStatus::Pending,
                    objective: obj,
                    success_criteria: vec![],
                    execution_state: TaskExecutionState::default(),
                    output_evidence: None,
                    priority: idx + 1,
                });
            }
            if !map.is_empty() {
                return Ok(TaskGraph { nodes: map });
            }
        }

        Ok(Self::create_fallback_graph(directive))
    }

    fn strip_terminal_nodes(graph: &mut TaskGraph) {
        let terminal_ids: Vec<String> = graph.nodes.iter()
            .filter(|(_, node)| {
                let obj = node.objective.trim().to_lowercase();
                obj == "halt" || obj == "halt." || obj == "halt execution" || obj == "halt execution." || obj.starts_with("halt execution") || obj.starts_with("stop execution") || obj.starts_with("end mission")
            })
            .map(|(id, _)| id.clone())
            .collect();

        for id in terminal_ids {
            graph.nodes.remove(&id);
        }
    }

    fn create_fallback_graph(directive: &str) -> TaskGraph {
        let mut nodes = HashMap::new();

        nodes.insert("task_1".to_string(), TaskNode {
            id: "task_1".to_string(),
            deps: vec![],
            required_agent: "CodeAnalyzer".to_string(),
            status: TaskStatus::Pending,
            objective: directive.to_string(),
            success_criteria: vec![],
            execution_state: TaskExecutionState::default(),
            output_evidence: None,
            priority: 1,
        });

        TaskGraph { nodes }
    }
}

use crate::orchestrator_v2::task_graph::{TaskGraph, TaskStatus};
use crate::orchestrator_v2::verifier::VerificationResult;

pub struct Scheduler;

/// v1.9 residual-driven priority.
/// Higher residual → expand first. impact/mission_priority weight the mission value.
pub fn residual_priority(
    residual: f32,
    impact: f32,
    mission_priority: f32,
    dependency_weight: f32,
) -> f32 {
    residual * impact * mission_priority * (1.0 + dependency_weight)
}

impl Scheduler {
    const MAX_RETRIES: u8 = 2;

    /// Transitions viable tasks from Pending/Retrying to Ready.
    /// Returns the IDs of tasks that are newly marked as Ready.
    pub fn prepare_ready_tasks(graph: &mut TaskGraph) -> Vec<String> {
        let mut newly_ready = Vec::new();

        // We must clone the keys to avoid mutably borrowing the graph while iterating
        let node_ids: Vec<String> = graph.nodes.keys().cloned().collect();

        for id in node_ids {
            let mut is_ready = false;

            // Check dependency status immutably
            if let Some(node) = graph.nodes.get(&id) {
                if node.status == TaskStatus::Pending || node.status == TaskStatus::Retrying {
                    let all_deps_met = node.deps.iter().all(|dep_id| {
                        graph
                            .nodes
                            .get(dep_id)
                            .map(|n| n.status == TaskStatus::Completed)
                            .unwrap_or(false)
                    });
                    if all_deps_met {
                        is_ready = true;
                    }
                }
            }

            // Apply state transition
            if is_ready {
                if let Some(node) = graph.nodes.get_mut(&id) {
                    node.status = TaskStatus::Ready;
                    newly_ready.push(id.clone());
                }
            }
        }
        newly_ready
    }

    pub fn mark_running(graph: &mut TaskGraph, task_id: &str) {
        if let Some(node) = graph.nodes.get_mut(task_id) {
            if node.status == TaskStatus::Ready {
                node.status = TaskStatus::Running;
            }
        }
    }

    pub fn process_verification(graph: &mut TaskGraph, task_id: &str, result: VerificationResult) {
        if let Some(node) = graph.nodes.get_mut(task_id) {
            match result {
                VerificationResult::Success => {
                    node.status = TaskStatus::Completed;
                }
                VerificationResult::Retry(reason) => {
                    node.execution_state.attempts += 1;
                    node.execution_state.last_error = Some(reason.clone());

                    if node.execution_state.attempts >= Self::MAX_RETRIES {
                        node.status = TaskStatus::Escalated; // Budget exhausted
                    } else {
                        node.status = TaskStatus::Retrying;
                    }
                }
                VerificationResult::Escalate(reason) => {
                    node.execution_state.last_error = Some(reason);
                    node.status = TaskStatus::Escalated;
                }
                VerificationResult::Failure(reason) => {
                    node.execution_state.last_error = Some(reason);
                    node.status = TaskStatus::Failed;
                }
            }
        }
    }
}

use anyhow::{anyhow, Result};
use serde::{Deserialize, Serialize};
#[allow(unused_imports)]
use std::collections::HashSet;
use std::collections::{HashMap, VecDeque};

#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub enum TaskStatus {
    Pending,
    Running,
    Success,
    Failed,
    Completed,
    Ready,
    Escalated,
    Retrying,
}

#[derive(Clone, Debug, Serialize, Deserialize, Default)]
pub struct TaskExecutionState {
    pub attempts: u8,
    pub last_error: Option<String>,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct TaskNode {
    pub id: String,

    #[serde(alias = "dependencies")]
    pub deps: Vec<String>,

    #[serde(alias = "assigned_agent")]
    pub required_agent: String,

    #[serde(default = "default_status")]
    pub status: TaskStatus,

    #[serde(default)]
    pub objective: String,

    #[serde(default)]
    pub success_criteria: Vec<String>,

    #[serde(default)]
    pub execution_state: TaskExecutionState,

    #[serde(default)]
    pub output_evidence: Option<String>,

    // HEFT Optimization field
    #[serde(default)]
    pub priority: usize,
}

fn default_status() -> TaskStatus {
    TaskStatus::Pending
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct TaskGraph {
    pub nodes: HashMap<String, TaskNode>,
}

impl TaskGraph {
    pub fn new() -> Self {
        Self { nodes: HashMap::new() }
    }

    pub fn add_task(&mut self, id: &str, deps: Vec<String>, required_agent: &str) {
        self.nodes.insert(
            id.to_string(),
            TaskNode {
                id: id.to_string(),
                deps,
                required_agent: required_agent.to_string(),
                status: TaskStatus::Pending,
                objective: String::new(),
                success_criteria: Vec::new(),
                execution_state: TaskExecutionState::default(),
                output_evidence: None,
                priority: 0,
            },
        );
    }

    pub fn compute_critical_path_priorities(&mut self) {
        let mut memo: HashMap<String, usize> = HashMap::new();

        fn b_level(node_id: &str, graph: &TaskGraph, memo: &mut HashMap<String, usize>) -> usize {
            if let Some(&cached) = memo.get(node_id) {
                return cached;
            }
            let mut max_child = 0;

            for (child_id, child) in &graph.nodes {
                if child.deps.contains(&node_id.to_string()) {
                    max_child = max_child.max(b_level(child_id, graph, memo));
                }
            }
            let level = max_child + 1;
            memo.insert(node_id.to_string(), level);
            level
        }

        let keys: Vec<String> = self.nodes.keys().cloned().collect();
        for id in keys {
            let prio = b_level(&id, self, &mut memo);
            if let Some(node) = self.nodes.get_mut(&id) {
                node.priority = prio;
            }
        }
    }

    pub fn resolve_execution_order(&self) -> Result<Vec<String>> {
        for node in self.nodes.values() {
            for dep in &node.deps {
                if !self.nodes.contains_key(dep.as_str()) {
                    return Err(anyhow!(
                        "DAG_ERROR: Task '{}' depends on unknown task '{}'.",
                        node.id,
                        dep
                    ));
                }
            }
        }

        let mut in_degree: HashMap<&str, usize> =
            self.nodes.keys().map(|k| (k.as_str(), 0)).collect();
        let mut dependents: HashMap<&str, Vec<&str>> = HashMap::new();

        for node in self.nodes.values() {
            for dep in &node.deps {
                dependents.entry(dep.as_str()).or_default().push(node.id.as_str());
                *in_degree.entry(node.id.as_str()).or_insert(0) += 1;
            }
        }

        let mut queue: VecDeque<&str> =
            in_degree.iter().filter(|(_, &d)| d == 0).map(|(&id, _)| id).collect();
        let mut order = Vec::with_capacity(self.nodes.len());

        while let Some(id) = queue.pop_front() {
            order.push(id.to_string());
            if let Some(kids) = dependents.get(id) {
                for &child in kids {
                    let deg = in_degree.get_mut(child).unwrap();
                    *deg -= 1;
                    if *deg == 0 {
                        queue.push_back(child);
                    }
                }
            }
        }

        if order.len() != self.nodes.len() {
            let resolved: HashSet<_> = order.iter().map(|s| s.as_str()).collect();
            let cycle_nodes: Vec<_> =
                self.nodes.keys().filter(|k| !resolved.contains(k.as_str())).cloned().collect();
            return Err(anyhow!(
                "DAG_DEADLOCK: Circular dependency detected among: {:?}",
                cycle_nodes
            ));
        }
        Ok(order)
    }
}

impl Default for TaskGraph {
    fn default() -> Self {
        Self::new()
    }
}

#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub enum VerificationResult {
    Success,
    Retry(String),
    Failure(String),
    Escalate(String),
}

pub struct Verifier;

impl Verifier {
    /// LocalDiagnosis Layer: Cheap, deterministic heuristic matching.
    pub fn diagnose_output(stderr: &str, stdout: &str) -> VerificationResult {
        let combined = format!("{} {}", stderr, stdout).to_lowercase();

        // 1. Transient Faults -> Yields a localized Retry
        if combined.contains("timeout")
            || combined.contains("connection reset")
            || combined.contains("rate limit")
            || combined.contains("503 service unavailable")
            || combined.contains("429 too many requests")
        {
            return VerificationResult::Retry("Transient network/API fault detected.".into());
        }

        // 2. Deterministic Faults -> Immediate Escalation
        if combined.contains("error:")
            || combined.contains("cannot find crate")
            || combined.contains("borrowed value")
            || combined.contains("syntax error")
            || combined.contains("test failed")
            || combined.contains("panicked at")
        {
            return VerificationResult::Escalate(
                "Deterministic compilation/logic fault detected.".into(),
            );
        }

        // 3. Ambiguous Anomaly -> Escalate
        if !stderr.trim().is_empty() {
            return VerificationResult::Escalate("Unknown stderr anomaly detected.".into());
        }

        VerificationResult::Success
    }

    pub async fn evaluate(stdout: &str, stderr: &str, criteria: &[String]) -> VerificationResult {
        let diagnosis = Self::diagnose_output(stderr, stdout);
        if diagnosis != VerificationResult::Success {
            return diagnosis;
        }

        if criteria.is_empty() {
            return VerificationResult::Success;
        }

        // TODO: Phase 2 Low-temp Gemma API semantic verification.
        VerificationResult::Success
    }
}

use std::io::{Error, ErrorKind, Result};
use std::process::{Command, Output};

pub struct PodmanShield;
impl PodmanShield {
    pub fn execute(command: &str) -> Result<Output> {
        if let Err(e) =
            crate::arbiter_agent::ArbiterAgent::validate_action("execute_command", command)
        {
            return Err(Error::new(ErrorKind::PermissionDenied, e));
        }
        // Use bash, not POSIX sh -- the Strategist routinely proposes bash-only
        // syntax (${PIPESTATUS[0]}, the `time` keyword, etc.) which fails under
        // sh with cryptic errors like "Bad substitution", wasting turns before
        // any working command is found.
        let mut child = Command::new("bash");
        child.arg("-c").arg(command);
        for (key, value) in std::env::vars() {
            if key != "CARGO_BUILD_TARGET" {
                child.env(key, value);
            }
        }
        child.env_remove("CARGO_BUILD_TARGET");
        let out = child.output()?;

        // REGRESSION FIX (reapplied): explicitly check command success and
        // expose stderr. Without this, a failing command (e.g. a broken
        // `cargo check`) is indistinguishable from a successful one -- both
        // get tagged [CMD_OK] by the executor, which the Cortical Loop's
        // is_terminal_cmd heuristic then treats as proof of success.
        if !out.status.success() {
            let stderr = String::from_utf8_lossy(&out.stderr);
            return Err(Error::other(format!(
                "Command failed with status: {}\nSTDERR:\n{}",
                out.status, stderr
            )));
        }

        Ok(out)
    }
}

    pub fn validate_intent(&self, action_type: &str) -> Result<(), String> {
        let required_token = match action_type {
            "read_file" | "read_file_chunk" | "search_code" | "search_memory" | "read_function" => {
                Capability::ReadSubstrate
            }
            "write_file" | "diff_patch" | "patch_function" | "lazarus_snapshot"
            | "export_report" => Capability::WriteSubstrate,
            "execute_command" | "git_rollback" => Capability::ExecuteSystemCall,
            "halt" => Capability::HaltCognition,
            _ => return Err(format!("SECURITY_FAULT: Unknown action taxonomy '{}'", action_type)),
        };

        if self.active_capabilities.contains(&required_token) {
            Ok(())
        } else {
            Err(format!("POLICY_VIOLATION: Agent lacks {:?} capability.", required_token))
        }
    }
}

use std::collections::{BTreeMap, HashMap};

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct SymbolId(pub String);

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
    pub line: usize,
}

impl Ord for Span {
    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
        self.line
            .cmp(&other.line)
            .then_with(|| self.start.cmp(&other.start))
            .then_with(|| self.end.cmp(&other.end))
    }
}

impl PartialOrd for Span {
    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
        Some(self.cmp(other))
    }
}

#[derive(Debug, Clone)]
pub struct SymbolInfo {
    pub id: SymbolId,
    pub name: String,
    pub kind: String,
    pub span: Span,
    pub file_path: String,
}

#[derive(Debug, Clone)]
pub struct SymbolReference {
    pub symbol_id: SymbolId,
    pub span: Span,
    pub file_path: String,
}

#[derive(Debug, Default, Clone)]
pub struct SymbolIndex {
    pub symbols: HashMap<String, SymbolInfo>,
    pub references: Vec<SymbolReference>,
    pub span_map: BTreeMap<Span, SymbolId>,
}

impl SymbolIndex {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn insert_symbol(&mut self, info: SymbolInfo) {
        self.span_map.insert(info.span.clone(), info.id.clone());
        self.symbols.insert(info.id.0.clone(), info);
    }

    pub fn add_reference(&mut self, reference: SymbolReference) {
        self.references.push(reference);
    }

    pub fn get_symbol(&self, id: &SymbolId) -> Option<&SymbolInfo> {
        self.symbols.get(&id.0)
    }

    pub fn find_by_span(&self, span: &Span) -> Option<&SymbolId> {
        self.span_map.get(span)
    }
}

    let _ = command_tx.send(ExecutiveCommand::InjectGoal {
        objective: raw_directive.to_string(),
        parent_id: None,
    }).await;

    tokio::spawn(executive_cortex.run_actor_loop());

    let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[EXECUTIVE CORTEX] Registered goal {}. Decomposing DAG...", goal_id)}).to_string());

    let mut graph = match PlannerAgent::generate_plan(&ctx.client, &ctx.api_key, raw_directive).await {
        Ok(g) => g,
        Err(e) => return format!("FRACTURE: Planner failed to decompose objective: {}", e),
    };

    graph.compute_critical_path_priorities();
    let order = match graph.resolve_execution_order() {
        Ok(o) => o,
        Err(e) => return format!("FRACTURE: Graph validation failed: {}", e),
    };

    let plan_id = uuid::Uuid::new_v4().to_string();
    let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[EXECUTIVE CORTEX] Plan {} persisted.", plan_id)}).to_string());

    let mut task_outputs: HashMap<String, String> = HashMap::new();
    let mut mission_had_failures = false;
    let world_model = Arc::new(WorldModel::new());

    for task_id in order {
        let node = graph.nodes.get(&task_id).unwrap().clone();
        let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[SCHEDULER] Dispatching DAG Node: {}\nvia Subagent: {}", node.objective, node.required_agent)}).to_string());

        let mut task_context_history = String::new();
        for dep in &node.deps {
            if let Some(out) = task_outputs.get(dep) {
                let bounded_out = if out.len() > 1500 {
                    format!("{}... [TRUNCATED FOR CONTEXT BOUNDS]", &out[..1500])
                } else {
                    out.clone()
                };
                task_context_history.push_str(&format!(
                    "---\n[DEPENDENCY FOCUS FROM {}]:\n{}\n---\n",
                    dep, bounded_out
                ));
            }
        }

        let combined_mission = format!(
            "OBJECTIVE: {}\nASSIGNED ROLE: {}\n\nLINEAGE CONTEXT:\n{}",
            node.objective, node.required_agent, task_context_history
        );

        let mut loop_cortex = CorticalLoop::new(
            ctx.api_key.clone(),
            world_model.clone(),
            ctx.memory.clone(),
            ctx.vmo_state.clone(),
            ctx.dist.clone(),
            ctx.tx.clone(),
        );

        let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[CORTEX] Igniting Epistemic Session for Node {}", task_id)}).to_string());

        let execution_evidence = match loop_cortex.ignite(&combined_mission).await {
            Ok(CortexOutcome::Proven { belief, evidence }) => {
                format!(
                    "[DAG NODE COMPLETE] Objective: {}\nBelief: {:.2}\nEvidence:\n{}",
                    node.objective, belief, evidence
                )
            }
            Ok(CortexOutcome::TurnsExhausted { best_belief, evidence }) => {
                mission_had_failures = true;
                let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[CORTEX INCOMPLETE] Node {} exhausted turns (best belief {:.2}).", task_id, best_belief)}).to_string());
                format!("[DAG NODE INCOMPLETE] Objective: {}\nBest belief: {:.2}\nPartial evidence:\n{}", node.objective, best_belief, evidence)
            }
            Err(e) => {
                mission_had_failures = true;
                let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[CORTEX FRACTURE] Node {} loop failed: {}", task_id, e)}).to_string());
                format!("[DAG NODE FAILED] Ancestor node '{}' failed: {}", node.objective, e)
            }
        };

        task_outputs.insert(task_id.clone(), execution_evidence.clone());

        // ⚡ FIX: Immediate Hard Abort if a dependency fails or runs out of turns
        if mission_had_failures || execution_evidence.contains("[CORTEX INCOMPLETE]") || execution_evidence.contains("[DAG NODE FAILED]") || execution_evidence.contains("[DAG NODE INCOMPLETE]") {
            let _ = ctx.tx.send(serde_json::json!({"type": "LOG_EVENT", "payload": format!("[DAG HALT] Aborting downstream execution. Dependency '{}' failed to complete.", task_id)}).to_string());
            break;
        }
    }

    if mission_had_failures {
        format!("DAG/GoT Execution FINISHED WITH INCOMPLETE NODES. Plan ID: {}", plan_id)
    } else {
        format!("DAG/GoT Hybrid Execution Complete. Plan ID: {}", plan_id)
    }
}

    pub path: String,
    pub content: String,
}

pub fn execute_write_file(input: WriteFileInput) -> Result<String> {
    let path = Path::new(&input.path);
    
    if let Some(parent) = path.parent() {
        if !parent.as_os_str().is_empty() {
            fs::create_dir_all(parent)
                .with_context(|| format!("Failed to create parent directories for {}", input.path))?;
        }
    }

    fs::write(path, &input.content)
        .with_context(|| format!("Failed to write file contents to {}", input.path))?;

    tracing::info!("[EVENT] Mission Complete. AST mutation seamlessly integrated.");

    Ok(format!("Successfully wrote {} bytes to target path: {}", input.content.len(), input.path))
}

#[derive(Debug, Clone)]
pub struct VerificationReport {
    pub evidence_quality: f32,
    pub mutation_detected: bool,
    pub command_completed: bool,
    pub security_violation_detected: bool,
    pub goal_verified: bool,
}

impl VerificationReport {
    pub fn from_evidence(evidence: &str, evaluator_confidence: f32) -> Self {
        let security_violation = evidence.contains("PATH_TRAVERSAL_DENIED")
            || evidence.contains("sandbox_violation")
            || evidence.contains("SECURITY_VIOLATION");

        let command_completed = (evidence.contains("[CMD_OK]")
            || evidence.contains("[READ_OK]")
            || evidence.contains("[WRITE_OK]"))
            && !security_violation;

        Self {
            evidence_quality: if security_violation { 0.0 } else { evaluator_confidence.max(0.0) },
            mutation_detected: evidence.contains("[WRITE_OK]")
                || evidence.contains("[PATCH_FUNCTION_OK]")
                || evidence.contains("[DIFF_PATCH_OK]"),
            command_completed,
            security_violation_detected: security_violation,
            goal_verified: evaluator_confidence > 0.5 && !security_violation,
        }
    }

    pub fn can_complete(&self, belief: f32) -> bool {
        !self.security_violation_detected
            && self.command_completed
            && belief >= 0.85
            && self.goal_verified
            && self.evidence_quality >= 0.5
    }
}

use anyhow::{anyhow, Result};
use base64::{engine::general_purpose, Engine as _};
use std::time::{SystemTime, UNIX_EPOCH};
use tokio::fs;

pub struct WasmForge;

/// Run wasm-opt -Oz when Binaryen is available. Returns original bytes on failure.
fn maybe_wasm_opt(wasm_bytes: &[u8]) -> Vec<u8> {
    use std::io::Write;
    use std::process::{Command, Stdio};
    use tempfile::NamedTempFile;

    // Fast path: wasm-opt not installed
    if Command::new("wasm-opt").arg("--version").output().is_err() {
        return wasm_bytes.to_vec();
    }

    let mut input = match NamedTempFile::new() {
        Ok(f) => f,
        Err(_) => return wasm_bytes.to_vec(),
    };
    if input.write_all(wasm_bytes).is_err() {
        return wasm_bytes.to_vec();
    }
    let output_path = input.path().with_extension("opt.wasm");

    let status = Command::new("wasm-opt")
        .args(["-Oz", "--enable-bulk-memory", "--strip-debug", "--strip-producers"])
        .arg(input.path())
        .arg("-o")
        .arg(&output_path)
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .status();

    match status {
        Ok(s) if s.success() => std::fs::read(&output_path).unwrap_or_else(|_| wasm_bytes.to_vec()),
        _ => wasm_bytes.to_vec(),
    }
}

impl WasmForge {
    pub async fn synthesize_cell(logic_code: &str) -> Result<String> {
        if logic_code.trim().starts_with("const std") || logic_code.contains("export fn") {
            return Self::synthesize_zig(logic_code).await;
        }
        Self::synthesize_rust(logic_code).await
    }

    async fn synthesize_zig(zig_code: &str) -> Result<String> {
        let ts = SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs();
        let tmp_path = format!("/tmp/nexus_zig_cell_{}.zig", ts);
        fs::write(&tmp_path, zig_code).await?;

        let output = tokio::process::Command::new("zig")
            .args([
                "build-exe",
                &tmp_path,
                "-target",
                "wasm32-freestanding",
                "-O",
                "ReleaseSmall",
                "--name",
                "cell",
            ])
            .current_dir("/tmp")
            .output()
            .await?;

        if !output.status.success() {
            let _ = fs::remove_file(&tmp_path).await;
            return Err(anyhow!("Zig compile error: {}", String::from_utf8_lossy(&output.stderr)));
        }

        let wasm_bytes = fs::read("/tmp/cell.wasm").await?;
        let wasm_bytes = maybe_wasm_opt(&wasm_bytes);
        let b64 = general_purpose::STANDARD.encode(&wasm_bytes);
        let _ = fs::remove_file(&tmp_path).await;
        Ok(b64)
    }

    // ⚡ REAL RUST TO WASM PIPELINE
    async fn synthesize_rust(logic_code: &str) -> Result<String> {
        let ts = SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs();
        let tmp_dir = format!("/tmp/nexus_rust_cell_{}", ts);
        fs::create_dir_all(&tmp_dir).await?;

        let cargo_toml = r#"[package]
name = "cell"
version = "0.1.0"
edition = "2021"
[lib]
crate-type = ["cdylib"]
"#;
        fs::write(format!("{}/Cargo.toml", tmp_dir), cargo_toml).await?;

        let src_dir = format!("{}/src", tmp_dir);
        fs::create_dir_all(&src_dir).await?;
        fs::write(format!("{}/lib.rs", src_dir), logic_code).await?;

        let output = tokio::process::Command::new("cargo")
            .args(["build", "--target", "wasm32-wasip1", "--release"])
            .current_dir(&tmp_dir)
            .output()
            .await?;

        if !output.status.success() {
            let stderr = String::from_utf8_lossy(&output.stderr);
            let _ = fs::remove_dir_all(&tmp_dir).await;
            return Err(anyhow!("Rust WASM compile error:\n{}", stderr));
        }

        let wasm_bytes =
            fs::read(format!("{}/target/wasm32-wasip1/release/cell.wasm", tmp_dir)).await?;
        let wasm_bytes = maybe_wasm_opt(&wasm_bytes);
        let b64 = general_purpose::STANDARD.encode(&wasm_bytes);
        let _ = fs::remove_dir_all(&tmp_dir).await;
        Ok(b64)
    }
}

use anyhow::{anyhow, Result};
use wasmtime::*;

pub struct WasmRuntime;

impl WasmRuntime {
    fn make_engine() -> Result<Engine> {
        let mut config = Config::new();
        // Size + speed balance for ARM64 / low-RAM nodes
        config.cranelift_opt_level(OptLevel::SpeedAndSize);
        // Hard limits so sidecars cannot starve the cortex
        config.consume_fuel(true);
        config.epoch_interruption(true);
        // Cheaper instantiation
        config.memory_init_cow(true);
        // Avoid memory spikes on constrained devices
        config.parallel_compilation(false);
        Ok(Engine::new(&config)?)
    }

    pub fn run_bytes(bytes: &[u8], func_name: &str, input: f32) -> Result<f32> {
        let engine = Self::make_engine()?;
        let module = Module::from_binary(&engine, bytes)?;
        let mut store = Store::new(&engine, ());
        store.set_fuel(50_000_000)?;
        let instance = Instance::new(&mut store, &module, &[])?;
        let func = instance.get_typed_func::<f32, f32>(&mut store, func_name)?;
        Ok(func.call(&mut store, input)?)
    }

    pub fn run_with_buffer(bytes: &[u8], func_name: &str, data: &[u8]) -> Result<f32> {
        let engine = Self::make_engine()?;
        let module =
            Module::from_binary(&engine, bytes).map_err(|e| anyhow!("WASM_LOAD_FAULT: {}", e))?;
        let mut store = Store::new(&engine, ());
        store.set_fuel(50_000_000)?;
        let instance = Instance::new(&mut store, &module, &[])
            .map_err(|e| anyhow!("WASM_INST_FAULT: {}", e))?;

        let memory = instance
            .get_memory(&mut store, "memory")
            .ok_or_else(|| anyhow!("WASM module must export memory"))?;

        let get_ptr = instance
            .get_typed_func::<(), i32>(&mut store, "get_buffer_ptr")
            .map_err(|e| anyhow!("Missing get_buffer_ptr export: {}", e))?;
        let ptr = get_ptr.call(&mut store, ())? as usize;

        if data.len() > 65_536 {
            return Err(anyhow!("VMO payload exceeds 64KB sandbox."));
        }

        // WASMtime 43+ API: Use .write() directly on the memory slice
        memory
            .write(&mut store, ptr, data)
            .map_err(|e| anyhow!("WASM_MEMORY_WRITE_FAULT: {}", e))?;

        let func = instance
            .get_typed_func::<i32, f32>(&mut store, func_name)
            .map_err(|e| anyhow!("WASM_FUNC_MISSING: {}", e))?;

        Ok(func.call(&mut store, data.len() as i32)?)
    }
}

var buffer: [65536]u8 = undefined;

export fn get_buffer_ptr() [*]u8 {
    return &buffer;
}

export fn sumBytes(len: i32) f32 {
    var total: u32 = 0;
    var i: usize = 0;
    const ulen = @as(usize, @intCast(len));
    while (i < ulen) : (i += 1) {
        total += buffer[i];
    }
    return @as(f32, @floatFromInt(total));
}

const std = @import("std");

// Compact Swarm Unit Definition
pub const SwarmUnit = struct {
    id: u32,
    status: u8,   // 0: Offline, 1: Idle, 2: Active
    load: u8,     // 0-100 percentage
};

// Fixed memory allocation for 68 units (The Hive)
var hive_memory: [68]SwarmUnit = undefined;

export fn init_hive() void {
    for (&hive_memory, 0..) |*unit, i| {
        unit.* = SwarmUnit{
            .id = @intCast(i),
            .status = 1, // Start in Idle
            .load = 0,
        };
    }
}

export fn get_hive_ptr() [*]SwarmUnit {
    return &hive_memory;
}

export fn update_unit(id: u32, status: u8, load: u8) void {
    if (id < 68) {
        hive_memory[id].status = status;
        hive_memory[id].load = load;
    }
}
