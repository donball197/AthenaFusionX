//! Full Gemma4DecoderLayer sandwich: 4 RMSNorms, self-attention setup, MLP,
//! PLE stage 2 injection, layer_scalar.
//!
//! NOTE: the actual QK^T / softmax / *V attention score computation is not
//! yet implemented -- that's the next roadmap milestone. Until it exists,
//! the attention branch's contribution is a placeholder zero vector.

use crate::attention::{rms_norm_in_place, AttentionBlock};
use crate::kv_cache::StaticKVCache;
use crate::neon_math::matvec_bf16_to_f32;
use half::bf16;

const HIDDEN_SIZE: usize = 1536;
const PER_LAYER_DIM: usize = 256;

/// Borrowed per-layer weights for one forward call. k_proj/v_proj/k_norm are
/// Option<> because it's UNVERIFIED whether KV-shared layers (the last
/// num_kv_shared_layers) even carry their own copies of these tensors in the
/// checkpoint -- pass None for shared layers regardless of which is true,
/// since they go unused either way.
pub struct DecoderLayerWeights<'a> {
    pub input_layernorm: &'a [f32],
    pub post_attention_layernorm: &'a [f32],
    pub pre_feedforward_layernorm: &'a [f32],
    pub post_feedforward_layernorm: &'a [f32],

    pub q_proj: &'a [bf16],
    pub q_norm: &'a [f32],
    pub k_proj: Option<&'a [bf16]>,
    pub v_proj: Option<&'a [bf16]>,
    pub k_norm: Option<&'a [f32]>,
    pub o_proj: &'a [bf16],

    pub gate_proj: &'a [bf16],
    pub up_proj: &'a [bf16],
    pub down_proj: &'a [bf16],
    pub mlp_intermediate_size: usize, // 6144

    pub per_layer_input_gate: &'a [bf16],
    pub per_layer_projection: &'a [bf16],
    pub post_per_layer_input_norm: &'a [f32],

    // Confirmed real (HF transformers issue #45482) but exact application
    // point unverified against primary source.
    pub layer_scalar: f32,
}

pub struct DecoderLayer {
    pub attn: AttentionBlock,
}

impl DecoderLayer {
    pub fn new(layer_idx: usize, is_global: bool) -> Self {
        Self { attn: AttentionBlock::new(layer_idx, is_global) }
    }

    /// Runs one token through the full layer sandwich. `hidden_state` is
    /// [HIDDEN_SIZE] and updated in place. `layer_ple_slice` is this layer's
    /// 256-dim slice of the Stage-1 per_layer_inputs tensor.
    ///
    /// `is_kv_source`: true if this layer computes and pushes its own K/V
    /// into `kv_cache`; false if this layer only projects Q and relies on
    /// `kv_cache` already holding this position's K/V (pushed earlier this
    /// same token-step by the actual source layer -- caller must pass the
    /// SAME StaticKVCache instance for both, per the routing table).
    pub fn forward(
        &self,
        hidden_state: &mut [f32],
        layer_ple_slice: &[f32],
        weights: &DecoderLayerWeights,
        position: usize,
        is_kv_source: bool,
        kv_cache: &mut StaticKVCache,
    ) {
        assert_eq!(hidden_state.len(), HIDDEN_SIZE);
        assert_eq!(layer_ple_slice.len(), PER_LAYER_DIM);

        // ---- Attention block ----
        let residual = hidden_state.to_vec();
        let mut x = hidden_state.to_vec();
        rms_norm_in_place(&mut x, weights.input_layernorm);

        let q = if is_kv_source {
            let k_proj = weights.k_proj.expect("is_kv_source=true requires k_proj");
            let v_proj = weights.v_proj.expect("is_kv_source=true requires v_proj");
            let k_norm = weights.k_norm.expect("is_kv_source=true requires k_norm");
            self.attn.project_and_cache(
                &x,
                weights.q_proj,
                k_proj,
                v_proj,
                weights.q_norm,
                k_norm,
                position,
                kv_cache,
            )
        } else {
            self.attn.project_q_only(&x, weights.q_proj, weights.q_norm, position)
        };

        // TODO(roadmap item 2): compute_attention_output(&_q, kv_cache) ->
        // [num_heads*head_dim] via softmax(QK^T/sqrt(head_dim))*V, then
        // o_proj back to HIDDEN_SIZE. Placeholder zero vector until then.
        let mut attn_out = self.attn.compute_attention_output(&q, weights.o_proj, kv_cache);

        rms_norm_in_place(&mut attn_out, weights.post_attention_layernorm);
        for i in 0..HIDDEN_SIZE {
            hidden_state[i] = residual[i] + attn_out[i] * weights.layer_scalar;
        }

        // ---- MLP block ----
        let residual = hidden_state.to_vec();
        let mut x = hidden_state.to_vec();
        rms_norm_in_place(&mut x, weights.pre_feedforward_layernorm);

        let inter = weights.mlp_intermediate_size;
        let x_bf16: Vec<bf16> = x.iter().map(|&v| bf16::from_f32(v)).collect();
        let mut gate = vec![0.0f32; inter];
        let mut up = vec![0.0f32; inter];
        matvec_bf16_to_f32(weights.gate_proj, &x_bf16, &mut gate, inter, HIDDEN_SIZE);
        matvec_bf16_to_f32(weights.up_proj, &x_bf16, &mut up, inter, HIDDEN_SIZE);

        let mut gated = vec![0.0f32; inter];
        for i in 0..inter {
            gated[i] = gelu_tanh(gate[i]) * up[i];
        }
        let gated_bf16: Vec<bf16> = gated.iter().map(|&v| bf16::from_f32(v)).collect();
        let mut mlp_out = vec![0.0f32; HIDDEN_SIZE];
        matvec_bf16_to_f32(weights.down_proj, &gated_bf16, &mut mlp_out, HIDDEN_SIZE, inter);

        rms_norm_in_place(&mut mlp_out, weights.post_feedforward_layernorm);
        for i in 0..HIDDEN_SIZE {
            hidden_state[i] = residual[i] + mlp_out[i] * weights.layer_scalar;
        }

        // ---- PLE stage 2 injection ----
        let x_bf16: Vec<bf16> = hidden_state.iter().map(|&v| bf16::from_f32(v)).collect();
        let mut ple_gate = vec![0.0f32; PER_LAYER_DIM];
        matvec_bf16_to_f32(
            weights.per_layer_input_gate,
            &x_bf16,
            &mut ple_gate,
            PER_LAYER_DIM,
            HIDDEN_SIZE,
        );
        for g in ple_gate.iter_mut() {
            *g = gelu_tanh(*g);
        }

        let mut injected = vec![0.0f32; PER_LAYER_DIM];
        for i in 0..PER_LAYER_DIM {
            injected[i] = ple_gate[i] * layer_ple_slice[i];
        }

        let injected_bf16: Vec<bf16> = injected.iter().map(|&v| bf16::from_f32(v)).collect();
        let mut ple_out = vec![0.0f32; HIDDEN_SIZE];
        matvec_bf16_to_f32(
            weights.per_layer_projection,
            &injected_bf16,
            &mut ple_out,
            HIDDEN_SIZE,
            PER_LAYER_DIM,
        );
        rms_norm_in_place(&mut ple_out, weights.post_per_layer_input_norm);
        for i in 0..HIDDEN_SIZE {
            hidden_state[i] += ple_out[i] * weights.layer_scalar;
        }
    }
}

fn gelu_tanh(x: f32) -> f32 {
    let inner = 0.797_884_6_f32 * (x + 0.044715_f32 * x.powi(3));
    0.5 * x * (1.0 + inner.tanh())
}

use crate::api_gateway::{AgentRequest, AgentResponse, ApiGateway};
use crate::sandbox::LocalSandbox;
use std::error::Error;

pub struct HybridDispatcher {
    gateway: ApiGateway,
    cascade: Vec<String>,
}

impl HybridDispatcher {
    pub fn new(cascade: Vec<String>) -> Self {
        Self { gateway: ApiGateway::new(), cascade }
    }

    pub fn process_task(
        &self,
        token_stream_summary: &str,
        prompt: &str,
        max_tokens: usize,
    ) -> Result<AgentResponse, Box<dyn Error>> {
        let mut enriched_prompt = format!(
            "[Local Context Summary: {}]\n\nTask Request: {}",
            token_stream_summary, prompt
        );

        let mut total_tokens = 0;
        let mut last_model;

        // Autonomous ReAct Loop: Max 3 iterations per task
        for _step in 1..=3 {
            let request =
                AgentRequest { prompt: enriched_prompt.clone(), temperature: 0.2, max_tokens };

            let response = self.gateway.dispatch_cascade(request, &self.cascade)?;
            total_tokens += response.tokens_used;
            last_model = response.model_used.clone();

            let mut command_to_run: Option<String> = None;

            // Interceptor 1: Custom XML tags (<execute>...</execute>)
            if let (Some(start), Some(end)) =
                (response.text.find("<execute>"), response.text.find("</execute>"))
            {
                let cmd_start = start + 9;
                if cmd_start < end {
                    command_to_run = Some(response.text[cmd_start..end].trim().to_string());
                }
            }
            // Interceptor 2: Natural Markdown code blocks (```bash ... ``` or ```sh ... ```)
            else if let Some(start_idx) =
                response.text.find("```bash").or_else(|| response.text.find("```sh"))
            {
                let block_start = response.text[start_idx..].find('\n').map(|i| start_idx + i + 1);
                if let Some(content_start) = block_start {
                    // Corrected to use Rust range syntax (..)
                    if let Some(end_idx) = response.text[content_start..].find("```") {
                        let cmd = response.text[content_start..content_start + end_idx]
                            .trim()
                            .to_string();
                        if !cmd.is_empty() {
                            command_to_run = Some(cmd);
                        }
                    }
                }
            }

            // If a command was successfully intercepted, execute it in the sandbox
            if let Some(cmd) = command_to_run {
                let host_output = LocalSandbox::execute(&cmd);

                enriched_prompt.push_str(&format!(
                    "\n\n[Agent Attempted Command]: {}\n[Sandbox Host Output]:\n{}\n\n[Directive]: Based on this real host output, provide the final summarized answer.",
                    cmd, host_output
                ));
                continue;
            }

            // If no execution block is found, the agent has concluded its reasoning
            return Ok(AgentResponse {
                text: response.text,
                tokens_used: total_tokens,
                model_used: last_model,
            });
        }

        Err("Task failed: Exceeded maximum ReAct sandbox iterations (3).".into())
    }
}

//! Full 35-Layer Forward Pass Orchestration for Gemma 4 E2B.
//! Owns the per-layer-type KV caches and the shared-KV routing table.
//!
//! num_kv_shared_layers=20: the last 20 of 35 layers reuse an earlier
//! layer's K/V (of the same attention type) instead of computing their own.
//! With layer_types alternating 4 sliding + 1 global (full_attention at
//! 0-indexed 4,9,14,19,24,29,34), the shared region is layers 15-34;
//! sliding layers in that region source from layer 13, global layers from
//! layer 14 (both derived generally below, not hardcoded).

use crate::decoder_layer::{DecoderLayer, DecoderLayerWeights};
use crate::kv_cache::StaticKVCache;
use crate::rope::AttentionMode;
use std::collections::HashMap;

pub const NUM_LAYERS: usize = 35;
pub const HIDDEN_SIZE: usize = 1536;
pub const PER_LAYER_DIM: usize = 256;
pub const NUM_KV_SHARED_LAYERS: usize = 20;
pub const SLIDING_WINDOW: usize = 512;
// Global (full_attention) layers must see the WHOLE sequence, not just the
// 512-token sliding window, or global attention silently loses context past
// this length. This is a real capacity decision -- the checkpoint doesn't
// specify a hard max context here -- set it to whatever max generation
// length you actually intend to support.
pub const GLOBAL_MAX_CONTEXT: usize = 8192;

/// Confirmed from config.json: full_attention at every 5th layer (0-indexed
/// offset 4) -- 7 global / 28 sliding of 35 total.
pub fn layer_types() -> [AttentionMode; NUM_LAYERS] {
    let mut types = [AttentionMode::Sliding; NUM_LAYERS];
    let mut i = 4;
    while i < NUM_LAYERS {
        types[i] = AttentionMode::Global;
        i += 5;
    }
    types
}

/// For each layer index, which cache index it reads/writes. Layers before
/// the shared boundary (NUM_LAYERS - NUM_KV_SHARED_LAYERS) own their cache
/// (source[i] == i, and are the only layers that ever push into it). Layers
/// at or past the boundary reuse the last prior layer's cache of the same
/// attention type -- general derivation, not hardcoded indices.
pub fn compute_kv_source_layers(types: &[AttentionMode; NUM_LAYERS]) -> [usize; NUM_LAYERS] {
    let boundary = NUM_LAYERS - NUM_KV_SHARED_LAYERS;
    let mut last_seen: HashMap<AttentionMode, usize> = HashMap::new();
    let mut source = [0usize; NUM_LAYERS];
    for i in 0..NUM_LAYERS {
        if i < boundary {
            source[i] = i;
            last_seen.insert(types[i], i);
        } else {
            source[i] = *last_seen
                .get(&types[i])
                .expect("no prior layer of this attention type before the shared-KV region");
        }
    }
    source
}

pub struct ForwardOrchestrator {
    layers: Vec<DecoderLayer>,
    kv_source: [usize; NUM_LAYERS],
    kv_boundary: usize,
    caches: Vec<StaticKVCache>, // indexed 0..kv_boundary, one per KV-source layer
}

impl ForwardOrchestrator {
    pub fn new() -> Self {
        let types = layer_types();
        let kv_source = compute_kv_source_layers(&types);
        let kv_boundary = NUM_LAYERS - NUM_KV_SHARED_LAYERS;

        let layers = (0..NUM_LAYERS)
            .map(|i| DecoderLayer::new(i, types[i] == AttentionMode::Global))
            .collect();

        let caches = (0..kv_boundary)
            .map(|i| match types[i] {
                AttentionMode::Sliding => StaticKVCache::new(SLIDING_WINDOW, 256),
                AttentionMode::Global => StaticKVCache::new(GLOBAL_MAX_CONTEXT, 512),
            })
            .collect();

        Self { layers, kv_source, kv_boundary, caches }
    }

    /// Runs one token through all 35 layers. `hidden_state` is [HIDDEN_SIZE],
    /// updated in place. `per_layer_inputs` is the full (35*256) Stage-1 PLE
    /// tensor for this token. `weights_for(layer_idx)` supplies that layer's
    /// weights -- a closure so this orchestrator doesn't own weight loading
    /// (that's inference_engine's job via EdgeLoader).
    ///
    /// NOTE: the weights_for closure signature/lifetime plumbing here is a
    /// design sketch, not compile-verified against a real EdgeLoader-backed
    /// implementation -- expect possible borrow-checker friction wiring this
    /// up for real, more so than the more mechanical files from before.
    /// Returns the number of tokens currently held in the KV cache at the
    /// given cache index (0..kv_boundary). Diagnostic only.
    pub fn kv_cache_len(&self, cache_idx: usize) -> Option<usize> {
        self.caches.get(cache_idx).map(|c| c.len())
    }

    /// Returns the total number of distinct KV caches (== kv_boundary).
    pub fn kv_cache_count(&self) -> usize {
        self.caches.len()
    }

    pub fn execute_token_step<'a>(
        &mut self,
        hidden_state: &mut [f32],
        per_layer_inputs: &[f32],
        position: usize,
        mut weights_for: impl FnMut(usize) -> DecoderLayerWeights<'a>,
    ) {
        assert_eq!(hidden_state.len(), HIDDEN_SIZE);
        assert_eq!(per_layer_inputs.len(), NUM_LAYERS * PER_LAYER_DIM);

        for layer_idx in 0..NUM_LAYERS {
            let weights = weights_for(layer_idx);
            let is_kv_source = layer_idx < self.kv_boundary;
            let cache_idx = self.kv_source[layer_idx];
            let layer_ple_slice =
                &per_layer_inputs[layer_idx * PER_LAYER_DIM..(layer_idx + 1) * PER_LAYER_DIM];

            self.layers[layer_idx].forward(
                hidden_state,
                layer_ple_slice,
                &weights,
                position,
                is_kv_source,
                &mut self.caches[cache_idx],
            );
        }
    }
}

impl Default for ForwardOrchestrator {
    fn default() -> Self {
        Self::new()
    }
}

//! High-Level Inference Engine for Gemma 4 E2B via nexus_edge_minion
//! Coordinates safetensors model weights loading, orchestrator setup, and autoregressive generation loops.

use crate::decoder_layer::DecoderLayerWeights;
use crate::forward_pass::ForwardOrchestrator;
use crate::safetensors::EdgeLoader;
use half::bf16;
use std::path::Path;

struct CachedNormWeights {
    input_layernorm: Vec<f32>,
    post_attention_layernorm: Vec<f32>,
    pre_feedforward_layernorm: Vec<f32>,
    post_feedforward_layernorm: Vec<f32>,
    q_norm: Vec<f32>,
    k_norm: Vec<f32>,
    post_per_layer_input_norm: Vec<f32>,
}

pub struct InferenceEngine {
    orchestrator: ForwardOrchestrator,
    loader: EdgeLoader,
    cached_norms: Vec<CachedNormWeights>,
}

impl InferenceEngine {
    pub fn new<P: AsRef<Path>>(weights_path: P) -> Result<Self, &'static str> {
        let path_str = weights_path.as_ref().to_str().ok_or("invalid weights path string")?;
        let loader = EdgeLoader::new(path_str)?;
        let orchestrator = ForwardOrchestrator::new();

        // Pre-cache small normalization weights as f32 vectors once at startup
        let mut cached_norms = Vec::with_capacity(35);
        for layer_idx in 0..35 {
            let prefix = format!("model.language_model.layers.{}", layer_idx);

            let load_f32 = |name: &str| -> Vec<f32> {
                if let Ok(ts) = loader.get_tensor(name) {
                    if let Ok(tt) = ts.as_bf16() {
                        return tt.data.iter().map(|v| v.to_f32()).collect();
                    }
                }
                Vec::new()
            };

            cached_norms.push(CachedNormWeights {
                input_layernorm: load_f32(&format!("{}.input_layernorm.weight", prefix)),
                post_attention_layernorm: load_f32(&format!(
                    "{}.post_attention_layernorm.weight",
                    prefix
                )),
                pre_feedforward_layernorm: load_f32(&format!(
                    "{}.pre_feedforward_layernorm.weight",
                    prefix
                )),
                post_feedforward_layernorm: load_f32(&format!(
                    "{}.post_feedforward_layernorm.weight",
                    prefix
                )),
                q_norm: load_f32(&format!("{}.self_attn.q_norm.weight", prefix)),
                k_norm: load_f32(&format!("{}.self_attn.k_norm.weight", prefix)),
                post_per_layer_input_norm: load_f32(&format!(
                    "{}.post_per_layer_input_norm.weight",
                    prefix
                )),
            });
        }

        Ok(Self { orchestrator, loader, cached_norms })
    }

    /// Runs a single autoregressive token generation step through the 35-layer orchestrator.
    pub fn step(
        &mut self,
        token_id: u32,
        position: usize,
        hidden_state: &mut [f32],
    ) -> Result<(), &'static str> {
        const HIDDEN: usize = 1536;
        const PLE_TOTAL: usize = 35 * 256;
        const PLE_LAYER_DIM: usize = 256;

        let embed_ts = self.loader.get_tensor("model.language_model.embed_tokens.weight")?;
        let embed_tt = embed_ts.as_bf16()?;
        if embed_tt.data.is_empty() {
            return Err("FATAL: Invalid embed_tokens tensor data.");
        }
        let vocab = embed_tt.shape[0] as usize;
        let tok = (token_id as usize) % vocab;
        let scale_h = (HIDDEN as f32).sqrt();
        for (i, v) in embed_tt.data[tok * HIDDEN..(tok + 1) * HIDDEN].iter().enumerate() {
            hidden_state[i] = v.to_f32() * scale_h;
        }
        eprintln!("[dbg] after token embed: hs[0..4]={:?}", &hidden_state[0..4]);

        let ple_ts =
            self.loader.get_tensor("model.language_model.embed_tokens_per_layer.weight")?;
        let ple_tt = ple_ts.as_bf16()?;
        let scale_ple = (PLE_LAYER_DIM as f32).sqrt();
        let raw_ple: Vec<f32> = ple_tt.data[tok * PLE_TOTAL..(tok + 1) * PLE_TOTAL]
            .iter()
            .map(|v| v.to_f32() * scale_ple)
            .collect();
        eprintln!(
            "[dbg] raw_ple[0..4]={:?} nonzero_count={}",
            &raw_ple[0..4],
            raw_ple.iter().filter(|v| **v != 0.0).count()
        );

        let proj_ts =
            self.loader.get_tensor("model.language_model.per_layer_model_projection.weight")?;
        let proj_tt = proj_ts.as_bf16()?;
        let hidden_bf16: Vec<bf16> = hidden_state.iter().map(|&v| bf16::from_f32(v)).collect();
        let mut proj = vec![0.0f32; PLE_TOTAL];
        crate::neon_math::matvec_bf16_to_f32(
            proj_tt.data,
            &hidden_bf16,
            &mut proj,
            PLE_TOTAL,
            HIDDEN,
        );
        eprintln!("[dbg] proj[0..4] pre-norm={:?}", &proj[0..4]);

        let norm_ts =
            self.loader.get_tensor("model.language_model.per_layer_projection_norm.weight")?;
        let norm_tt = norm_ts.as_bf16()?;
        let norm_w: Vec<f32> = norm_tt.data.iter().map(|v| v.to_f32()).collect();
        for chunk in proj.chunks_mut(PLE_LAYER_DIM) {
            crate::attention::rms_norm_in_place(chunk, &norm_w);
        }
        eprintln!("[dbg] proj[0..4] post-norm={:?}", &proj[0..4]);

        let ple1 = crate::ple::Stage1PLE::new(35, PLE_LAYER_DIM);
        let mut per_layer_inputs = vec![0.0f32; PLE_TOTAL];
        ple1.blend_stage1(&raw_ple, &proj, &mut per_layer_inputs);
        eprintln!("[dbg] per_layer_inputs[0..4]={:?}", &per_layer_inputs[0..4]);
        eprintln!("[dbg] hidden_state pre-layers[0..4]={:?}", &hidden_state[0..4]);

        let total_bytes_touched = std::cell::Cell::new(0usize);
        let tensor_fetch_count = std::cell::Cell::new(0usize);

        let get_mmap_bf16 = |name: &str| -> &[bf16] {
            if let Ok(ts) = self.loader.get_tensor(name) {
                if let Ok(tt) = ts.as_bf16() {
                    let ptr = tt.data.as_ptr();
                    let len = tt.data.len();
                    total_bytes_touched.set(total_bytes_touched.get() + len * 2); // bf16 = 2 bytes
                    tensor_fetch_count.set(tensor_fetch_count.get() + 1);
                    return unsafe { std::slice::from_raw_parts(ptr, len) };
                }
            }
            eprintln!("[dbg][WARN] tensor fetch returned empty slice for: {}", name);
            &[]
        };

        self.orchestrator.execute_token_step(
            hidden_state,
            &per_layer_inputs,
            position,
            |layer_idx| {
                let prefix = format!("model.language_model.layers.{}", layer_idx);
                let norms = &self.cached_norms[layer_idx];

                let gate_proj = get_mmap_bf16(&format!("{}.mlp.gate_proj.weight", prefix));
                let mlp_intermediate_size =
                    if !gate_proj.is_empty() { gate_proj.len() / 1536 } else { 12288 };

                let q_proj = get_mmap_bf16(&format!("{}.self_attn.q_proj.weight", prefix));
                if layer_idx == 0 {
                    eprintln!(
                        "[dbg] layer 0 q_proj.len()={} gate_proj.len()={}",
                        q_proj.len(),
                        gate_proj.len()
                    );
                }

                let layer_scalar = self
                    .loader
                    .get_tensor(&format!("{}.layer_scalar", prefix))
                    .ok()
                    .and_then(|ts| ts.as_bf16().ok())
                    .and_then(|tt| tt.data.first().map(|v| v.to_f32()))
                    .unwrap_or(1.0);
                if layer_idx == 0 {
                    eprintln!("[dbg] layer 0 layer_scalar={}", layer_scalar);
                }

                DecoderLayerWeights {
                    input_layernorm: &norms.input_layernorm,
                    post_attention_layernorm: &norms.post_attention_layernorm,
                    pre_feedforward_layernorm: &norms.pre_feedforward_layernorm,
                    post_feedforward_layernorm: &norms.post_feedforward_layernorm,

                    q_proj,
                    q_norm: &norms.q_norm,
                    k_proj: Some(get_mmap_bf16(&format!("{}.self_attn.k_proj.weight", prefix))),
                    v_proj: Some(get_mmap_bf16(&format!("{}.self_attn.v_proj.weight", prefix))),
                    k_norm: Some(&norms.k_norm),
                    o_proj: get_mmap_bf16(&format!("{}.self_attn.o_proj.weight", prefix)),

                    gate_proj,
                    up_proj: get_mmap_bf16(&format!("{}.mlp.up_proj.weight", prefix)),
                    down_proj: get_mmap_bf16(&format!("{}.mlp.down_proj.weight", prefix)),
                    mlp_intermediate_size,

                    per_layer_input_gate: get_mmap_bf16(&format!(
                        "{}.per_layer_input_gate.weight",
                        prefix
                    )),
                    per_layer_projection: get_mmap_bf16(&format!(
                        "{}.per_layer_projection.weight",
                        prefix
                    )),
                    post_per_layer_input_norm: &norms.post_per_layer_input_norm,

                    layer_scalar,
                }
            },
        );

        eprintln!("[dbg] hidden_state post-layers[0..4]={:?}", &hidden_state[0..4]);
        eprintln!(
            "[dbg] weight bytes touched across all layers: {} bytes ({:.2} MB) across {} tensor fetches",
            total_bytes_touched.get(),
            total_bytes_touched.get() as f64 / (1024.0 * 1024.0),
            tensor_fetch_count.get()
        );

        // Final RMSNorm before any LM-head / logits step (model.language_model.norm.weight)
        if let Ok(final_norm_ts) = self.loader.get_tensor("model.language_model.norm.weight") {
            if let Ok(final_norm_tt) = final_norm_ts.as_bf16() {
                let final_norm_w: Vec<f32> =
                    final_norm_tt.data.iter().map(|v| v.to_f32()).collect();
                if final_norm_w.len() == hidden_state.len() {
                    crate::attention::rms_norm_in_place(hidden_state, &final_norm_w);
                    eprintln!("[dbg] hidden_state post-final-norm[0..4]={:?}", &hidden_state[0..4]);
                } else {
                    eprintln!(
                        "[dbg][WARN] final norm weight len {} != hidden_state len {}, skipping",
                        final_norm_w.len(),
                        hidden_state.len()
                    );
                }
            }
        } else {
            eprintln!(
                "[dbg][WARN] model.language_model.norm.weight not found, skipping final norm"
            );
        }

        Ok(())
    }

    /// Diagnostic: KV cache length at cache index 0 (layer 0's own cache,
    /// since layer 0 is always a KV-source layer -- before the shared-KV
    /// boundary). Grows by 1 per step() call, up to that cache's capacity
    /// (sliding_window=512 or GLOBAL_MAX_CONTEXT=8192 depending on type).
    pub fn kv_cache_len0(&self) -> Option<usize> {
        self.orchestrator.kv_cache_len(0)
    }

    /// Computes logits from a (post-final-norm) hidden state via the tied LM
    /// head -- config has tie_word_embeddings=true, so there is no separate
    /// output projection tensor; the head reuses embed_tokens.weight
    /// [vocab_size, hidden_size] directly as [out_dim, in_dim] for the matvec.
    /// Applies final_logit_softcapping=30.0 per config.json afterward.
    pub fn compute_logits(&self, hidden_state: &[f32]) -> Result<Vec<f32>, &'static str> {
        let embed_ts = self.loader.get_tensor("model.language_model.embed_tokens.weight")?;
        let embed_tt = embed_ts.as_bf16()?;
        let vocab = embed_tt.shape[0] as usize;
        let hidden_dim = embed_tt.shape[1] as usize;

        if hidden_dim != hidden_state.len() {
            return Err("FATAL: hidden_state length mismatch with embed_tokens hidden_dim.");
        }

        let hidden_bf16: Vec<bf16> = hidden_state.iter().map(|&v| bf16::from_f32(v)).collect();
        let mut logits = vec![0.0f32; vocab];
        crate::neon_math::matvec_bf16_to_f32(
            embed_tt.data,
            &hidden_bf16,
            &mut logits,
            vocab,
            hidden_dim,
        );

        const SOFTCAP: f32 = 30.0;
        for l in logits.iter_mut() {
            *l = SOFTCAP * (*l / SOFTCAP).tanh();
        }

        Ok(logits)
    }
}

use half::bf16;

/// Fixed-capacity ring buffer for K/V.
/// One allocation at construction → no further heap growth.
pub struct StaticKVCache {
    k_data: Vec<bf16>,
    v_data: Vec<bf16>,
    max_seq_len: usize,
    head_dim: usize,
    cursor: usize,
    active_tokens: usize,
}

impl StaticKVCache {
    pub fn new(max_seq_len: usize, head_dim: usize) -> Self {
        let total = max_seq_len * head_dim;
        Self {
            k_data: vec![bf16::from_f32(0.0); total],
            v_data: vec![bf16::from_f32(0.0); total],
            max_seq_len,
            head_dim,
            cursor: 0,
            active_tokens: 0,
        }
    }

    #[inline(always)]
    pub fn push(&mut self, k: &[bf16], v: &[bf16]) {
        debug_assert_eq!(k.len(), self.head_dim);
        debug_assert_eq!(v.len(), self.head_dim);

        let off = self.cursor * self.head_dim;
        self.k_data[off..off + self.head_dim].copy_from_slice(k);
        self.v_data[off..off + self.head_dim].copy_from_slice(v);

        self.cursor = (self.cursor + 1) % self.max_seq_len;
        if self.active_tokens < self.max_seq_len {
            self.active_tokens += 1;
        }
    }

    #[inline(always)]
    pub fn k_at(&self, pos: usize) -> &[bf16] {
        let off = pos * self.head_dim;
        &self.k_data[off..off + self.head_dim]
    }

    #[inline(always)]
    pub fn v_at(&self, pos: usize) -> &[bf16] {
        let off = pos * self.head_dim;
        &self.v_data[off..off + self.head_dim]
    }

    pub fn len(&self) -> usize {
        self.active_tokens
    }

    pub fn is_empty(&self) -> bool {
        self.active_tokens == 0
    }

    pub fn capacity(&self) -> usize {
        self.max_seq_len
    }
}

//! High-performance ARM64 NEON matvec for BF16 → F32 (Stable / Universal)

use half::bf16;
use std::arch::aarch64::*;

#[inline(always)]
pub fn matvec_bf16_to_f32(w: &[bf16], x: &[bf16], y: &mut [f32], m: usize, n: usize) {
    assert_eq!(w.len(), m * n);
    assert_eq!(x.len(), n);
    assert_eq!(y.len(), m);

    #[cfg(target_arch = "aarch64")]
    unsafe {
        matvec_bf16_neon_v2(w, x, y, m, n);
    }

    #[cfg(not(target_arch = "aarch64"))]
    {
        for i in 0..m {
            let mut sum = 0.0f32;
            let row = i * n;
            for j in 0..n {
                sum += w[row + j].to_f32() * x[j].to_f32();
            }
            y[i] = sum;
        }
    }
}

#[cfg(target_arch = "aarch64")]
#[target_feature(enable = "neon")]
unsafe fn matvec_bf16_neon_v2(w: &[bf16], x: &[bf16], y: &mut [f32], m: usize, n: usize) {
    for i in 0..m {
        let mut acc0 = vdupq_n_f32(0.0);
        let mut acc1 = vdupq_n_f32(0.0);
        let mut acc2 = vdupq_n_f32(0.0);
        let mut acc3 = vdupq_n_f32(0.0);

        let row_ptr = w.as_ptr().add(i * n);
        let x_ptr = x.as_ptr();
        let mut j = 0usize;

        while j + 16 <= n {
            let vw0 = vld1q_u16(row_ptr.add(j) as *const u16);
            let vx0 = vld1q_u16(x_ptr.add(j) as *const u16);

            let w0l = vreinterpretq_f32_u32(vshll_n_u16(vget_low_u16(vw0), 16));
            let w0h = vreinterpretq_f32_u32(vshll_high_n_u16(vw0, 16));
            let x0l = vreinterpretq_f32_u32(vshll_n_u16(vget_low_u16(vx0), 16));
            let x0h = vreinterpretq_f32_u32(vshll_high_n_u16(vx0, 16));

            acc0 = vfmaq_f32(acc0, w0l, x0l);
            acc1 = vfmaq_f32(acc1, w0h, x0h);

            let vw1 = vld1q_u16(row_ptr.add(j + 8) as *const u16);
            let vx1 = vld1q_u16(x_ptr.add(j + 8) as *const u16);

            let w1l = vreinterpretq_f32_u32(vshll_n_u16(vget_low_u16(vw1), 16));
            let w1h = vreinterpretq_f32_u32(vshll_high_n_u16(vw1, 16));
            let x1l = vreinterpretq_f32_u32(vshll_n_u16(vget_low_u16(vx1), 16));
            let x1h = vreinterpretq_f32_u32(vshll_high_n_u16(vx1, 16));

            acc2 = vfmaq_f32(acc2, w1l, x1l);
            acc3 = vfmaq_f32(acc3, w1h, x1h);

            j += 16;
        }

        let sum01 = vaddq_f32(acc0, acc1);
        let sum23 = vaddq_f32(acc2, acc3);
        let sum = vaddq_f32(sum01, sum23);
        let mut total = vaddvq_f32(sum);

        while j < n {
            total += w[i * n + j].to_f32() * x[j].to_f32();
            j += 1;
        }

        y[i] = total;
    }
}

//! Stage 1 Per-Layer Embeddings (PLE) for Gemma 4 E2B
//! Master table: embed_tokens_per_layer [262144, 8960] (35 layers * 256 dims).
//! Produces the base per-layer input tensor (seq, 35, 256) blended with model projections.

pub struct Stage1PLE {
    num_layers: usize,
    layer_dim: usize,
}

impl Stage1PLE {
    pub fn new(num_layers: usize, layer_dim: usize) -> Self {
        Self { num_layers, layer_dim }
    }

    /// Computes Stage 1 PLE vector for a single token ID.
    /// Extracts the 8960-dim slice from embed_tokens_per_layer and splits into 35 chunks of 256-dim.
    pub fn compute_raw_ple(
        &self,
        token_id: u32,
        embed_tokens_per_layer_weights: &[f32], // Flattened [262144 * 8960] or row view
        out_ple: &mut [f32],                    // Output buffer [35 * 256]
    ) {
        let total_dim = self.num_layers * self.layer_dim;
        assert_eq!(total_dim, 8960);
        assert_eq!(out_ple.len(), total_dim);

        let row_start = (token_id as usize) * total_dim;
        let row_end = row_start + total_dim;

        // Ensure bounds on weight slice if checking flat buffer
        if embed_tokens_per_layer_weights.len() >= row_end {
            out_ple.copy_from_slice(&embed_tokens_per_layer_weights[row_start..row_end]);
        }
    }

    /// Blends raw PLE with the projected model representation scaled by 1/sqrt(2).
    pub fn blend_stage1(&self, raw_ple: &[f32], proj: &[f32], out_blended: &mut [f32]) {
        let len = raw_ple.len();
        assert_eq!(proj.len(), len);
        assert_eq!(out_blended.len(), len);

        let scale = 1.0f32 / (2.0f32).sqrt();
        for i in 0..len {
            out_blended[i] = (proj[i] + raw_ple[i]) * scale;
        }
    }
}

/// Stage 2 Per-Layer Conditioning and Residual Integration
pub struct Stage2PLE;

impl Stage2PLE {
    /// Applies GELU activation, conditioning scale, and residual addition.
    pub fn forward_layer(hidden_states: &mut [f32], residual: &[f32], conditioning_scale: f32) {
        assert_eq!(hidden_states.len(), residual.len());
        for i in 0..hidden_states.len() {
            let x = hidden_states[i];
            // Approximate GELU activation: 0.5 * x * (1.0 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
            let inner = 0.797_884_6_f32 * (x + 0.044715_f32 * x.powi(3));
            let gelu = 0.5_f32 * x * (1.0_f32 + inner.tanh());

            // Apply conditioning scale and add residual
            hidden_states[i] = (gelu * conditioning_scale) + residual[i];
        }
    }
}

//! Proportional RoPE Kernel Specification for Gemma 4 E2B.
//! head_dim differs by attention mode -- confirmed against real weight shapes:
//!   Sliding: head_dim=256, rotary_dim=256 (full rotary), theta=1e4, rope_type=default
//!   Global:  head_dim=512, rotary_dim=128 (0.25 partial rotary), theta=1e6, rope_type=proportional
//! Frequency basis denominator uses head_dim (not rotary_dim) per Gemma's
//! "proportional" RoPE variant. No silent fallbacks -- validate() enforces
//! both known configurations explicitly.

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum AttentionMode {
    Global,
    Sliding,
}

#[derive(Debug, Clone, Copy)]
pub struct RoPEConfig {
    pub head_dim: usize,
    pub rotary_dim: usize,
    pub theta: f32,
}

impl RoPEConfig {
    pub fn new(mode: AttentionMode) -> Self {
        match mode {
            // Confirmed via real layer.4 tensor shapes: q_proj[4096,1536] /
            // k_proj[512,1536] / o_proj[1536,4096] / q_norm[512] / k_norm[512]
            // -> head_dim=512 (8 heads * 512 = 4096), matching config's
            // global_head_dim. partial_rotary_factor=0.25 of 512 = 128.
            AttentionMode::Global => Self { head_dim: 512, rotary_dim: 128, theta: 1_000_000.0 },
            // Confirmed via real layer.0 tensor shapes: q_proj[2048,1536] /
            // k_proj[256,1536] / o_proj[1536,2048] / q_norm[256] / k_norm[256]
            // -> head_dim=256 (8 heads * 256 = 2048), full rotary.
            AttentionMode::Sliding => Self { head_dim: 256, rotary_dim: 256, theta: 10_000.0 },
        }
    }

    pub fn validate(&self) {
        assert!(
            self.head_dim == 256 || self.head_dim == 512,
            "FATAL: head_dim must be 256 (sliding) or 512 (global) per confirmed model spec, got {}",
            self.head_dim
        );
        assert!(
            self.rotary_dim <= self.head_dim,
            "FATAL: Rotary dimension cannot exceed head dimension."
        );
        assert!(
            self.theta == 1_000_000.0 || self.theta == 10_000.0,
            "FATAL: Invalid theta configuration. Must be explicitly 1e6 or 1e4."
        );
    }
}

/// Authoritative scalar reference implementation for Proportional RoPE.
/// Transforms Q or K vectors in place. Dimensions beyond rotary_dim act as identity.
pub fn apply_rope_scalar(vec: &mut [f32], position: usize, config: &RoPEConfig) {
    config.validate();
    assert_eq!(
        vec.len(),
        config.head_dim,
        "FATAL: Vector length mismatch with head_dim specification."
    );

    let half_rot = config.rotary_dim / 2;

    for i in 0..half_rot {
        // CRITICAL: Frequency basis denominator explicitly uses head_dim,
        // NOT rotary_dim -- this is the "proportional" RoPE variant.
        let exponent = (2 * i) as f32 / config.head_dim as f32;
        let inv_freq = 1.0 / config.theta.powf(exponent);
        let freq = position as f32 * inv_freq;

        let cos_val = freq.cos();
        let sin_val = freq.sin();

        let x1 = vec[i];
        let x2 = vec[i + half_rot];

        // NeoX-style split-half rotation pairing
        vec[i] = x1 * cos_val - x2 * sin_val;
        vec[i + half_rot] = x1 * sin_val + x2 * cos_val;
    }
    // Dimensions from rotary_dim .. head_dim remain untouched (identity rotation).
}

#[cfg(target_arch = "aarch64")]
/// # Safety
///
/// `vec` must be a valid, properly aligned slice of at least
/// `config.head_dim` elements; `position` and `config` must describe a
/// layout consistent with `vec`'s length.
pub unsafe fn apply_rope_neon(vec: &mut [f32], position: usize, config: &RoPEConfig) {
    // Phase 1 initial state: delegate to scalar reference until NEON vector layout is verified
    apply_rope_scalar(vec, position, config);
}

use std::ffi::CString;
use std::os::raw::{c_char, c_void};
use std::ptr;
use std::slice;

#[repr(C)]
struct CTensorView {
    data_ptr: *const u8,
    byte_size: u64,
    dtype_code: u8,
    shape_ptr: *const i64,
    shape_len: usize,
    internal_view: *mut c_void,
}

extern "C" {
    fn nexus_st_init(file_path: *const c_char) -> *mut c_void;
    fn nexus_st_get(handle: *mut c_void, name: *const c_char, out_view: *mut CTensorView) -> bool;
    fn nexus_st_free_view(handle: *mut c_void, view: *mut CTensorView);
    fn nexus_st_deinit(handle: *mut c_void);
}

pub struct EdgeLoader {
    handle: *mut c_void,
}

impl EdgeLoader {
    pub fn new(path: &str) -> Result<Self, &'static str> {
        let c_path = CString::new(path).map_err(|_| "path contained interior nul")?;
        let handle = unsafe { nexus_st_init(c_path.as_ptr()) };
        if handle.is_null() {
            return Err("nexus_st_init failed");
        }
        Ok(Self { handle })
    }

    pub fn get_tensor<'a>(&'a self, name: &str) -> Result<TensorSlice<'a>, &'static str> {
        let c_name = CString::new(name).map_err(|_| "tensor name contained interior nul")?;
        let mut c_view = CTensorView {
            data_ptr: ptr::null(),
            byte_size: 0,
            dtype_code: 0,
            shape_ptr: ptr::null(),
            shape_len: 0,
            internal_view: ptr::null_mut(),
        };

        let ok = unsafe { nexus_st_get(self.handle, c_name.as_ptr(), &mut c_view) };
        if !ok || c_view.data_ptr.is_null() {
            return Err("tensor not found");
        }

        // Read the fields we need BEFORE moving c_view
        let data = unsafe { slice::from_raw_parts(c_view.data_ptr, c_view.byte_size as usize) };
        let shape = unsafe { slice::from_raw_parts(c_view.shape_ptr, c_view.shape_len) };
        let dtype = c_view.dtype_code;

        Ok(TensorSlice {
            engine: self,
            c_view, // move happens here
            data,
            shape,
            dtype, // already copied
        })
    }
}

impl Drop for EdgeLoader {
    fn drop(&mut self) {
        if !self.handle.is_null() {
            unsafe { nexus_st_deinit(self.handle) };
        }
    }
}

pub struct TensorSlice<'a> {
    engine: &'a EdgeLoader,
    c_view: CTensorView,
    pub data: &'a [u8],
    pub shape: &'a [i64],
    pub dtype: u8,
}

impl<'a> Drop for TensorSlice<'a> {
    fn drop(&mut self) {
        unsafe { nexus_st_free_view(self.engine.handle, &mut self.c_view) };
    }
}

impl<'a> TensorSlice<'a> {
    pub fn as_bf16(&self) -> Result<TypedTensor<'a, half::bf16>, &'static str> {
        if self.dtype != 3 {
            return Err("not BF16");
        }
        if !self.data.len().is_multiple_of(2) {
            return Err("odd length");
        }
        let ptr = self.data.as_ptr() as *const half::bf16;
        let len = self.data.len() / 2;
        let data = unsafe { slice::from_raw_parts(ptr, len) };
        Ok(TypedTensor { data, shape: self.shape })
    }
}

pub struct TypedTensor<'a, T> {
    pub data: &'a [T],
    pub shape: &'a [i64],
}

use std::process::Command;

pub struct LocalSandbox;

impl LocalSandbox {
    pub fn execute(command: &str) -> String {
        println!("    [Sandbox] Intercepted command. Executing on Host: `{}`", command);

        match Command::new("sh").arg("-c").arg(command).output() {
            Ok(output) => {
                let mut res = String::new();
                let stdout = String::from_utf8_lossy(&output.stdout).trim().to_string();
                let stderr = String::from_utf8_lossy(&output.stderr).trim().to_string();

                if !stdout.is_empty() {
                    res.push_str(&stdout);
                }
                if !stderr.is_empty() {
                    if !res.is_empty() {
                        res.push('\n');
                    }
                    res.push_str(&format!("[STDERR] {}", stderr));
                }

                if res.is_empty() {
                    "[Command executed successfully with no output]".to_string()
                } else {
                    res
                }
            }
            Err(e) => format!("[SANDBOX FATAL] Failed to spawn shell: {}", e),
        }
    }
}

//! Thin wrapper around HuggingFace's `tokenizers` crate, loading the real
//! Gemma 4 tokenizer.json shipped with the checkpoint (SentencePiece-backed,
//! not reimplemented by hand -- avoids subtle mismatches with the real
//! vocab/merges the model was trained on).

use tokenizers::Tokenizer;

pub struct NexusTokenizer {
    inner: Tokenizer,
}

impl NexusTokenizer {
    pub fn from_file(path: &str) -> Result<Self, String> {
        let inner = Tokenizer::from_file(path).map_err(|e| e.to_string())?;
        Ok(Self { inner })
    }

    /// true for a fresh prompt (adds BOS etc. per tokenizer_config.json).
    pub fn encode(&self, text: &str, add_special_tokens: bool) -> Result<Vec<u32>, String> {
        let enc = self.inner.encode(text, add_special_tokens).map_err(|e| e.to_string())?;
        Ok(enc.get_ids().to_vec())
    }

    /// Decodes token IDs back to text. `skip_special_tokens` should
    /// generally be true when rendering output for a person to read.
    pub fn decode(&self, ids: &[u32], skip_special_tokens: bool) -> Result<String, String> {
        self.inner.decode(ids, skip_special_tokens).map_err(|e| e.to_string())
    }
}

    for i in 0..512 {
        assert!(
            (vec[i] - original[i]).abs() < 1e-6,
            "Position 0 failed at index {}: got {}, expected {}",
            i,
            vec[i],
            original[i]
        );
    }
}

#[test]
fn test_rope_global_partial_rotary_bounds() {
    let config = RoPEConfig::new(AttentionMode::Global);
    config.validate();

    let r_dim = config.rotary_dim;
    let h_dim = config.head_dim;

    let mut vec: Vec<f32> = (0..h_dim).map(|i| i as f32 + 1.0).collect();
    let original = vec.clone();

    apply_rope_scalar(&mut vec, 5, &config);

    // First r_dim dimensions must be modified (rotated)
    let mut modified_count = 0;
    for i in 0..r_dim {
        if (vec[i] - original[i]).abs() > 1e-5 {
            modified_count += 1;
        }
    }
    assert!(
        modified_count > 0,
        "Global attention failed: expected rotation in the first {} dimensions.",
        r_dim
    );

    // Dimensions r_dim..h_dim must remain strictly invariant (identity rotation)
    for i in r_dim..h_dim {
        assert!(
            (vec[i] - original[i]).abs() < 1e-6,
            "Global attention invariant failure at index {}: got {}, expected {}",
            i,
            vec[i],
            original[i]
        );
    }
}

use anyhow::{Context, Result};
use futures_util::StreamExt;
use reqwest::Client;
use serde::{Deserialize, Serialize};
use serde_json::{json, Value};
use std::collections::VecDeque;
use std::sync::Mutex;
use std::time::Instant;
use tokio::sync::broadcast;

const MAX_HISTORY_TURNS: usize = 12;
const SOFT_TOKEN_LIMIT: usize = 110_000;
const HARD_TOKEN_LIMIT: usize = 140_000;
const RESERVED_TOKENS: usize = 8192;

#[derive(Default, Serialize, Deserialize, Clone)]
pub struct ContextPacket {
    pub system_prompt: Option<String>,
    pub model_override: Option<String>,
    pub file_content: Option<String>,
    pub terminal_log: Option<String>,
    pub memory_recall: Option<String>,
    pub system_stats: Option<String>,
}

#[derive(Debug, thiserror::Error)]
pub enum BuilderError {
    #[error("Missing mandatory field: {0}")]
    MissingField(String),
}

#[derive(Default)]
pub struct MicroAgentBuilder {
    system_prompt: Option<String>,
    model: Option<String>,
}

impl MicroAgentBuilder {
    pub fn new() -> Self {
        Self { system_prompt: None, model: None }
    }

    pub fn system_prompt(mut self, prompt: &str) -> Self {
        self.system_prompt = Some(prompt.to_string());
        self
    }

    pub fn model(mut self, model: &str) -> Self {
        self.model = Some(model.to_string());
        self
    }

    pub fn build(self, tx: broadcast::Sender<String>) -> Result<MicroAgent, BuilderError> {
        if self.system_prompt.is_none() {
            return Err(BuilderError::MissingField("system_prompt".to_string()));
        }
        if self.model.is_none() {
            return Err(BuilderError::MissingField("model".to_string()));
        }

        Ok(MicroAgent { memory: Mutex::new(VecDeque::with_capacity(MAX_HISTORY_TURNS * 2)), tx })
    }
}

pub struct MicroAgent {
    pub memory: Mutex<VecDeque<Value>>,
    pub tx: broadcast::Sender<String>,
}

impl MicroAgent {
    pub fn builder() -> MicroAgentBuilder {
        MicroAgentBuilder::new()
    }

    pub fn init(tx: broadcast::Sender<String>) -> Result<Self> {
        Ok(MicroAgent { memory: Mutex::new(VecDeque::with_capacity(MAX_HISTORY_TURNS * 2)), tx })
    }

    pub async fn prepare_and_run(
        &self,
        client: &Client,
        api_key: &str,
        user_text: &str,
        context: ContextPacket,
    ) -> Result<(String, u128)> {
        let start = Instant::now();
        let default_model = std::env::var("NEXUS_HEAVY_MODEL")
            .unwrap_or_else(|_| "nvidia/nemotron-3-ultra-550b-a55b".to_string());
        let model = context.model_override.as_deref().unwrap_or(&default_model);
        let url = format!(
            "https://generativelanguage.googleapis.com/v1beta/models/{}:streamGenerateContent?alt=sse&key={}",
            model,
            api_key.trim()
        );

        let system_instruction =
            context.system_prompt.unwrap_or_else(|| "You are NEXUS.".to_string());

        let mut enriched = format!(
            "SYSTEM: {}

USER REQUEST:
{}",
            system_instruction, user_text
        );
        if let Some(code) = &context.file_content {
            enriched += &format!(
                "

ACTIVE FILE CONTENT:
```
{}
```",
                code
            );
        }
        if let Some(logs) = &context.terminal_log {
            enriched += &format!(
                "

RECENT TERMINAL LOGS:
```
{}
```",
                logs
            );
        }

        let mut history: Vec<Value> = self.memory.lock().unwrap().iter().cloned().collect();
        trim_history_token_aware(&mut history, &enriched);

        history.push(json!({
            "role": "user",
            "parts": [{"text": enriched}]
        }));

        let payload = json!({
            "contents": history,
            "generationConfig": {
                "maxOutputTokens": 8192,
                "temperature": 0.15
            }
        });

        let response =
            client.post(&url).json(&payload).send().await.context("API request failed")?;
        let mut full_response = String::new();
        let mut byte_buffer = Vec::new();
        let mut stream = response.bytes_stream();

        while let Some(chunk_result) = stream.next().await {
            if let Ok(chunk) = chunk_result {
                byte_buffer.extend_from_slice(&chunk);
                while let Some(pos) = byte_buffer.iter().position(|&b| b == b'\n') {
                    let line_bytes: Vec<u8> = byte_buffer.drain(..=pos).collect();
                    let line = String::from_utf8_lossy(&line_bytes);
                    let clean = line.trim_start_matches("data: ").trim();
                    if clean.is_empty() || clean == "[DONE]" {
                        continue;
                    }

                    if let Ok(val) = serde_json::from_str::<Value>(clean) {
                        if let Some(parts) =
                            val.pointer("/candidates/0/content/parts").and_then(|v| v.as_array())
                        {
                            for part in parts {
                                if part.get("thought").and_then(|t| t.as_bool()).unwrap_or(false) {
                                    if let Some(token) = part.get("text").and_then(|t| t.as_str()) {
                                        let _ = self.tx.send(
                                            json!({"type": "NEURAL_THOUGHT", "payload": token})
                                                .to_string(),
                                        );
                                    }
                                    continue;
                                }

                                if let Some(token) = part.get("text").and_then(|t| t.as_str()) {
                                    full_response.push_str(token);
                                    let _ = self.tx.send(
                                        json!({"type": "NEURAL_STREAM", "payload": token})
                                            .to_string(),
                                    );
                                }
                            }
                        }
                    }
                }
            }
        }

        {
            let mut mem = self.memory.lock().unwrap();
            mem.push_back(json!({"role": "user", "parts": [{"text": user_text}]}));
            if !full_response.is_empty() {
                mem.push_back(json!({"role": "model", "parts": [{"text": &full_response}]}));
            }
            while mem.len() > MAX_HISTORY_TURNS * 2 {
                mem.pop_front();
            }
        }

        Ok((full_response, start.elapsed().as_millis()))
    }
}

fn rough_token_count(text: &str) -> usize {
    let chars = text.chars().count();
    let words = text.split_whitespace().count();
    let is_code_heavy = text.contains("```") || text.contains("{") || text.contains("fn ");
    let factor = if is_code_heavy { 3.6 } else { 4.1 };
    ((chars as f64 / factor) + (words as f64 * 0.3)).ceil() as usize
}

fn estimate_message_tokens(msg: &Value) -> usize {
    let empty_vec = Vec::new();
    let parts = msg.pointer("/parts").and_then(|v| v.as_array()).unwrap_or(&empty_vec);
    parts
        .iter()
        .filter_map(|p| {
            p.get("text")
                .and_then(|t| t.as_str())
                .or_else(|| p.get("functionCall").and_then(|fc| fc.as_str()))
                .or_else(|| p.get("functionResponse").and_then(|fr| fr.as_str()))
        })
        .map(rough_token_count)
        .sum::<usize>()
        + 10
}

fn trim_history_token_aware(history: &mut Vec<Value>, current_user_msg: &str) {
    if history.is_empty() {
        return;
    }

    let current_tokens = rough_token_count(current_user_msg) + RESERVED_TOKENS;
    let mut total_tokens: usize = history.iter().map(estimate_message_tokens).sum();
    total_tokens += current_tokens;

    if total_tokens <= SOFT_TOKEN_LIMIT {
        return;
    }

    while total_tokens > SOFT_TOKEN_LIMIT && history.len() >= 4 {
        if let (Some(old_user), Some(old_model)) = (history.first(), history.get(1)) {
            let removed = estimate_message_tokens(old_user) + estimate_message_tokens(old_model);
            total_tokens = total_tokens.saturating_sub(removed);
        }
        history.drain(0..2);
    }

    while total_tokens > HARD_TOKEN_LIMIT && !history.is_empty() {
        let removed = estimate_message_tokens(&history[0]);
        total_tokens = total_tokens.saturating_sub(removed);
        history.remove(0);
    }
}

use anyhow::{anyhow, Result};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ProviderKind {
    Google,
    OpenAiCompatible, // Covers NVIDIA NIM, Ollama, vLLM, Groq, Together
}

impl ProviderKind {
    pub fn detect(model_or_provider: &str) -> Self {
        let env_provider = env::var("NEXUS_PROVIDER").unwrap_or_default().to_lowercase();
        if env_provider == "nvidia" || env_provider == "openai" || env_provider == "ollama" {
            return ProviderKind::OpenAiCompatible;
        }

        if model_or_provider.contains('/')
            || model_or_provider.contains("llama")
            || model_or_provider.contains("deepseek")
            || model_or_provider.contains("nvidia")
            || model_or_provider.contains("mistral")
        {
            ProviderKind::OpenAiCompatible
        } else {
            ProviderKind::Google
        }
    }
}

pub struct UniversalProvider;

impl UniversalProvider {
    pub async fn generate(
        client: &Client,
        api_key: &str,
        model: &str,
        system_prompt: Option<&str>,
        user_prompt: &str,
        temperature: f32,
        max_tokens: u32,
    ) -> Result<String> {
        let kind = ProviderKind::detect(model);

        match kind {
            ProviderKind::Google => {
                let base_url = env::var("NEXUS_API_GATEWAY")
                    .unwrap_or_else(|_| "https://generativelanguage.googleapis.com".to_string());
                let url = format!(
                    "{}/v1beta/models/{}:generateContent?key={}",
                    base_url.trim_end_matches('/'),
                    model.replace("models/", ""),
                    api_key.trim()
                );

                let mut contents = Vec::new();
                if let Some(sys) = system_prompt {
                    contents.push(json!({
                        "role": "user",
                        "parts": [{"text": format!("SYSTEM INSTRUCTION: {}", sys)}]
                    }));
                }
                contents.push(json!({
                    "role": "user",
                    "parts": [{"text": user_prompt}]
                }));

                let payload = json!({
                    "contents": contents,
                    "generationConfig": {
                        "temperature": temperature,
                        "maxOutputTokens": max_tokens
                    }
                });

                let res = client.post(&url).json(&payload).send().await?;
                let body: Value = res.json().await.unwrap_or_default();

                if let Some(err) = body.pointer("/error/message").and_then(|v| v.as_str()) {
                    return Err(anyhow!("Google API Error: {}", err));
                }

                let text = body
                    .pointer("/candidates/0/content/parts/0/text")
                    .and_then(|v| v.as_str())
                    .unwrap_or_default();

                Ok(text.replace("```json", "").replace("```", "").trim().to_string())
            }

            ProviderKind::OpenAiCompatible => {
                let base_url = env::var("NVIDIA_API_GATEWAY")
                    .or_else(|_| env::var("NEXUS_OPENAI_GATEWAY"))
                    .unwrap_or_else(|_| {
                        "https://generativelanguage.googleapis.com/v1beta".to_string()
                    });

                let url = format!("{}/chat/completions", base_url.trim_end_matches('/'));

                let mut messages = Vec::new();
                if let Some(sys) = system_prompt {
                    messages.push(json!({"role": "system", "content": sys}));
                }
                messages.push(json!({"role": "user", "content": user_prompt}));

                let payload = json!({
                    "model": model,
                    "messages": messages,
                    "temperature": temperature,
                    "max_tokens": max_tokens
                });

                let auth_key = env::var("NVIDIA_API_KEY").unwrap_or_else(|_| api_key.to_string());

                let res = client
                    .post(&url)
                    .header("Authorization", format!("Bearer {}", auth_key.trim()))
                    .header("Content-Type", "application/json")
                    .json(&payload)
                    .send()
                    .await?;

                let body: Value = res.json().await.unwrap_or_default();

                if let Some(err) = body.pointer("/error/message").and_then(|v| v.as_str()) {
                    return Err(anyhow!("OpenAI/NIM API Error: {}", err));
                }

                let text = body
                    .pointer("/choices/0/message/content")
                    .and_then(|v| v.as_str())
                    .unwrap_or_default();

                Ok(text.replace("```json", "").replace("```", "").trim().to_string())
            }
        }
    }
}

use std::collections::HashMap;
use std::path::Path;
use turbovec::TurboQuantIndex;

pub struct VectorVault {
    index: TurboQuantIndex,
    id_map: HashMap<usize, u64>, // Maps TurboVec Index to SQLite ID!
    current_idx: usize,
    dimension: usize,
    #[allow(dead_code)]
    bit_width: usize,
}

impl VectorVault {
    pub fn new(dimension: usize, bit_width: usize) -> Self {
        Self {
            index: TurboQuantIndex::new(dimension, bit_width),
            id_map: HashMap::new(),
            current_idx: 0,
            dimension,
            bit_width,
        }
    }

    pub fn commit_state(&mut self, id: u64, embedding: &[f32]) -> Result<(), String> {
        if embedding.len() != self.dimension {
            return Err(format!(
                "Dimension mismatch: expected {}, got {}",
                self.dimension,
                embedding.len()
            ));
        }
        self.index.add(embedding);
        self.id_map.insert(self.current_idx, id);
        self.current_idx += 1;
        Ok(())
    }

    pub fn recall_context(
        &self,
        query_embedding: &[f32],
        k: usize,
    ) -> Result<(Vec<f32>, Vec<u64>), String> {
        if query_embedding.len() != self.dimension {
            return Err("Query embedding mismatch".into());
        }
        let results = self.index.search(query_embedding, k);
        let mut resolved_ids = Vec::new();
        for raw_idx in results.indices {
            if let Some(&real_id) = self.id_map.get(&(raw_idx as usize)) {
                resolved_ids.push(real_id);
            }
        }
        Ok((results.scores, resolved_ids))
    }

    pub fn persist_vault<P: AsRef<Path>>(&self, path: P) -> Result<(), String> {
        let path_str = path.as_ref().to_string_lossy().into_owned();
        self.index.write(&path_str).map_err(|e| e.to_string())?;
        Ok(())
    }
}

//! Async bridge for SQLite using parking_lot::Mutex (no poisoning).

use parking_lot::Mutex;
use std::sync::Arc;
use thiserror::Error;

#[derive(Debug, Error)]
pub enum DbError {
    #[error("SQLite error: {0}")]
    Sqlite(#[from] rusqlite::Error),
    #[error("Join error: {0}")]
    Join(#[from] tokio::task::JoinError),
}

pub async fn db_exec<F, R>(
    conn: Arc<Mutex<rusqlite::Connection>>,
    f: F,
) -> Result<R, DbError>
where
    F: FnOnce(&rusqlite::Connection) -> Result<R, rusqlite::Error> + Send + 'static,
    R: Send + 'static,
{
    tokio::task::spawn_blocking(move || {
        let guard = conn.lock();
        f(&guard).map_err(DbError::Sqlite)
    })
    .await
    .map_err(DbError::Join)?
}

pub async fn db_exec_mut<F, R>(
    conn: Arc<Mutex<rusqlite::Connection>>,
    f: F,
) -> Result<R, DbError>
where
    F: FnOnce(&mut rusqlite::Connection) -> Result<R, rusqlite::Error> + Send + 'static,
    R: Send + 'static,
{
    tokio::task::spawn_blocking(move || {
        let mut guard = conn.lock();
        f(&mut guard).map_err(DbError::Sqlite)
    })
    .await
    .map_err(DbError::Join)?
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GenomeRecord {
    pub objective: String,
    pub wasm_b64: String,
    pub composite_score: f32,
    pub execution_time_ms: u64,
    pub memory_delta_kb: f32,
    pub thermal_impact: f32,
    pub timestamp: i64,
}

pub struct GenomeVault {
    genomes: Arc<Mutex<HashMap<String, GenomeRecord>>>,
    max_retain: usize,
}

impl GenomeVault {
    pub fn new(max_retain: usize) -> Self {
        Self {
            genomes: Arc::new(Mutex::new(HashMap::new())),
            max_retain,
        }
    }

    pub async fn store_high_fitness(&self, objective: &str, record: GenomeRecord) {
        let mut guard = self.genomes.lock().await;
        guard.insert(objective.to_string(), record);

        if guard.len() > self.max_retain {
            let mut sorted: Vec<_> = guard.drain().collect();
            sorted.sort_by(|a, b| b.1.composite_score.partial_cmp(&a.1.composite_score).unwrap());
            sorted.truncate(self.max_retain);
            *guard = sorted.into_iter().collect();
        }
    }

    pub async fn get_best_for_objective(&self, objective: &str) -> Option<GenomeRecord> {
        let guard = self.genomes.lock().await;
        guard.get(objective).cloned()
    }

    pub async fn get_global_best(&self) -> Option<GenomeRecord> {
        let guard = self.genomes.lock().await;
        guard.values()
            .max_by(|a, b| a.composite_score.partial_cmp(&b.composite_score).unwrap())
            .cloned()
    }
}

impl Default for GenomeVault {
    fn default() -> Self {
        Self::new(50)
    }
}

// ⚡ 100% Safe Pure-Rust Virtual Memory Registry (Replaces Zig)
pub static VMO_REGISTRY: Lazy<DashMap<u32, Vec<u8>>> = Lazy::new(DashMap::new);
pub static NEXT_HANDLE: AtomicU32 = AtomicU32::new(100);

pub fn get_active_vmos() -> u32 {
    VMO_REGISTRY.len() as u32
}

pub fn calculate_entropy(handle: u32) -> f32 {
    if let Some(vmo) = VMO_REGISTRY.get(&handle) {
        let data = vmo.value();
        if data.is_empty() {
            return 0.0;
        }

        let mut counts = [0u32; 256];
        for &b in data {
            counts[b as usize] += 1;
        }

        let mut entropy = 0.0f32;
        let len = data.len() as f32;
        for &c in &counts {
            if c > 0 {
                let p = c as f32 / len;
                entropy -= p * p.log2();
            }
        }
        entropy / 8.0
    } else {
        0.0
    }
}

use bytes::BytesMut;

#[derive(Debug)]
pub enum BufferError {
    Overflow,
}

pub struct SafeBuffer {
    buf: BytesMut,
    max_capacity: usize,
}

impl SafeBuffer {
    pub fn new(max_capacity: usize) -> Self {
        Self { buf: BytesMut::with_capacity(max_capacity), max_capacity }
    }

    pub fn write_safe(&mut self, data: &[u8]) -> Result<(), BufferError> {
        if self.buf.len() + data.len() > self.max_capacity {
            return Err(BufferError::Overflow);
        }
        self.buf.extend_from_slice(data);
        Ok(())
    }
}

use crate::kernel::error::KernelError;
use crate::kernel::{NEXT_HANDLE, VMO_REGISTRY};
use std::sync::atomic::Ordering;

pub struct SafeKernel;

impl SafeKernel {
    pub fn create_vmo(size: usize) -> Result<u32, KernelError> {
        if size == 0 || size > 16_777_216 {
            return Err(KernelError::InvalidArgs);
        }
        let handle = NEXT_HANDLE.fetch_add(1, Ordering::Relaxed);
        VMO_REGISTRY.insert(handle, vec![0u8; size]);
        Ok(handle)
    }

    pub fn close_handle(handle: u32) {
        VMO_REGISTRY.remove(&handle);
    }
}

use crate::kernel::error::KernelError;
use crate::kernel::safe_kernel::SafeKernel;
use crate::kernel::VMO_REGISTRY;

pub struct SafeVmo {
    handle: u32,
    pub size: usize,
}

impl SafeVmo {
    pub fn new(size: usize) -> Result<Self, KernelError> {
        let handle = SafeKernel::create_vmo(size)?;
        Ok(Self { handle, size })
    }

    pub fn write(&self, offset: usize, data: &[u8]) -> Result<(), KernelError> {
        if self.size == 0 {
            return Err(KernelError::InvalidArgs);
        }
        let end = offset.checked_add(data.len()).ok_or(KernelError::InvalidArgs)?;
        if end > self.size {
            return Err(KernelError::InvalidArgs);
        }

        if let Some(mut vmo) = VMO_REGISTRY.get_mut(&self.handle) {
            vmo[offset..end].copy_from_slice(data);
            Ok(())
        } else {
            Err(KernelError::InvalidArgs)
        }
    }

    pub fn read_into(&self, offset: usize, buffer: &mut [u8]) -> Result<(), KernelError> {
        let end = offset.checked_add(buffer.len()).ok_or(KernelError::InvalidArgs)?;
        if end > self.size {
            return Err(KernelError::InvalidArgs);
        }

        if let Some(vmo) = VMO_REGISTRY.get(&self.handle) {
            buffer.copy_from_slice(&vmo[offset..end]);
            Ok(())
        } else {
            Err(KernelError::InvalidArgs)
        }
    }

    pub fn handle(&self) -> u32 {
        self.handle
    }
}

impl Drop for SafeVmo {
    fn drop(&mut self) {
        SafeKernel::close_handle(self.handle);
    }
}

        }
        let connect_options = sqlx::sqlite::SqliteConnectOptions::new()
            .filename(db_path)
            .journal_mode(sqlx::sqlite::SqliteJournalMode::Wal)
            .synchronous(sqlx::sqlite::SqliteSynchronous::Normal)

        let vault = Arc::new(Mutex::new(VectorVault::new(768, 4)));

#[derive(Debug, Clone)]
pub struct ModelQuota {
    pub rpm_limit: u32,
    pub tpm_limit: u32,
}

impl ModelQuota {
    fn for_model(model: &str) -> Self {
        match model {
            m if m.contains("flash-lite") => Self { rpm_limit: 10, tpm_limit: 100_000 },
            m if m.contains("flash") => Self { rpm_limit: 5, tpm_limit: 60_000 },
            m if m.contains("31b") => Self { rpm_limit: 3, tpm_limit: 50_000 },
            m if m.contains("pro") => Self { rpm_limit: 2, tpm_limit: 30_000 },
            m if m.contains("research") => Self { rpm_limit: 1, tpm_limit: 10_000 },
            _ => Self { rpm_limit: 3, tpm_limit: 30_000 },
        }
    }
}

#[derive(Debug)]
struct WindowEntry {
    timestamp: Instant,
    tokens: u32,
}

#[derive(Debug, Default)]
struct ModelWindow {
    entries: Vec<WindowEntry>,
}

impl ModelWindow {
    fn purge_stale(&mut self) {
        let cutoff = Instant::now() - Duration::from_secs(60);
        self.entries.retain(|e| e.timestamp > cutoff);
    }
    fn token_sum(&self) -> u32 {
        self.entries.iter().map(|e| e.tokens).sum()
    }
    fn ms_until_slot_free(&self) -> u64 {
        if self.entries.is_empty() {
            return 0;
        }
        let oldest = self.entries.iter().map(|e| e.timestamp).min().unwrap();
        (Duration::from_secs(60).saturating_sub(oldest.elapsed())).as_millis() as u64 + 100
    }
}

pub struct TokenLedger {
    windows: Mutex<HashMap<String, ModelWindow>>,
}

impl TokenLedger {
    pub fn new() -> Self {
        Self { windows: Mutex::new(HashMap::new()) }
    }

    pub fn check_and_reserve(&self, model: &str, estimated_tokens: u32) -> Result<(), u64> {
        let quota = ModelQuota::for_model(model);
        let mut map = self.windows.lock().unwrap();
        let win = map.entry(model.to_string()).or_default();
        win.purge_stale();
        if win.entries.len() as u32 >= quota.rpm_limit {
            return Err(win.ms_until_slot_free());
        }
        if win.token_sum() + estimated_tokens > quota.tpm_limit {
            return Err(win.ms_until_slot_free());
        }
        win.entries.push(WindowEntry { timestamp: Instant::now(), tokens: estimated_tokens });
        Ok(())
    }
}

pub static LEDGER: Lazy<TokenLedger> = Lazy::new(TokenLedger::new);

/// Returns Ok(()) when a quota slot is reserved.
/// Returns Err(message) after MAX_ATTEMPTS — caller MUST NOT fire the API on Err.
pub async fn wait_for_quota(model: &str, estimated_tokens: u32) -> Result<(), String> {
    const MAX_ATTEMPTS: usize = 10;
    for attempt in 0..MAX_ATTEMPTS {
        match LEDGER.check_and_reserve(model, estimated_tokens) {
            Ok(()) => return Ok(()),
            Err(wait_ms) => {
                let jitter = generate_seed() % 500;
                tracing::warn!(
                    "[LEDGER] {} ceiling hit — waiting {}ms (attempt {}/{}).",
                    model,
                    wait_ms + jitter,
                    attempt + 1,
                    MAX_ATTEMPTS
                );
                tokio::time::sleep(Duration::from_millis(wait_ms + jitter)).await;
            }
        }
    }
    Err(format!("[LEDGER] Quota for '{}' unavailable after {} attempts.", model, MAX_ATTEMPTS))
}

fn http_rpm_limit() -> u32 {
    std::env::var("NEXUS_HTTP_RPM").ok().and_then(|v| v.parse().ok()).unwrap_or(60)
}

static HTTP_WINDOWS: Lazy<Mutex<HashMap<String, Vec<Instant>>>> =
    Lazy::new(|| Mutex::new(HashMap::new()));

pub fn http_check(ip: &str) -> Result<(), u64> {
    let limit = http_rpm_limit();
    let cutoff = Instant::now() - Duration::from_secs(60);
    let mut map = HTTP_WINDOWS.lock().unwrap();
    let win = map.entry(ip.to_string()).or_default();
    win.retain(|&ts| ts > cutoff);
    if win.len() as u32 >= limit {
        let oldest = *win.first().unwrap();
        return Err(
            (Duration::from_secs(60).saturating_sub(oldest.elapsed())).as_millis() as u64 + 100
        );
    }
    win.push(Instant::now());
    Ok(())
}

fn generate_seed() -> u64 {
    use std::hash::{BuildHasher, Hasher};
    let t = SystemTime::now().duration_since(SystemTime::UNIX_EPOCH).unwrap_or_default().as_nanos()
        as u64;
    let mut h = std::collections::hash_map::RandomState::new().build_hasher();
    h.write_u64(t);
    h.finish()
}

pub async fn get_embedding(api_key: &str, text: &str) -> Result<Vec<f32>> {

    let text_owned = text.to_string();
    if let Ok(Some(vector)) = tokio::task::spawn_blocking(move || {
        let mut guard = LOCAL_EMBEDDER.lock();
        if let Some(ref mut model) = *guard {
            model.embed(vec![text_owned], None).ok().and_then(|mut v| v.pop())
        } else {
            None
        }
    })
    .await
    {
        return Ok(vector);
    }

    // 2. FALLBACK TO CLOUD IF LOCAL MODEL FAILED OR UNLOADED
    let _ = crate::token_ledger::wait_for_quota("gemini-embedding-2", 500).await;

    let external_client = Client::new();
    let url =
        "https://generativelanguage.googleapis.com/v1beta/models/gemini-embedding-2:embedContent";

    let payload = json!({
        "model": "models/gemini-embedding-2",
        "content": { "parts": [{ "text": text }] }
    });

    let res = external_client
        .post(url)
        .header("x-goog-api-key", api_key.trim())
        .json(&payload)
        .send()
        .await?;
    let body: serde_json::Value = res.json().await?;

    let mut vector = Vec::new();
    if let Some(values) = body.pointer("/embedding/values").and_then(|v| v.as_array()) {

                vector.push(f as f32);
            }
        }
    } else {
        return Err(anyhow::anyhow!("API REJECTED EMBEDDING. Raw response: {}", body));
    }

    Ok(vector)
}

pub fn cosine_similarity(a: &[f32], b: &[f32]) -> f32 {
    if a.len() != b.len() || a.is_empty() {
        return 0.0;
    }
    let dot: f32 = a.iter().zip(b.iter()).map(|(x, y)| x * y).sum();
    let norm_a: f32 = a.iter().map(|x| x * x).sum::<f32>().sqrt();
    let norm_b: f32 = b.iter().map(|x| x * x).sum::<f32>().sqrt();
    if norm_a == 0.0 || norm_b == 0.0 {
        return 0.0;
    }
    dot / (norm_a * norm_b)
}

//! Lock-free VMO registry using DashMap.
//!
//! Previously used `RwLock<HashMap<VmoHandle, Arc<SafeVmo>>>` which serialised
//! every write (allocate/release) behind a single global lock. Under concurrent
//! agent workloads this became a hot contention point.
//!
//! DashMap shards the map into N buckets (default: 2× logical CPUs). Each
//! shard has its own RwLock, so concurrent operations on different handles
//! proceed in parallel with no cross-shard blocking.
//!
//! API is fully backward-compatible — callers need zero changes.

use dashmap::DashMap;
use std::sync::{
    atomic::{AtomicU32, Ordering},
    Arc,
};

pub type VmoHandle = u32;

pub struct SafeVmo {
    pub handle: VmoHandle,
    pub data: Vec<u8>,
    pub timestamp: u64,
}

pub struct VmoHandler {
    /// Sharded concurrent map — no global lock on reads or cross-shard writes.
    registry: DashMap<VmoHandle, Arc<SafeVmo>>,
    /// Atomic counter replaces the second RwLock entirely.
    next_handle: AtomicU32,
}

impl VmoHandler {
    pub fn new() -> Self {
        Self { registry: DashMap::new(), next_handle: AtomicU32::new(1) }
    }

    pub fn allocate_cognitive_window(&self, raw_data: &[u8]) -> Result<VmoHandle, String> {
        // fetch_add is lock-free — no mutex involved at all for handle allocation.
        let handle = self.next_handle.fetch_add(1, Ordering::Relaxed);
        let vmo = Arc::new(SafeVmo {
            handle,
            data: raw_data.to_vec(),
            timestamp: std::time::SystemTime::now()
                .duration_since(std::time::UNIX_EPOCH)
                .unwrap_or_default()
                .as_secs(),
        });
        self.registry.insert(handle, vmo);
        Ok(handle)
    }

    pub fn read_cognitive_window(&self, handle: VmoHandle) -> Result<Vec<u8>, String> {
        self.registry
            .get(&handle)
            .map(|v| v.data.clone())
            .ok_or_else(|| format!("Invalid handle: {}", handle))
    }

    pub fn export_provenance_checkpoint(
        &self,
        handle: VmoHandle,
    ) -> Result<(VmoHandle, Vec<u8>, u64), String> {
        self.registry
            .get(&handle)
            .map(|v| (v.handle, v.data.clone(), v.timestamp))
            .ok_or_else(|| format!("Handle not found: {}", handle))
    }

    pub fn release_cognitive_window(&self, handle: VmoHandle) -> Result<(), String> {
        self.registry
            .remove(&handle)
            .map(|_| ())
            .ok_or_else(|| format!("Handle not found on release: {}", handle))
    }

    /// Returns the number of live VMO allocations.
    pub fn active_count(&self) -> usize {
        self.registry.len()
    }
}

impl Default for VmoHandler {
    fn default() -> Self {
        Self::new()
    }
}

use crate::agent_protocol::AgentAction;

/// When the Translator LLM produces an empty action list, this parses the
/// Strategist's raw proposed-action text directly as a fallback. Pure string
/// parsing -- no side effects, no borrows of CorticalLoop state -- which is
/// exactly why it's the first piece pulled out of the ~500-line
/// ignite_with_evidence function in cortical_loop.rs.
///
/// Returns None if the text doesn't match any known tool prefix or labeled
/// multi-line form; the caller falls back further to raw ExecuteCommand
/// (see cortical_loop.rs).
///
/// NOTE: deliberately does NOT include a write_file branch. That branch was
/// removed from cortical_loop.rs earlier this session (see [CORTEX SAFETY]
/// log line) because raw, un-JSON-escaped Strategist text written straight
/// to disk caused real file corruption. Do not re-add it here.
pub fn intercept_fallback_action(action_str: &str) -> Option<AgentAction> {
    let action_str = action_str.trim();

    // --- Prefix form (translator / single-line) ---
    if let Some(cmd) = action_str.strip_prefix("execute_command") {
        let clean_cmd = cmd.trim().trim_matches('\'').trim_matches('"').to_string();
        return Some(AgentAction::ExecuteCommand {
            command: clean_cmd,
            reasoning: "Intercepted fallback execute_command".to_string(),
            confidence: 0.9,
        });
    }
    if action_str.starts_with("search_code") {
        let parts: Vec<&str> = action_str.split_whitespace().collect();
        if parts.len() >= 3 {
            let path = parts[1].to_string();
            let query = parts[2..].join(" ").trim_matches('\'').trim_matches('"').to_string();
            return Some(AgentAction::SearchCode {
                path,
                query,
                reasoning: "Intercepted fallback search_code".to_string(),
            });
        }
        return None;
    }
    if action_str.starts_with("read_file_chunk") {
        let parts: Vec<&str> = action_str.split_whitespace().collect();
        if parts.len() >= 2 {
            let path = parts[1].to_string();
            let start_line = parts.get(2).and_then(|s| s.parse().ok()).unwrap_or(0);
            let max_lines = parts.get(3).and_then(|s| s.parse().ok()).unwrap_or(500);
            return Some(AgentAction::ReadFileChunk {
                path,
                start_line,
                max_lines,
                reasoning: "Intercepted fallback read_file_chunk".to_string(),
            });
        }
        return None;
    }
    if let Some(stripped) = action_str.strip_prefix("diff_patch ") {
        let parts: Vec<&str> = stripped.split_whitespace().collect();
        if parts.len() >= 3 {
            let path = parts[0].to_string();
            let search = parts[1].to_string();
            let replace = parts[2..].join(" ");
            return Some(AgentAction::DiffPatch {
                path,
                search,
                replace,
                reasoning: "Intercepted fallback diff_patch".to_string(),
            });
        }
        return None;
    }
    if let Some(stripped) = action_str.strip_prefix("patch_function ") {
        let parts: Vec<&str> = stripped.split_whitespace().collect();
        if parts.len() >= 3 {
            let path = parts[0].to_string();
            let function_name = parts[1].to_string();
            let new_body = parts[2..].join(" ");
            return Some(AgentAction::PatchFunction {
                path,
                function_name,
                new_body,
                reasoning: "Intercepted fallback patch_function".to_string(),
            });
        }
        return None;
    }
    if let Some(path) = action_str.strip_prefix("read_file ") {
        return Some(AgentAction::ReadFile {
            path: path.trim().to_string(),
            reasoning: "Intercepted fallback read_file".to_string(),
        });
    }
    if let Some(raw_query) = action_str.strip_prefix("search_memory") {
        let query = raw_query.trim().trim_matches('\'').trim_matches('"').to_string();
        return Some(AgentAction::SearchMemory {
            query,
            reasoning: "Intercepted fallback search_memory".to_string(),
        });
    }

    // --- Labeled multi-line form (Strategist NL) ---
    // Matches proposals like:
    //   Tool: read_file
    //   Target Path: /tmp/openfang/SECURITY.md
    // Does not handle write_file (same safety contract as above).
    if let Some(action) = intercept_labeled_proposal(action_str) {
        return Some(action);
    }

    None
}

/// Strategist-style labeled proposals (Tool: / Target Path: / Command: / Query:).
fn intercept_labeled_proposal(text: &str) -> Option<AgentAction> {
    let lower = text.to_lowercase();

    // read_file
    if lower.contains("tool: read_file")
        || lower.contains("use the `read_file`")
        || lower.contains("use the read_file")
    {
        if let Some(path) = labeled_value(text, &["Target Path:", "target path:", "Path:", "path:"])
        {
            if looks_like_path(&path) {
                return Some(AgentAction::ReadFile {
                    path,
                    reasoning: "Intercepted labeled read_file".to_string(),
                });
            }
        }
    }

    // search_code
    if lower.contains("tool: search_code")
        || lower.contains("use the `search_code`")
        || lower.contains("use the search_code")
    {
        let path = labeled_value(text, &["Path:", "path:"]);
        let query = labeled_value(text, &["Query:", "query:"]);
        if let (Some(path), Some(query)) = (path, query) {
            if looks_like_path(&path) && !query.is_empty() {
                return Some(AgentAction::SearchCode {
                    path,
                    query,
                    reasoning: "Intercepted labeled search_code".to_string(),
                });
            }
        }
    }

    // execute_command — require Tool/Command labels, not bare "git clone" anywhere
    if lower.contains("tool: execute_command")
        || lower.contains("use the execute_command")
        || lower.contains("use the `execute_command`")
    {
        if let Some(cmd) = labeled_value(text, &["Command:", "command:"]) {
            let cmd = cmd.trim().trim_matches('`').to_string();
            if !cmd.is_empty() && !cmd.to_lowercase().starts_with("tool") {
                return Some(AgentAction::ExecuteCommand {
                    command: cmd,
                    reasoning: "Intercepted labeled execute_command".to_string(),
                    confidence: 0.85,
                });
            }
        }
    }

    None
}

fn labeled_value(text: &str, markers: &[&str]) -> Option<String> {
    for marker in markers {
        for line in text.lines() {
            let trimmed = line.trim();
            if trimmed.len() >= marker.len() && trimmed[..marker.len()].eq_ignore_ascii_case(marker)
            {
                let rest = trimmed[marker.len()..].trim().trim_matches('`').trim_matches('"');
                if !rest.is_empty() {
                    return Some(rest.to_string());
                }
            }
        }
    }
    None
}

fn looks_like_path(p: &str) -> bool {
    p.starts_with('/')
        || p.starts_with("./")
        || p.ends_with(".rs")
        || p.ends_with(".toml")
        || p.ends_with(".md")
        || p.ends_with(".json")
        || p.contains('/')
}

#[derive(Clone)]
pub struct AgenticRuntime {
    cmd_sender: mpsc::Sender<AgentCommand>,
}

impl AgenticRuntime {
    pub fn new(execution_budget_ms: u64) -> Self {
        let (cmd_sender, mut cmd_receiver) = mpsc::channel::<AgentCommand>(32);
        tokio::spawn(async move {
            while let Some(cmd) = cmd_receiver.recv().await {
                match cmd {
                    AgentCommand::DispatchTask { prompt, respond_to } => {
                        let result =
                            Self::process_execution_loop(&prompt, execution_budget_ms).await;
                        let _ = respond_to.send(result);
                    }
                }
            }
        });
        Self { cmd_sender }
    }

    pub async fn dispatch(&self, prompt: &str) -> RuntimeResult<()> {
        let (tx, rx) = oneshot::channel();
        self.cmd_sender
            .send(AgentCommand::DispatchTask { prompt: prompt.to_string(), respond_to: tx })
            .await
            .map_err(|e| format!("Failed to queue task: {}", e))?;
        rx.await.map_err(|e| format!("Runtime channel dropped: {}", e))?
    }

    async fn process_execution_loop(prompt: &str, _budget: u64) -> RuntimeResult<()> {
        let raw_source = "<html><body><div id='editor' contenteditable='true'></div></body></html>";
        let _compressed_context = DomBridge::minify_dom(raw_source)?;

    }

    Store(String),
    Recall(String),
    Shutdown,
}

    tx: mpsc::Sender<MemoryEvent>,
}

impl MemoryActor {
    pub fn start() -> (Self, JoinHandle<()>) {
        let (tx, mut rx) = mpsc::channel::<MemoryEvent>(64);

        let handle = tokio::spawn(async move {
            while let Some(event) = rx.recv().await {
                match event {
                    MemoryEvent::Store(data) => {
                        println!("[MEMORY] storing: {}", data);
                    }

                    MemoryEvent::Recall(query) => {
                        println!("[MEMORY] recalling: {}", query);
                    }

                    MemoryEvent::Shutdown => {
                        println!("[MEMORY] shutdown");
                        break;
                    }
                }
            }
        });

        (Self { tx }, handle)
    }

    pub async fn send(
        &self,
        event: MemoryEvent,
    ) -> Result<(), mpsc::error::SendError<MemoryEvent>> {
        self.tx.send(event).await
    }
}

use nexus_model_hub::neural_agent::MicroAgent;
use std::sync::Arc;

pub struct ArbiterAgent {
    pub agent: Arc<MicroAgent>,
}

impl ArbiterAgent {
    pub fn new(agent: Arc<MicroAgent>) -> Self {
        Self { agent }
    }

    pub fn validate_action(action_type: &str, command: &str) -> Result<(), String> {
        let dangerous_patterns = [
            "rm -rf /",
            "rm -rf ~",
            "mkfs",
            "> /dev/sda",
            "> /dev/nvme",
            "chmod 777 /",
            "/etc/shadow",
            "mv /etc/",
            "mv /usr/",
            ":(){ :|:& };:",
            "dd if=/dev/zero",
            "fork bomb",
            "sudo su",
            "sudo -i",
            "sudo bash",
            "sudo sh",
            "passwd root",
            "chown root",
            "$'\\x",
            "\\u0072\\u006d",
            "base64 -d |",
            "eval $(",
            "exec(",
            "curl | bash",
            "wget | bash",
            "curl | sh",
            "history -c",
            "unset HISTFILE",
        ];

        let normalized = command.to_lowercase();

        // Narrow, explicit exception: allow `rm -rf /tmp/...` for legitimate
        // local temp-folder cleanup without loosening the general root-path
        // rm -rf block (which the broad "rm -rf /" pattern below restores).
        let is_scoped_tmp_cleanup =
            normalized.contains("rm -rf /tmp/") && !normalized.contains("rm -rf /tmp/../");

        for pattern in dangerous_patterns.iter() {
            if normalized.contains(pattern) {
                if *pattern == "rm -rf /" && is_scoped_tmp_cleanup {
                    continue;
                }
                return Err(format!(
                    "SANDBOX_VIOLATION: Threat pattern '{}' detected in {}.",
                    pattern, action_type
                ));
            }
        }

        if normalized.contains("../../../") || normalized.contains("/etc/passwd") {
            return Err("SANDBOX_VIOLATION: Directory traversal attempt blocked.".to_string());
        }

        if (normalized.contains("| bash")
            || normalized.contains("| sh")
            || normalized.contains("|bash")
            || normalized.contains("|sh"))
            && !normalized.starts_with("echo")
        {
            return Err("SANDBOX_VIOLATION: Pipe-to-shell pattern blocked.".to_string());
        }

        Ok(())
    }

    pub fn evaluate_entropy_delta(base_entropy: f32, new_entropy: f32) -> Result<(), String> {
        let delta = (new_entropy - base_entropy).abs();
        if delta > 0.5 {
            return Err(format!("ENTROPY_ANOMALY: Delta of {:.4} exceeds 0.5 threshold.", delta));
        }
        Ok(())
    }

    pub fn resolve_consensus(alpha_vector: &[f32], beta_vector: &[f32]) -> bool {
        if alpha_vector.len() != beta_vector.len() {
            return false;
        }
        let diff: f32 =
            alpha_vector.iter().zip(beta_vector.iter()).map(|(a, b)| (a - b).abs()).sum();
        diff < 0.5
    }

    pub fn bft_consensus(vectors: &[Vec<f32>], epsilon: f32) -> Option<Vec<f32>> {
        if vectors.is_empty() {
            return None;
        }
        let n = vectors.len();
        let threshold = (2 * n / 3) + 1;
        let dim = vectors[0].len();
        if vectors.iter().any(|v| v.len() != dim) {
            return None;
        }

        let (best_idx, neighbor_count) = (0..n)
            .map(|i| {
                let neighbors = (0..n)
                    .filter(|&j| {
                        vectors[i]
                            .iter()
                            .zip(vectors[j].iter())
                            .map(|(a, b)| (a - b).abs())
                            .sum::<f32>()
                            < epsilon
                    })
                    .count();
                (i, neighbors)
            })
            .max_by_key(|&(_, count)| count)?;

        if neighbor_count < threshold {
            return None;
        }

        let agreeing: Vec<&Vec<f32>> = (0..n)
            .filter(|&j| {
                vectors[best_idx]
                    .iter()
                    .zip(vectors[j].iter())
                    .map(|(a, b)| (a - b).abs())
                    .sum::<f32>()
                    < epsilon
            })
            .map(|j| &vectors[j])
            .collect();

        let centroid: Vec<f32> = (0..dim)
            .map(|k| agreeing.iter().map(|v| v[k]).sum::<f32>() / agreeing.len() as f32)
            .collect();

        Some(centroid)
    }
}

use anyhow::{anyhow, Result};
use std::fs;
use tree_sitter::{Parser, Query, QueryCursor};
pub struct AstEditor;
impl AstEditor {
    /// update_function: Finds a function by name and replaces its entire body.
    /// This is the "Semantic Precision" upgrade to replace simple string diffs.
    pub fn read_function(path: &str, function_name: &str) -> Result<String> {
        let source_code = fs::read_to_string(path)?;
        let mut parser = Parser::new();
        parser.set_language(&tree_sitter_rust::LANGUAGE.into())?;
        let tree = parser
            .parse(&source_code, None)
            .ok_or_else(|| anyhow!("AST_PARSE_FAIL: Could not parse {}", path))?;
        let query_str = format!(
            "(function_item name: (identifier) @name (#eq? @name \"{}\")) @func",
            function_name
        );
        let query = Query::new(&tree_sitter_rust::LANGUAGE.into(), &query_str)?;
        let mut cursor = QueryCursor::new();
        let matches = cursor.matches(&query, tree.root_node(), source_code.as_bytes());
        for m in matches {
            for capture in m.captures {
                if capture.index == 1 {
                    let node = capture.node;
                    let start = node.start_byte();
                    let end = node.end_byte();
                    return Ok(source_code[start..end].to_string());
                }
            }
        }
        Err(anyhow!("FUNCTION_NOT_FOUND: Could not find function '{}' in {}", function_name, path))
    }

    /// Validates that `new_body` is exactly one function definition named
    /// `function_name` and nothing else -- no top-level `use` statements, no
    /// extra functions, no `mod` blocks alongside it. Without this, a
    /// Strategist proposal whose `new_body` accidentally (or confidently)
    /// contains a whole re-typed file gets spliced in verbatim: real
    /// incident -- new_body containing `use axum::{...}`, a duplicate
    /// `#[tokio::main]`, and a `mod tests` block all landed inside what was
    /// meant to be just the function body, producing duplicate-import
    /// compile errors. A naive "does it parse" check does not catch this,
    /// since each individual piece (a use statement, a second `fn main`) is
    /// syntactically valid Rust on its own -- duplicate-definition is a
    /// semantic error tree-sitter's grammar-only parser never sees. This
    /// check instead requires new_body to parse as a standalone fragment
    /// containing exactly one named top-level item, and that item must be a
    /// function_item matching function_name.
    fn validate_single_function_body(new_body: &str, function_name: &str) -> Result<()> {
        let mut parser = Parser::new();
        parser.set_language(&tree_sitter_rust::LANGUAGE.into())?;
        let fragment_tree = parser.parse(new_body, None).ok_or_else(|| {
            anyhow!(
                "PATCH_VALIDATION_FAILED: new_body for '{}' could not be parsed as Rust",
                function_name
            )
        })?;

        let root = fragment_tree.root_node();
        let mut walker = root.walk();
        let mut named_children = root.named_children(&mut walker);
        let first = named_children.next().ok_or_else(|| {
            anyhow!(
                "PATCH_VALIDATION_FAILED: new_body for '{}' is empty or contains no recognizable Rust item",
                function_name
            )
        })?;

        if named_children.next().is_some() {
            return Err(anyhow!(
                "PATCH_VALIDATION_FAILED: new_body for '{}' contains more than one top-level item \
                 (e.g. extra use statements, functions, or a mod block splice into the middle of a \
                 single-function patch) -- new_body must contain exactly one function definition",
                function_name
            ));
        }

        if first.kind() != "function_item" {
            return Err(anyhow!(
                "PATCH_VALIDATION_FAILED: new_body for '{}' is a '{}', not a function_item -- \
                 patch_function only replaces function bodies, not other item kinds",
                function_name,
                first.kind()
            ));
        }

        let name_node = first.child_by_field_name("name").ok_or_else(|| {
            anyhow!("PATCH_VALIDATION_FAILED: could not read the name of the function in new_body")
        })?;
        let actual_name = &new_body[name_node.start_byte()..name_node.end_byte()];
        if actual_name != function_name {
            return Err(anyhow!(
                "PATCH_VALIDATION_FAILED: new_body defines function '{}' but the patch target is '{}' -- \
                 refusing to rename a function via patch_function",
                actual_name,
                function_name
            ));
        }

        Ok(())
    }

    pub fn update_function(path: &str, function_name: &str, new_body: &str) -> Result<String> {
        Self::validate_single_function_body(new_body, function_name)?;

        let source_code = fs::read_to_string(path)?;
        let mut parser = Parser::new();
        parser.set_language(&tree_sitter_rust::LANGUAGE.into())?;
        let tree = parser
            .parse(&source_code, None)
            .ok_or_else(|| anyhow!("AST_PARSE_FAIL: Could not parse {}", path))?;
        // Query to find the specific function node
        let query_str = format!(
            "(function_item name: (identifier) @name (#eq? @name \"{}\")) @func",
            function_name
        );
        let query = Query::new(&tree_sitter_rust::LANGUAGE.into(), &query_str)?;
        let mut cursor = QueryCursor::new();
        let matches = cursor.matches(&query, tree.root_node(), source_code.as_bytes());
        for m in matches {
            for capture in m.captures {
                if capture.index == 1 {
                    // The @func capture
                    let node = capture.node;
                    let start = node.start_byte();
                    let end = node.end_byte();
                    let mut new_code = source_code.clone();
                    new_code.replace_range(start..end, new_body);

                    // Safety net beyond the standalone-fragment check above:
                    // confirm the *spliced* result still parses cleanly.
                    // Catches cases the isolated fragment check can't see,
                    // e.g. new_body being syntactically fine alone but
                    // producing a parse error once merged with surrounding
                    // context (mismatched braces at the splice boundary).
                    let mut verify_parser = Parser::new();
                    verify_parser.set_language(&tree_sitter_rust::LANGUAGE.into())?;
                    if let Some(verify_tree) = verify_parser.parse(&new_code, None) {
                        if verify_tree.root_node().has_error() {
                            return Err(anyhow!(
                                "PATCH_VALIDATION_FAILED: applying new_body to '{}' would leave {} with a Rust syntax error",
                                function_name,
                                path
                            ));
                        }
                    }

                    return Ok(new_code);
                }
            }
        }
        Err(anyhow!("FUNCTION_NOT_FOUND: Could not find function '{}' in {}", function_name, path))
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BeliefDelta {
    pub previous_confidence: f32,

    pub evidence_strength: f32,

    pub failure_penalty: f32,

    pub reflection_gain: f32,

    pub new_confidence: f32,

    pub explanation: String,
}

impl BeliefDelta {
    pub fn calculate(
        previous: f32,
        evidence: f32,
        failure: f32,
        reflection: f32,
        explanation: String,
    ) -> Self {
        let updated = (previous + evidence - failure + reflection).clamp(0.0, 1.0);

        Self {
            previous_confidence: previous,
            evidence_strength: evidence,
            failure_penalty: failure,
            reflection_gain: reflection,
            new_confidence: updated,
            explanation,
        }
    }
}

use crate::evidence::BeliefVector;
use serde::{Deserialize, Serialize};
use uuid::Uuid;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BeliefState {
    pub id: Uuid,
    pub proposition: String,
    pub confidence: f32,
    pub evidence_count: usize,
    pub stability: f32,
    /// Rich multidimensional view (canonical path going forward).
    pub vector: BeliefVector,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BeliefEvidence {
    pub source: String,
    pub observation: String,
    pub reliability: f32,
}

pub struct BeliefEngine;

impl BeliefEngine {
    pub fn new() -> Self {
        Self
    }

    /// Compatibility entry point – routes through BeliefVector::bayesian_update.
    pub fn update(&self, current: Option<BeliefState>, evidence: BeliefEvidence) -> BeliefState {
        let likelihood = evidence.reliability.clamp(0.0, 1.0);
        match current {
            Some(mut belief) => {
                // Canonical update
                belief.vector = belief.vector.bayesian_update(likelihood, 0.85);
                // Keep scalar fields in sync for legacy callers
                belief.confidence = belief.vector.scalar_score();
                belief.evidence_count += 1;
                belief.stability = (belief.stability + 0.05).min(1.0);
                belief
            }
            None => {
                let vector = BeliefVector::default().bayesian_update(likelihood, 0.85);
                BeliefState {
                    id: Uuid::new_v4(),
                    proposition: evidence.observation,
                    confidence: vector.scalar_score(),
                    evidence_count: 1,
                    stability: 0.5,
                    vector,
                }
            }
        }
    }
}

impl Default for BeliefEngine {
    fn default() -> Self {
        Self::new()
    }
}

impl BeliefState {
    pub fn update_confidence(&mut self, delta: f32) {
        self.confidence = (self.confidence + delta).clamp(0.0, 1.0);
    }

    pub fn apply_delta(&mut self, delta: crate::belief_delta::BeliefDelta) {
        self.confidence = delta.new_confidence;
        // Keep vector roughly in sync
        self.vector.implementation_confidence = delta.new_confidence;
    }

    /// Hard gate – prefer this over raw confidence checks.
    pub fn may_prove(&self, required_artifacts: u8, threshold: f32) -> bool {
        self.vector.may_prove(required_artifacts, threshold)
    }
}

    Adapt,

    Abandon,

pub struct BeliefDecision {
    pub id: Uuid,

    pub belief_id: Uuid,

    pub confidence: f32,

    pub action: BeliefAction,

    pub reason: String,

    pub timestamp: i64,
}

pub struct BeliefDecisionPolicy;

impl BeliefDecisionPolicy {
    pub fn evaluate(belief_id: Uuid, confidence: f32) -> BeliefDecision {
        let (action, reason) = if confidence >= 0.75 {
            (BeliefAction::Continue, "Confidence sufficient to continue objective".to_string())
        } else if confidence >= 0.45 {
            (
                BeliefAction::Adapt,
                "Confidence degraded, strategy adjustment recommended".to_string(),
            )
        } else if confidence >= 0.20 {
            (BeliefAction::Suspend, "Confidence too low, suspend objective".to_string())
        } else {
            (BeliefAction::Abandon, "Evidence no longer supports objective".to_string())
        };

        Self::decision(belief_id, confidence, action, reason)
    }

    fn decision(
        belief_id: Uuid,
        confidence: f32,
        action: BeliefAction,
        reason: String,
    ) -> BeliefDecision {
        BeliefDecision {
            id: Uuid::new_v4(),

            belief_id,

            confidence,

            action,

            reason,

            timestamp: chrono::Utc::now().timestamp(),
        }
    }
}

}

        node.last_seen = now_secs();
        let mut guard = self.nodes.write();
        if let Some(existing) = guard.iter_mut().find(|n| n.id == node.id) {
            *existing = node;
        } else {
            guard.push(node);
        }
    }

        self.nodes.read().clone()
    }

        let now = now_secs();
        self.nodes.read().iter().filter(|n| now.saturating_sub(n.last_seen) < 40).cloned().min_by(
            |a, b| {
                (a.cpu + a.ram).partial_cmp(&(b.cpu + b.ram)).unwrap_or(std::cmp::Ordering::Equal)
            },
        )
    }

use parking_lot::RwLock;
use std::collections::HashMap;
use std::sync::Arc;
use std::time::{SystemTime, UNIX_EPOCH};

#[derive(Clone, Debug)]
pub struct WeightedFact {
    pub confidence: f32,
    pub last_updated: u64,
}

/// V5 BeliefEngine: Maintains persistent environment facts (e.g., "gemini_api_healthy")
/// with automatic temporal decay for staleness.
#[derive(Clone, Default)]
pub struct BeliefEngine {
    facts: Arc<RwLock<HashMap<String, WeightedFact>>>,
}

impl BeliefEngine {
    pub fn new() -> Self {
        Self { facts: Arc::new(RwLock::new(HashMap::new())) }
    }

    /// Modifies a belief based on real-world evidence.
    /// Applies decay calculation before adjustment.
    pub fn update_belief(&self, key: &str, delta: f32) -> f32 {
        let mut lock = self.facts.write();
        let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs();
        let fact = lock
            .entry(key.to_string())
            .or_insert(WeightedFact { confidence: 0.5, last_updated: now });

        let elapsed_hours = now.saturating_sub(fact.last_updated) as f32 / 3600.0;
        let decay = elapsed_hours * 0.05; // 5% decay per hour of staleness

        fact.confidence = (fact.confidence - decay + delta).clamp(0.0, 1.0);
        fact.last_updated = now;
        fact.confidence
    }

    /// Retrieves current confidence of a belief, factoring in decay since last update.
    pub fn get_belief(&self, key: &str) -> f32 {
        let lock = self.facts.read();
        if let Some(fact) = lock.get(key) {
            let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs();
            let elapsed_hours = now.saturating_sub(fact.last_updated) as f32 / 3600.0;
            let decay = elapsed_hours * 0.05;
            (fact.confidence - decay).clamp(0.0, 1.0)
        } else {
            0.5 // Unknown facts rest at neutral parity
        }
    }
}

}

impl EpistemicSession {
    pub fn new(mission: &str) -> Self {
        Self {
            session_id: Uuid::new_v4(),
            mission: mission.to_string(),
            contract: SubgoalCompletionEngine::infer_contract("session_node", mission),
            budget: CognitiveBudget::default_for_node(),
            graph: CognitiveGraph::new(),
            local_evidence: String::new(),
            executed_actions: HashSet::new(),
            max_observed_belief: 0.0,
            best_hypothesis_id: None,
            consecutive_mutation_vetoes: 0,
        }
    }

    pub fn apply_event(&mut self, event: &CognitiveEvent) {
        match event {
            CognitiveEvent::HypothesisCreated { hypothesis_id, hypothesis, .. } => {
                self.graph.add_node(NodeKind::Hypothesis, hypothesis, 0.5, 100);
                if self.best_hypothesis_id.is_none() {
                    self.best_hypothesis_id = Some(*hypothesis_id);
                }
            }
            CognitiveEvent::ObservationRecorded { evidence, modifier, .. } => {
                self.local_evidence.push_str(evidence);
                self.local_evidence.push('\n');

                if let Some(h_id) = self.best_hypothesis_id {
                    let obs_id = self.graph.add_node(NodeKind::Observation, evidence, 1.0, 50);
                    let relation = if *modifier > 0.0 {
                        EdgeRelation::Supports
                    } else {
                        EdgeRelation::Contradicts
                    };
                    self.graph.add_edge(obs_id, h_id, relation, modifier.abs());
                    self.graph.propagate_beliefs();

                    if let Some(node) = self.graph.nodes.get(&h_id) {
                        if node.belief_score > self.max_observed_belief {
                            self.max_observed_belief = node.belief_score;
                        }
                    }
                }
            }
            CognitiveEvent::BeliefUpdated { hypothesis_id, new_belief, .. } => {
                if let Some(node) = self.graph.nodes.get_mut(hypothesis_id) {
                    node.belief_score = *new_belief;
                    if *new_belief > self.max_observed_belief {
                        self.max_observed_belief = *new_belief;
                    }
                }
            }
            _ => {}
        }
    }

    pub fn record_action_attempt(&mut self, action_key: &str) {
        self.executed_actions.insert(action_key.to_string());
    }

    pub fn is_action_attempted(&self, action_key: &str) -> bool {
        self.executed_actions.contains(action_key)
    }

            modifier: eval.confidence_modifier,
        };

        self.apply_event(&event);
    }

    pub fn is_subgoal_satisfied(&self, current_belief: f32) -> bool {
        if current_belief < 0.85 {
            return false;
        }
        SubgoalCompletionEngine::evaluate(&self.contract, &self.local_evidence, current_belief)
    }

    pub fn collect_evidence(&self, h_id: Uuid) -> String {

                }
            }

        } else {

        }
    }

/// Pluggable history backend (Vec today, SQLite/RocksDB later).
#[async_trait]
pub trait EventStore: Send + Sync {
    async fn append(&self, event: &CognitiveEvent);
    async fn recent(&self, n: usize) -> Vec<CognitiveEvent>;
}

#[derive(Default)]
pub struct InMemoryEventStore {
    events: RwLock<Vec<CognitiveEvent>>,
    max: usize,
}

impl InMemoryEventStore {
    pub fn new(max: usize) -> Self {
        Self { events: RwLock::new(Vec::new()), max }
    }
}

#[async_trait]
impl EventStore for InMemoryEventStore {
    async fn append(&self, event: &CognitiveEvent) {
        let mut guard = self.events.write();
        guard.push(event.clone());
        if guard.len() > self.max {
            let excess = guard.len() - self.max;
            guard.drain(0..excess);
        }
    }

    async fn recent(&self, n: usize) -> Vec<CognitiveEvent> {
        let guard = self.events.read();
        guard.iter().rev().take(n).cloned().collect()
    }
}

#[derive(Clone)]
pub struct EventBus {
    tx: broadcast::Sender<CognitiveEvent>,
    store: Arc<dyn EventStore>,
}

impl EventBus {
    pub fn new() -> Self {
        let (tx, _) = broadcast::channel(256);
        Self { tx, store: Arc::new(InMemoryEventStore::new(256)) }
    }

    pub fn with_store(store: Arc<dyn EventStore>) -> Self {
        let (tx, _) = broadcast::channel(256);
        Self { tx, store }
    }

    pub fn subscribe(&self) -> broadcast::Receiver<CognitiveEvent> {
        self.tx.subscribe()
    }

    pub async fn publish(&self, event: CognitiveEvent) {
        self.store.append(&event).await;
        // Ignore lagging receivers – cognitive runtime must not block on slow listeners.
        let _ = self.tx.send(event);
    }

    pub async fn recent(&self, n: usize) -> Vec<CognitiveEvent> {
        self.store.recent(n).await
    }
}

impl Default for EventBus {
    fn default() -> Self {
        Self::new()
    }

//! Executive Cortex – Phase 8B v2
//! Staged decision pipeline:
//!   observe → evaluate_policy → evaluate_goals → evaluate_risk → choose → emit

use super::event_bus::{CognitiveEvent, CognitiveEventKind, EventBus, EventMeta};
use super::policy_engine::PolicyEngine;
use super::traits::*;
use std::time::{SystemTime, UNIX_EPOCH};
use uuid::Uuid;

#[derive(Clone)]
pub struct ExecutiveCortex {
    bus: EventBus,
    policy: PolicyEngine,
}

#[derive(Debug, Default)]
struct Evaluation {
    policy_ok: bool,
    goal_pressure: f32,   // 0 = none, 1 = urgent
    risk: f32,            // 0 = safe, 1 = critical
    resource_pressure: f32,
}

impl ExecutiveCortex {
    pub fn new(bus: EventBus) -> Self {
        Self {
            bus,
            policy: PolicyEngine::new(),
        }
    }

    pub fn with_policy(bus: EventBus, policy: PolicyEngine) -> Self {
        Self { bus, policy }
    }

    // ── Pipeline stages ────────────────────────────────────────────────────

    fn observe(&self, state: &MissionState) -> Evaluation {
        let mut ev = Evaluation::default();
        ev.policy_ok = true; // policy checked against actions, not mission state
        ev.risk = 1.0 - state.health.clamp(0.0, 1.0);
        ev.resource_pressure = 0.0; // filled by scheduler later
        ev.goal_pressure = state.priority.clamp(0.0, 1.0);
        ev
    }

    fn evaluate_policy(&self, ev: &mut Evaluation, _state: &MissionState) {
        // Placeholder: mission-level policy always passes in 8B.
        // Action-level checks happen via PolicyEngine::check_action.
        ev.policy_ok = true;
    }

    fn evaluate_goals(&self, ev: &mut Evaluation, state: &MissionState) {
        // Higher priority + active status → higher pressure to continue.
        if state.status == MissionStatus::Active {
            ev.goal_pressure = state.priority.clamp(0.0, 1.0);
        } else {
            ev.goal_pressure = 0.0;
        }
    }

    fn evaluate_risk(&self, ev: &mut Evaluation, state: &MissionState) {
        ev.risk = 1.0 - state.health.clamp(0.0, 1.0);
    }

    fn choose(&self, ev: &Evaluation, state: &MissionState) -> (ExecutiveDecisionKind, String, f32) {
        match state.status {
            MissionStatus::Completed | MissionStatus::Failed | MissionStatus::Aborted => {
                (ExecutiveDecisionKind::Reflect, "mission terminal".into(), 0.95)
            }
            MissionStatus::Paused => {
                (ExecutiveDecisionKind::Continue, "resume paused mission".into(), 0.7)
            }
            MissionStatus::Active => {
                if !ev.policy_ok {
                    (ExecutiveDecisionKind::Abort, "policy violation".into(), 0.9)
                } else if ev.risk > 0.75 {
                    (
                        ExecutiveDecisionKind::Abort,
                        format!("risk critical ({:.2})", ev.risk),
                        0.9,
                    )
                } else if ev.risk > 0.5 {
                    (
                        ExecutiveDecisionKind::Replan,
                        format!("risk elevated ({:.2})", ev.risk),
                        0.8,
                    )
                } else if ev.resource_pressure > 0.8 {
                    (
                        ExecutiveDecisionKind::Pause,
                        "resource pressure high".into(),
                        0.75,
                    )
                } else {
                    (ExecutiveDecisionKind::Continue, "mission healthy".into(), 0.85)
                }
            }
        }
    }

    // ── Public API ─────────────────────────────────────────────────────────

    pub async fn decide(&self, state: &MissionState) -> ExecutiveDecision {
        let mut ev = self.observe(state);
        self.evaluate_policy(&mut ev, state);
        self.evaluate_goals(&mut ev, state);
        self.evaluate_risk(&mut ev, state);
        let (kind, rationale, confidence) = self.choose(&ev, state);

        let decision = ExecutiveDecision {
            kind,
            reason: DecisionReason {
                confidence,
                supporting_evidence: vec![
                    format!("health={:.2}", state.health),
                    format!("risk={:.2}", ev.risk),
                    format!("goal_pressure={:.2}", ev.goal_pressure),
                ],
                policy_trigger: Some("phase8b_v2_pipeline".into()),
                expected_cost: 0.0,
                rationale,
            },
        };

        self.bus
            .publish(CognitiveEvent {
                meta: EventMeta::new("executive_cortex"),
                mission_id: Some(state.id),
                kind: CognitiveEventKind::DecisionMade {
                    decision_summary: format!("{:?}", decision.kind),
                },
            })
            .await;

        decision
    }

    pub fn start_mission(&self, directive: &str, priority: f32) -> MissionState {
        MissionState {
            id: Uuid::new_v4(),
            directive: directive.to_string(),
            priority,
            health: 1.0,
            status: MissionStatus::Active,
            intent: Some(directive.to_string()),
        }
    }

    pub fn policy(&self) -> &PolicyEngine {
        &self.policy
    }
}

#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub enum NodeKind {
    Observation,
    Verification,
    Reflection,
    Hypothesis,
    ProposedAction { action_json: String },

}

}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CognitiveNode {
    pub id: Uuid,
    pub kind: NodeKind,
    pub content: String,
    pub belief_score: f32,
    pub computation_cost: u32,
    pub is_active: bool,
}

#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub enum EdgeRelation {
    Supports,
    Contradicts,
    Prerequisite,
    DerivesFrom,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GraphEdge {
    pub source: Uuid,
    pub target: Uuid,
    pub relation: EdgeRelation,
    pub weight: f32,
}

#[derive(Debug, Clone, Default)]
pub struct CognitiveGraph {
    pub nodes: HashMap<Uuid, CognitiveNode>,
    pub edges: Vec<GraphEdge>,
    pub frontiers: HashSet<Uuid>,
}

impl CognitiveGraph {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn add_node(
        &mut self,
        kind: NodeKind,
        content: &str,
        initial_belief: f32,
        cost: u32,
    ) -> Uuid {
        let id = Uuid::new_v4();
        let node = CognitiveNode {
            id,
            kind,
            content: content.to_string(),
            belief_score: initial_belief,
            computation_cost: cost,
            is_active: false,
        };
        self.nodes.insert(id, node);
        self.frontiers.insert(id);
        id
    }

    pub fn add_edge(&mut self, source: Uuid, target: Uuid, relation: EdgeRelation, weight: f32) {
        self.edges.push(GraphEdge { source, target, relation, weight });
    }

    pub fn propagate_beliefs(&mut self) {
        // Accumulate signed impacts first so multiple edges targeting the same
        // hypothesis are combined instead of overwriting each other.
        let mut delta: HashMap<Uuid, f32> = HashMap::new();
        const DAMPING: f32 = 0.85;

        for edge in &self.edges {
            let Some(source_node) = self.nodes.get(&edge.source) else {
                continue;
            };
            let impact = source_node.belief_score * edge.weight;
            let signed = match edge.relation {
                EdgeRelation::Supports => impact,
                EdgeRelation::Contradicts => -impact,
                // Prerequisite / DerivesFrom intentionally ignored for score updates
                _ => continue,
            };
            *delta.entry(edge.target).or_insert(0.0) += signed;
        }

        for (id, d) in delta {
            if let Some(node) = self.nodes.get_mut(&id) {
                let candidate = (node.belief_score + d).clamp(0.0, 1.0);
                // Keep a fraction of the previous score (PageRank-style damping)
                node.belief_score = DAMPING * candidate + (1.0 - DAMPING) * node.belief_score;
            }
        }
    }

    pub fn select_best_hypothesis(&self) -> Option<&CognitiveNode> {
        self.nodes
            .values()
            // ⚡ FIX: Filter out dead hypotheses (belief < 0.15).
            // v1.9: When BeliefVector is available on the node, prefer may_prove() over raw score.
            .filter(|n| matches!(n.kind, NodeKind::Hypothesis) && n.belief_score >= 0.15)
            .max_by(|a, b| {
                let score_a = a.belief_score / (a.computation_cost as f32).max(1.0);
                let score_b = b.belief_score / (b.computation_cost as f32).max(1.0);
                score_a.partial_cmp(&score_b).unwrap_or(std::cmp::Ordering::Equal)
            })
    }
}

//! Policy Engine – Phase 8B v2
//! Capability-based authorization.

use std::collections::HashSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum Capability {
    FilesystemRead,
    FilesystemWrite,
    Network,
    Shell,
    Patch,
    Delete,
    MemoryWrite,
    SpawnSubagent,
}

#[derive(Debug, Clone)]
pub struct PolicyEngine {
    allowed: HashSet<Capability>,
}

impl Default for PolicyEngine {
    fn default() -> Self {
        let mut allowed = HashSet::new();
        allowed.insert(Capability::FilesystemRead);
        allowed.insert(Capability::FilesystemWrite);
        allowed.insert(Capability::Shell);
        allowed.insert(Capability::Patch);
        allowed.insert(Capability::MemoryWrite);
        // Delete and unrestricted Network denied by default
        Self { allowed }
    }
}

impl PolicyEngine {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn allows(&self, cap: Capability) -> bool {
        self.allowed.contains(&cap)
    }

    pub fn grant(&mut self, cap: Capability) {
        self.allowed.insert(cap);
    }

    pub fn revoke(&mut self, cap: Capability) {
        self.allowed.remove(&cap);
    }

    /// Map a raw action string to required capabilities (best-effort).
    pub fn required_capabilities(action: &str) -> Vec<Capability> {
        let lower = action.to_lowercase();
        let mut caps = Vec::new();
        if lower.contains("write_file") || lower.contains("diff_patch") || lower.contains("patch_function") {
            caps.push(Capability::FilesystemWrite);
            caps.push(Capability::Patch);
        }
        if lower.contains("read_file") || lower.contains("search_code") || lower.contains("read_file_chunk") {
            caps.push(Capability::FilesystemRead);
        }
        if lower.contains("execute_command") || lower.starts_with("rg ") || lower.starts_with("find ") {
            caps.push(Capability::Shell);
        }
        if lower.contains("http") || lower.contains("curl") || lower.contains("wget") {
            caps.push(Capability::Network);
        }
        if lower.contains("rm ") || lower.contains("delete") {
            caps.push(Capability::Delete);
        }
        if lower.contains("spawn") || lower.contains("delegate") {
            caps.push(Capability::SpawnSubagent);
        }
        if caps.is_empty() {
            caps.push(Capability::Shell); // conservative default
        }
        caps
    }

    pub fn check_action(&self, action: &str) -> bool {
        Self::required_capabilities(action)
            .into_iter()
            .all(|c| self.allows(c))
    }
}

use super::traits::{Prediction, PredictionEvaluation};
use parking_lot::RwLock;
use std::collections::HashMap;
use std::sync::Arc;
use std::time::{SystemTime, UNIX_EPOCH};
use uuid::Uuid;

#[derive(Clone, Debug, Default)]
pub struct ActionHistory {
    pub success_count: u32,
    pub total_count: u32,
    pub avg_cost: f32,
}

/// V5 PredictionEngine: Operates a fast nearest-neighbor lookup on historical logs
/// using action-verb + target-extension signatures.
#[derive(Clone, Default)]
pub struct PredictionEngine {
    active_predictions: Arc<RwLock<HashMap<Uuid, Prediction>>>,
    history: Arc<RwLock<HashMap<String, ActionHistory>>>,
}

impl PredictionEngine {
    pub fn new() -> Self {
        Self {
            active_predictions: Arc::new(RwLock::new(HashMap::new())),
            history: Arc::new(RwLock::new(HashMap::new())),
        }
    }

    /// Generates a heuristic signature: e.g. "read_file src/main.rs" -> "read_file_rs"
    fn get_signature(action_str: &str) -> String {
        let parts: Vec<&str> = action_str.split_whitespace().collect();
        let verb = parts.first().unwrap_or(&"unknown");
        let ext = parts.get(1).and_then(|p| p.split('.').next_back()).unwrap_or("none");
        format!("{}_{}", verb, ext)
    }

    pub fn generate_prediction(&self, action_id: Uuid, action_str: &str) -> Prediction {
        let sig = Self::get_signature(action_str);
        let hist_lock = self.history.read();

        let (confidence, cost) = if let Some(h) = hist_lock.get(&sig) {
            let conf =
                if h.total_count > 0 { h.success_count as f32 / h.total_count as f32 } else { 0.5 };
            (conf, h.avg_cost)
        } else {
            (0.5, 0.1) // Baseline default
        };

        let ts = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs();
        let pred = Prediction {
            action_id,
            expected_state: sig, // We store the signature here to match against upon evaluation
            confidence,
            cost_estimate: cost,
            timestamp: ts,
        };

        self.active_predictions.write().insert(action_id, pred.clone());
        pred
    }

    pub fn evaluate_prediction(
        &self,
        action_id: Uuid,
        actual_cost: f32,
        success: bool,
    ) -> PredictionEvaluation {
        let mut active = self.active_predictions.write();
        if let Some(pred) = active.remove(&action_id) {
            let mut hist_lock = self.history.write();
            let h = hist_lock.entry(pred.expected_state.clone()).or_default();

            h.total_count += 1;
            if success {
                h.success_count += 1;
            }

            // Rolling average update
            h.avg_cost =
                (h.avg_cost * (h.total_count - 1) as f32 + actual_cost) / h.total_count as f32;

            let accuracy = if success == (pred.confidence >= 0.5) { 1.0 } else { 0.0 };
            let error = (actual_cost - pred.cost_estimate).abs();

            PredictionEvaluation { accuracy, error }
        } else {
            PredictionEvaluation { accuracy: 0.0, error: 0.0 }
        }
    }
}

//! Working Memory – Phase 8B v2
//! Partitioned: facts / observations / reasoning / hypotheses.

use parking_lot::RwLock;
use std::collections::HashMap;
use std::sync::Arc;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum MemoryPartition {
    Facts,
    Observations,
    Reasoning,
    Hypotheses,
}

#[derive(Clone, Default)]
pub struct WorkingMemory {
    partitions: Arc<RwLock<HashMap<MemoryPartition, HashMap<String, String>>>>,
}

impl WorkingMemory {
    pub fn new() -> Self {
        let mut map = HashMap::new();
        for p in [
            MemoryPartition::Facts,
            MemoryPartition::Observations,
            MemoryPartition::Reasoning,
            MemoryPartition::Hypotheses,
        ] {
            map.insert(p, HashMap::new());
        }
        Self {
            partitions: Arc::new(RwLock::new(map)),
        }
    }

    pub fn store(&self, partition: MemoryPartition, key: &str, value: &str) {
        if let Some(slot) = self.partitions.write().get_mut(&partition) {
            slot.insert(key.to_string(), value.to_string());
        }
    }

    pub fn recall(&self, partition: MemoryPartition, key: &str) -> Option<String> {
        self.partitions
            .read()
            .get(&partition)
            .and_then(|s| s.get(key).cloned())
    }

    pub fn clear_partition(&self, partition: MemoryPartition) {
        if let Some(slot) = self.partitions.write().get_mut(&partition) {
            slot.clear();
        }
    }

    pub fn clear_all(&self) {
        for slot in self.partitions.write().values_mut() {
            slot.clear();
        }
    }

    pub fn snapshot(&self, partition: MemoryPartition) -> HashMap<String, String> {
        self.partitions
            .read()
            .get(&partition)
            .cloned()
            .unwrap_or_default()
    }
}

fn has_word(text: &str, word: &str) -> bool {
    text.split_whitespace()
        .any(|w| w.trim_matches(|c: char| !c.is_alphanumeric()) == word)
}

fn any_word(text: &str, words: &[&str]) -> bool {
    words.iter().any(|w| has_word(text, w))
}

fn has_word_unnegated(text: &str, word: &str) -> bool {
    let negations = ["not", "never", "no", "cannot", "n't"];
    let tokens: Vec<&str> = text.split_whitespace().collect();
    for (i, tok) in tokens.iter().enumerate() {
        let clean = tok.trim_matches(|c: char| !c.is_alphanumeric());
        if clean == word {
            let start = i.saturating_sub(3);
            let negated = tokens[start..i].iter().any(|w| {
                let wl = w.trim_matches(|c: char| !c.is_alphanumeric()).to_lowercase();
                negations.contains(&wl.as_str()) || wl.ends_with("n't")
            });
            if !negated {
                return true;
            }
        }
    }
    false
}

fn any_word_unnegated(text: &str, words: &[&str]) -> bool {
    words.iter().any(|w| has_word_unnegated(text, w))
}

impl SubgoalCompletionEngine {
    pub fn infer_contract(node_id: &str, objective: &str) -> SubgoalContract {
        let obj_lower = objective.trim().to_lowercase();
        let classify_text = obj_lower
            .strip_prefix("objective:")
            .or_else(|| obj_lower.strip_prefix("objective"))
            .unwrap_or(&obj_lower)
            .trim();

        let mut conditions = Vec::new();
        let mut is_mutation_task = false;

        let first_word = classify_text.split_whitespace().next().unwrap_or("");
        let is_recon_intent = [
            "read", "analyze", "inspect", "search", "examine", "locate", "profile", "benchmark",
        ]
        .contains(&first_word);

        if is_recon_intent {
            conditions.push(CompletionCondition::ReadMarkerAcquired);
            conditions.push(CompletionCondition::MinBeliefScore(0.65));
        } else if any_word_unnegated(
            classify_text,
            &["write", "export", "save", "implement", "integrate", "add", "create"],
        ) {
            conditions.push(CompletionCondition::WriteMarkerAcquired);
            is_mutation_task = true;

            if any_word(classify_text, &["roadmap", "audit", "comparison"]) {
                conditions.push(CompletionCondition::SourceCodeEvidenceRequired);
            }
        } else if any_word_unnegated(classify_text, &["patch", "apply", "fix", "remediate"]) {
            conditions.push(CompletionCondition::PatchMarkerAcquired);
            is_mutation_task = true;
        } else if any_word(classify_text, &["search", "find", "locate"]) {
            conditions.push(CompletionCondition::SearchMarkerAcquired);
            conditions.push(CompletionCondition::MinBeliefScore(0.65));
        } else if any_word(classify_text, &["read", "inspect", "audit"]) {
            conditions.push(CompletionCondition::ReadMarkerAcquired);
            conditions.push(CompletionCondition::MinBeliefScore(0.65));
        } else {
            conditions.push(CompletionCondition::CommandMarkerAcquired);
            conditions.push(CompletionCondition::MinBeliefScore(0.85));
        }

        SubgoalContract {
            node_id: node_id.to_string(),
            objective: objective.to_string(),
            conditions,
            is_mutation_task,
        }
    }

    pub fn evaluate(
        contract: &SubgoalContract, 
        local_evidence: &str, 
        belief_score: f32
    ) -> bool {
        // ⚡ DETERMINISTIC VERIFIER INTERFACE
        // Passes the contract conditions and belief score to the strict verifier
        let verifier_result = crate::outcome_verifier::OutcomeVerifier::verify(
            &contract.node_id, 
            local_evidence,
            belief_score,
            &contract.conditions
        );
        matches!(verifier_result, crate::outcome_verifier::CompletionStatus::Approved)
    }
}

use std::sync::atomic::Ordering;
use tokio::sync::broadcast;

const ESTIMATED_TOKENS: u32 = 2_000;

pub struct Conductor;

pub struct TurnContext<'a> {
    pub orch: &'a Orchestrator,
    pub client: &'a Client,
    pub api_key: &'a str,
    pub tx: Option<broadcast::Sender<String>>,
}

impl Conductor {
    pub async fn run_turn(
        profile: &AgentProfile,
        prompt: String,
        turn: usize,
        ctx: TurnContext<'_>,
        errors: usize,
    ) -> Result<String, String> {
        METRICS.api_calls.fetch_add(1, Ordering::Relaxed);

        let model = AgentFactory::get_model_for_turn(turn, errors);
        let _ = wait_for_quota(&model, ESTIMATED_TOKENS).await;

        let neural_ctx = ContextPacket {
            system_prompt: Some(profile.persona.clone()),
            model_override: Some(model.to_string()),
            ..Default::default()
        };

        if let Some(ref sender) = ctx.tx {
            let _ = sender.send(
                json!({"type": "NEURAL_THOUGHT", "payload": format!("[CONDUCTOR] turn {} → {}", turn, model)}).to_string()
            );
        }

        let result =
            ctx.orch.run(prompt.clone(), neural_ctx, ctx.client, ctx.api_key, ctx.tx.clone()).await;

        if let Err(ref e) = result {
            if e.contains("STALE")
                || e.contains("500")
                || e.contains("RESOURCE_EXHAUSTED")
                || e.contains("INTERNAL")
                || e.contains("CONTEXT_OVERFLOW")
            {
                METRICS.task_failures.fetch_add(1, Ordering::Relaxed);

                let failover = AgentFactory::failover_model(&model);
                tracing::warn!("[CONDUCTOR] {} failed → failing over to {}", model, failover);
                let _ = wait_for_quota(&failover, ESTIMATED_TOKENS).await;

                let fail_ctx = ContextPacket {
                    system_prompt: Some(profile.persona.clone()),
                    model_override: Some(failover.to_string()),
                    ..Default::default()
                };
                return ctx.orch.run(prompt, fail_ctx, ctx.client, ctx.api_key, ctx.tx).await;
            }
        }
        result
    }
}

#[derive(Debug, Clone, PartialEq)]
pub enum ContractStatus {
    Satisfied,
    RejectedMutation,
    PartialEvidence,
    NoEvidence,
}

pub struct ContractEvaluator;

impl ContractEvaluator {
    pub fn evaluate(
        is_mutation_task: bool,
        is_write_task: bool,
        local_evidence: &str,
        current_belief: f32,
        mission: &str,
    ) -> ContractStatus {
        let has_mutation_marker = local_evidence.lines().any(|l| {
            l.starts_with("[WRITE_OK]") || 
            l.starts_with("[PATCH_FUNCTION_OK]") || 
            l.starts_with("[REPLACE_LINES_OK]") ||
            l.starts_with("[DIFF_PATCH_OK]")
        });

        let has_recon_marker = local_evidence.lines().any(|l| {
            l.starts_with("[READ_CHUNK_OK]") || 
            l.starts_with("[SEARCH_CODE_OK]") || 
            l.starts_with("[READ_OK]") || 
            l.starts_with("[CMD_OK]")
        });

        if (is_mutation_task || is_write_task) && !has_mutation_marker {
            return if has_recon_marker { 
                ContractStatus::PartialEvidence 
            } else { 
                ContractStatus::NoEvidence 
            };
        }

        if Self::is_compound_objective_satisfied(mission, local_evidence) {
            if current_belief >= 0.95 {
                return ContractStatus::Satisfied;
            }
            return ContractStatus::PartialEvidence;
        }

        if has_mutation_marker || has_recon_marker {
            return if current_belief >= 0.95 {
                ContractStatus::Satisfied
            } else {
                ContractStatus::PartialEvidence
            };
        }

        ContractStatus::NoEvidence
    }

    fn is_compound_objective_satisfied(mission: &str, evidence: &str) -> bool {
        let mission_lower = mission.to_lowercase();
        let objective_part = mission_lower
            .lines()
            .find(|line| line.starts_with("objective:"))
            .unwrap_or(mission_lower.as_str());
            
        let evidence_lower = evidence.to_lowercase();
        
        let and_segments: Vec<&str> = objective_part
            .split(" and ")
            .map(|s| s.trim())
            .filter(|s| !s.is_empty())
            .collect();

        if and_segments.is_empty() {
            return false;
        }

        and_segments.iter().all(|segment| {
            let or_parts: Vec<&str> = segment
                .split(" or ")
                .map(|s| s.trim())
                .filter(|s| !s.is_empty())
                .collect();

            if or_parts.len() <= 1 {
                segment.split_whitespace()
                    .filter(|w| w.len() > 3)
                    .any(|w| evidence_lower.contains(w))
            } else {
                or_parts.iter().any(|part| {
                    part.split_whitespace()
                        .filter(|w| w.len() > 3)
                        .any(|w| evidence_lower.contains(w))
                })
            }
        })
    }
}

    pub async fn ignite_with_evidence(

        mission: &str,
        ancestor_evidence: Option<&str>,
    ) -> Result<CortexOutcome> {
        self.log("\x1b[1;36m>> [CORTEX] Igniting Epistemic Reasoning Session\x1b[0m".to_string());

                context_chunks.push(format!("\n[ANCESTOR NODE EVIDENCE]\n{}", evidence));
            }
        }
        let world_context = Self::compress_context(&context_chunks.join("\n"), 12000);

        let mut session = EpistemicSession::new(mission);

        self.log("\x1b[36m>> [CORTEX] Generating competing strategies...\x1b[0m".to_string());
        session.budget.record_call();
        let hypotheses = EpistemicEngine::generate_hypotheses(&self.client, &self.api_key, mission, &world_context).await?;

        for h in hypotheses {
            session.graph.add_node(NodeKind::Hypothesis, &h, 0.5, 100);
            self.log(format!("\x1b[90m  -> Strategy: {}\x1b[0m", h));
        }

        let is_mutation_task = session.contract.is_mutation_task;
        let is_write_task = is_mutation_task
            && (session.contract.objective.to_lowercase().contains("write")
                || session.contract.objective.to_lowercase().contains("export")
                || session.contract.objective.to_lowercase().contains(".md")
                || session.contract.objective.to_lowercase().contains(".json"));

        let mut state = CortexState::Strategize;
        let mut turn = 0;

        loop {
            turn += 1;
            self.log(format!("\n\x1b[1;33m>> [CORTEX] Cognitive Turn {} / {}\x1b[0m", turn, session.budget.max_turns));

            if let Some(best_node) = session.graph.select_best_hypothesis().cloned() {
                let current_belief = best_node.belief_score;
                if session.budget.is_exhausted(turn) {
                    if current_belief >= 0.55 && session.consecutive_mutation_vetoes < 3 && turn < 300 {
                        self.log(format!("\x1b[1;35m>> [COGNITIVE MOMENTUM] Call limit reached, but Belief is high ({:.2}). Dynamically extending budget.\x1b[0m", current_belief));
                        session.budget.max_turns += 20;
                        session.budget.max_model_calls += 80;
                    } else {
                        self.log("\x1b[1;31m>> [SUBGOAL_BUDGET_EXHAUSTED] Cognitive call limit reached with low belief or stagnation. Node failed.\x1b[0m".to_string());
                        return Ok(CortexOutcome::TurnsExhausted { best_belief: current_belief, evidence: session.collect_evidence(best_node.id) });
                    }
                }
            } else if session.budget.is_exhausted(turn) {
                return Ok(CortexOutcome::TurnsExhausted { best_belief: 0.0, evidence: "No viable hypotheses left.".to_string() });
            }

            state = match state {
                CortexState::Strategize => self.handle_strategize(&mut session, mission, &world_context, is_mutation_task, is_write_task).await?,
                CortexState::Critique { proposal, belief } => self.handle_critic(&mut session, mission, &world_context, proposal, belief, is_mutation_task, is_write_task).await?,
                CortexState::Translate { proposal, belief } => self.handle_translate(&mut session, proposal, belief).await?,
                CortexState::Execute { actions, proposal, belief } => self.handle_execute(&mut session, actions, proposal, belief).await?,
                CortexState::Evaluate { evidence, proposal, belief, is_halt_action, mutation_path } => self.handle_evaluate(&mut session, mission, evidence, proposal, belief, is_mutation_task, is_write_task, is_halt_action, mutation_path).await?,
                CortexState::Complete { belief, evidence } => return Ok(CortexOutcome::CompletionProposed { belief, evidence }),
                CortexState::Fail { belief, evidence } => return Ok(CortexOutcome::TurnsExhausted { best_belief: belief, evidence }),
            };
        }
    }

    async fn handle_strategize(&mut self, session: &mut EpistemicSession, mission: &str, world_context: &str, is_mutation_task: bool, is_write_task: bool) -> Result<CortexState> {
        let best_node = match session.graph.select_best_hypothesis().cloned() {
            Some(node) => node,
            None => return Ok(CortexState::Fail { belief: 0.0, evidence: "No viable hypotheses left.".to_string() })
        };

        let h_id = best_node.id;
        let h_content = best_node.content.clone();
        let current_belief = best_node.belief_score;

        self.log(format!("\x1b[32m>> [CORTEX] Exploring Top Strategy (Belief: {:.2}):\x1b[0m {}", current_belief, h_content));

        let status = ContractEvaluator::evaluate(is_mutation_task, is_write_task, &session.local_evidence, current_belief, mission);

        if status == ContractStatus::Satisfied {
            self.log("\x1b[1;36m>> [CORTEX OMNI-GUARD] Contract mathematically satisfied. Stripping LLM authority and forcing deterministic node completion.\x1b[0m".to_string());
            return Ok(CortexState::Complete { belief: current_belief, evidence: session.collect_evidence(h_id) });
        }

        let evidence_history = Self::compress_context(&session.collect_evidence(h_id), 64000);
        session.budget.record_call();
        
        match EpistemicEngine::propose_action(&self.client, &self.api_key, mission, &h_content, world_context, &evidence_history).await {
            Ok(proposal) => {
                self.log(format!("\x1b[35m>> [STRATEGIST PROPOSAL]:\x1b[0m {}", proposal));
                let prop_lower = proposal.trim().to_lowercase();
                if prop_lower == "halt" || prop_lower.starts_with("halt ") {
                    if status == ContractStatus::Satisfied || status == ContractStatus::PartialEvidence {
                        self.log("\x1b[1;36m>> [CORTEX] DETERMINISTIC GUARD: Halt proposed following successful local evidence. Subtask complete.\x1b[0m".to_string());
                        return Ok(CortexState::Complete { belief: 1.0, evidence: session.collect_evidence(h_id) });
                    } else if (is_mutation_task || is_write_task) && status == ContractStatus::NoEvidence {
                        if current_belief > 0.90 {
                            self.log("\x1b[1;33m>> [CORTEX GUARD OVERRIDE] High belief (>.90) halt proposed without mutation. Allowing Evaluator to assess validity.\x1b[0m".to_string());
                        } else {
                            self.log("\x1b[1;31m>> [CORTEX FAULT] Halt rejected: Objective requires patching/writing, but no mutation success marker found.\x1b[0m".to_string());
                            let obs_id = session.graph.add_node(NodeKind::Observation, "[HALT_REJECTED] The directive requires modifying files. You proposed halt without successfully patching anything. You MUST use diff_patch, patch_function, or write_file.", 1.0, 50);
                            session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.6);
                            session.graph.propagate_beliefs();
                            return Ok(CortexState::Strategize);
                        }
                    } else {
                        self.log("\x1b[1;33m>> [CORTEX] DETERMINISTIC GUARD SUPPRESSED: Halt proposed but evidence covers only part of a compound objective; continuing to explore.\x1b[0m".to_string());
                    }
                }
                Ok(CortexState::Critique { proposal, belief: current_belief })
            }
            Err(e) => {
                self.log(format!("\x1b[31m>> [STRATEGIST FAULT]\x1b[0m {}", e));
                Ok(CortexState::Strategize)
            }
        }
    }

    async fn handle_critic(&mut self, session: &mut EpistemicSession, mission: &str, world_context: &str, proposal: String, belief: f32, is_mutation_task: bool, is_write_task: bool) -> Result<CortexState> {
        self.log("\x1b[31m>> [RED TEAM] Criticizing proposal...\x1b[0m".to_string());
        
        let action_key = proposal.trim().to_string();
        if session.is_action_attempted(&action_key) {
            let status = ContractEvaluator::evaluate(is_mutation_task, is_write_task, &session.local_evidence, belief, mission);
            if status == ContractStatus::Satisfied || status == ContractStatus::PartialEvidence {
                self.log("\x1b[1;36m>> [SUBGOAL_COMPLETE] Loop detected but evidence is sufficient. Halting.\x1b[0m".to_string());
                return Ok(CortexState::Complete { belief, evidence: session.collect_evidence(session.best_hypothesis_id.unwrap_or_default()) });
            } else {
                self.log(format!("\x1b[1;31m>> [ENTERPRISE CORTEX] Anti-Loop Guard Triggered: Action [{}] already attempted.\x1b[0m", action_key));
                let obs_id = session.graph.add_node(NodeKind::Observation, "[CRITICAL VETO] You are repeating a failed action! YOU MUST MUTATE YOUR STRATEGY, USE A DIFFERENT TOOL, OR HALT.", 1.0, 95);
                if let Some(h_id) = session.best_hypothesis_id {
                    session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.99);
                    session.graph.propagate_beliefs();
                }
                return Ok(CortexState::Strategize);
            }
        }
        session.record_action_attempt(&action_key);

        session.budget.record_call();
        match EpistemicEngine::criticize_proposal(&self.client, &self.api_key, mission, &proposal, world_context, &session.local_evidence, &session.local_evidence).await {
            Err(e) => {
                self.log(format!("\x1b[31m>> [CORTEX FATAL] Red Team critic parser panicked: {}\x1b[0m", e));
                if let Some(h_id) = session.best_hypothesis_id {
                    let obs_id = session.graph.add_node(NodeKind::Observation, "[CRITIC_PARSE_FAILURE]", 1.0, 50);
                    session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.15);
                    session.graph.propagate_beliefs();
                }
                Ok(CortexState::Strategize)
            }
            Ok(mut critic) => {
                self.log(format!("\x1b[31m>> [CRITIC] Verdict:\x1b[0m {} (Modifier: {:.2})", critic.critique, critic.confidence_modifier));

                let is_mutation_shaped_proposal = proposal.contains("write_file") || proposal.contains("patch_function") || proposal.contains("diff_patch") || proposal.contains("replace_lines");
                let contract_still_unmet = (is_mutation_task || is_write_task) && !session.local_evidence.lines().any(|l| l.starts_with("[WRITE_OK]") || l.starts_with("[PATCH_FUNCTION_OK]") || l.starts_with("[REPLACE_LINES_OK]") || l.starts_with("[DIFF_PATCH_OK]"));

                if critic.confidence_modifier <= -0.2 && is_mutation_shaped_proposal && contract_still_unmet {
                    session.consecutive_mutation_vetoes += 1;
                    if session.consecutive_mutation_vetoes >= 3 {
                        self.log(format!("\x1b[1;35m>> [CIRCUIT_BREAKER_OVERRIDE] Critic vetoed {} consecutive mutation proposals while the contract's mutation marker is still missing. Overriding veto and executing.\x1b[0m", session.consecutive_mutation_vetoes));
                        critic.confidence_modifier = 0.1;
                        session.consecutive_mutation_vetoes = 0;
                    }
                } else {
                    session.consecutive_mutation_vetoes = 0;
                }

                if critic.confidence_modifier > -0.2 {
                    Ok(CortexState::Translate { proposal, belief })
                } else {
                    Ok(CortexState::Strategize)
                }
            }
        }
    }

    async fn handle_translate(&mut self, session: &mut EpistemicSession, proposal: String, belief: f32) -> Result<CortexState> {
        self.log("\x1b[36m>> [CORTEX] Translating intent to physical action...\x1b[0m".to_string());
        session.budget.record_call();

        let parsed_actions = if let Ok(direct_actions) = crate::json_parser::extract_actions(&proposal) {
            if !direct_actions.is_empty() {
                self.log("\x1b[32m>> [CORTEX INTERCEPT] Direct JSON Proposal Intercept: Bypassing Translator LLM.\x1b[0m".to_string());
                Ok(direct_actions)
            } else {
                if let Ok(raw_json) = EpistemicEngine::translate_to_action(&self.client, &self.api_key, &proposal, &proposal).await {
                    crate::json_parser::extract_actions(&raw_json)
                } else {
                    Ok(vec![])
                }
            }
        } else {
            if let Ok(raw_json) = EpistemicEngine::translate_to_action(&self.client, &self.api_key, &proposal, &proposal).await {
                crate::json_parser::extract_actions(&raw_json)
            } else {
                Ok(vec![])
            }
        };

        let mut actions = parsed_actions.unwrap_or_default();
        if actions.is_empty() && !proposal.trim().is_empty() {
            if let Some(intercepted) = crate::action_intercept::intercept_fallback_action(&proposal) {
                self.log("\x1b[32m>> [CORTEX INTERCEPT] Native Tool Intercept: Handling action directly.\x1b[0m".to_string());
                actions.push(intercepted);
            } else {
                self.log("\x1b[31m>> [CORTEX FAULT] Translator produced empty JSON and proposal text matched no known tool. Refusing raw fallback execution.\x1b[0m".to_string());
                if let Some(h_id) = session.best_hypothesis_id {
                    let obs_id = session.graph.add_node(NodeKind::Observation, "[TRANSLATION_UNRECOGNIZED]", 1.0, 50);
                    session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.3);
                    session.graph.propagate_beliefs();
                }
                return Ok(CortexState::Strategize);
            }
        }

        if actions.is_empty() {
            self.log("\x1b[31m>> [CORTEX FAULT] Translator produced an empty action list.\x1b[0m".to_string());
            if let Some(h_id) = session.best_hypothesis_id {
                let obs_id = session.graph.add_node(NodeKind::Observation, "[TRANSLATION_EMPTY]", 1.0, 50);
                session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.15);
                session.graph.propagate_beliefs();
            }
            Ok(CortexState::Strategize)
        } else {
            Ok(CortexState::Execute { actions, proposal, belief })
        }
    }

    async fn handle_execute(
        &mut self, 
        session: &mut EpistemicSession, 
        actions: Vec<AgentAction>, 
        proposal: String,
        belief: f32
    ) -> Result<CortexState> {
        let mut executor = RoboticExecutor::new(self.api_key.clone(), self.tx.clone());
        let mut latest_evidence = String::new();
        let mut mutation_paths: Vec<String> = Vec::new();
        let mut last_is_halt = false;
        let mut rust_files_mutated = false;
        let evidence_checkpoint = session.local_evidence.len();

        for action in actions {
            let intent = IntentContract::new(&proposal, belief);
            let capability = match &action {
                AgentAction::ExecuteCommand { .. } => Capability::ExecuteSystemCall,
                AgentAction::WriteFile { .. } | AgentAction::DiffPatch { .. } | AgentAction::PatchFunction { .. } => Capability::WriteSubstrate,
                AgentAction::Halt { .. } => Capability::HaltCognition,
                _ => Capability::ReadSubstrate,
            };

            let mutation_path = match &action {
                AgentAction::WriteFile { path, .. } | AgentAction::DiffPatch { path, .. } | AgentAction::PatchFunction { path, .. } => Some(path.clone()),
                _ => None,
            };
            
            if let Some(path) = mutation_path {
                mutation_paths.push(path.clone());
                if path.ends_with(".rs") {
                    rust_files_mutated = true;
                }
            }

            let is_halt_action = matches!(action, AgentAction::Halt { .. }) || proposal.trim().to_lowercase().starts_with("halt");
            if is_halt_action {
                last_is_halt = true;
            }

            let motor_cmd = MotorCortex::translate(&intent, None, action, capability, proposal.clone());

            if let Err(gate_err) = self.action_gate.approve(&motor_cmd) {
                self.log(format!("\x1b[1;31m>> [ACTION GATE REJECTED]: {}\x1b[0m", gate_err));
                if let Some(h_id) = session.best_hypothesis_id {
                    let obs_id = session.graph.add_node(NodeKind::Observation, &format!("[ACTION_GATE_BLOCKED] {}", gate_err), 1.0, 50);
                    session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.50);
                    session.graph.propagate_beliefs();
                }
                continue;
            }

            self.log("\x1b[35m>> [MOTOR CORTEX] Emitting authorized MotorCommand to RoboticExecutor...\x1b[0m".to_string());
            let evidence = match executor.execute_motor_command(motor_cmd, self.memory.clone(), self.vmo_state.clone(), self.dist.clone()).await {
                Ok(ev) => ev,
                Err(e) => format!("[ACTION FAILED] OS rejected action: {}", e)
            };

            latest_evidence.push_str(&evidence);
            latest_evidence.push('\n');
            session.local_evidence.push_str(&evidence);
            session.local_evidence.push('\n');

            let preview = evidence.lines().take(5).collect::<Vec<_>>().join("\n");
            self.log(format!("\x1b[32m>> [EVIDENCE ACQUIRED]\x1b[0m\n{}", preview));
        }

        if latest_evidence.trim().is_empty() {
            return Ok(CortexState::Strategize);
        }

        // ⚡ COMPILER AS A CRITIC + DETERMINISTIC AUTO-HEALER
        if rust_files_mutated {
            self.log("\x1b[36m>> [COMPILER CRITIC] Running cargo check to validate Rust mutation...\x1b[0m".to_string());
            
            let mut attempts = 0;
            loop {
                attempts += 1;
                if attempts > 3 {
                    self.log("\x1b[1;31m>> [COMPILER CRITIC] Max auto-heal attempts reached. Reverting.\x1b[0m".to_string());
                    let compile_fail_evidence = "[COMPILE_FAIL] Auto-healer attempted 3 fixes but errors persist.".to_string();
                    for path in &mutation_paths {
                        let _ = crate::podman::PodmanShield::execute(&format!("git checkout -- {}", path));
                    }
                    session.local_evidence.truncate(evidence_checkpoint);
                    session.local_evidence.push_str(&compile_fail_evidence);
                    session.local_evidence.push('\n');
                    if let Some(h_id) = session.best_hypothesis_id {
                        let obs_id = session.graph.add_node(NodeKind::Observation, "[CRITICAL VETO] Persistent compilation errors after 3 auto-heal attempts.", 1.0, 90);
                        session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.99);
                        session.graph.propagate_beliefs();
                    }
                    return Ok(CortexState::Strategize);
                }

                let compile_output = tokio::process::Command::new("cargo")
                    .arg("check")
                    .arg("--message-format=json")
                    .output()
                    .await;

                match compile_output {
                    Ok(output) if output.status.success() => {
                        latest_evidence.push_str("\n[COMPILE_OK] Workspace compiles successfully.");
                        self.log("\x1b[32m>> [COMPILER CRITIC] Compile OK. Proceeding to LLM Evaluator.\x1b[0m".to_string());
                        break; // Exit loop, proceed to Evaluate
                    }
                    Ok(output) => {
                        let stdout = String::from_utf8_lossy(&output.stdout);
                        let mut applied_heals = 0;
                        
                        // Parse cargo JSON output line by line
                        for line in stdout.lines() {
                            if let Some(diag) = nexus_ast_tool::engine::parse_compiler_line(line) {
                                if let Some(action) = nexus_ast_tool::engine::determine_policy(&diag) {
                                    let file_path = std::path::PathBuf::from(&action.file);
                                    if file_path.exists() {
                                        if let Ok(source) = std::fs::read_to_string(&file_path) {
                                            if let Some(strategy) = nexus_ast_tool::mutation::get_strategy(&action.strategy) {
                                                match strategy.apply(&source, &action.symbol, &action.payload) {
                                                    Ok(new_source) => {
                                                        let _ = std::fs::write(&file_path, new_source);
                                                        applied_heals += 1;
                                                        self.log(format!("\x1b[1;36m>> [AUTO-HEAL] Applied {} to {}\x1b[0m", action.strategy, action.file));
                                                    }
                                                    Err(e) => self.log(format!("\x1b[33m>> [AUTO-HEAL FAIL] {}: {}\x1b[0m", action.strategy, e))
                                                }
                                            }
                                        }
                                    }
                                }
                            }
                        }

                        if applied_heals > 0 {
                            latest_evidence.push_str(&format!("\n[AUTO_HEAL] Applied {} deterministic fixes. Re-checking...", applied_heals));
                            session.local_evidence.push_str(&latest_evidence);
                            continue; // Re-run cargo check
                        }

                        // No healable errors found, standard failure path
                        let stderr = String::from_utf8_lossy(&output.stderr);
                        let err_preview = stderr.lines().take(15).collect::<Vec<_>>().join("\n");
                        
                        self.log("\x1b[1;31m>> [COMPILER CRITIC] Compile FAILED. Reverting mutations and bypassing LLM Evaluator.\x1b[0m".to_string());
                        
                        let compile_fail_evidence = format!("[COMPILE_FAIL] Your patch caused a compilation error:\n{}", err_preview);
                        
                        for path in &mutation_paths {
                            self.log(format!("\x1b[1;31m>> [CORTEX SAFETY] Reverting {} via git checkout.\x1b[0m", path));
                            let _ = crate::podman::PodmanShield::execute(&format!("git checkout -- {}", path));
                        }

                        session.local_evidence.truncate(evidence_checkpoint);
                        let stripped_this_turn = latest_evidence
                            .replace("[WRITE_OK]", "[MUTATION_REVERTED]")
                            .replace("[DIFF_PATCH_OK]", "[MUTATION_REVERTED]")
                            .replace("[PATCH_FUNCTION_OK]", "[MUTATION_REVERTED]")
                            .replace("[REPLACE_LINES_OK]", "[MUTATION_REVERTED]");
                        
                        session.local_evidence.push_str(&stripped_this_turn);
                        session.local_evidence.push_str(&compile_fail_evidence);
                        session.local_evidence.push('\n');
                        
                        if let Some(h_id) = session.best_hypothesis_id {
                            let obs_id = session.graph.add_node(
                                NodeKind::Observation, 
                                "[CRITICAL VETO] Your patch caused a compilation error. Files reverted. Try again with valid syntax.", 
                                1.0, 
                                90 
                            );
                            session.graph.add_edge(obs_id, h_id, EdgeRelation::Contradicts, 0.99);
                            session.graph.propagate_beliefs();
                        }
                        
                        return Ok(CortexState::Strategize);
                    }
                    Err(e) => {
                        self.log(format!("\x1b[33m>> [COMPILER CRITIC] Failed to run cargo check: {}. Proceeding with LLM Evaluator.\x1b[0m", e));
                        break;
                    }
                }
            }
        }

        Ok(CortexState::Evaluate { 
            evidence: latest_evidence, 
            proposal: proposal.clone(), 
            belief, 
            is_halt_action: last_is_halt, 
            mutation_path: mutation_paths.last().cloned()
        })
    }

    async fn handle_evaluate(&mut self, session: &mut EpistemicSession, _mission: &str, evidence: String, proposal: String, belief: f32, _is_mutation_task: bool, _is_write_task: bool, is_halt_action: bool, mutation_path: Option<String>) -> Result<CortexState> {
        self.log("\x1b[33m>> [CORTEX] Evaluating physical evidence against strategy...\x1b[0m".to_string());
        session.budget.record_call();
        
        let h_id = session.best_hypothesis_id.unwrap_or_default();
        let eval_evidence_context = format!("Current Action Output:\n{}\n\nPrior Evidence in Task:\n{}", evidence, session.local_evidence);

        match EpistemicEngine::evaluate_evidence(&self.client, &self.api_key, &proposal, &proposal, &eval_evidence_context).await {
            Err(e) => {
                self.log(format!("\x1b[31m>> [CORTEX FATAL] Evaluator panicked: {}\x1b[0m", e));
                Ok(CortexState::Strategize)
            }
            Ok(eval) => {
                self.log(format!("\x1b[33m>> [EVALUATION]\x1b[0m {} (Modifier: {:.2})", eval.critique, eval.confidence_modifier));
                session.record_evidence(&evidence, &eval, h_id);

                if is_halt_action && eval.confidence_modifier >= 0.0 && belief > 0.85 {
                    self.log("\x1b[1;36m>> [CORTEX TERMINAL] Valid halt executed and verified. Submitting evidence to ContractVerifier.\x1b[0m".to_string());
                    return Ok(CortexState::Complete { belief: belief.max(0.95), evidence: session.collect_evidence(h_id) });
                }

                if eval.confidence_modifier < 0.0 {
                    if let Some(path) = &mutation_path {
                        self.log(format!("\x1b[1;31m>> [CORTEX SAFETY] Mutation to '{}' scored negative ({:.2}) -- reverting via git checkout.\x1b[0m", path, eval.confidence_modifier));
                        let _ = crate::podman::PodmanShield::execute(&format!("git checkout -- {}", path));
                        session.local_evidence = session.local_evidence.replace(&evidence, "");
                        let reverted_tag = format!("[MUTATION_REVERTED] {} -- Evaluator rejected this mutation ({:.2}): {}\n", path, eval.confidence_modifier, eval.critique);
                        session.local_evidence.push_str(&reverted_tag);
                    }
                }

                if let Some(updated_node) = session.graph.nodes.get(&h_id) {
                    self.log(format!("\x1b[1;32m>> [CORTEX] Strategy Belief updated to {:.2}\x1b[0m", updated_node.belief_score));
                    if session.is_subgoal_satisfied(updated_node.belief_score) {
                        self.log("\x1b[1;32m>> [CORTEX] Contract criteria met. Awaiting explicit Agent HALT.\x1b[0m".to_string());
                    }
                }
                Ok(CortexState::Strategize)
            }
        }
    }
}

use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct CriticVerdict {
    pub approved: bool,
    pub confidence: f32,
    pub reason: String,
    pub suggested_recovery: Option<String>,
}

pub struct Critic;

impl Critic {
    pub async fn evaluate(
        _objective: &str,
        observations: &str,
        _history: &str,
    ) -> Result<CriticVerdict, Box<dyn std::error::Error>> {
        // If there is no observation history yet, we can't be done
        if observations.trim().is_empty() {
            return Ok(CriticVerdict {
                approved: false,
                confidence: 0.1,
                reason: "No actions have been executed yet. Cannot halt.".into(),
                suggested_recovery: Some("Execute an action first.".into()),
            });
        }

        // We do NOT check for [HALT] here, because the Critic is invoked specifically
        // to validate the Halt intent *before* it gets added to the log.
        let has_write = observations.contains("[WRITE_OK]")
            || observations.contains("[DIFF_PATCH_OK]")
            || observations.contains("[PATCH_FUNCTION_OK]")
            || observations.contains("success");
        let has_read = observations.contains("[READ_OK]");
        let has_cmd = observations.contains("[CMD_OK]");
        let has_search = observations.contains("[SEARCH_OK]");

        let approved = has_write || has_read || has_cmd || has_search;
        let confidence = if approved { 0.95 } else { 0.4 };

        Ok(CriticVerdict {
            approved,
            confidence,
            reason: if approved {
                "Concrete success indicators found in action history.".into()
            } else {
                "Missing concrete success indicators (Read/Write/Cmd).".into()
            },
            suggested_recovery: if !approved {
                Some("Perform a read, write, or command execution to gather evidence.".into())
            } else {
                None
            },
        })
    }
}
