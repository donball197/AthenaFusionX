// ResonatorAudio.ts — v3.0 (uses shared AudioEngine)
import { ctx } from '$lib/AudioEngine';

let _lastClick = 0;

export function initAudio() { ctx(); }

export function playResonance(ramUsage: string | number) {
    const c = ctx();
    if (!c) return;
    const now    = c.currentTime;
    const ramInt = typeof ramUsage === 'string' ? parseInt(ramUsage.replace('MB','')) : ramUsage;
    const interval = Math.max(0.05, 1.0 - ((ramInt || 0) / 2000));
    if (now - _lastClick < interval) return;
    try {
        const osc  = c.createOscillator();
        const gain = c.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime((ramInt || 0) > 1800 ? 880 : 440, now);
        gain.gain.setValueAtTime(0.1, now);
        gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.1);
        osc.connect(gain); gain.connect(c.destination);
        osc.start(); osc.stop(now + 0.1);
        _lastClick = now;
    } catch (e: any) { console.warn('[RESONATOR] error:', e.message); }
}

// agentPipeline.js — v1.6
// FIX: extractJSON now handles single-object results (not only arrays).
// FIX: isSafePath extended with .nexus, .env patterns and null-byte check.
// FIX: normalizeResponse handles non-string input gracefully.

export function normalizeResponse(response) {
    if (response == null) return '';
    if (typeof response !== 'string') {
        try { return JSON.stringify(response); } catch { return ''; }
    }
    return response.trim();
}

/**
 * extractJSON: Parse all <ACTION> blocks from a response string.
 * Handles arrays and single objects. Returns null if nothing valid found.
 */
export function extractJSON(str) {
    const cleaned = normalizeResponse(str);
    const actions = [];

    const actionRegex = /<ACTION>([\s\S]*?)<\/ACTION>/g;
    let match;
    while ((match = actionRegex.exec(cleaned)) !== null) {
        try {
            const rawJson = match[1].trim();
            const parsed  = JSON.parse(rawJson);
            if (Array.isArray(parsed)) {
                actions.push(...parsed);
            } else if (parsed && typeof parsed === 'object') {
                // Single action object
                actions.push(parsed);
            }
        } catch (e) {
            console.error('[PIPELINE] Malformed JSON in <ACTION> block:', e.message);
        }
    }

    // Fallback: last [...] block in response
    if (actions.length === 0) {
        const end = cleaned.lastIndexOf(']');
        if (end !== -1) {
            const start = cleaned.lastIndexOf('[', end);
            if (start !== -1) {
                try {
                    const parsed = JSON.parse(cleaned.slice(start, end + 1));
                    if (Array.isArray(parsed)) actions.push(...parsed);
                } catch { /* ignore */ }
            }
        }
    }

    return actions.length > 0 ? actions : null;
}

/**
 * isSafePath: Returns false for dangerous or sensitive paths.
 * Checks for null bytes, traversal components, and blacklisted prefixes.
 */
export function isSafePath(path) {
    if (!path || typeof path !== 'string') return false;
    if (path.includes('\0')) return false;

    const normalized = path.toLowerCase().replace(/\\/g, '/');

    const forbidden = [
        '../',
        '.env',
        '.ssh/',
        '/etc/',
        'target/',
        'node_modules/',
        '.nexus_memory.db',
        '.nexus_wasm_registry.db',
        '.nexus/vault',
        '.git/',
    ];
    return !forbidden.some(f => normalized.includes(f));
}

    import { ide } from '$lib/stores/ide.svelte';
    import { onMount, onDestroy } from 'svelte';
    import { basicSetup } from 'codemirror';
    import { EditorState } from '@codemirror/state';
    import { EditorView, keymap } from '@codemirror/view';
    import { indentWithTab } from '@codemirror/commands';
    import { oneDark } from '@codemirror/theme-one-dark';

    let editorContainer = $state<HTMLElement | null>(null);
    let view = $state<EditorView | null>(null);
    let currentPath = $state<string | null>(null);

    let activeFile = $derived(ide.openTabs.find(t => t.path === ide.activeTab));

    $effect(() => {
        if (!editorContainer) return;

        if (ide.activeTab !== currentPath) {
            currentPath = ide.activeTab;
            
            if (ide.activeTab && activeFile) {
                if (!view) {
                    const state = EditorState.create({
                        doc: activeFile.content,
                        extensions: [
                            basicSetup,
                            oneDark,
                            keymap.of([indentWithTab]),
                            EditorView.updateListener.of(update => {
                                if (update.docChanged && view?.hasFocus) {
                                    ide.updateContent(ide.activeTab as string, update.state.doc.toString());
                                }
                            })
                        ]
                    });
                    view = new EditorView({ state, parent: editorContainer });
                } else {
                    view.dispatch({
                        changes: { from: 0, to: view.state.doc.length, insert: activeFile.content }
                    });
                }
            }
        } else if (view && activeFile && activeFile.content !== view.state.doc.toString() && !view.hasFocus) {
            view.dispatch({
                changes: { from: 0, to: view.state.doc.length, insert: activeFile.content }
            });
        }
    });

    onDestroy(() => { if (view) view.destroy(); });
</script>

    function execute(cmd: typeof COMMANDS[0]) {

    function onKey(e: KeyboardEvent) {

    import { oneDark } from '@codemirror/theme-one-dark';
    import { vim } from '@replit/codemirror-vim';
    import { Compartment } from '@codemirror/state';

    let { path = '', vimEnabled = true } = $props();
    let container = $state();
    let view;
    // Compartment allows dynamically reconfiguring extensions without recreating the editor
    let vimCompartment = new Compartment();

    onMount(() => {
        const tab = ide.openTabs.find(t => t.path === path);
        const startState = EditorState.create({
            doc: tab?.content ?? '',
            extensions: [
                basicSetup,
                rust(),
                oneDark,
                vimCompartment.of(vimEnabled ? vim() : []),
                EditorView.updateListener.of((update) => {
                    if (update.docChanged) {
                        ide.updateContent(path, update.state.doc.toString());
                    }
                })
            ]
        });

        view = new EditorView({
            state: startState,
            parent: container
        });
    });

    onDestroy(() => { if (view) view.destroy(); });

    // React to Vim toggle
    $effect(() => {
        if (view) {
            view.dispatch({
                effects: vimCompartment.reconfigure(vimEnabled ? vim() : [])
            });
        }
    });

    // React to tab switching/content updates
    $effect(() => {
        if (view && path) {
            const tab = ide.openTabs.find(t => t.path === path);
            const currentDoc = view.state.doc.toString();
            if (tab && tab.content !== currentDoc) {
                view.dispatch({
                    changes: { from: 0, to: currentDoc.length, insert: tab.content }
                });
            }
        }
    });
</script>

    let breadcrumb  = $state(['.']);

    const FILE_ICONS = {
        rs:'🦀', js:'🟨', ts:'🔷', svelte:'🔶', toml:'⚙',
        md:'📝', json:'📋', zig:'⚡', css:'🎨', html:'🌐',
        sh:'💻', py:'🐍', lock:'🔒', env:'🔑',
    };

    function fileIcon(name) {
        const ext = name.split('.').at(-1) ?? '';
        return FILE_ICONS[ext] ?? '📄';
    }

    async function scan(path = currentPath) {
        scanning  = true;
        createErr = '';
        try {
            const r = await uplink.fetch(`/api/fs/list?path=${encodeURIComponent(path)}`);
            if (r.ok) {
                files       = await r.json();
                currentPath = path;
            }
        } catch (e) {
            console.error('SIDEBAR_FAULT', e);
        } finally {
            scanning = false;
        }
    }

    function enterDir(path) {
        breadcrumb = [...breadcrumb, path];
        scan(path);
    }

    function goUp() {
        if (breadcrumb.length <= 1) return;
        const b = [...breadcrumb];
        b.pop();
        breadcrumb = b;
        scan(b.at(-1));
    }

    async function handleCreate(e) {
        e.preventDefault();
        const name = newFileName.trim();
        if (!name || creating) return;
        const fullPath = currentPath === '.' ? name : `${currentPath}/${name}`;
        creating  = true;
        createErr = '';
        try {
            if (onCreateFile) {
                await onCreateFile(fullPath);
            } else {
                // FIX: use uplink.post instead of raw fetch with manual headers
                await uplink.post('/api/fs/write', { path: fullPath, content: '' });
            }
            newFileName = '';
            await scan();
        } catch (e) {
            createErr = e.message;
        } finally {
            creating = false;
        }
    }

    $effect(() => { if (fileTreeTrigger.value) scan(); });
    onMount(() => scan('.'));
</script>

    // Auto-scroll when logs or ghost buffer change
    $effect(() => {
        const _len = terminal.logs.length;
        const _buf = terminal.ghostBuffer;
        if (autoScroll && logEl) {
            tick().then(() => { if (logEl) logEl.scrollTop = logEl.scrollHeight; });
        }
    });

    async function sendDirective() {
        const d = inputVal.trim();
        if (!d) return;
        inputVal = '';
        TaskService.runDirective(d, 'USER_TERMINAL');
    }

    function onKey(e: KeyboardEvent) {
        if (e.key === 'Enter' && !e.shiftKey) {
            e.preventDefault();
            sendDirective();
        }
    }

    function onScroll(e: Event) {
        const el = e.target as HTMLDivElement;
        // Re-enable auto-scroll if user scrolls back to bottom
        autoScroll = el.scrollHeight - el.scrollTop - el.clientHeight < 40;
    }

    function clearLogs() {
        // Use the store helper — direct array mutation breaks Svelte 5 reactivity
        terminal.logs = [];
        clearGhost();
    }

    async function handleGenesis(event: Event) {
        const b64 = (event as CustomEvent<string>).detail;
        try {
            TaskService.log(">> [TRINITY] Intercepting WASM logic core...", "text-purple-400");
            const binaryString = window.atob(b64);
            const bytes = new Uint8Array(binaryString.length);
            for (let i = 0; i < binaryString.length; i++) bytes[i] = binaryString.charCodeAt(i);

            const { instance } = await WebAssembly.instantiate(bytes.buffer);
            const exports = instance.exports as Record<string, unknown>;

            // Discover function exports
            const fns = Object.entries(exports)
                .filter(([, v]) => typeof v === 'function')
                // skip internal WASM bookkeeping exports
                .filter(([k]) => !k.startsWith('__') && !k.startsWith('_'))
                .map(([k]) => k);

            if (fns.length === 0) {
                warnMsg = 'No exported functions found in this WASM cell.';
                TaskService.log("!! [TRINITY] No usable exports in cell.", "text-rose-400");
                return;
            }

            wasmEngine  = exports as Record<string, Function>;
            exportNames = fns;
            warnMsg     = fns.length < 2 ? `Only 1 export found: ${fns[0]}` : '';
            TaskService.log(
                `>> [TRINITY] Bound ${fns.length} export(s): ${fns.slice(0,3).join(', ')}`,
                "text-emerald-400"
            );
        } catch (e: any) {
            warnMsg = `V8 Engine Fault: ${e.message}`;
            TaskService.log(`!! [TRINITY] ${e.message}`, "text-rose-400");

    import { uplink } from '$lib/services/uplink.js';

    let loading = $state(false);

    $effect(() => {

    });

    async function loadFile(path) {
        loading = true;
        try {
            const data = await uplink.json(`/api/fs/read?path=${encodeURIComponent(path)}`);
            fileContent = data.content !== undefined ? data.content : '';
        } catch (e) { 
            TaskService.log(`!! [EDITOR] Read error: ${e.message}`, 'text-rose-400');
        } finally { 
            loading = false; 
        }
    }

    async function saveFile() {
        if (!activeFile || saving) return;
        saving = true;
        try {
            await uplink.post(`/api/fs/write`, { path: activeFile, content: fileContent });
            TaskService.log(`>> [EDITOR] Commited changes to ${activeFile}`, 'text-emerald-400');
        } catch (e) { 
            TaskService.log(`!! [EDITOR] Save error: ${e.message}`, 'text-rose-400');
        } finally { 
            saving = false; 
        }
    }

    function handleKeydown(e) {
        if (e.key === 'Tab') {
            e.preventDefault();
            const start = e.target.selectionStart;
            const end = e.target.selectionEnd;
            fileContent = fileContent.substring(0, start) + "    " + fileContent.substring(end);
            setTimeout(() => {
                e.target.selectionStart = e.target.selectionEnd = start + 4;
            }, 0);
        }
        if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 's') {
            e.preventDefault();
            saveFile();
        }
    }

    async function loadDirectory(targetPath) {
        loading = true;
        currentPath = targetPath;
        try {
            const res = await uplink.fetch(`/api/fs/list?path=${encodeURIComponent(targetPath)}`);
            if (res.ok) {
                let data = await res.json();
                data.sort((a, b) => {
                    if (a.is_dir === b.is_dir) return a.name.localeCompare(b.name);
                    return a.is_dir ? -1 : 1;
                });
                files = data;
            }
        } catch (e) { console.error("FS_EXPLORER_FAULT:", e); } 
        finally { loading = false; }
    }

    function navigateUp() {
        if (currentPath === '.') return;
        const parts = currentPath.split('/');
        parts.pop();
        loadDirectory(parts.length > 0 ? parts.join('/') : '.');
    }

// socket.ts — v2.1
// FIX: Added .telemetry property so +page.svelte references don't silently explode
// FIX: Capped retries at 15 (was already there, kept)

type Listener = (msg: any) => void;

export interface Telemetry {
    status:      string;
    active_vmos: number;
    resonance:   number;
}

class NexusSocket {
    private socket:       WebSocket | null = null;
    private url:          string;
    private listeners:    Set<Listener> = new Set();
    private retries     = 0;
    private retryTimer?: ReturnType<typeof setTimeout>;
    private watchdogTimer?: ReturnType<typeof setInterval>;
    private lastHeartbeat = Date.now();

    public status:    'DISCONNECTED' | 'CONNECTED' = 'DISCONNECTED';
    public telemetry: Telemetry = { status: 'IDLE', active_vmos: 0, resonance: 0 };

    constructor(path: string) {
        if (typeof window === 'undefined') {
            this.url = `ws://localhost:3000${path}`;
        } else {
            const proto = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
            this.url = `${proto}//${window.location.host}${path}`;
        }
    }

    connect() {
        if (this.socket?.readyState === WebSocket.OPEN || this.socket?.readyState === WebSocket.CONNECTING) return;
        this.socket = new WebSocket(this.url);

        this.socket.onopen = () => {
            this.status = 'CONNECTED';
            this.telemetry.status = 'CONNECTED';
            this.retries = 0;
            this.lastHeartbeat = Date.now();
            this.emit({ _internal: 'CONNECTION_STATE', status: 'CONNECTED' });
            this.startWatchdog();
        };

        this.socket.onmessage = (e) => {
            this.lastHeartbeat = Date.now();
            try {
                const msg = JSON.parse(e.data);
                // Heartbeat pass
                // Mirror telemetry fields
                if (msg.active_vmos !== undefined) this.telemetry.active_vmos = msg.active_vmos;
                if (msg.resonance   !== undefined) this.telemetry.resonance   = msg.resonance;
                this.emit(msg);
            } catch {
                console.warn('[SOCKET] Dropped invalid JSON frame');
            }
        };

        this.socket.onclose = () => {
            this.status = 'DISCONNECTED';
            this.telemetry.status = 'DISCONNECTED';
            this.emit({ _internal: 'CONNECTION_STATE', status: 'DISCONNECTED' });
            clearInterval(this.watchdogTimer);
            if (this.retries < 15) this.retries++;
            const delay = Math.min(1000 * 2 ** this.retries, 16000);
            this.retryTimer = setTimeout(() => this.connect(), delay);
        };
    }

    send(payload: any) {
        if (this.socket?.readyState === WebSocket.OPEN) {
            const data = typeof payload === 'string' ? { directive: payload } : payload;
            this.socket.send(JSON.stringify(data));
        }
    }

    subscribe(fn: Listener) {
        this.listeners.add(fn);
        return () => this.listeners.delete(fn);
    }

    private emit(msg: any) { this.listeners.forEach(l => l(msg)); }

    private startWatchdog() {
        clearInterval(this.watchdogTimer);
        this.watchdogTimer = setInterval(() => {
            if (this.status === 'CONNECTED' && Date.now() - this.lastHeartbeat > 15000) {
                console.warn('[WATCHDOG] Socket frozen — forcing reconnect.');
                this.socket?.close();
            }
        }, 5000);
    }

    destroy() {
        clearTimeout(this.retryTimer);
        clearInterval(this.watchdogTimer);
        this.socket?.close();
        this.listeners.clear();
    }
}

    const headers = { 'Content-Type': 'application/json', ...extra };
    if (auth.token) {
        // SUTURE: Added 'Bearer ' prefix to align with standard Axum/JWT extractors
        // Ensure backend middleware strips this or verify_token handles it
        headers['Authorization'] = `Bearer ${auth.token}`;
    }
    return headers;
}

async function request(url, options = {}) {
    const { headers: extraHeaders, ...rest } = options;
    const headers = buildHeaders(extraHeaders ?? {});
    
    try {
        const res = await fetch(url, { ...rest, headers });
        
        if (res.status === 401) {
            console.warn(`!! [UPLINK] 401 Unauthorized at ${url}. Token invalid or missing.`);
        }
        
        if (!res.ok) throw new Error(`HTTP ${res.status} — ${url}`);
        return res;
    } catch (e) {
        throw e;
    }
}

export const uplink = {
    fetch:    (url, options = {})     => request(url, options),
    post:     (url, body, signal)     => request(url, { method: 'POST',  body: JSON.stringify(body), signal }),
    put:      (url, body, signal)     => request(url, { method: 'PUT',   body: JSON.stringify(body), signal }),
    delete:   (url, signal)           => request(url, { method: 'DELETE', signal }),
    get:      (url, signal)           => request(url, { signal }),
    json:     async (url, opts)       => (await request(url, opts)).json(),
};

export const auth = $state({ 
    token: '', 
    isInitialized: false 
});

export async function bootAuth(): Promise<void> {
    if (auth.token) {
        auth.isInitialized = true;
        return;
    }
    try {
        const res = await fetch('/api/auth/bootstrap');
        if (res.ok) {
            const data = await res.json();
            auth.token = data.token;
        }
    } catch (e) {
        console.warn('[AUTH] Bootstrap failed:', e);
    } finally {
        // Handshake clears regardless of success (allows login page to show)
        auth.isInitialized = true;
    }
}

export async function login(key: string): Promise<boolean> {
    try {
        const res = await uplink.post('/api/auth/login', { key });
        const data = await res.json();
        if (data.token) {
            auth.token = data.token;
            return true;
        }
        return false;
    } catch {
        return false;
    }
}

// tasks.svelte.ts — v1.2 [CAPPED + UPLINK + CREATED_AT]
// FIX: Task array capped at MAX_TASKS — was unbounded.
// FIX: Uses uplink instead of secureFetch for consistent auth header.
// FIX: createdAt timestamp added for sorting/eviction.
// FIX: Active tasks protected from eviction when pruning.
import { uplink } from '$lib/services/uplink.js';

export type Task = {
    id:        string;
    directive: string;
    status:    'queued' | 'running' | 'done' | 'failed';
    createdAt: number;
};

const MAX_TASKS = 200;

export const taskQueue = $state({ tasks: [] as Task[] });

function prune() {
    if (taskQueue.tasks.length <= MAX_TASKS) return;
    // Evict oldest completed tasks first, never evict active ones
    const active    = taskQueue.tasks.filter(t => t.status === 'queued' || t.status === 'running');
    const completed = taskQueue.tasks.filter(t => t.status === 'done'   || t.status === 'failed');
    const keep      = completed.slice(-(MAX_TASKS - active.length));
    taskQueue.tasks = [...active, ...keep].sort((a, b) => a.createdAt - b.createdAt);
}

export async function dispatchTask(directive: string): Promise<void> {
    const task: Task = {
        id:        crypto.randomUUID(),
        directive,
        status:    'queued',
        createdAt: Date.now(),
    };
    taskQueue.tasks = [...taskQueue.tasks, task];
    prune();

    try {
        task.status = 'running';
        const res = await uplink.post('/api/task', { directive });
        task.status = res.ok ? 'done' : 'failed';
    } catch {
        task.status = 'failed';
    }
}

/**
 * AudioEngine.ts — single shared AudioContext for the entire app.
 * Import `ctx()` instead of creating your own AudioContext.
 * Both ResonatorAudio.ts and SoundFx.ts should use this.
 */

let _sharedCtx: AudioContext | null = null;

export function ctx(): AudioContext | null {
    if (!_sharedCtx) {
        try {
            const Ctor = window.AudioContext || (window as any).webkitAudioContext;
            _sharedCtx = new Ctor();
        } catch (e: any) {
            console.warn('[AUDIO] AudioContext unavailable:', e.message);
            return null;
        }
    }
    if (_sharedCtx.state === 'suspended') {
        _sharedCtx.resume().catch(() => {});
    }
    return _sharedCtx;
}

export function playTone(
    freq:     number,
    type:     OscillatorType,
    duration: number,
    volume:   number,
) {
    const c = ctx();
    if (!c) return;
    try {
        const osc  = c.createOscillator();
        const gain = c.createGain();
        osc.type = type;
        osc.frequency.setValueAtTime(freq, c.currentTime);
        gain.gain.setValueAtTime(volume, c.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.0001, c.currentTime + duration);
        osc.connect(gain);
        gain.connect(c.destination);
        osc.start();
        osc.stop(c.currentTime + duration);
    } catch (e: any) {
        console.warn('[AUDIO] Tone playback failed:', e.message);
    }
}
