// src/bin/global_audit.rs - Native Hierarchical Map-Reduce Auditor
use anyhow::Result;
use std::fs;
use std::path::Path;
use std::sync::Arc;
use tokio::sync::broadcast;
use nexus_singularity::neural_agent::{MicroAgent, ContextPacket};

struct Domain {
    name: &'static str,
    paths: Vec<&'static str>,
}

fn read_domain_files(paths: &[&str]) -> String {
    let mut payload = String::new();
    for path_str in paths {
        let path = Path::new(path_str);
        if path.is_file() {
            if let Ok(content) = fs::read_to_string(path) {
                payload.push_str(&format!("\n--- FILE: {} ---\n{}\n", path_str, content));
            }
        } else if path.is_dir() {
            // Basic flat directory read for UI/Kernel
            if let Ok(entries) = fs::read_dir(path) {
                for entry in entries.flatten() {
                    let file_path = entry.path();
                    if file_path.is_file() {
                        if let Some(ext) = file_path.extension().and_then(|s| s.to_str()) {
                            if matches!(ext, "rs" | "svelte" | "js" | "ts" | "zig") {
                                if let Ok(content) = fs::read_to_string(&file_path) {
                                    payload.push_str(&format!("\n--- FILE: {} ---\n{}\n", file_path.display(), content));
                                }
                            }
                        }
                    }
                }
            }
        }
    }
    payload
}

#[tokio::main]
async fn main() -> Result<()> {
    dotenvy::dotenv().ok();
    let api_key = std::env::var("GEMINI_API_KEY").expect("GEMINI_API_KEY missing");
    let client = reqwest::Client::new();
    let (tx, _) = broadcast::channel(16);
    let agent = Arc::new(MicroAgent::init(tx)?);

    println!("\x1b[35m>> [NEXUS] INITIATING NATIVE HIERARCHICAL AUDIT (RATE-LIMITED)\x1b[0m");

    let domains = vec![
        Domain { name: "UI", paths: vec!["ui/src"] },
        Domain { name: "KERNEL", paths: vec!["src/kernel", "src/reflex"] },
        Domain { name: "CORTEX", paths: vec!["src/neural_agent.rs", "src/task_manager.rs", "src/orchestrator.rs"] },
        Domain { name: "INFRASTRUCTURE", paths: vec!["src/server.rs", "src/executor.rs", "src/lazarus_vault.rs"] },
    ];

    let mut grand_synthesis_payload = String::from("=== GLOBAL SYSTEM ARCHITECTURE ===\n\n");

    // 1. THE MAP PHASE
    for domain in domains {
        println!("\n\x1b[36m[*] Mapping Domain: {}\x1b[0m", domain.name);
        let code_payload = read_domain_files(&domain.paths);
        
        if code_payload.is_empty() {
            println!("    -> No readable code found. Skipping.");
            continue;
        }

        println!("    -> Compiling {} sector via Gemma API...", domain.name);
        
        let prompt = format!(
            "Analyze the following codebase section for the {} domain.\n\
            Provide a dense, highly technical summary of its architectural role, key structs/functions, and any logic bottlenecks.\n\n\
            CODE:\n{}", domain.name, code_payload
        );

        let ctx = ContextPacket {
            system_prompt: Some("You are a Senior Systems Architect. Output a concise technical summary. No pleasantries.".into()),
            ..Default::default()
        };

        match agent.prepare_and_run(&client, &api_key, &prompt, ctx, 0.3, None).await {
            Ok((summary, _)) => {
                grand_synthesis_payload.push_str(&format!("--- {} DOMAIN SUMMARY ---\n{}\n\n", domain.name, summary));
                println!("    -> Domain summarized successfully.");
            }
            Err(e) => println!("\x1b[31m    -> API Failure on {}: {}\x1b[0m", domain.name, e),
        }

        println!("\x1b[90m    [+] Cooling down API token window (15s)...\x1b[0m");
        tokio::time::sleep(tokio::time::Duration::from_secs(15)).await;
    }

    // 2. THE REDUCE PHASE
    println!("\n\x1b[35m[*] Initiating Grand Synthesis...\x1b[0m");
    
    let synth_prompt = format!(
        "You are the Lead Architect. Review these domain summaries and provide a unified 'Grand Synthesis' of the entire system architecture. \
        Identify how the domains interact and point out any cross-domain fracture points.\n\n{}", 
        grand_synthesis_payload
    );

    let synth_ctx = ContextPacket {
        system_prompt: Some("You are the NEXUS Sovereign AI. Output a highly structured, systemic analysis.".into()),
        ..Default::default()
    };

    match agent.prepare_and_run(&client, &api_key, &synth_prompt, synth_ctx, 0.7, None).await {
        Ok((final_report, _)) => {
            println!("\n\x1b[32m=== [GLOBAL SYNTHESIS REPORT] ===\x1b[0m\n{}", final_report);
            std::fs::write(".nexus_global_synthesis.md", final_report)?;
            println!("\n\x1b[32m>> [GLOBAL AUDIT COMPLETE] Report saved to .nexus_global_synthesis.md\x1b[0m");
        }
        Err(e) => println!("\x1b[31m>> Synthesis API Failure: {}\x1b[0m", e),
    }

    Ok(())
}

// src/broker.rs - v2.0 (Indexes structs, traits, impls — not just functions)
use chrono;
use once_cell::sync::Lazy;
use regex::Regex;
use serde::{Deserialize, Serialize};
use std::fs;
use std::sync::Arc;
use tokio::sync::RwLock;
use walkdir::WalkDir;

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct AtlasEntry {
    pub name: String,
    pub kind: String,
    pub path: String,
    pub signature: String,
}

#[derive(Serialize, Deserialize, Clone, Default)]
pub struct ProjectAtlas {
    pub entries: Vec<AtlasEntry>,
    pub timestamp: i64,
}

static CACHED_ATLAS: Lazy<Arc<RwLock<ProjectAtlas>>> =
    Lazy::new(|| Arc::new(RwLock::new(ProjectAtlas::default())));

pub struct KnowledgeBroker;

impl KnowledgeBroker {
    pub async fn refresh_atlas() {
        let mut entries = Vec::new();

        // Match: pub fn, pub async fn
        let re_fn    = Regex::new(r"pub (?:async )?fn ([a-zA-Z0-9_]+)").unwrap();
        // Match: pub struct Foo
        let re_struct = Regex::new(r"pub struct ([a-zA-Z0-9_]+)").unwrap();
        // Match: pub trait Foo
        let re_trait  = Regex::new(r"pub trait ([a-zA-Z0-9_]+)").unwrap();
        // Match: impl Foo or impl<T> Foo
        let re_impl   = Regex::new(r"^impl(?:<[^>]+>)?\s+([a-zA-Z0-9_]+)").unwrap();
        // Match: pub enum Foo
        let re_enum   = Regex::new(r"pub enum ([a-zA-Z0-9_]+)").unwrap();

        let walk_result: Result<Vec<_>, _> = WalkDir::new("src").into_iter().collect();
        if let Ok(entries_raw) = walk_result {
            for entry in entries_raw
                .into_iter()
                .filter(|e| e.path().extension().is_some_and(|ext| ext == "rs"))
            {
                let path = entry.path().to_string_lossy().to_string();
                if let Ok(content) = fs::read_to_string(entry.path()) {
                    for cap in re_fn.captures_iter(&content) {
                        entries.push(AtlasEntry {
                            name: cap[1].to_string(),
                            kind: "function".into(),
                            path: path.clone(),
                            signature: cap[0].to_string(),
                        });
                    }
                    for cap in re_struct.captures_iter(&content) {
                        entries.push(AtlasEntry {
                            name: cap[1].to_string(),
                            kind: "struct".into(),
                            path: path.clone(),
                            signature: cap[0].to_string(),
                        });
                    }
                    for cap in re_trait.captures_iter(&content) {
                        entries.push(AtlasEntry {
                            name: cap[1].to_string(),
                            kind: "trait".into(),
                            path: path.clone(),
                            signature: cap[0].to_string(),
                        });
                    }
                    for cap in re_enum.captures_iter(&content) {
                        entries.push(AtlasEntry {
                            name: cap[1].to_string(),
                            kind: "enum".into(),
                            path: path.clone(),
                            signature: cap[0].to_string(),
                        });
                    }
                    for line in content.lines() {
                        if let Some(cap) = re_impl.captures(line.trim()) {
                            entries.push(AtlasEntry {
                                name: cap[1].to_string(),
                                kind: "impl".into(),
                                path: path.clone(),
                                signature: line.trim().to_string(),
                            });
                        }
                    }
                }
            }
        }

        let mut cache = CACHED_ATLAS.write().await;
        *cache = ProjectAtlas { entries, timestamp: chrono::Utc::now().timestamp() };
    }

    pub async fn get_atlas() -> ProjectAtlas {
        let cache = CACHED_ATLAS.read().await;
        (*cache).clone()
    }

    pub async fn get_compressed_context() -> String {
        let cache = CACHED_ATLAS.read().await;
        let mut summary = String::from("PROJECT_ATLAS:\n");
        for e in cache.entries.iter().take(50) {
            summary.push_str(&format!("[{}] {} in {}\n", e.kind, e.name, e.path));
        }
        summary
    }
}

use bitflags::bitflags;
use tokio::sync::mpsc;
use crate::agent_protocol::AgentAction;

bitflags! {
    #[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
    pub struct Capabilities: u32 {
        const NONE = 0;
        const READ = 1 << 0;
        const WRITE = 1 << 1;
        const EXEC = 1 << 2;
        const NETWORK = 1 << 3;
        const ALL = Self::READ.bits() | Self::WRITE.bits() | Self::EXEC.bits() | Self::NETWORK.bits();
    }
}

pub struct AgentHandle {
    pub tx: mpsc::UnboundedSender<String>,
    pub id: String,
    pub capabilities: Capabilities,
}

impl AgentHandle {
    pub async fn execute_actions(&self, actions: Vec<AgentAction>) -> Result<Vec<String>, String> {
        let mut results = Vec::new();
        for action in actions {
            // Route the legacy action straight to the async loop
            let _ = self.tx.send(format!("{:?}", action));
            results.push("ACTION_ROUTED_TO_CORTEX".to_string());
        }
        Ok(results)
    }
}

pub struct AgentComponent;

impl AgentComponent {
    pub fn spawn(id: &str, _caps: Capabilities) -> AgentHandle {
        let (tx, mut rx) = mpsc::unbounded_channel::<String>();
        let agent_id = id.to_string();
        
        tokio::spawn(async move {
            while let Some(msg) = rx.recv().await {
                println!("\x1b[36m>> [MICRO-AGENT {}] Directive received: {}\x1b[0m", agent_id, msg);
            }
        });

        AgentHandle { tx, id: id.to_string(), capabilities: Capabilities::ALL }
    }

    // Generic wrapper to swallow the 3 extra legacy arguments
    pub fn spawn_legacy<A, T, M>(id: &str, caps: Capabilities, _agent: A, _tx: T, _memory: M) -> AgentHandle {
        Self::spawn(id, caps) 
    }
}

use anyhow::{anyhow, Result};

pub struct DiffEngine;

impl DiffEngine {
    pub fn apply(original: &str, search: &str, replace: &str) -> Result<String> {
        if search.is_empty() {
            return Ok(original.to_string());
        }

        // STRICT PROTOCOL: Exact match only, preserving all whitespace/indentation.
        if !original.contains(search) {
            return Err(anyhow!(
                "PATTERN_NOT_FOUND: The exact search string (including indentation) was not located."
            ));
        }

        Ok(original.replace(search, replace))
    }

    pub fn unified_diff(_old: &str, _new: &str, path: &str) -> String {
        format!("--- a/{}\n+++ b/{}\n[MODIFICATION APPLIED]", path, path)
    }
}

const ALLOWED_BINARIES: &[&str] = &["cargo", "ls", "cat", "grep", "find", "echo", "mkdir", "rm", "zig"];
const PROTECTED_CORE: &[&str] = &["src/lib.rs", "src/main.rs"];
const OVERRIDE_KEY: &str = "OVERRIDE_SOVEREIGN_GUARD";

static REFLEX: Lazy<Mutex<Option<litert_bridge::GemmaReflex>>> = Lazy::new(|| Mutex::new(None));

pub struct Executor;

impl Executor {
    fn check_guard(path: &str, content: Option<&str>) -> Result<()> {
        if PROTECTED_CORE.contains(&path) {
            match content {
                Some(c) if c.contains(OVERRIDE_KEY) => Ok(()),
                _ => Err(anyhow!("RESTRICTED: Modification of '{}' requires OVERRIDE_KEY.", path))
            }
        } else { Ok(()) }
    }

    pub async fn execute_handle(actions: Vec<AgentAction>, _memory: Arc<Mutex<MemoryPalace>>) -> Result<Vec<String>> {
        let mut obs = Vec::new();
        for action in actions {
            match action {
                AgentAction::LogicCheck { input: _input } => {
                    let mut rg = REFLEX.lock().await;
                    if rg.is_none() { 
                        *rg = Some(litert_bridge::GemmaReflex::init("models/gemma-4-e2b.litertlm")?); 
                    }
                    obs.push(format!("LOGIC_SUCCESS: {}", rg.as_ref().unwrap().execute_logic_strand()?));
                }
                AgentAction::WriteFile { path, content } => {
                    Self::check_guard(&path, Some(&content))?;
                    fs::write(&path, &content)?;
                    obs.push(format!("WRITE_SUCCESS: {}", path));
                }
                AgentAction::ExecuteCommand { command } => {
                    let cmd = command.split_whitespace().next().unwrap_or("");
                    if ALLOWED_BINARIES.contains(&cmd) {
                        let out = Command::new("sh").args(["-c", &command]).output()?;
                        obs.push(format!("EXEC_SUCCESS: {}", String::from_utf8_lossy(&out.stdout)));
                    } else { obs.push(format!("SECURITY_FAIL: {} restricted.", cmd)); }
                }
                _ => obs.push("ACTION_PROCESSED".into()),
            }
        }
        Ok(obs)
    }
}

pub fn extract_actions(response: &str) -> Result<Vec<AgentAction>> {
    let start_tag = "<ACTION>";
    let end_tag = "</ACTION>";

    if let Some(start_pos) = response.find(start_tag) {
        if let Some(end_pos) = response.find(end_tag) {
            let raw_json = &response[start_pos + start_tag.len()..end_pos].trim();
            
            // SUTURE: Resiliency against Markdown backticks
            let mut cleaned = raw_json.to_string();
            if cleaned.starts_with("```") {
                cleaned = cleaned.trim_start_matches('`')
                    .trim_start_matches("json")
                    .trim_end_matches('`')
                    .trim()
                    .to_string();
            }

            let actions: Vec<AgentAction> = serde_json::from_str(&cleaned)
                .map_err(|e| anyhow!("JSON_PARSE_ERR: {} | RAW: {}", e, cleaned))?;
            return Ok(actions);
        }
    }
    Err(anyhow!("PROTOCOL_VIOLATION: Missing <ACTION> tags."))
}

use std::fmt;

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum Severity {
    Critical, // CVSS 9.0-10.0
    High,     // CVSS 7.0-8.9
    Medium,   // CVSS 4.0-6.9
    Low,      // CVSS 0.1-3.9
}

#[derive(Debug, Clone)]
pub enum KernelError {
    VmoCloneFailed { parent: u32, size: usize },
    VmarMapFailed { vmar_handle: u32, size: usize },
    InvalidHandle(u32),
    AlignmentAnomaly { address: usize },
    SecurityViolation { message: String, cvss: f32 },
}

impl KernelError {
    pub fn severity(&self) -> Severity {
        match self {
            Self::SecurityViolation { cvss, .. } => {
                if *cvss >= 9.0 {
                    Severity::Critical
                } else if *cvss >= 7.0 {
                    Severity::High
                } else {
                    Severity::Medium
                }
            }
            Self::VmarMapFailed { .. } => Severity::High,
            Self::VmoCloneFailed { .. } => Severity::High,
            Self::AlignmentAnomaly { .. } => Severity::Medium,
            Self::InvalidHandle(_) => Severity::Low,
        }
    }
}

impl fmt::Display for KernelError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        let label = match self.severity() {
            Severity::Critical => "![CRITICAL]!",
            Severity::High => "[HIGH]",
            Severity::Medium => "[MEDIUM]",
            Severity::Low => "[LOW]",
        };
        match self {
            Self::VmoCloneFailed { parent, size } => {
                write!(f, "{} VMO_CLONE_FAIL: parent={} size={}", label, parent, size)
            }
            Self::VmarMapFailed { vmar_handle, size } => {
                write!(f, "{} VMAR_MAP_FAIL: handle=0x{:x} size={}", label, vmar_handle, size)
            }
            Self::InvalidHandle(h) => write!(f, "{} INVALID_HANDLE: 0x{:x}", label, h),
            Self::AlignmentAnomaly { address } => {
                write!(f, "{} ALIGN_FAIL: address=0x{:x}", label, address)
            }
            Self::SecurityViolation { message, cvss } => {
                write!(f, "{} SECURITY: {} (CVSS: {:.1})", label, message, cvss)
            }
        }
    }
}

impl std::error::Error for KernelError {}

/// Vmo: Virtual Memory Object owning a kernel handle.
pub struct Vmo {
    pub handle: u32,
    pub size: usize,
    ops: Arc<dyn KernelOps>,
}

impl Vmo {
    /// clone: Generates a child VMO using the provided hardware interface.
    pub fn clone(ops: Arc<dyn KernelOps>, parent: u32, size: usize) -> Result<Self, KernelError> {
        if parent == 0 {
            return Err(KernelError::InvalidHandle(0));
        }

        let handle = ops.vmo_clone(parent, size);
        if handle == 0 {
            Err(KernelError::VmoCloneFailed { parent, size })
        } else {
            Ok(Self { handle, size, ops })
        }
    }
}

impl Drop for Vmo {
    fn drop(&mut self) {
        self.ops.handle_close(self.handle);
    }
}

#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct Snapshot {
    pub hash: String,
    pub path: String,
    pub backup_path: String,
    pub timestamp: u64,
}

pub struct LazarusVault;

impl LazarusVault {
    const VAULT_DIR: &'static str = ".lazarus_vault";
    const LEDGER: &'static str = ".lazarus_vault/ledger.json";

    pub fn auto_snapshot(path: &str, _tag: &str) -> anyhow::Result<Snapshot> {
        let content = fs::read(path)?;
        let mut hasher = Sha256::new();
        hasher.update(&content);
        let hash = hasher.finalize().iter().map(|b| format!("{:02x}", b)).collect::<String>();

        let vault = Path::new(Self::VAULT_DIR);
        if !vault.exists() {
            fs::create_dir_all(vault)?;
        }

        let backup_path = vault.join(format!("{}.bak", &hash[..12]));
        fs::write(&backup_path, &content)?;

        let snap = Snapshot {
            hash: hash.clone(),
            path: path.to_string(),
            backup_path: backup_path.to_string_lossy().to_string(),
            timestamp: std::time::SystemTime::now()
                .duration_since(std::time::UNIX_EPOCH)?
                .as_secs(),
        };

        // Append to ledger
        let mut snaps = Self::list_snapshots().unwrap_or_default();
        snaps.retain(|s| s.hash != hash); // deduplicate
        snaps.push(snap.clone());
        if snaps.len() > 50 {
            snaps.drain(0..snaps.len() - 50);
        }
        fs::write(Self::LEDGER, serde_json::to_string_pretty(&snaps)?)?;

        Ok(snap)
    }

    pub fn list_snapshots() -> anyhow::Result<Vec<Snapshot>> {
        let p = Path::new(Self::LEDGER);
        if !p.exists() {
            return Ok(vec![]);
        }
        let data = fs::read_to_string(p)?;
        Ok(serde_json::from_str(&data).unwrap_or_default())
    }

    pub fn restore_snapshot(hash_prefix: &str) -> anyhow::Result<String> {
        let snaps = Self::list_snapshots()?;
        let snap = snaps
            .iter()
            .find(|s| s.hash.starts_with(hash_prefix))
            .ok_or_else(|| anyhow::anyhow!("Snapshot not found: {}", hash_prefix))?;
        let content = fs::read(&snap.backup_path)?;
        fs::write(&snap.path, content)?;
        Ok(format!(">> [LAZARUS] Restored {} from {}", snap.path, &snap.hash[..8]))
    }
}

        unsafe {
            // SUTURE: Explicit cast to *const i8 applied here for ARM64 DNA
            let m_ptr = TfLiteModelCreateFromFile(c_path.as_ptr() as *const i8);
            if m_ptr == 0 { return Err(anyhow!("LOAD_FAIL")); }
            let options = TfLiteInterpreterOptionsCreate();
            let i_ptr = TfLiteInterpreterCreate(m_ptr, options);

    }

impl MemoryPalace {
    pub async fn new(db_path: &str) -> Result<Self, sqlx::Error> {
        if !Path::new(db_path).exists() {
            std::fs::File::create(db_path).ok();
        }
        let pool = SqlitePoolOptions::new()
            .max_connections(5)
            .connect(&format!("sqlite:{}", db_path))
            .await?;

        sqlx::query("PRAGMA journal_mode=WAL;").execute(&pool).await?;

        let palace = Self { pool };
        palace.init_full_schema().await?;
        Ok(palace)
    }

    pub async fn init_full_schema(&self) -> Result<(), sqlx::Error> {
        sqlx::query("CREATE TABLE IF NOT EXISTS neural_context (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            origin TEXT NOT NULL,
            directive TEXT NOT NULL,
            result TEXT NOT NULL,
            timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
        );").execute(&self.pool).await?;

        sqlx::query("CREATE VIRTUAL TABLE IF NOT EXISTS memory_search USING fts5(
            directive, result, content='neural_context', content_rowid='id'
        );").execute(&self.pool).await?;

        sqlx::query("CREATE TABLE IF NOT EXISTS healing_metrics (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            layer TEXT NOT NULL,
            success INTEGER NOT NULL,
            timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
            cvss REAL,
            notes TEXT
        );").execute(&self.pool).await?;
        Ok(())
    }

    pub async fn store_memory(&self, origin: &str, directive: &str, result: &str) -> Result<(), sqlx::Error> {
        let res = sqlx::query("INSERT INTO neural_context (origin, directive, result) VALUES (?, ?, ?)")
            .bind(origin).bind(directive).bind(result).execute(&self.pool).await?;
        let row_id = res.last_insert_rowid();
        sqlx::query("INSERT INTO memory_search (rowid, directive, result) VALUES (?, ?, ?)")
            .bind(row_id).bind(directive).bind(result).execute(&self.pool).await?;
        Ok(())
    }

    // --- SUTURE: DAYDREAM HOOKS ---
    pub async fn fetch_random_strands(&self, limit: i64) -> Result<Vec<AgentMemory>, sqlx::Error> {
        let rows = sqlx::query("SELECT id, origin, directive, result, timestamp FROM neural_context ORDER BY RANDOM() LIMIT ?")
            .bind(limit).fetch_all(&self.pool).await?;
        Ok(rows.into_iter().map(|row| AgentMemory {
            id: row.get("id"), origin: row.get("origin"), directive: row.get("directive"),
            result: row.get("result"), timestamp: row.get("timestamp"),
        }).collect())
    }

    // --- SUTURE: PRUNING HOOKS ---

        let rows = sqlx::query("SELECT id, origin, directive, result, timestamp FROM neural_context ORDER BY id ASC LIMIT ?")
            .bind(limit).fetch_all(&self.pool).await?;
        Ok(rows.into_iter().map(|row| AgentMemory {

            result: row.get("result"), timestamp: row.get("timestamp"),
        }).collect())
    }

    pub async fn update_strand_summary(&self, id: i64, summary: &str) -> Result<(), sqlx::Error> {
        sqlx::query("UPDATE neural_context SET result = ? WHERE id = ?")
            .bind(summary).bind(id).execute(&self.pool).await?;
        // Update the FTS table to match the new summarized content
        sqlx::query("INSERT INTO memory_search(memory_search, rowid, directive, result) VALUES('delete', ?, ?, ?)")
            .bind(id).execute(&self.pool).await.ok(); // Attempt delete of old index
        sqlx::query("INSERT INTO memory_search(rowid, directive, result) SELECT id, directive, result FROM neural_context WHERE id = ?")
            .bind(id).execute(&self.pool).await?;
        Ok(())
    }

    pub async fn search_memories(&self, query: &str) -> Result<Vec<AgentMemory>, sqlx::Error> {
        let rows = sqlx::query("SELECT neural_context.id, neural_context.origin, neural_context.directive, neural_context.result, neural_context.timestamp
                                FROM neural_context
                                JOIN memory_search ON neural_context.id = memory_search.rowid
                                WHERE memory_search MATCH ? ORDER BY rank LIMIT 5")
            .bind(query).fetch_all(&self.pool).await?;

        Ok(rows.into_iter().map(|r| AgentMemory {
            id: r.get("id"), origin: r.get("origin"), directive: r.get("directive"),
            result: r.get("result"), timestamp: r.get("timestamp")

            .fetch_all(&self.pool).await?;

            id: row.get("id"), origin: row.get("origin"), directive: row.get("directive"),

    }

        match result {
            Ok(Some(row)) => row.get::<Option<f32>, _>("rate").unwrap_or(0.78),
            _ => 0.78,
        }
    }

pub struct NexusMetrics {
    pub tasks_total: AtomicU64,
    pub task_failures: AtomicU64,
    pub arbiter_blocks: AtomicU64,
    pub heal_successes: AtomicU64,
    pub heal_failures: AtomicU64,
    pub api_latency_ms_sum: AtomicU64,
    pub api_calls: AtomicU64,
    pub circuit_opens: AtomicU64,
}

impl NexusMetrics {
    const fn new() -> Self {
        Self {
            tasks_total: AtomicU64::new(0),
            task_failures: AtomicU64::new(0),
            arbiter_blocks: AtomicU64::new(0),
            heal_successes: AtomicU64::new(0),
            heal_failures: AtomicU64::new(0),
            api_latency_ms_sum: AtomicU64::new(0),
            api_calls: AtomicU64::new(0),
            circuit_opens: AtomicU64::new(0),
        }
    }

    pub fn record_api_call(&self, latency_ms: u128) {
        self.api_calls.fetch_add(1, Ordering::Relaxed);
        self.api_latency_ms_sum.fetch_add(latency_ms as u64, Ordering::Relaxed);
    }

    pub fn avg_latency_ms(&self) -> u64 {
        let calls = self.api_calls.load(Ordering::Relaxed);
        if calls == 0 {
            return 0;
        }
        self.api_latency_ms_sum.load(Ordering::Relaxed) / calls
    }

    pub fn inc(&self) {
        self.tasks_total.fetch_add(1, Ordering::Relaxed);
    }
}

// src/mission_planner.rs - v3.2 (Memory Flow Integration)
use crate::agent_protocol::AgentAction;
use crate::component_runner::{AgentHandle, Capabilities};
use crate::server::AppState;
use serde::{Deserialize, Serialize};
use std::sync::Arc;

#[derive(Serialize, Deserialize, Clone, Debug, PartialEq)]
pub enum StepStatus {
    Pending,
    Running,
    Done,
    Failed(String),
}

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct MissionStep {
    pub id: usize,
    pub description: String,
    pub action: Option<AgentAction>,
    pub status: StepStatus,
    pub result: Option<String>,
}

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct Mission {
    pub id: String,
    pub goal: String,
    pub steps: Vec<MissionStep>,
    pub created_at: String,
    pub completed: bool,
}

pub struct MissionPlanner;

impl MissionPlanner {
    pub async fn execute(state: &Arc<AppState>, mut mission: Mission) -> anyhow::Result<Mission> {
        let (lifecycle_tx, _) = tokio::sync::broadcast::channel::<String>(16);

        // SUTURE: Added state.memory.clone() to satisfy new spawn signature
        let handle: AgentHandle = crate::component_runner::AgentComponent::spawn_legacy(
            &format!("mission-{}", mission.id),
            Capabilities::READ | Capabilities::WRITE,
            state.agent.clone(),
            lifecycle_tx,
            state.memory.clone(),
        );

        for step in mission.steps.iter_mut() {
            if step.status != StepStatus::Pending {
                continue;
            }
            step.status = StepStatus::Running;

            if let Some(action) = step.action.clone() {
                match handle.execute_actions(vec![action]).await {
                    Ok(results) => {
                        step.status = StepStatus::Done;
                        step.result = Some(results.join("\n"));
                    }
                    Err(e) => {
                        step.status = StepStatus::Failed(e.to_string());
                        let _ = state.tx.send(format!("Mission step {} failed: {}", step.id, e));
                        break;
                    }
                }
            }
        }

        mission.completed = mission.steps.iter().all(|s| matches!(s.status, StepStatus::Done));
        Ok(mission)
    }
}

/// THERMAL-PULSE: Hardware-level temperature extraction.
export fn extract_thermal_pulse_raw() i32 {
    const paths = [_][]const u8{
        "/sys/class/thermal/thermal_zone0/temp",
        "/sys/class/thermal/thermal_zone1/temp",
        "/sys/class/hwmon/hwmon0/temp1_input",
        "/sys/class/hwmon/hwmon1/temp1_input",
    };
    var buf: [16]u8 = undefined;
    for (paths) |path| {
        const file = std.fs.openFileAbsolute(path, .{}) catch continue;
        defer file.close();
        const bytes_read = file.readAll(&buf) catch continue;
        if (bytes_read == 0) continue;

}

/// ALPHA-SCOUT: High-speed payload extraction.
export fn extract_payload_strands(
    input_ptr: [*]const u8,
    input_len: usize,
    target_ptr: [*]const u8,
    target_len: usize,
    output_ptr: [*]u8,
    output_max: usize,
) usize {
    if (input_len == 0 or target_len == 0 or output_max == 0) return 0;
    const input = input_ptr[0..input_len];
    const target = target_ptr[0..target_len];
    if (std.mem.indexOf(u8, input, target)) |start_idx| {
        const data_start = start_idx + target_len;
        var end_idx = data_start;
        while (end_idx < input_len and input[end_idx] != '"' and input[end_idx] != '<' and input[end_idx] != '\n') {
            end_idx += 1;
        }
        const extracted_len = end_idx - data_start;
        if (extracted_len > 0 and extracted_len <= output_max) {
            std.mem.copyForwards(u8, output_ptr[0..extracted_len], input[data_start..end_idx]);
            return extracted_len;
        }
    }
    return 0;
}

pub fn validate_path(requested_path: &str) -> Result<PathBuf> {
    let workspace_root = env::current_dir().context("Failed to resolve workspace root")?;

    // SUTURE: Allow read-only access to specific system diagnostics
    let is_diagnostic = requested_path == "/proc/meminfo" || requested_path == "/proc/loadavg";

    if !is_diagnostic && (
        requested_path.contains("..")
        || requested_path.contains('\0')
        || requested_path.starts_with('/')
        || requested_path.starts_with('~')
        || ["/sys", "/dev", "/etc", "/home"].iter().any(|p| requested_path.contains(p))
    ) {
        return Err(anyhow!("SECURITY VIOLATION: Forbidden path pattern"));
    }

    if is_diagnostic {
        return Ok(PathBuf::from(requested_path));
    }

    let target = Path::new(requested_path);
    let absolute = if target.is_absolute() { target.to_path_buf() } else { workspace_root.join(target) };
    let mut check_path = absolute.clone();
    if !check_path.exists() {
        if let Some(parent) = check_path.parent() { check_path = parent.to_path_buf(); }
    }

    let canonical = check_path.canonicalize().map_err(|e| anyhow!("Path resolution failed: {}", e))?;
    let canonical_root = workspace_root.canonicalize().context("Failed to canonicalize workspace root")?;

    if !canonical.starts_with(&canonical_root) {
        return Err(anyhow!("[CRITICAL] Path escape detected"));
    }

    Ok(absolute)
}

    memory: Arc<Mutex<MemoryPalace>>,
    governor: ThermalGovernor,
}

impl SelfHealingEngine {
    pub fn new(state: Arc<AppState>, memory: Arc<Mutex<MemoryPalace>>) -> Self {
        Self { state, memory, governor: ThermalGovernor::new() }
    }

    pub async fn ignite_autonomic_loop(self) {
        println!("\x1b[36m>> [SELF-HEALING v2.1] Autonomic Loop Started (4GB safe)\x1b[0m");

        loop {
            self.governor.apply_cooldown().await;

            let vitals = self.state.monitor.get_vitals();

            // Resource guard for Lenovo Duet 4GB
            if vitals.cpu_load > 0.65 || vitals.temperature > 68.0 || vitals.ram_gb > 3.2 {
                sleep(Duration::from_secs(90)).await;
                continue;
            }

            let issues = self.diagnose().await;

            for issue in issues {
                let priority = self.calculate_priority(&issue, &vitals).await;

                if priority <= 0.52 {
                    println!(
                        ">> [SELF-HEALING] Issue priority {:.2} below threshold — skipping.",
                        priority
                    );
                    continue;
                }

                // FIX: Actually use the Arbiter result to decide whether to heal
                let arbiter = ArbiterAgent::new(
                    self.state.agent.clone(),
                    Some(self.state.orchestrator.clone()),
                );

                match arbiter.analyze_directive(&self.state.api_key, &issue.description).await {
                    Ok(report) => {
                        if !report.can_proceed {
                            println!(
                                "\x1b[33m>> [ARBITER] Healing blocked (score {:.2}): {}\x1b[0m",
                                report.risk_score, report.recommendation
                            );
                            continue;
                        }
                        println!(
                            "\x1b[32m>> [ARBITER] Authorized (score {:.2}): {}\x1b[0m",
                            report.risk_score, report.recommendation
                        );
                    }
                    Err(e) => {
                        println!(
                            "\x1b[33m>> [ARBITER] Analysis failed ({}), using fallback gate.\x1b[0m",
                            e
                        );
                        // Fallback: only proceed if CVSS is below critical
                        if issue.cvss >= 9.0 {
                            println!(">> [ARBITER] CVSS >= 9.0 — blocking without Arbiter confirmation.");
                            continue;
                        }
                    }
                }

                let mission = self.create_mission(&issue).await;
                let _ = crate::mission_planner::MissionPlanner::execute(&self.state, mission).await;
            }

            self.record_cycle().await;
            sleep(Duration::from_secs(75)).await;
        }
    }

    async fn diagnose(&self) -> Vec<HealingIssue> {
        let mut issues = Vec::new();

        if let Ok(log) = Inquisitor::audit_backend() {
            if log != "NOMINAL" {
                issues.push(HealingIssue {
                    layer: HealingLayer::Backend,
                    description: log,
                    cvss: 7.5,
                });
            }
        }
        if let Ok(log) = Inquisitor::audit_ui() {
            if log != "NOMINAL" {
                issues.push(HealingIssue {
                    layer: HealingLayer::Frontend,
                    description: log,
                    cvss: 6.8,
                });
            }
        }
        issues
    }

    async fn calculate_priority(
        &self,
        issue: &HealingIssue,
        vitals: &crate::system_monitor::Vitals,
    ) -> f32 {
        let base = issue.cvss / 10.0;
        let thermal_factor = if vitals.temperature > 65.0 { 0.6 } else { 1.0 };
        base * thermal_factor * 0.78
    }

    async fn create_mission(&self, issue: &HealingIssue) -> Mission {
        Mission {
            id: format!("heal-{}", Utc::now().timestamp()),
            goal: format!("Repair {:?}", issue.layer),
            steps: vec![
                MissionStep {
                    id: 1,
                    description: "Lazarus pre-snapshot".into(),
                    action: None,
                    status: StepStatus::Pending,
                    result: None,
                },
                MissionStep {
                    id: 2,
                    description: format!("Fix {:?}", issue.layer),
                    action: None,
                    status: StepStatus::Pending,
                    result: None,
                },
                MissionStep {
                    id: 3,
                    description: "Validate repair".into(),
                    action: None,
                    status: StepStatus::Pending,
                    result: None,
                },
            ],
            created_at: Utc::now().to_rfc3339(),
            completed: false,
        }
    }

    async fn record_cycle(&self) {
        let palace = self.memory.lock().await;
        let _ = palace
            .store_memory(
                "SelfHealingEngine",
                "autonomic_cycle",
                "Completed diagnostic + repair pass with Arbiter gating",
            )
            .await;
    }
}

async fn run_cli_loop(state: Arc<AppState>) {
    println!("\x1b[1;36m>> [NEXUS CLI] v4.2.3 Active.\x1b[0m");
    println!("\x1b[38;5;240m   Commands: task <directive> | status | memories | exit\x1b[0m\n");

    loop {
        let thermal = get_thermal_prompt(&state.monitor);
        print!("{} \x1b[38;5;51m❯\x1b[0m ", thermal);
        io::stdout().flush().unwrap_or_default();

        let mut input = String::new();
        match io::stdin().read_line(&mut input) {
            Ok(0) | Err(_) => { println!("\n>> [NEXUS] EOF — shutting down."); break; }
            Ok(_) => {}
        }

        let trimmed = input.trim();
        if trimmed.is_empty() { continue; }

        match trimmed {
            "exit" | "quit" | "q" => {
                println!("\x1b[33m>> [NEXUS] Shutdown initiated.\x1b[0m");
                break;
            }
            "status" => {
                let v = state.monitor.get_vitals();
                println!(
                    "\x1b[36m>> [STATUS] TEMP: {:.1}°C | CPU: {:.1}% | RAM: {:.2} GB\x1b[0m",
                    v.temperature, v.cpu_load * 100.0, v.ram_gb
                );
            }
            // FIX: AgentMemory fields are: id, origin, directive, result, timestamp
            // The field is .directive not .topic
            "memories" => {
                let palace = state.memory.lock().await;
                match palace.fetch_all().await {
                    Ok(mems) => {
                        println!("\x1b[35m>> [MEMORY] {} strand(s):\x1b[0m", mems.len());
                        for m in mems.iter().take(10) {
                            println!("   [{}] {}", m.origin, m.directive);
                        }
                    }
                    Err(e) => eprintln!("\x1b[31m!! [MEMORY_FAULT] {}\x1b[0m", e),
                }
            }
            // FIX: send_task returns String, not Result — no if let Err needed
            cmd if cmd.starts_with("task ") => {
                let directive = trimmed["task ".len()..].trim();
                if directive.is_empty() { continue; }
                println!("\x1b[38;5;51m>> [CLI_UPLINK] {}\x1b[0m", directive);
                task_manager::send_task(
                    state.agent.clone(), &state.client, &state.api_key,
                    directive, state.memory.clone(), state.tx.clone(),
                ).await;
                println!("\x1b[32m>> [CLI] Task complete.\x1b[0m");
            }
            other => {
                println!("\x1b[38;5;240m>> Treating as directive: {}\x1b[0m", other);
                task_manager::send_task(
                    state.agent.clone(), &state.client, &state.api_key,
                    other, state.memory.clone(), state.tx.clone(),
                ).await;
            }
        }
    }
}

fn get_thermal_prompt(monitor: &crate::system_monitor::SystemMonitor) -> String {
    let v = monitor.get_vitals();
    let temp = v.temperature;
    let color = if temp > 75.0 { "\x1b[31m" } else if temp > 60.0 { "\x1b[33m" } else { "\x1b[32m" };
    format!("{color}[{:.1}°C]\x1b[0m", temp)
}

async fn handle_exterior_vision(
    State(state): State<Arc<AppState>>,
    Json(payload): Json<SensoryInput>,
) -> impl IntoResponse {
    let frames = payload.image_frames.unwrap_or_default();
    let base64_image: Option<String> = frames.into_iter().next();
    let agent    = state.agent.clone();
    let client   = state.client.clone();
    let api_key  = state.api_key.clone();
    let tx       = state.tx.clone();
    let directive = payload.directive.clone().unwrap_or_default();
    let source    = payload.source.clone().unwrap_or_default();

    tokio::spawn(async move {
        let _ = tx.send(format!(">> [VISION] OPTICAL OVERRIDE: \"{}\"\n", directive));

        // Encode the image into the system prompt text since ContextPacket
        // does not have image_base64 / vision_source fields yet.
        let image_context = if let Some(b64) = base64_image {
            format!("\n[IMAGE_BASE64_ATTACHED from {}]\nData length: {} chars\n", source, b64.len())
        } else {
            format!("\n[NO_IMAGE from {}]\n", source)
        };

        let ctx = crate::neural_agent::ContextPacket {
            system_prompt: Some(format!(
                "You are the Ghost of NEXUS. You are an autonomous agent running on ARM64 hardware.{}",
                image_context
            )),
            ..Default::default()
        };

        let current_prompt = format!("OBJECTIVE: {}\n\nDescribe precisely what you see.", directive);
        let _ = tx.send(">> [NEURAL_PULSE] Processing Sensory Input...\n".to_string());

        // FIX: 6th argument None = use default model (no override)
        match agent.prepare_and_run(&client, &api_key, &current_prompt, ctx, 0.0, None).await {
            Ok((_, lat)) => { let _ = tx.send(format!(">> [METRICS] Latency: {}ms\n", lat)); }
            Err(e)       => { let _ = tx.send(format!(">> [VISION_FAULT] {}\n", e)); }
        }
        let _ = tx.send(">> [NEXUS_HALT] Sensory task complete.\n".to_string());
    });

    StatusCode::ACCEPTED
}

async fn handle_fs_list(Query(p): Query<ListParams>) -> Response {
    let path = p.path.as_deref().unwrap_or(".");
    match sandbox::validate_path(path) {
        Ok(safe_path) => {
            let mut entries = Vec::new();
            if let Ok(rd) = std::fs::read_dir(&safe_path) {
                for entry in rd.flatten() {
                    let name = entry.file_name().to_string_lossy().to_string();
                    if !name.starts_with('.') && name != "target" && name != "node_modules" {
                        entries.push(FsEntry {
                            name,
                            path: entry.path().to_string_lossy().to_string(),
                            is_dir: entry.path().is_dir(),
                        });
                    }
                }
                // Sort: dirs first, then files, both alphabetically
                entries.sort_by(|a, b| {
                    b.is_dir.cmp(&a.is_dir).then(a.name.cmp(&b.name))
                });
            }
            (StatusCode::OK, Json(entries)).into_response()
        }
        Err(e) => (StatusCode::FORBIDDEN, Json(json!({"error": e.to_string()}))).into_response(),
    }
}

async fn handle_read_file(Query(p): Query<FileParams>) -> Response {
    match sandbox::validate_path(&p.path) {
        Ok(safe_path) => match std::fs::read_to_string(&safe_path) {
            Ok(c)  => (StatusCode::OK, Json(json!({ "content": c }))).into_response(),
            Err(_) => (StatusCode::NOT_FOUND, "File not found").into_response(),
        },
        Err(e) => (StatusCode::FORBIDDEN, Json(json!({"error": e.to_string()}))).into_response(),
    }
}

async fn handle_write_file(Json(payload): Json<FileContent>) -> Response {
    match sandbox::validate_path(&payload.path) {
        Ok(safe_path) => {
            if let Some(p) = safe_path.parent() { let _ = std::fs::create_dir_all(p); }
            match std::fs::write(&safe_path, &payload.content) {
                Ok(_)  => (StatusCode::OK, Json(json!({"saved": true}))).into_response(),
                Err(e) => (StatusCode::INTERNAL_SERVER_ERROR, e.to_string()).into_response(),
            }
        }
        Err(e) => (StatusCode::FORBIDDEN, Json(json!({"error": e.to_string()}))).into_response(),
    }
}

async fn handle_delete_file(Query(p): Query<FileParams>) -> Response {
    match sandbox::validate_path(&p.path) {
        Ok(safe_path) => {
            let res = if safe_path.is_dir() {
                std::fs::remove_dir_all(&safe_path)
            } else {
                std::fs::remove_file(&safe_path)
            };
            match res {
                Ok(_)  => StatusCode::OK.into_response(),
                Err(e) => (StatusCode::INTERNAL_SERVER_ERROR, e.to_string()).into_response(),
            }
        }
        Err(e) => (StatusCode::FORBIDDEN, Json(json!({"error": e.to_string()}))).into_response(),
    }
}

async fn handle_socket(mut socket: WebSocket, state: Arc<AppState>) {
    let mut rx = state.tx.subscribe();
    let mut heartbeat = tokio::time::interval(std::time::Duration::from_secs(3));

    loop {
        tokio::select! {
            Ok(msg) = rx.recv() => {
                if socket.send(Message::Text(msg.into())).await.is_err() { break; }
            }
            _ = heartbeat.tick() => {
                let v = state.monitor.get_vitals();
                let payload = json!({
                    "type": "telemetry",
                    "payload": {
                        "temperature": v.temperature,
                        "cpu_load":    v.cpu_load,
                        "ram_gb":      v.ram_gb,
                        "resonance":   0.0
                    }
                });
                if socket.send(Message::Text(payload.to_string().into())).await.is_err() { break; }
            }
            Some(Ok(msg)) = socket.recv() => {
                if let Message::Text(text) = msg {
                    if let Ok(parsed) = serde_json::from_str::<serde_json::Value>(&text) {
                        match parsed["type"].as_str() {
                            Some("directive") => {
                                let cmd = parsed["payload"].as_str().unwrap_or_default().to_string();
                                if !cmd.is_empty() {
                                    let s  = state.clone();
                                    let tx = state.tx.clone();
                                    // FIX: send_task returns String, not Result
                                    tokio::spawn(async move {
                                        task_manager::send_task(
                                            s.agent.clone(), &s.client, &s.api_key,
                                            &cmd, s.memory.clone(), tx,
                                        ).await;
                                    });
                                }
                            }
                            Some("resonance_update") => {
                                if let Some(val) = parsed["data"].as_u64() {
                                    state.current_resonance.store(
                                        val as u32,
                                        std::sync::atomic::Ordering::Relaxed,
                                    );
                                }
                            }
                            Some("ping") => {
                                let _ = socket.send(Message::Text(
                                    json!({"type":"pong"}).to_string().into()
                                )).await;
                            }
                            _ => {}
                        }
                    }
                }
            }
            else => break,
        }
    }
}

// ═══════════════════════════════════════════════════════════════
//  STATIC FILE HANDLER (SPA fallback)
// ═══════════════════════════════════════════════════════════════

async fn static_handler(uri: Uri) -> impl IntoResponse {
    let path = uri.path().trim_start_matches('/').to_string();
    let final_path = if path.is_empty() { "index.html".to_string() } else { path };

    match Assets::get(&final_path) {
        Some(content) => (
            [(header::CONTENT_TYPE, mime_guess::from_path(&final_path).first_or_octet_stream().as_ref())],
            content.data,
        ).into_response(),
        None => {
            if !final_path.contains('.') {
                if let Some(index) = Assets::get("index.html") {
                    return ([(header::CONTENT_TYPE, "text/html")], index.data).into_response();
                }

use bytes::{Bytes, BytesMut};
use futures_util::Stream;
use std::pin::Pin;
use std::task::{Context, Poll};

pub struct NeuralChunker<S> {
    stream: S,
    buffer: BytesMut,
    chunk_size: usize,
}

impl<S> NeuralChunker<S> {
    pub fn new(stream: S, chunk_size: usize) -> Self {
        Self {
            stream,
            buffer: BytesMut::with_capacity(chunk_size * 2),
            chunk_size,
        }
    }
}

impl<S> Stream for NeuralChunker<S>
where
    S: Stream<Item = Result<Bytes, reqwest::Error>> + Unpin,
{
    type Item = Result<String, String>;

    fn poll_next(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
        // Continue polling the inner stream until we reach chunk_size or EOF
        while self.buffer.len() < self.chunk_size {
            match Pin::new(&mut self.stream).poll_next(cx) {
                Poll::Ready(Some(Ok(bytes))) => self.buffer.extend_from_slice(&bytes),
                Poll::Ready(Some(Err(e))) => return Poll::Ready(Some(Err(e.to_string()))),
                Poll::Ready(None) => break,
                Poll::Pending => return Poll::Pending,
            }
        }

        if self.buffer.is_empty() {
            return Poll::Ready(None);
        }

        // Split the buffer at the chunk boundary or remaining length
        let split_at = std::cmp::min(self.buffer.len(), self.chunk_size);
        let chunk = self.buffer.split_to(split_at);
        
        Poll::Ready(Some(Ok(String::from_utf8_lossy(&chunk).to_string())))
    }
}

use std::fs;
use serde::Serialize;

#[derive(Debug, Clone, Default, Serialize)]
pub struct Vitals {
    pub ram_available_mb: u64,
    pub ram_gb: f32,
    pub temperature: f32,
    pub cpu_load: f32,
    pub is_throttled: bool,
}

#[derive(Clone)]
pub struct SystemMonitor {
    pub memory_threshold_bytes: u64,
    pub thermal_limit_celsius: f32,
    active_thermal_path: String,
}

impl SystemMonitor {
    pub fn new() -> Self {
        let mut best_path = "/sys/class/thermal/thermal_zone0/temp".to_string();
        
        // Probe zones 0-9 for real data (millidegrees > 1000)
        for i in 0..10 {
            let p = format!("/sys/class/thermal/thermal_zone{}/temp", i);
            if let Ok(val) = fs::read_to_string(&p) {
                if let Ok(temp) = val.trim().parse::<f32>() {
                    if temp > 1000.0 {
                        best_path = p;
                        break;
                    }
                }
            }
        }

        Self {
            memory_threshold_bytes: 800 * 1024 * 1024,
            thermal_limit_celsius: 42.5,
            active_thermal_path: best_path,
        }
    }

    pub fn get_vitals(&self) -> Vitals {
        let (mem_avail, total_mem) = self.get_ram_stats();
        let cpu_temp = self.get_cpu_temp();
        let load = self.get_cpu_load();
        
        let ram_gb = (total_mem as f32 - mem_avail as f32) / 1024.0 / 1024.0;
        let is_throttled = cpu_temp > self.thermal_limit_celsius;

        Vitals {
            ram_available_mb: mem_avail / 1024,
            ram_gb,
            temperature: cpu_temp,
            cpu_load: load,
            is_throttled,
        }
    }

    fn get_ram_stats(&self) -> (u64, u64) {
        let meminfo = fs::read_to_string("/proc/meminfo").unwrap_or_default();
        let mut avail = 0;
        let mut total = 0;
        for line in meminfo.lines() {
            if line.starts_with("MemTotal:") {
                total = line.split_whitespace().nth(1).and_then(|v| v.parse::<u64>().ok()).unwrap_or(0) * 1024;
            }
            if line.starts_with("MemAvailable:") {
                avail = line.split_whitespace().nth(1).and_then(|v| v.parse::<u64>().ok()).unwrap_or(0) * 1024;
            }
        }
        (avail, total)
    }

    fn get_cpu_load(&self) -> f32 {
        let loadavg = fs::read_to_string("/proc/loadavg").unwrap_or_default();
        loadavg.split_whitespace().next().and_then(|v| v.parse::<f32>().ok()).map(|v| v / 8.0).unwrap_or(0.0)
    }

    pub fn get_cpu_temp(&self) -> f32 {
        fs::read_to_string(&self.active_thermal_path)
            .ok()
            .and_then(|s| s.trim().parse::<f32>().ok())
            .map(|t| t / 1000.0)
            .unwrap_or(-1.0) 
    }
}

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum CoolingState {
    Normal,
    Throttled,
    Critical,
}

pub struct ThermalGovernor {
    pub warn_threshold: f32,
    pub crit_threshold: f32,
}

impl Default for ThermalGovernor {
    fn default() -> Self {
        Self { warn_threshold: 68.0, crit_threshold: 78.0 }
    }
}

impl ThermalGovernor {
    pub fn new() -> Self {
        Self::default()
    }

    fn read_temp() -> f32 {
        for zone in 0..5 {
            let path = format!("/sys/class/thermal/thermal_zone{}/temp", zone);
            if let Ok(raw) = fs::read_to_string(&path) {
                if let Ok(t) = raw.trim().parse::<f32>() {
                    if t > 0.0 {
                        return t / 1000.0;
                    }
                }
            }
        }
        0.0
    }

    pub fn get_state(&self) -> CoolingState {
        let t = Self::read_temp();
        if t >= self.crit_threshold {
            CoolingState::Critical
        } else if t >= self.warn_threshold {
            CoolingState::Throttled
        } else {
            CoolingState::Normal
        }
    }

    pub async fn apply_cooldown(&self) {
        match self.get_state() {
            CoolingState::Critical => {
                println!(
                    "\x1b[31m>> [THERMAL] CRITICAL {}°C — Aggressive cooldown\x1b[0m",
                    Self::read_temp()
                );
                sleep(Duration::from_secs(12)).await;
            }
            CoolingState::Throttled => {
                println!(
                    "\x1b[33m>> [THERMAL] WARN {}°C — Light cooldown\x1b[0m",
                    Self::read_temp()
                );
                sleep(Duration::from_secs(5)).await;
            }
            _ => {}
        }
    }
}

use std::fs;
use std::path::{Path, PathBuf};

pub struct VoidWalker;

impl VoidWalker {
    /// Deep-scans the project to ensure every .rs file is linked to the module tree.
    pub fn identify_orphans() -> Vec<String> {
        let mut orphans = Vec::new();
        let mut declared_mods = Vec::new();

        // 1. Map the entire Module Tree starting from roots
        let root_files = ["src/main.rs", "src/lib.rs"];
        for root in &root_files {
            if let Ok(content) = fs::read_to_string(root) {
                Self::trace_module_tree(Path::new(root), &content, &mut declared_mods);
            }
        }

        // 2. Recursive Scan of the physical filesystem
        let mut all_files = Vec::new();
        let _ = Self::visit_dirs(Path::new("src"), &mut all_files);

        // 3. Cross-reference absolute paths to prevent name collisions
        for file_path in all_files {
            let path_str = file_path.to_string_lossy().to_string();
            
            // Safe Zone: Entry points and standalone binaries
            if path_str.contains("/bin/") || path_str.ends_with("main.rs") || path_str.ends_with("lib.rs") || path_str.ends_with("mod.rs") {
                continue;
            }

            if !declared_mods.iter().any(|m| path_str.ends_with(m)) {
                orphans.push(path_str);
            }
        }
        orphans
    }

    /// Recursively traces mod declarations to find their associated files
    fn trace_module_tree(current_file: &Path, content: &str, whitelist: &mut Vec<String>) {
        for line in content.lines() {
            let trimmed = line.trim();
            if trimmed.starts_with("mod ") || trimmed.starts_with("pub mod ") {
                if let Some(mod_name) = trimmed.split_whitespace().nth(if trimmed.starts_with("pub") { 2 } else { 1 }) {
                    let name = mod_name.trim_end_matches(';');
                    
                    // Predict potential file paths for this module
                    let parent = current_file.parent().unwrap_or(Path::new(""));
                    let possible_paths = vec![
                        parent.join(format!("{}.rs", name)),
                        parent.join(name).join("mod.rs"),
                        parent.join(current_file.file_stem().unwrap_or_default()).join(format!("{}.rs", name)),
                    ];

                    for path in possible_paths {
                        if path.exists() {
                            let path_key = path.to_string_lossy().to_string();
                            if !whitelist.contains(&path_key) {
                                whitelist.push(path_key);
                                // Follow the rabbit hole: Read and trace the sub-module
                                if let Ok(sub_content) = fs::read_to_string(&path) {
                                    Self::trace_module_tree(&path, &sub_content, whitelist);
                                }
                            }
                        }
                    }
                }
            }
        }
    }

    fn visit_dirs(dir: &Path, files: &mut Vec<PathBuf>) -> std::io::Result<()> {
        if dir.is_dir() {
            for entry in fs::read_dir(dir)? {
                let entry = entry?;
                let path = entry.path();
                if path.is_dir() {
                    Self::visit_dirs(&path, files)?;
                } else if path.extension().and_then(|s| s.to_str()) == Some("rs") {
                    files.push(path);
                }
            }
        }
        Ok(())
    }
}
