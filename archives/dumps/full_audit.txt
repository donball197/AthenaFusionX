                    return Ok(vec![action]);
                }

    }

pub struct Executor;

impl Executor {
    pub async fn execute(actions: Vec<AgentAction>) -> Result<Vec<String>, NexusError> {
        let mut results = vec![];
        for action in actions {
            match action {
                AgentAction::WriteFile { path, content } => {
                    if let Err(e) = SyntaxSentinel::validate_safety(&path, &content) {
                        results.push(format!("REJECTED: {}", e));
                        continue;
                    }

                    // SUTURE: Ensure parent directory exists before writing
                    if let Some(parent) = Path::new(&path).parent() {
                        let _ = fs::create_dir_all(parent);
                    }

                    let _ = LazarusVault::seal_patch(&path, &content, "SWARM_WRITE");
                    if fs::write(&path, &content).is_ok() {
                        results.push(format!("SUCCESS: Updated {}", path));
                    } else {
                        results.push(format!("WRITE_FAIL: Check permissions for {}", path));
                    }
                }
                AgentAction::PatchFile { path, search, replace } => {
                    let original = fs::read_to_string(&path)
                        .map_err(|_| NexusError::NotFound(format!("File not found: {}", path)))?;

                    let patched = DiffEngine::apply(&original, &search, &replace).unwrap_or(original);

                    let _ = LazarusVault::seal_patch(&path, &patched, "SWARM_PATCH");
                    if fs::write(&path, &patched).is_ok() {
                        results.push(format!("SUCCESS: Patched {}", path));
                    } else {
                        results.push("WRITE_FAIL after patch".into());
                    }
                }
                AgentAction::RunCommand { command } => {
                    let cmd = command.trim();
                    if cmd.starts_with("cat ") {
                        let path = cmd.replace("cat ", "").trim().to_string();
                        match fs::read_to_string(&path) {
                            Ok(code) => results.push(format!("FILE_CONTENT [{}]:\n{}", path, code)),
                            Err(_) => results.push(format!("READ_FAIL: {}", path)),
                        }
                    } else if ["cargo check", "ls", "cargo fmt"].iter().any(|&c| cmd.starts_with(c)) {
                        results.push(format!("CMD_EXEC_MOCKED: {}", cmd));
                    } else {
                        results.push(format!("CMD_REJECTED: '{}'", command));
                    }
                }
                AgentAction::Message { message } => {
                    results.push(format!("MSG: {}", message));
                }
            }
        }
        Ok(results)
    }
}

use serde::{Deserialize, Deserializer, Serialize};
use serde_json::Value;

#[derive(Debug, Serialize, Clone)]
#[serde(tag = "action")]
pub enum AgentAction {
    WriteFile { path: String, content: String },
    PatchFile { path: String, search: String, replace: String },
    RunCommand { command: String },
    Message { message: String },
}

impl<'de> Deserialize<'de> for AgentAction {
    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
    where
        D: Deserializer<'de>,
    {
        let v: Value = Deserialize::deserialize(deserializer)?;
        
        let action_type = v.get("action")
            .or_else(|| v.get("command"))
            .and_then(Value::as_str)
            .unwrap_or("Message");

        let args = v.get("arguments").unwrap_or(&v);

        let get_str = |key: &str, alt: &str| -> String {
            args.get(key)
                .or_else(|| args.get(alt))
                .and_then(Value::as_str)
                .unwrap_or("")
                .to_string()
        };

        match action_type {
            "WriteFile" => Ok(AgentAction::WriteFile {
                path: get_str("path", "file"),
                content: get_str("content", "text"),
            }),
            "PatchFile" => Ok(AgentAction::PatchFile {
                path: get_str("path", "file"),
                search: get_str("search", "diff"), // Fallback if LLM uses 'diff'
                replace: get_str("replace", "content"),
            }),
            "RunCommand" => Ok(AgentAction::RunCommand {
                command: get_str("command", "cmd"),
            }),
            _ => Ok(AgentAction::Message {
                message: get_str("message", "text"),
            }),
        }
    }
}

    let parts: Vec<&str> = cmd_line.split_whitespace().collect();
    let command = parts[0].to_lowercase();

    let response = match command.as_str() {

        },

        },

            if let Ok(rd) = std::fs::read_dir(target) {
                for entry in rd.flatten() {
                    let name = entry.file_name().to_string_lossy().to_string();
                    if !name.starts_with('.') && name != "target" { 
                        let suffix = if entry.path().is_dir() { "/" } else { "" };
                        out.push_str(&format!("   {}{}\n", name, suffix)); 
                    }
                }
            } else {
                out.push_str("   [Directory not found or inaccessible]\n");
            }
            json!({"type": "output", "data": out}).to_string()
        },

        },

                if let Some(snap) = snapshots.iter().find(|s| s.hash.starts_with(hash)) {
                    if let Ok(current) = fs::read_to_string(&snap.path) {
                        let snapshot_content = fs::read_to_string(&snap.backup_path).unwrap_or_default();
                        let unified = DiffEngine::unified_diff(&snapshot_content, &current, &snap.path);

                    } else {

                    }
                } else {
                    json!({"type": "output", "data": format!(">> ERR: Hash {} not found.", hash)}).to_string()
                }
            } else {
                json!({"type": "output", "data": ">> Usage: diff <hash>"}).to_string()
            }
        },

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct Snapshot {
    pub timestamp: String,
    pub path: String,
    pub author: String,
    pub hash: String,
    pub backup_path: String,
}

pub struct LazarusVault;

impl LazarusVault {
    pub fn list_snapshots() -> Result<Vec<Snapshot>> {
        let ledger_path = Path::new(".lazarus/ledger.json");
        if !ledger_path.exists() { return Ok(vec![]); }
        let data = fs::read_to_string(ledger_path)?;
        Ok(serde_json::from_str(&data)?)
    }

    pub fn auto_snapshot(target: &str, reason: &str) -> Result<String> {
        let content = fs::read_to_string(target)
            .with_context(|| format!("Failed to read target {}", target))?;
        let author = format!("AUTO_SHIELD: {}", reason);
        Self::seal_patch(target, &content, &author)
    }

    pub fn seal_patch(target: &str, content: &str, author: &str) -> Result<String> {
        let root = Path::new(".lazarus/snapshots");
        fs::create_dir_all(root)?;

        let mut hasher = Sha256::new();
        hasher.update(content);
        let hash = hasher.finalize().iter().map(|b| format!("{:02x}", b)).collect::<String>();

        let timestamp = Utc::now().to_rfc3339();
        let backup_name = format!("{}.snap", Utc::now().timestamp());
        let backup_path = root.join(&backup_name);
        fs::write(&backup_path, content)?;
        
        let mut ledger = Self::list_snapshots()?;
        ledger.push(Snapshot {
            timestamp,
            path: target.to_string(),
            author: author.to_string(),
            hash: hash.clone(),
            backup_path: backup_path.to_string_lossy().to_string(),
        });
        
        fs::write(".lazarus/ledger.json", serde_json::to_string_pretty(&ledger)?)?;
        Ok(hash)
    }
}

use std::fs;
use syn::visit::Visit;
use syn::parse_file;

pub struct SyntaxSentinel;

struct SafetyVisitor {
    pub violations: Vec<String>,
}

impl<'ast> Visit<'ast> for SafetyVisitor {
    fn visit_expr_call(&mut self, node: &syn::ExprCall) {
        let call_str = quote::quote!(#node).to_string();
        let banned = [
            "std :: process :: Command",
            "std :: fs :: remove_file",
            "std :: fs :: remove_dir",
            "process :: exit",
        ];
        for b in &banned {
            if call_str.contains(b) {
                self.violations.push(format!("AST: Banned call detected — {}", b));
            }
        }
        syn::visit::visit_expr_call(self, node);
    }

    fn visit_macro(&mut self, node: &syn::Macro) {
        let mac_str = quote::quote!(#node).to_string();
        if mac_str.contains("include_str") || mac_str.contains("include_bytes") {
            self.violations.push("AST: Dynamic include macro detected".into());
        }
        syn::visit::visit_macro(self, node);
    }
}

impl SyntaxSentinel {
    pub fn validate_safety(path: &str, content: &str) -> Result<(), String> {
        // Layer 1: fast string pre-screen
        const DANGEROUS: &[&str] = &[
            "rm -rf",
            "unlink(",
            "exec(",
            "system(",
        ];
        for pat in DANGEROUS {
            if content.contains(pat) {
                return Err(format!("Blocked: Dangerous pattern '{}'", pat));
            }
        }

        // Layer 2: infinite loop protection
        if content.contains("loop {")
            && !content.contains("sleep")
            && !content.contains("break")
            && !content.contains("return")
        {
            return Err("Blocked: Infinite loop without yield (ARM64 risk)".into());
        }

        // Layer 3: structural collapse protection
        if path.ends_with(".rs") {
            if let Ok(existing) = fs::read_to_string(path) {
                let old_lines = existing.lines().count();
                let new_lines = content.lines().count();
                if old_lines > 60 && new_lines < (old_lines / 3) {
                    return Err(format!(
                        "Structural collapse blocked: reduced {} from {} to {} lines",
                        path, old_lines, new_lines
                    ));
                }
            }
        }

        // Layer 4: AST analysis (Rust files only)
        if path.ends_with(".rs") {
            match parse_file(content) {
                Ok(ast) => {
                    let mut visitor = SafetyVisitor { violations: vec![] };
                    visitor.visit_file(&ast);
                    if !visitor.violations.is_empty() {
                        return Err(format!(
                            "AST Safety Block:\n{}",
                            visitor.violations.join("\n")
                        ));
                    }
                }
                Err(e) => {
                    return Err(format!("Parse error (will not write broken code): {}", e));
                }
            }
        }

        // Layer 5: architecture protection
        if path.ends_with("main.rs") && content.contains("mod nexus_core") {
            return Err("Architectural sabotage: forbidden module nesting".into());
        }

        Ok(())
    }
}

    Lazy::new(|| Arc::new(RwLock::new(ProjectAtlas::default())));

pub struct KnowledgeBroker;

impl KnowledgeBroker {
    pub async fn refresh_atlas() {
        let mut entries = Vec::new();
        let re_fn = Regex::new(r"pub (?:async )?fn ([a-zA-Z0-9_]+)").unwrap();

        if let Ok(it) = WalkDir::new("src")
            .into_iter()
            .collect::<Result<Vec<_>, _>>()
        {
            for entry in it
                .into_iter()
                .filter(|e| e.path().extension().is_some_and(|ext| ext == "rs"))
            {
                let path = entry.path().to_string_lossy().to_string();
                if let Ok(content) = fs::read_to_string(entry.path()) {
                    for cap in re_fn.captures_iter(&content) {
                        entries.push(AtlasEntry {
                            name: cap[1].to_string(),
                            kind: "function".into(),
                            path: path.clone(),
                            signature: cap[0].to_string(),
                        });
                    }
                }
            }
        }

        let mut cache = CACHED_ATLAS.write().await;
        *cache = ProjectAtlas {
            entries,
            timestamp: chrono::Utc::now().timestamp(),
        };
    }

    pub async fn get_atlas() -> ProjectAtlas {
        let cache = CACHED_ATLAS.read().await;
        (*cache).clone()
    }

    }

            .iter()

            timestamp: Utc::now().to_rfc3339(),
            prompt,
            solution_code: code,
        };

        if let Ok(bytes) = serde_json::to_vec(&memory) {
            let _ = self.db.insert(key, bytes);
        }

        // Retention limit: 20 memories for ARM64 RAM safety
        let count = self.db.len();
        if count > 20 {
            self.purge_to_archive("Auto-maintenance trigger".to_string());
        }
    }

    pub fn purge_to_archive(&self, summary: String) {
        let memories = self.load_all();
        let timestamp = Utc::now().format("%Y%m%d_%H%M%S");
        let archive_content = format!(
            "# Dream Cycle: {}\n\nSummary: {}\n\n---\n{:?}",
            timestamp, summary, memories
        );

        let _ = std::fs::create_dir_all("archives");
        let _ = std::fs::write(
            format!("archives/dream_{}.md", timestamp),
            archive_content,
        );

        // Clear the sled tree
        let _ = self.db.clear();
        let _ = self.db.flush();
        println!(">> [MEMORY] Dream cycle complete. {} memories archived.", memories.len());
    }

            detected_stack: "UNKNOWN".into(),
        };
    };

    println!(">> [POLYGLOT] Building {} project at {}...", stack, path);

    match output {

                String::from_utf8_lossy(&out.stdout),
                String::from_utf8_lossy(&out.stderr)
            );
            BuildResult {
                success: out.status.success(),
                log,
                detected_stack: stack.into(),
            }
        }
        Err(e) => BuildResult {

        },
    }

pub fn generate_project_tree() -> String {
    let mut tree = String::from("PROJECT STRUCTURE:\n");
    for entry in WalkDir::new(".")
        .max_depth(2)
        .into_iter()
        .filter_map(|e| e.ok())
    {
        let path = entry.path();
        let display = path.display().to_string();

        if display.contains("/target")
            || display.contains("/.git")
            || display.contains("/node_modules")
            || display.contains("/dist")
        {
            continue;
        }

        let depth = display.matches('/').count();
        let indent = "  ".repeat(depth);
        tree.push_str(&format!(
            "{}{}\n",
            indent,
            path.file_name().unwrap_or_default().to_string_lossy()
        ));
    }
    tree
}

    }

pub struct SystemMonitor {
    sys: Arc<Mutex<System>>,
}

impl SystemMonitor {
    pub fn new() -> Self {
        SystemMonitor { sys: Arc::new(Mutex::new(System::new_all())) }
    }

    pub fn get_vitals(&self) -> VitalsSnapshot {
        let mut sys = self.sys.lock().unwrap();
        sys.refresh_all();
        let used = sys.used_memory() as f32 / 1024.0 / 1024.0 / 1024.0;
        let total = sys.total_memory() as f32 / 1024.0 / 1024.0 / 1024.0;
        let cpu = sys.global_cpu_info().cpu_usage();

        VitalsSnapshot {
            status: if used > (total * 0.9) { "CRITICAL" } else { "ONLINE" }.into(),
            cpu_usage: cpu,
            cpu_load: cpu,
            ram_usage: format!("{:.2}GB / {:.1}GB", used, total),
            ram_used_gb: used,
            resonance: 0.98,
            temperature: crate::thermal_governor::ThermalGovernor::read_temp(),
        }
    }

    pub fn get_vital_signs_string(&self) -> String {
        let v = self.get_vitals();
        format!("CPU: {:.1}% | RAM: {}", v.cpu_usage, v.ram_usage)
    }
}

#[derive(Debug)]
pub enum NexusError {
    Safety(String),
    Patch(String),
    Io(std::io::Error),
    Json(serde_json::Error),
    Command(String),
    NotFound(String),
    Other(String),
}

impl fmt::Display for NexusError {
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        match self {
            NexusError::Safety(msg) => write!(f, "Safety violation: {}", msg),
            NexusError::Patch(msg) => write!(f, "Patch failed: {}", msg),
            NexusError::Io(e) => write!(f, "IO error: {}", e),
            NexusError::Json(e) => write!(f, "JSON error: {}", e),
            NexusError::Command(msg) => write!(f, "Command error: {}", msg),
            NexusError::NotFound(msg) => write!(f, "Not found: {}", msg),
            NexusError::Other(msg) => write!(f, "{}", msg),
        }
    }
}

impl std::error::Error for NexusError {}

impl From<std::io::Error> for NexusError {
    fn from(e: std::io::Error) -> Self { NexusError::Io(e) }
}

impl From<serde_json::Error> for NexusError {
    fn from(e: serde_json::Error) -> Self { NexusError::Json(e) }
}

        Self { warn_threshold: 68.0, crit_threshold: 78.0 }
    }
}

impl ThermalGovernor {
    pub fn new() -> Self { Self::default() }

    pub fn read_temp() -> f32 {
        fs::read_to_string("/sys/class/thermal/thermal_zone0/temp")
            .ok()
            .and_then(|s| s.trim().parse::<f32>().ok())
            .map(|t| t / 1000.0)
            .unwrap_or(0.0)
    }

    pub fn get_state(&self) -> CoolingState {
        let t = Self::read_temp();
        if t >= self.crit_threshold { CoolingState::Critical }
        else if t >= self.warn_threshold { CoolingState::Throttled }
        else { CoolingState::Normal }
    }

    pub async fn apply_cooldown(&self) {
        match self.get_state() {
            CoolingState::Critical => { sleep(Duration::from_secs(8)).await; }
            CoolingState::Throttled => { sleep(Duration::from_secs(3)).await; }
            _ => {}
        }
    }
}

use crate::nexus_error::NexusError;

pub struct DiffEngine;

impl DiffEngine {
    pub fn apply(original: &str, search: &str, replace: &str) -> Result<String, NexusError> {
        let clean_search = search.trim();
        if clean_search.is_empty() { return Ok(original.to_string()); }
        if original.contains(search) { return Ok(original.replace(search, replace)); }

        // Fuzzy match by normalized lines
        let orig_lines: Vec<&str> = original.lines().collect();
        let search_lines: Vec<String> = search.lines()
            .map(|l| l.trim().to_string())
            .filter(|l| !l.is_empty())
            .collect();

        if search_lines.is_empty() { return Ok(original.to_string()); }

        for i in 0..=orig_lines.len().saturating_sub(search_lines.len()) {
            let mut match_count = 0;
            for (j, sline) in search_lines.iter().enumerate() {
                if orig_lines[i + j].trim() == sline { match_count += 1; }
            }
            if match_count >= (search_lines.len() * 9 / 10).max(1) {
                let mut new_lines = orig_lines.clone();
                new_lines.splice(i..i + search_lines.len(), replace.lines());
                return Ok(new_lines.join("\n"));
            }
        }
        Err(NexusError::Patch("DNA mismatch: Fuzzy search block not found.".into()))
    }

    pub fn unified_diff(old: &str, new: &str, path: &str) -> String {
        let mut output = vec![format!("--- a/{}", path), format!("+++ b/{}", path)];
        let old_l: Vec<&str> = old.lines().collect();
        let new_l: Vec<&str> = new.lines().collect();
        // Simplified diff for UI stability
        output.push(format!("@@ -1,{} +1,{} @@", old_l.len(), new_l.len()));
        for l in old_l { output.push(format!("-{}", l)); }
        for l in new_l { output.push(format!("+{}", l)); }
        output.join("\n")
    }
}

use serde::{Deserialize, Serialize};
use std::fs;

use chrono::Utc;
use crate::agent_protocol::AgentAction;
use crate::neural_agent::ContextPacket;
use crate::server::AppState;
use std::sync::Arc;

#[derive(Serialize, Deserialize, Clone, Debug, PartialEq)]
pub enum StepStatus {
    Pending,
    Running,
    Done,
    Failed(String),
}

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct MissionStep {
    pub id: usize,
    pub description: String,
    pub action: Option<AgentAction>,
    pub status: StepStatus,
    pub result: Option<String>,
}

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct Mission {
    pub id: String,
    pub goal: String,
    pub steps: Vec<MissionStep>,
    pub created_at: String,
    pub completed: bool,
}

pub struct MissionPlanner;

impl MissionPlanner {
    /// Decompose a high-level goal into ordered steps using the LLM
    pub async fn decompose(
        state: &Arc<AppState>,
        goal: &str,
    ) -> anyhow::Result<Mission> {
        let prompt = format!(
            "You are NEXUS mission planner. Decompose this goal into 3-7 ordered steps.\n\
            GOAL: {}\n\n\
            Reply ONLY with a JSON array. Each step must have:\n\
            - description: string (what this step does)\n\
            - action: one of WriteFile | PatchFile | RunCommand | Message\n\
            - path: string (file path if applicable, else null)\n\
            - command: string (if RunCommand)\n\
            Example:\n\
            [\n\
              {{\"description\": \"Read current server.rs\", \"action\": \"RunCommand\", \"command\": \"cat src/server.rs\", \"path\": null}},\n\
              {{\"description\": \"Add auth middleware\", \"action\": \"PatchFile\", \"path\": \"src/server.rs\", \"command\": null}}\n\
            ]",
            goal
        );

        let ctx = ContextPacket::default();
        let (raw, _) = state
            .agent
            .prepare_and_run(&state.client, &state.api_key, &prompt, ctx, 0.3)
            .await?;

        let clean = raw.replace("```json", "").replace("```", "");
        let json_str = clean.trim();

        #[derive(Deserialize)]
        struct RawStep {
            description: String,
            action: String,
            path: Option<String>,
            command: Option<String>,
        }

        let raw_steps: Vec<RawStep> = serde_json::from_str(json_str)
            .unwrap_or_default();

        let steps = raw_steps
            .into_iter()
            .enumerate()
            .map(|(i, s)| {
                let action = match s.action.as_str() {
                    "RunCommand" => s.command.map(|c| AgentAction::RunCommand { command: c }),
                    "WriteFile" => s.path.clone().map(|p| AgentAction::WriteFile {
                        path: p,
                        content: String::new(),
                    }),
                    "PatchFile" => s.path.clone().map(|p| AgentAction::PatchFile {
                        path: p,
                        search: String::new(),
                        replace: String::new(),
                    }),
                    _ => Some(AgentAction::Message {
                        message: s.description.clone(),
                    }),
                };
                MissionStep {
                    id: i,
                    description: s.description,
                    action,
                    status: StepStatus::Pending,
                    result: None,
                }
            })
            .collect();

        let mission = Mission {
            id: Utc::now().timestamp_micros().to_string(),
            goal: goal.to_string(),
            steps,
            created_at: Utc::now().to_rfc3339(),
            completed: false,
        };

        Self::persist(&mission)?;
        Ok(mission)
    }

    /// Execute all pending steps in order, broadcasting progress
    pub async fn execute(
        state: &Arc<AppState>,
        mut mission: Mission,
    ) -> anyhow::Result<Mission> {
        use crate::executor::Executor;

        for step in mission.steps.iter_mut() {
            if step.status != StepStatus::Pending {
                continue;
            }

            step.status = StepStatus::Running;
            let _ = state.tx.send(format!(
                ">> [MISSION {}] Step {}: {}",
                &mission.id[..8], step.id, step.description
            ));

            if let Some(action) = step.action.clone() {
                match Executor::execute(vec![action]).await {
                    Ok(results) => {
                        let result_str = results.join("\n");
                        let failed = result_str.contains("REJECTED")
                            || result_str.contains("FAIL");

                        if failed {
                            step.status = StepStatus::Failed(result_str.clone());
                            step.result = Some(result_str);
                            let _ = state.tx.send(format!(
                                ">> [MISSION] Step {} FAILED — halting mission.",
                                step.id
                            ));
                            break;
                        } else {
                            step.status = StepStatus::Done;
                            step.result = Some(result_str);
                        }
                    }
                    Err(e) => {
                        step.status = StepStatus::Failed(e.to_string());
                        break;
                    }
                }
            } else {
                step.status = StepStatus::Done;
                step.result = Some("No action required".into());
            }

            tokio::time::sleep(tokio::time::Duration::from_millis(200)).await;
        }

        mission.completed = mission
            .steps
            .iter()
            .all(|s| s.status == StepStatus::Done);

        Self::persist(&mission)?;
        Ok(mission)
    }

    fn persist(mission: &Mission) -> anyhow::Result<()> {
        let _ = fs::create_dir_all("missions");
        let path = format!("missions/{}.json", mission.id);
        let json = serde_json::to_string_pretty(mission)?;
        fs::write(path, json)?;
        Ok(())
    }

    pub fn load_all() -> Vec<Mission> {
        let Ok(entries) = fs::read_dir("missions") else { return vec![] };
        entries
            .flatten()
            .filter(|e| e.path().extension().is_some_and(|x| x == "json"))
            .filter_map(|e| fs::read_to_string(e.path()).ok())
            .filter_map(|s| serde_json::from_str(&s).ok())
            .collect()
    }
}

    ) -> Result<(String, u128)> {

        let url = format!("https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:streamGenerateContent?key={}", api_key);

        let payload = serde_json::json!({

            "generationConfig": { "temperature": temp }
        });

        let mut full_text = String::new();

        while let Some(item) = stream.next().await {
            if let Ok(bytes) = item {
                // In a real implementation, you'd parse the SSE chunks here.
                // For now, we'll collect the raw response to clear the build error.

async fn handle_file_tree() -> Json<Vec<DirEntry>> {
    let mut entries = Vec::new();
    if let Ok(rd) = std::fs::read_dir("src") {
        for entry in rd.flatten() {
            let p = entry.path();
            entries.push(DirEntry {
                name: entry.file_name().to_string_lossy().to_string(),
                is_dir: p.is_dir(), path: p.to_string_lossy().to_string(), children: vec![],
            });
        }
    }
    Json(entries)
}

async fn handle_read_file(Query(p): Query<FileParams>) -> impl IntoResponse {
    match std::fs::read_to_string(&p.path) {
        Ok(c) => (StatusCode::OK, Json(FileContent { path: p.path.clone(), content: c })).into_response(),
        Err(_) => (StatusCode::NOT_FOUND, "File not found").into_response(),
    }
}

async fn handle_write_file(Json(payload): Json<FileContent>) -> impl IntoResponse {
    let path = Path::new(&payload.path);
    if let Some(p) = path.parent() { let _ = std::fs::create_dir_all(p); }
    match std::fs::write(path, &payload.content) {
        Ok(_) => StatusCode::OK.into_response(),
        Err(_) => StatusCode::INTERNAL_SERVER_ERROR.into_response(),
    }
}

async fn static_handler(uri: Uri) -> impl IntoResponse {
    let mut path = uri.path().trim_start_matches('/').to_string();
    if path.is_empty() { path = "index.html".to_string(); }
    match Assets::get(&path) {
        Some(content) => {
            let mime = mime_guess::from_path(&path).first_or_octet_stream();
            ([(header::CONTENT_TYPE, mime.as_ref())], content.data).into_response()
        }
        None => {
            let index = Assets::get("index.html").expect("index.html missing");
            ([(header::CONTENT_TYPE, "text/html")], index.data).into_response()
        }
    }
}

async fn ws_ghost_handler(ws: WebSocketUpgrade, State(state): State<Arc<AppState>>) -> impl IntoResponse {
    ws.on_upgrade(move |socket| async move {
        let mut rx = state.tx.subscribe();
        let mut socket = socket;
        while let Ok(msg) = rx.recv().await { 
            if socket.send(Message::Text(msg)).await.is_err() { break; } 
        }
    })
}

                    ));

                    // Back off exponentially up to 10 minutes
                    backoff_secs = (backoff_secs * 2).min(600);

                    let _ = state.tx.send(format!(">> [HEALER] cargo check failed to run: {}", e));

use serde::{Serialize, Deserialize};
use std::sync::Arc;

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct ArbiterDecision {
    pub goal: String,
    pub confidence: f32,
    pub risk_score: f32,
    pub recommended_action: String,
    pub rationale: String,
    pub lazarus_hash: Option<String>,
    pub alternatives: Vec<String>,
    pub timestamp: String,
}

pub struct ArbiterAgent {
    pub core: Arc<MicroAgent>,
    pub orchestrator: Option<Arc<Orchestrator>>,
}

impl ArbiterAgent {
    pub fn new(core: Arc<MicroAgent>, orchestrator: Option<Arc<Orchestrator>>) -> Self {
        Self { core, orchestrator }
    }

    pub async fn arbitrate(
        &self,
        goal: &str,
        constraints: Vec<String>,
        client: &reqwest::Client,
        api_key: &str,
    ) -> Result<ArbiterDecision> {
        let prompt = format!("Analyze risk for GOAL: {}\nCONSTRAINTS: {:?}", goal, constraints);
        let ctx = ContextPacket { system_prompt: Some("Risk-aware Arbiter".into()), ..Default::default() };

        let (response, _) = self.core.prepare_and_run(client, api_key, &prompt, ctx, 0.25).await?;
        let mut decision: ArbiterDecision = serde_json::from_str(&response).unwrap_or_else(|_| ArbiterDecision {
            goal: goal.to_string(), confidence: 0.5, risk_score: 0.5,
            recommended_action: response.clone(), rationale: "Fallback".into(),
            lazarus_hash: None, alternatives: vec![], timestamp: chrono::Utc::now().to_rfc3339(),
        });

        if decision.risk_score > 0.7 {
            if let Ok(hash) = LazarusVault::auto_snapshot("src/main.rs", "High-risk trigger") {
                decision.lazarus_hash = Some(hash);
            }
        }

        Ok(decision)
    }
}
